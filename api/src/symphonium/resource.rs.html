5/// A resource of raw f32 audio samples stored in deinterleaved format.
6///
7/// This struct stores samples
8/// in their native sample format when possible to save memory.
9#[derive(Clone)]
10pub struct DecodedAudioF32 {
11    pub data: Vec<Vec<f32>>,
12    pub sample_rate: u32,
13}
14
15impl DecodedAudioF32 {
16    pub fn new(data: Vec<Vec<f32>>, sample_rate: u32) -> Self {
17        let frames = data[0].len();
18
19        for ch in data.iter().skip(1) {
20            assert_eq!(ch.len(), frames);
21        }
22
23        Self { data, sample_rate }
24    }
25
26    /// The number of channels in this resource.
27    pub fn channels(&self) -> usize {
28        self.data.len()
29    }
30
31    /// The length of this resource in frames (length of a single channel in
32    /// samples).
33    pub fn frames(&self) -> usize {
34        self.data[0].len()
35    }
36
37    /// Get the data in interleaved format.
38    pub fn as_interleaved(&self) -> Vec<f32> {
39        fast_interleave::interleave_to_vec_variable(
40            &self.data,
41            0..self.frames(),
42            NonZeroUsize::new(self.data.len()).unwrap(),
43        )
44    }
45}

73/// A resource of raw audio samples stored in deinterleaved format.
74///
75/// This struct stores samples
76/// in their native sample format when possible to save memory.
77#[derive(Debug, Clone)]
78pub struct DecodedAudio {
79    resource_type: DecodedAudioType,
80    sample_rate: u32,
81    channels: usize,
82    frames: usize,
83}
84
85/// The format of the raw audio samples stored in deinterleaved format.
86///
87/// Note that there is no option for U32/I32. This is because in processing
88/// we ultimately use `f32` for everything anyway. We only store the other
89/// types to save memory.
90///
91/// Also note there is no option for `I24` (signed 24 bit). This is because
92/// converting two's compliment numbers from 32 bit to 24 bit and vice-versa
93/// is tricky and less performant. Instead, just convert the data into `u24`
94/// format.
95#[derive(Clone)]
96pub enum DecodedAudioType {
97    U8(Vec<Vec<u8>>),
98    U16(Vec<Vec<u16>>),
99    /// The endianness of the samples must be the native endianness of the
100    /// target platform.
101    U24(Vec<Vec<[u8; 3]>>),
102    S8(Vec<Vec<i8>>),
103    S16(Vec<Vec<i16>>),
104    F32(Vec<Vec<f32>>),
105    F64(Vec<Vec<f64>>),
106}
107
108impl DecodedAudio {
109    pub fn new(resource_type: DecodedAudioType, sample_rate: u32) -> Self {
110        let (channels, frames) = match &resource_type {
111            DecodedAudioType::U8(b) => {
112                let len = b[0].len();
113
114                for ch in b.iter().skip(1) {
115                    assert_eq!(ch.len(), len);
116                }
117
118                (b.len(), len)
119            }
120            DecodedAudioType::U16(b) => {
121                let len = b[0].len();
122
123                for ch in b.iter().skip(1) {
124                    assert_eq!(ch.len(), len);
125                }
126
127                (b.len(), len)
128            }
129            DecodedAudioType::U24(b) => {
130                let len = b[0].len();
131
132                for ch in b.iter().skip(1) {
133                    assert_eq!(ch.len(), len);
134                }
135
136                (b.len(), len)
137            }
138            DecodedAudioType::S8(b) => {
139                let len = b[0].len();
140
141                for ch in b.iter().skip(1) {
142                    assert_eq!(ch.len(), len);
143                }
144
145                (b.len(), len)
146            }
147            DecodedAudioType::S16(b) => {
148                let len = b[0].len();
149
150                for ch in b.iter().skip(1) {
151                    assert_eq!(ch.len(), len);
152                }
153
154                (b.len(), len)
155            }
156            DecodedAudioType::F32(b) => {
157                let len = b[0].len();
158
159                for ch in b.iter().skip(1) {
160                    assert_eq!(ch.len(), len);
161                }
162
163                (b.len(), len)
164            }
165            DecodedAudioType::F64(b) => {
166                let len = b[0].len();
167
168                for ch in b.iter().skip(1) {
169                    assert_eq!(ch.len(), len);
170                }
171
172                (b.len(), len)
173            }
174        };

177            resource_type,
178            sample_rate,
179            channels,
180            frames,
181        }

204    /// Fill the buffer with samples from the given `channel`, starting from the
205    /// given `frame`.
206    ///
207    /// If the length of the buffer exceeds the length of the PCM resource, then
208    /// the remaining samples will be filled with zeros.
209    ///
210    /// This returns the number of frames that were copied into the buffer. (If
211    /// this number is less than the length of `buf`, then it means that the
212    /// remaining samples were filled with zeros.)
213    ///
214    /// The will return an error if the given channel does not exist.
215    pub fn fill_channel(&self, channel: usize, frame: usize, buf: &mut [f32]) -> Result<usize, ()> {
216        if channel >= self.channels {
217            return Err(());
218        }
219
220        if frame >= self.frames {
221            // Out of range, fill with zeros instead.
222            buf.fill(0.0);
223            return Ok(0);
224        }
225
226        let fill_frames = if frame + buf.len() > self.frames {
227            // Fill the out-of-range part with zeros.
228            let fill_frames = self.frames - frame;
229            buf[fill_frames..].fill(0.0);
230            fill_frames
231        } else {
232            buf.len()
233        };
234
235        let buf_part = &mut buf[0..fill_frames];
236
237        match &self.resource_type {
238            DecodedAudioType::U8(pcm) => {
239                let pcm_part = &pcm[channel][frame..frame + fill_frames];
240
241                for i in 0..fill_frames {
242                    buf_part[i] = FromSample::from_sample(pcm_part[i]);
243                }
244            }
245            DecodedAudioType::U16(pcm) => {
246                let pcm_part = &pcm[channel][frame..frame + fill_frames];
247
248                for i in 0..fill_frames {
249                    buf_part[i] = FromSample::from_sample(pcm_part[i]);
250                }
251            }
252            DecodedAudioType::U24(pcm) => {
253                let pcm_part = &pcm[channel][frame..frame + fill_frames];
254
255                for i in 0..fill_frames {
256                    let b = &pcm_part[i];
257
258                    let s = if cfg!(target_endian = "little") {
259                        // In little-endian the MSB is the last byte.
260                        u24(u32::from_le_bytes([b[0], b[1], b[2], 0]))
261                    } else {
262                        // In big-endian the MSB is the first byte.
263                        u24(u32::from_be_bytes([0, b[1], b[2], b[3]]))
264                    };
265
266                    buf_part[i] = FromSample::from_sample(s);
267                }
268            }
269            DecodedAudioType::S8(pcm) => {
270                let pcm_part = &pcm[channel][frame..frame + fill_frames];
271
272                for i in 0..fill_frames {
273                    buf_part[i] = FromSample::from_sample(pcm_part[i]);
274                }
275            }
276            DecodedAudioType::S16(pcm) => {
277                let pcm_part = &pcm[channel][frame..frame + fill_frames];
278
279                for i in 0..fill_frames {
280                    buf_part[i] = FromSample::from_sample(pcm_part[i]);
281                }
282            }
283            DecodedAudioType::F32(pcm) => {
284                let pcm_part = &pcm[channel][frame..frame + fill_frames];
285
286                buf_part.copy_from_slice(pcm_part);
287            }
288            DecodedAudioType::F64(pcm) => {
289                let pcm_part = &pcm[channel][frame..frame + fill_frames];
290
291                for i in 0..fill_frames {
292                    buf_part[i] = pcm_part[i] as f32;
293                }
294            }
295        }
296
297        Ok(fill_frames)
298    }

300    /// Fill the stereo buffer with samples, starting from the given `frame`.
301    ///
302    /// If this resource has only one channel, then both channels will be
303    /// filled with the same data.
304    ///
305    /// If the length of the buffer exceeds the length of the PCM resource, then
306    /// the remaining samples will be filled with zeros.
307    ///
308    /// This returns the number of frames that were copied into the buffer. (If
309    /// this number is less than the length of `buf`, then it means that the
310    /// remaining samples were filled with zeros.)
311    pub fn fill_stereo(&self, frame: usize, buf_l: &mut [f32], buf_r: &mut [f32]) -> usize {
312        let buf_len = buf_l.len().min(buf_r.len());
313
314        if self.channels == 1 {
315            let fill_frames = self.fill_channel(0, frame, buf_l).unwrap();
316            buf_r.copy_from_slice(buf_l);
317            return fill_frames;
318        }
319
320        if frame >= self.frames {
321            // Out of range, fill with zeros instead.
322            buf_l.fill(0.0);
323            buf_r.fill(0.0);
324            return 0;
325        }
326
327        let fill_frames = if frame + buf_len > self.frames {
328            // Fill the out-of-range part with zeros.
329            let fill_frames = self.frames - frame;
330            buf_l[fill_frames..].fill(0.0);
331            buf_r[fill_frames..].fill(0.0);
332            fill_frames
333        } else {
334            buf_len
335        };
336
337        let buf_l_part = &mut buf_l[0..fill_frames];
338        let buf_r_part = &mut buf_r[0..fill_frames];
339
340        match &self.resource_type {
341            DecodedAudioType::U8(pcm) => {
342                let pcm_l_part = &pcm[0][frame..frame + fill_frames];
343                let pcm_r_part = &pcm[1][frame..frame + fill_frames];
344
345                for i in 0..fill_frames {
346                    buf_l_part[i] = FromSample::from_sample(pcm_l_part[i]);
347                    buf_r_part[i] = FromSample::from_sample(pcm_r_part[i]);
348                }
349            }
350            DecodedAudioType::U16(pcm) => {
351                let pcm_l_part = &pcm[0][frame..frame + fill_frames];
352                let pcm_r_part = &pcm[1][frame..frame + fill_frames];
353
354                for i in 0..fill_frames {
355                    buf_l_part[i] = FromSample::from_sample(pcm_l_part[i]);
356                    buf_r_part[i] = FromSample::from_sample(pcm_r_part[i]);
357                }
358            }
359            DecodedAudioType::U24(pcm) => {
360                let pcm_l_part = &pcm[0][frame..frame + fill_frames];
361                let pcm_r_part = &pcm[1][frame..frame + fill_frames];
362
363                for i in 0..fill_frames {
364                    let b_l = &pcm_l_part[i];
365                    let b_r = &pcm_r_part[i];
366
367                    let (s_l, s_r) = if cfg!(target_endian = "little") {
368                        // In little-endian the MSB is the last byte. Drop it.
369                        (
370                            u24(u32::from_le_bytes([b_l[0], b_l[1], b_l[2], 0])),
371                            u24(u32::from_le_bytes([b_r[0], b_r[1], b_r[2], 0])),
372                        )
373                    } else {
374                        // In big-endian the MSB is the first byte. Drop it.
375                        (
376                            u24(u32::from_be_bytes([0, b_l[1], b_l[2], b_l[3]])),
377                            u24(u32::from_be_bytes([0, b_r[1], b_r[2], b_r[3]])),
378                        )
379                    };
380
381                    buf_l_part[i] = FromSample::from_sample(s_l);
382                    buf_r_part[i] = FromSample::from_sample(s_r);
383                }
384            }
385            DecodedAudioType::S8(pcm) => {
386                let pcm_l_part = &pcm[0][frame..frame + fill_frames];
387                let pcm_r_part = &pcm[1][frame..frame + fill_frames];
388
389                for i in 0..fill_frames {
390                    buf_l_part[i] = FromSample::from_sample(pcm_l_part[i]);
391                    buf_r_part[i] = FromSample::from_sample(pcm_r_part[i]);
392                }
393            }
394            DecodedAudioType::S16(pcm) => {
395                let pcm_l_part = &pcm[0][frame..frame + fill_frames];
396                let pcm_r_part = &pcm[1][frame..frame + fill_frames];
397
398                for i in 0..fill_frames {
399                    buf_l_part[i] = FromSample::from_sample(pcm_l_part[i]);
400                    buf_r_part[i] = FromSample::from_sample(pcm_r_part[i]);
401                }
402            }
403            DecodedAudioType::F32(pcm) => {
404                let pcm_l_part = &pcm[0][frame..frame + fill_frames];
405                let pcm_r_part = &pcm[1][frame..frame + fill_frames];
406
407                buf_l_part.copy_from_slice(pcm_l_part);
408                buf_r_part.copy_from_slice(pcm_r_part);
409            }
410            DecodedAudioType::F64(pcm) => {
411                let pcm_l_part = &pcm[0][frame..frame + fill_frames];
412                let pcm_r_part = &pcm[1][frame..frame + fill_frames];
413
414                for i in 0..fill_frames {
415                    buf_l_part[i] = pcm_l_part[i] as f32;
416                    buf_r_part[i] = pcm_r_part[i] as f32;
417                }
418            }
419        }
420
421        fill_frames
422    }

480mod tests {
481    use super::*;

483    #[test]
484    fn pcm_fill_range_test() {
485        let test_pcm =
486            DecodedAudio::new(DecodedAudioType::F32(vec![vec![1.0, 2.0, 3.0, 4.0]]), 44100);
487
488        let mut out_buf: [f32; 8] = [10.0; 8];
489
490        let fill_frames = test_pcm.fill_channel(0, 0, &mut out_buf[0..4]);
491        assert_eq!(fill_frames, Ok(4));
492        assert_eq!(&out_buf[0..4], &[1.0, 2.0, 3.0, 4.0]);
493
494        out_buf = [10.0; 8];
495        let fill_frames = test_pcm.fill_channel(0, 0, &mut out_buf[0..5]);
496        assert_eq!(fill_frames, Ok(4));
497        assert_eq!(&out_buf[0..5], &[1.0, 2.0, 3.0, 4.0, 0.0]);
498
499        out_buf = [10.0; 8];
500        let fill_frames = test_pcm.fill_channel(0, 2, &mut out_buf[0..4]);
501        assert_eq!(fill_frames, Ok(2));
502        assert_eq!(&out_buf[0..4], &[3.0, 4.0, 0.0, 0.0]);
503
504        out_buf = [10.0; 8];
505        let fill_frames = test_pcm.fill_channel(0, 3, &mut out_buf[0..4]);
506        assert_eq!(fill_frames, Ok(1));
507        assert_eq!(&out_buf[0..4], &[4.0, 0.0, 0.0, 0.0]);
508
509        out_buf = [10.0; 8];
510        let fill_frames = test_pcm.fill_channel(0, 4, &mut out_buf[0..4]);
511        assert_eq!(fill_frames, Ok(0));
512        assert_eq!(&out_buf[0..4], &[0.0, 0.0, 0.0, 0.0]);
513
514        out_buf = [10.0; 8];
515        let fill_frames = test_pcm.fill_channel(0, 1, &mut out_buf[0..2]);
516        assert_eq!(fill_frames, Ok(2));
517        assert_eq!(&out_buf[0..2], &[2.0, 3.0]);
518
519        out_buf = [10.0; 8];
520        let fill_frames = test_pcm.fill_channel(0, 1, &mut out_buf[0..4]);
521        assert_eq!(fill_frames, Ok(3));
522        assert_eq!(&out_buf[0..4], &[2.0, 3.0, 4.0, 0.0]);
523    }
