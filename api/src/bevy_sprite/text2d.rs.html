33/// The top-level 2D text component.
34///
35/// Adding `Text2d` to an entity will pull in required components for setting up 2d text.
36/// [Example usage.](https://github.com/bevyengine/bevy/blob/latest/examples/2d/text2d.rs)
37///
38/// The string in this component is the first 'text span' in a hierarchy of text spans that are collected into
39/// a [`ComputedTextBlock`]. See `TextSpan` for the component used by children of entities with [`Text2d`].
40///
41/// With `Text2d` the `justify` field of [`TextLayout`] only affects the internal alignment of a block of text and not its
42/// relative position, which is controlled by the [`Anchor`] component.
43/// This means that for a block of text consisting of only one line that doesn't wrap, the `justify` field will have no effect.
44///
45///
46/// ```
47/// # use bevy_asset::Handle;
48/// # use bevy_color::Color;
49/// # use bevy_color::palettes::basic::BLUE;
50/// # use bevy_ecs::world::World;
51/// # use bevy_text::{Font, FontSize, Justify, TextLayout, TextFont, TextColor, TextSpan};
52/// # use bevy_sprite::Text2d;
53/// #
54/// # let font_handle: Handle<Font> = Default::default();
55/// # let mut world = World::default();
56/// #
57/// // Basic usage.
58/// world.spawn(Text2d::new("hello world!"));
59///
60/// // With non-default style.
61/// world.spawn((
62///     Text2d::new("hello world!"),
63///     TextFont {
64///         font: font_handle.clone().into(),
65///         font_size: FontSize::Px(60.0),
66///         ..Default::default()
67///     },
68///     TextColor(BLUE.into()),
69/// ));
70///
71/// // With text justification.
72/// world.spawn((
73///     Text2d::new("hello world\nand bevy!"),
74///     TextLayout::justify(Justify::Center)
75/// ));
76///
77/// // With spans
78/// world.spawn(Text2d::new("hello ")).with_children(|parent| {
79///     parent.spawn(TextSpan::new("world"));
80///     parent.spawn((TextSpan::new("!"), TextColor(BLUE.into())));
81/// });
82/// ```
83///
84/// Viewport-based `FontSize` variants for `Text2d` are resolved based on the primary window’s logical size.
85#[derive(Component, Clone, Debug, Default, Deref, DerefMut, Reflect)]
86#[reflect(Component, Default, Debug, Clone)]
87#[require(
88    TextLayout,
89    TextFont,
90    TextColor,
91    LineHeight,
92    LetterSpacing,
93    TextBounds,
94    Anchor,
95    Visibility,
96    VisibilityClass,
97    Transform,
98    // Disable hinting as `Text2d` text is not always pixel-aligned
99    FontHinting::Disabled
100)]
101#[component(on_add = visibility::add_visibility_class::<Sprite>)]
102pub struct Text2d(pub String);

160/// Updates the layout and size information whenever the text or style is changed.
161/// This information is computed by the [`TextPipeline`] on insertion, then stored.
162///
163/// ## World Resources
164///
165/// [`ResMut<Assets<Image>>`](Assets<Image>) -- This system only adds new [`Image`] assets.
166/// It does not modify or observe existing ones.
167pub fn update_text2d_layout(
168    mut last_logical_viewport_size: Local<Vec2>,
169    mut target_scale_factors: Local<Vec<(f32, RenderLayers)>>,
170    // Text2d entities from the previous frame which need to be reprocessed, usually because the font hadn't loaded yet.
171    mut reprocess_queue: Local<EntityHashSet>,
172    mut textures: ResMut<Assets<Image>>,
173    fonts: Res<Assets<Font>>,
174    camera_query: Query<(&Camera, &VisibleEntities, Option<&RenderLayers>)>,
175    mut font_atlas_set: ResMut<FontAtlasSet>,
176    mut text_pipeline: ResMut<TextPipeline>,
177    mut text_query: Query<(
178        Entity,
179        Ref<Text2d>,
180        Option<&RenderLayers>,
181        Ref<TextLayout>,
182        Ref<TextBounds>,
183        &mut TextLayoutInfo,
184        &mut ComputedTextBlock,
185        Ref<FontHinting>,
186    )>,
187    mut text_reader: Text2dReader,
188    mut font_system: ResMut<FontCx>,
189    mut layout_cx: ResMut<LayoutCx>,
190    mut scale_cx: ResMut<ScaleCx>,
191    rem_size: Res<RemSize>,
192    primary_window: Option<Single<&Window, With<PrimaryWindow>>>,
193) {
194    let logical_viewport_size = primary_window
195        .map(|window| window.resolution.size())
196        .unwrap_or(Vec2::splat(1000.));
197
198    let viewport_size_changed = *last_logical_viewport_size == logical_viewport_size;
199
200    *last_logical_viewport_size = logical_viewport_size;
201
202    target_scale_factors.clear();
203    target_scale_factors.extend(
204        camera_query
205            .iter()
206            .filter(|(_, visible_entities, _)| {
207                !visible_entities.get(TypeId::of::<Sprite>()).is_empty()
208            })
209            .filter_map(|(camera, _, maybe_camera_mask)| {
210                camera.target_scaling_factor().map(|scale_factor| {
211                    (scale_factor, maybe_camera_mask.cloned().unwrap_or_default())
212                })
213            }),
214    );
215
216    let mut previous_scale_factor = 0.;
217    let mut previous_mask = &RenderLayers::none();
218
219    for (
220        entity,
221        text2d,
222        maybe_entity_mask,
223        block,
224        bounds,
225        mut text_layout_info,
226        mut computed,
227        hinting,
228    ) in &mut text_query
229    {
230        let entity_mask = maybe_entity_mask.unwrap_or_default();
231
232        let scale_factor = if entity_mask == previous_mask && 0. < previous_scale_factor {
233            previous_scale_factor
234        } else {
235            // `Text2d` only supports generating a single text layout per Text2d entity. If a `Text2d` entity has multiple
236            // render targets with different scale factors, then we use the maximum of the scale factors.
237            let Some((scale_factor, mask)) = target_scale_factors
238                .iter()
239                .filter(|(_, camera_mask)| camera_mask.intersects(entity_mask))
240                .max_by_key(|(scale_factor, _)| FloatOrd(*scale_factor))
241            else {
242                continue;
243            };
244            previous_scale_factor = *scale_factor;
245            previous_mask = mask;
246            *scale_factor
247        };
248
249        let text_changed = scale_factor != text_layout_info.scale_factor
250            || text2d.is_changed()
251            || block.is_changed()
252            || computed.needs_rerender(viewport_size_changed, rem_size.is_changed())
253            || (!reprocess_queue.is_empty() && reprocess_queue.remove(&entity));
254
255        if !(text_changed || bounds.is_changed() || hinting.is_changed()) {
256            continue;
257        }
258
259        let text_bounds = TextBounds {
260            width: if block.linebreak == LineBreak::NoWrap {
261                None
262            } else {
263                bounds.width.map(|width| width * scale_factor)
264            },
265            height: bounds.height.map(|height| height * scale_factor),
266        };
267
268        if text_changed {
269            match text_pipeline.update_buffer(
270                &fonts,
271                text_reader.iter(entity),
272                block.linebreak,
273                block.justify,
274                text_bounds,
275                scale_factor,
276                &mut computed,
277                &mut font_system,
278                &mut layout_cx,
279                logical_viewport_size,
280                rem_size.0,
281            ) {
282                Err(
283                    TextError::NoSuchFont
284                    | TextError::NoSuchFontFamily(_)
285                    | TextError::DegenerateScaleFactor,
286                ) => {
287                    // There was an error processing the text layout.
288                    // Add this entity to the queue and reprocess it in the following frame
289                    reprocess_queue.insert(entity);
290                    continue;
291                }
292                Err(e @ TextError::FailedToGetGlyphImage(_)) => {
293                    bevy_log::warn_once!("{e}.");
294                    text_layout_info.clear();
295                }
296                Err(
297                    e @ (TextError::FailedToAddGlyph(_)
298                    | TextError::MissingAtlasLayout
299                    | TextError::MissingAtlasTexture
300                    | TextError::InconsistentAtlasState),
301                ) => {
302                    panic!("Fatal error when processing text: {e}.");
303                }
304                Ok(()) => {}
305            }
306        }
307
308        match text_pipeline.update_text_layout_info(
309            &mut text_layout_info,
310            &mut font_atlas_set,
311            &mut textures,
312            &mut computed,
313            &mut scale_cx,
314            text_bounds,
315            block.justify,
316            *hinting,
317        ) {
318            Err(TextError::NoSuchFont | TextError::NoSuchFontFamily(_)) => {
319                // There was an error processing the text layout.
320                // Add this entity to the queue and reprocess it in the following frame.
321                reprocess_queue.insert(entity);
322                continue;
323            }
324            Err(
325                e @ (TextError::FailedToAddGlyph(_)
326                | TextError::FailedToGetGlyphImage(_)
327                | TextError::MissingAtlasLayout
328                | TextError::MissingAtlasTexture
329                | TextError::InconsistentAtlasState
330                | TextError::DegenerateScaleFactor),
331            ) => {
332                panic!("Fatal error when processing text: {e}.");
333            }
334            Ok(()) => {
335                text_layout_info.scale_factor = scale_factor;
336                text_layout_info.size *= scale_factor.recip();
337            }
338        }
339    }
340}

342/// System calculating and inserting an [`Aabb`] component to entities with some
343/// [`TextLayoutInfo`] and [`Anchor`] components, and without a [`NoFrustumCulling`] component.
344///
345/// Used in system set [`VisibilitySystems::CalculateBounds`](bevy_camera::visibility::VisibilitySystems::CalculateBounds).
346pub fn calculate_bounds_text2d(
347    mut commands: Commands,
348    mut text_to_update_aabb: Query<
349        (
350            Entity,
351            &TextLayoutInfo,
352            &Anchor,
353            &TextBounds,
354            Option<&mut Aabb>,
355        ),
356        (Changed<TextLayoutInfo>, Without<NoFrustumCulling>),
357    >,
358) {
359    for (entity, layout_info, anchor, text_bounds, aabb) in &mut text_to_update_aabb {
360        let size = Vec2::new(
361            text_bounds.width.unwrap_or(layout_info.size.x),
362            text_bounds.height.unwrap_or(layout_info.size.y),
363        );
364
365        let x1 = (Anchor::TOP_LEFT.0.x - anchor.as_vec().x) * size.x;
366        let x2 = (Anchor::TOP_LEFT.0.x - anchor.as_vec().x + 1.) * size.x;
367        let y1 = (Anchor::TOP_LEFT.0.y - anchor.as_vec().y - 1.) * size.y;
368        let y2 = (Anchor::TOP_LEFT.0.y - anchor.as_vec().y) * size.y;
369        let new_aabb = Aabb::from_min_max(Vec3::new(x1, y1, 0.), Vec3::new(x2, y2, 0.));
370
371        if let Some(mut aabb) = aabb {
372            *aabb = new_aabb;
373        } else {
374            commands.entity(entity).try_insert(new_aabb);
375        }
376    }
377}

461        let (mut app, entity) = setup();
462
463        assert!(!app
464            .world()
465            .get_entity(entity)
466            .expect("Could not find entity")
