5/// Helper macro to define a dummy implementation of the sample trait if a
6/// feature is not supported.
7macro_rules! interpolator {
8    (
9    #[cfg($($cond:tt)*)]
10    mod $mod:ident;
11    trait $trait:ident;
12    ) => {
13        #[cfg($($cond)*)]
14        pub mod $mod;
15
16        #[cfg(not($($cond)*))]
17        pub mod $mod {
18            use crate::Sample;
19
20            /// Dummy trait when not supported.
21            pub trait $trait {
22            }
23
24            /// Dummy impl of trait when not supported.
25            impl<T> $trait for T where T: Sample {
26            }
27        }
28
29        pub use self::$mod::$trait;
30    }
31}

51/// Functions for making the scalar product with a sinc.
52pub trait SincInterpolator<T>: Send {
53    /// Make the scalar product between the waveform starting at `index` and the sinc of `subindex`.
54    fn get_sinc_interpolated(&self, wave: &[T], index: usize, subindex: usize) -> T;
55
56    /// Get sinc length.
57    fn len(&self) -> usize;
58
59    /// Check if sincs are empty.
60    fn is_empty(&self) -> bool {
61        self.len() == 0
62    }
63
64    /// Get number of sincs used for oversampling.
65    fn nbr_sincs(&self) -> usize;
66}
67
68/// A plain scalar interpolator.
69pub struct ScalarInterpolator<T> {
70    sincs: Vec<Vec<T>>,
71    length: usize,
72    nbr_sincs: usize,
73}
74
75impl<T> SincInterpolator<T> for ScalarInterpolator<T>
76where
77    T: Sample,
78{
79    /// Calculate the scalar produt of an input wave and the selected sinc filter.
80    fn get_sinc_interpolated(&self, wave: &[T], index: usize, subindex: usize) -> T {
81        assert!(
82            (index + self.length) < wave.len(),
83            "Tried to interpolate for index {}, max for the given input is {}",
84            index,
85            wave.len() - self.length - 1
86        );
87        assert!(
88            subindex < self.nbr_sincs,
89            "Tried to use sinc subindex {}, max is {}",
90            subindex,
91            self.nbr_sincs - 1
92        );
93        let wave_cut = &wave[index..(index + self.sincs[subindex].len())];
94        let sinc = &self.sincs[subindex];
95        unsafe {
96            let mut acc0 = T::zero();
97            let mut acc1 = T::zero();
98            let mut acc2 = T::zero();
99            let mut acc3 = T::zero();
100            let mut acc4 = T::zero();
101            let mut acc5 = T::zero();
102            let mut acc6 = T::zero();
103            let mut acc7 = T::zero();
104            let mut idx = 0;
105            for _ in 0..wave_cut.len() / 8 {
106                acc0 += *wave_cut.get_unchecked(idx) * *sinc.get_unchecked(idx);
107                acc1 += *wave_cut.get_unchecked(idx + 1) * *sinc.get_unchecked(idx + 1);
108                acc2 += *wave_cut.get_unchecked(idx + 2) * *sinc.get_unchecked(idx + 2);
109                acc3 += *wave_cut.get_unchecked(idx + 3) * *sinc.get_unchecked(idx + 3);
110                acc4 += *wave_cut.get_unchecked(idx + 4) * *sinc.get_unchecked(idx + 4);
111                acc5 += *wave_cut.get_unchecked(idx + 5) * *sinc.get_unchecked(idx + 5);
112                acc6 += *wave_cut.get_unchecked(idx + 6) * *sinc.get_unchecked(idx + 6);
113                acc7 += *wave_cut.get_unchecked(idx + 7) * *sinc.get_unchecked(idx + 7);
114                idx += 8;
115            }
116            acc0 + acc1 + acc2 + acc3 + acc4 + acc5 + acc6 + acc7
117        }
118    }
119
120    fn len(&self) -> usize {
121        self.length
122    }
123
124    fn nbr_sincs(&self) -> usize {
125        self.nbr_sincs
126    }
127}

129impl<T> ScalarInterpolator<T>
130where
131    T: Sample,
132{
133    /// Create a new ScalarInterpolator.
134    ///
135    /// Parameters are:
136    /// - `sinc_len`: Length of sinc functions.
137    /// - `oversampling_factor`: Number of intermediate sincs (oversampling factor).
138    /// - `f_cutoff`: Relative cutoff frequency.
139    /// - `window`: Window function to use.
140    pub fn new(
141        sinc_len: usize,
142        oversampling_factor: usize,
143        f_cutoff: f32,
144        window: WindowFunction,
145    ) -> Self {
146        assert!(sinc_len % 8 == 0, "Sinc length must be a multiple of 8");
147        let sincs = make_sincs(sinc_len, oversampling_factor, f_cutoff, window);
148        Self {
149            sincs,
150            length: sinc_len,
151            nbr_sincs: oversampling_factor,
152        }
153    }
154}

165    fn get_sinc_interpolated<T: Float>(wave: &[T], index: usize, sinc: &[T]) -> T {
166        let wave_cut = &wave[index..(index + sinc.len())];
167        wave_cut
168            .iter()
169            .zip(sinc.iter())
170            .fold(T::zero(), |acc, (x, y)| acc + *x * *y)
171    }

173    #[test]
174    fn test_scalar_interpolator_64() {
175        let mut rng = rand::thread_rng();
176        let mut wave = Vec::new();
177        for _ in 0..2048 {
178            wave.push(rng.gen::<f64>());
179        }
180        let sinc_len = 256;
181        let f_cutoff = 0.9473371669037001;
182        let oversampling_factor = 256;
183        let window = WindowFunction::BlackmanHarris2;
184
185        let interpolator =
186            ScalarInterpolator::<f64>::new(sinc_len, oversampling_factor, f_cutoff, window);
187        let value = interpolator.get_sinc_interpolated(&wave, 333, 123);
188        let check = get_sinc_interpolated(&wave, 333, &interpolator.sincs[123]);
189        assert!((value - check).abs() < 1.0e-9);
190    }
