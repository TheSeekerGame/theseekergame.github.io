9const MARGIN: i32 = 7;
10const ROW: i32 = 11;
11const KEYS_LEFT: i32 = 66;
12const ACCENT: Accent = Accent { bright: Color::srgb(0.80, 0.84, 0.88), dim: Color::srgb(0.24, 0.27, 0.32) };

14/// Each action, its keys, and its gamepad buttons.
15const CONTROLS: [(&str, &[&str], &str); 12] = [
16    ("MOVE", &["A", "D", "<", ">"], "L STICK"),
17    ("JUMP", &["W", "^", "SPACE"], "L2"),
18    ("FALL", &["S", "v"], "D-PAD DOWN"),
19    ("SKILLS", &["J", "K", "L", ";", "'"], "X R2 A B"),
20    ("", &["1", "2", "3", "4", "5"], ""),
21    ("UP VARIANT", &["I"], ""),
22    ("INTERACT", &["F"], "Y"),
23    ("INVENTORY", &["M"], ""),
24    ("CONTROLS", &["C"], "START"),
25    ("MENU", &["ESC"], ""),
26    ("PAUSE", &["P"], ""),
27    ("SCROLL UI", &["PGUP", "PGDN"], "SHIFT: SIDE"),
28];

30pub(super) fn plugin(app: &mut App) {
31    app.init_resource::<ControlsShown>()
32        .add_systems(OnEnter(AppState::InGame), |mut shown: ResMut<ControlsShown>| shown.0 = false)
33        .add_systems(Update, (toggle_controls.after(GameTickSet::Post), present_controls.after(super::sync_ui_scale)).chain().run_if(in_state(AppState::InGame)));
34}

36#[derive(Resource, Default)]
37struct ControlsShown(bool, PanelScroll);
38
39#[derive(Component)]
40struct ControlsPanel;

42fn toggle_controls(actions: Query<&ActionState<PlayerAction>>, mut shown: ResMut<ControlsShown>) {
43    if actions.single().is_ok_and(|actions| actions.just_pressed(&PlayerAction::ToggleControlOverlay)) {
44        shown.0 = !shown.0;
45    }
46}

48fn present_controls(
49    mut commands: Commands,
50    mut shown: ResMut<ControlsShown>,
51    keys: Res<ButtonInput<KeyCode>>,
52    library: Res<ArtLibrary>,
53    mut images: ResMut<Assets<Image>>,
54    mut panel: Query<(Entity, &ImageNode, &mut Node), With<ControlsPanel>>,
55    grid: UiPixelGrid,
56) {
57    let content = shown.0.then(|| draw_controls(&library.art(&images))).and_then(|raster| {
58        grid.scrollable(raster, &mut shown.1, &keys, None, Vec2::splat(0.5))
59    });
60    show_raster(&mut commands, &mut images, panel.single_mut().ok(), content, |image, node| {
61        (Name::new("ControlsPanel"), ControlsPanel, image, node, ZIndex(PANEL_Z_INDEX), StateDespawnMarker)
62    });
63}

65fn draw_controls(art: &Art) -> Raster {
66    let keys_width = |keys: &[&str]| keys.iter().map(|key| key_hint_width(key, "") - 1).sum::<i32>();
67    let pad_left = KEYS_LEFT + CONTROLS.iter().map(|(_, keys, _)| keys_width(keys)).max().unwrap_or(0) + 6;
68    let width = pad_left + CONTROLS.iter().map(|(.., pad)| text_width(pad)).max().unwrap_or(0) + MARGIN;
69    let height = 18 + CONTROLS.len() as i32 * ROW + 4;
70    let mut canvas = Canvas::new(UVec2::new(width as u32, height as u32), art);
71    draw_frame(&mut canvas, IVec2::new(width, height), ACCENT, &[14]);
72    canvas.text(IVec2::new(MARGIN, 4), "CONTROLS", ACCENT.bright);
73    canvas.text(IVec2::new(pad_left, 4), "GAMEPAD", MUTED);
74    for (index, (action, keys, pad)) in CONTROLS.into_iter().enumerate() {
75        let y = 18 + index as i32 * ROW;
76        canvas.text(IVec2::new(MARGIN, y + 1), action, TEXT);
77        let mut x = KEYS_LEFT;
78        for key in keys {
79            x += keycap(&mut canvas, IVec2::new(x, y), key) + 2;
80        }
81        canvas.text(IVec2::new(pad_left, y + 1), pad, MUTED);
82    }
83    canvas.fill(rect(pad_left - 4, 15, 1, height - 16), ACCENT.dim);
84    canvas.finish()
85}
