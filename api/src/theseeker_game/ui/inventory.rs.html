45#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
46enum InventoryTab {
47    #[default]
48    Skills,
49    Passives,
50    Weapons,
51    Items,
52}

54impl InventoryTab {
55    const ALL: [Self; 4] = [Self::Skills, Self::Passives, Self::Weapons, Self::Items];
56    const LABELS: [&'static str; 4] = ["SKILLS", "PASSIVES", "WEAPONS", "ITEMS"];
57}

59#[derive(Resource, Debug, Default)]
60pub(crate) struct InventoryState {
61    open: bool,
62    tab: InventoryTab,
63    tabs_focused: bool,
64    row: usize,
65    column: usize,
66    /// Why the last action failed, cleared when the selection moves.
67    failure: Option<&'static str>,
68    scroll: PanelScroll,
69}
70
71impl InventoryState {
72    pub(crate) fn is_open(&self) -> bool {
73        self.open
74    }
75
76    fn select(&mut self, row: usize, column: usize) {
77        if (row, column) != (self.row, self.column) {
78            (self.row, self.column, self.failure) = (row, column, None);
79        }
80    }
81}

99/// A labeled grid of wells; an equipped row shows exactly its capacity.
100struct Section {
101    label: &'static str,
102    visible_rows: usize,
103    equipped_row: bool,
104    cells: Vec<Option<Cell>>,
105}
106
107impl Section {
108    fn row_count(&self) -> usize {
109        if self.equipped_row { 1 } else { self.cells.len().div_ceil(COLUMNS * self.visible_rows).max(1) * self.visible_rows }
110    }
111
112    fn row_width(&self) -> usize {
113        if self.equipped_row { self.cells.len() } else { COLUMNS }
114    }
115}

120struct Loadout {
121    passives: &'static mut Passives,
122    skills: &'static mut SkillInventory,
123    weapons: &'static mut WeaponInventory,
124    materials: &'static MaterialInventory,

127struct View<'a> {
128    passives: &'a Passives,
129    skills: &'a SkillInventory,
130    weapons: &'a WeaponInventory,
131    materials: &'a MaterialInventory,
132    melee: PlayerMeleeWeapon,
133}
134
135impl View<'_> {
136    fn sections(&self, tab: InventoryTab) -> Vec<Section> {
137        let equipped = |cells: Vec<Option<Item>>| Section {
138            label: "EQUIPPED",
139            visible_rows: 1,
140            equipped_row: true,
141            cells: cells.into_iter().map(|item| item.map(|item| Cell { item, equipped: true })).collect(),
142        };
143        let stored = |label, visible_rows, mut items: Vec<Item>| {
144            items.sort_by_key(|item| item.order());
145            let cells = items.into_iter().map(|item| Some(Cell { item, equipped: false })).collect();
146            Section { label, visible_rows, equipped_row: false, cells }
147        };
148        match tab {
149            InventoryTab::Skills => {
150                let capacity = SkillInventory::max_slots_for(self.passives);
151                vec![
152                    equipped(padded(self.skills.equipped.iter().map(|skill| Item::Skill(*skill)), capacity)),
153                    stored("STORED", 3, self.skills.inventory.iter().map(|skill| Item::Skill(*skill)).collect()),
154                ]
155            },
156            InventoryTab::Passives => vec![
157                equipped(padded(self.passives.equipped.iter().map(|passive| Item::Passive(*passive)), Passives::MAX_EQUIPPED)),
158                stored("STORED", 3, self.passives.inventory.iter().map(|passive| Item::Passive(*passive)).collect()),
159            ],
160            InventoryTab::Weapons => {
161                let (ranged, ammo) = (self.weapons.equipped_ranged(), self.weapons.equipped_ammo());
162                let melee = PlayerMeleeWeapon::ALL.into_iter().filter(|weapon| *weapon != self.melee).map(Item::Melee);
163                let owned = self.weapons.owned_ranged().iter().copied().filter(|weapon| Some(*weapon) != ranged);
164                let stocked = PlayerAmmo::ALL.into_iter().filter(|kind| Some(*kind) != ammo && self.weapons.count_ammo(*kind) > 0);
165                vec![
166                    equipped(vec![Some(Item::Melee(self.melee)), ranged.map(Item::Ranged), ammo.map(Item::Ammo)]),
167                    stored("STORED", 3, melee.chain(owned.map(Item::Ranged)).chain(stocked.map(Item::Ammo)).collect()),
168                ]
169            },
170            InventoryTab::Items => {
171                let carried = CraftMaterial::STORAGE.into_iter().filter(|material| self.materials.backpack_count(*material) > 0);
172                vec![stored("BACKPACK", 4, carried.map(Item::Material).collect())]
173            },
174        }
175    }
176
177    fn icon_path(&self, item: Item) -> &'static str {
178        match item {
179            Item::Skill(skill) => skill.icon_path_with_melee_weapon(self.melee),
180            Item::Passive(passive) => passive.icon_path(),
181            Item::Melee(weapon) => weapon.icon_path(),
182            Item::Ranged(weapon) => weapon.icon_path(),
183            Item::Ammo(ammo) => ammo.icon_path(),
184            Item::Material(material) => material.icon_path(),
185        }
186    }
187
188    /// The stock a stacked item shows.
189    fn count(&self, item: Item) -> Option<u32> {
190        match item {
191            Item::Skill(SkillId::BowAttack) => Some(self.weapons.count_ammo(PlayerAmmo::Arrows)),
192            Item::Skill(skill) => skill.stock_material().map(|material| self.materials.backpack_count(material)),
193            Item::Ammo(ammo) => Some(self.weapons.count_ammo(ammo)),
194            Item::Material(material) => Some(self.materials.backpack_count(material)),
195            Item::Passive(_) | Item::Melee(_) | Item::Ranged(_) => None,
196        }
197    }
198
199    /// Name, kind, and description; passive descriptions show their current rolled value.
200    fn describe(&self, item: Item) -> (&'static str, &'static str, String) {
201        match item {
202            Item::Skill(skill) => (skill.name(), "SKILL", skill.description().to_owned()),
203            Item::Passive(passive) => {
204                (passive.name(), "PASSIVE", passive.describe(self.passives.passive_effective_value(passive)))
205            },
206            Item::Melee(weapon) => (weapon.name(), "MELEE", weapon.description().to_owned()),
207            Item::Ranged(weapon) => (weapon.name(), "RANGED", weapon.description().to_owned()),
208            Item::Ammo(ammo) => (ammo.name(), "AMMO", ammo.description().to_owned()),
209            Item::Material(material) => (material.name(), "MATERIAL", material.description().to_owned()),
210        }
211    }
212}

214impl Item {
215    /// Stored items keep a fixed order so equipping one never shuffles the rest.
216    fn order(self) -> (u8, u16) {
217        match self {
218            Self::Skill(skill) => (0, skill as u16),
219            Self::Passive(passive) => (1, passive as u16),
220            Self::Melee(weapon) => (2, weapon as u16),
221            Self::Ranged(weapon) => (3, weapon as u16),
222            Self::Ammo(ammo) => (4, ammo as u16),
223            Self::Material(material) => (5, material as u16),
224        }
225    }
226}

228fn padded(items: impl Iterator<Item = Item>, capacity: usize) -> Vec<Option<Item>> {
229    let mut cells: Vec<_> = items.map(Some).collect();
230    cells.resize(cells.len().max(capacity), None);
231    cells
232}

234/// The flat list of navigable rows: (section, row within it).
235fn rows(sections: &[Section]) -> Vec<(usize, usize)> {
236    sections.iter().enumerate().flat_map(|(index, section)| (0..section.row_count()).map(move |row| (index, row))).collect()
237}
238
239fn selected_cell(sections: &[Section], row: usize, column: usize) -> Option<Cell> {
240    let &(section, local) = rows(sections).get(row)?;
241    sections[section].cells.get(local * COLUMNS + column).copied().flatten()
242}

244fn handle_inventory_input(
245    keys: Res<ButtonInput<KeyCode>>,
246    mut state: ResMut<InventoryState>,
247    capsule: Res<CapsuleUiState>,
248    menu: Res<super::mainmenu::MenuState>,
249    mut player: Query<(&ActionState<PlayerAction>, Loadout), With<Player>>,
250    mut melee: ResMut<PlayerMeleeWeapon>,
251) {
252    if menu.blocks_panel_input() { return; }
253    let Ok((actions, mut loadout)) = player.single_mut() else { return };
254    let toggled = actions.just_pressed(&PlayerAction::TogglePassiveInventory);
255    if state.open && (toggled || close_pressed(&keys)) || capsule.is_open() {
256        state.open = false;
257        return;
258    }
259    if !state.open {
260        state.open = toggled;
261        return;
262    }
263    if !panel_key_pressed(&keys) {
264        return;
265    }
266    let view = View {
267        passives: &loadout.passives,
268        skills: &loadout.skills,
269        weapons: &loadout.weapons,
270        materials: loadout.materials,
271        melee: *melee,
272    };
273    let sections = view.sections(state.tab);
274    navigate(&mut state, &sections, horizontal_delta(&keys), vertical_delta(&keys));
275    let cell = selected_cell(&sections, state.row, state.column).filter(|_| !state.tabs_focused);
276    if let Some(cell) = cell.filter(|_| keys.just_pressed(KeyCode::KeyF)) {
277        state.failure = perform(cell, &mut loadout, &mut melee).err();
278    }
279}

281fn navigate(state: &mut InventoryState, sections: &[Section], horizontal: i32, vertical: i32) {
282    if state.tabs_focused {
283        let tab = (state.tab as usize).saturating_add_signed(horizontal as isize).min(InventoryTab::ALL.len() - 1);
284        if InventoryTab::ALL[tab] != state.tab {
285            (state.tab, state.row, state.column, state.failure) = (InventoryTab::ALL[tab], 0, 0, None);
286        }
287        state.tabs_focused = vertical <= 0;
288        return;
289    }
290    let rows = rows(sections);
291    if vertical < 0 && state.row == 0 {
292        state.tabs_focused = true;
293        return;
294    }
295    let row = state.row.saturating_add_signed(vertical as isize).min(rows.len().saturating_sub(1));
296    let width = sections[rows[row].0].row_width();
297    let column = state.column.saturating_add_signed(horizontal as isize).min(width.saturating_sub(1));
298    state.select(row, column);
299}

301/// Equips or removes the selected item; the error says why it could not.
302fn perform(cell: Cell, loadout: &mut LoadoutItem, melee: &mut ResMut<PlayerMeleeWeapon>) -> Result<(), &'static str> {
303    let capacity = SkillInventory::max_slots_for(&loadout.passives);
304    let done = match (cell.item, cell.equipped) {
305        (Item::Skill(skill), true) => loadout.skills.unequip_skill(skill),
306        (Item::Skill(skill), false) => loadout.skills.equip_skill_with_capacity(skill, capacity),
307        (Item::Passive(passive), true) => loadout.passives.unequip_passive(passive),
308        (Item::Passive(passive), false) => loadout.passives.equip_passive(passive),
309        (Item::Melee(weapon), false) => {
310            **melee = weapon;
311            true
312        },
313        (Item::Ranged(weapon), false) => loadout.weapons.equip_ranged(weapon),
314        (Item::Ammo(ammo), false) => {
315            loadout.weapons.equip_ammo(ammo);
316            true
317        },
318        _ => true,
319    };
320    if done { Ok(()) } else { Err("NO FREE SLOT") }
321}

323/// What F does to a cell, if anything.
324fn action_label(cell: Cell) -> Option<&'static str> {
325    match (cell.item, cell.equipped) {
326        (Item::Skill(_) | Item::Passive(_), true) => Some("REMOVE"),
327        (Item::Material(_), _) | (_, true) => None,
328        (_, false) => Some("EQUIP"),
329    }
330}

335#[expect(clippy::too_many_arguments, reason = "The panel draws player items with their art at the pixel grid.")]
336fn present_inventory(
337    mut commands: Commands,
338    mut state: ResMut<InventoryState>,
339    keys: Res<ButtonInput<KeyCode>>,
340    player: Query<LoadoutReadOnly, With<Player>>,
341    melee: Res<PlayerMeleeWeapon>,
342    unlocks: Res<UnlockedSkillPresentationState>,
343    mut library: ResMut<ArtLibrary>,
344    server: Res<AssetServer>,
345    mut images: ResMut<Assets<Image>>,
346    mut panel: Query<(Entity, &ImageNode, &mut Node), With<InventoryPanel>>,
347    grid: UiPixelGrid,
348) {
349    let content = player.single().ok().filter(|_| state.open).and_then(|loadout| {
350        let view = View {
351            passives: loadout.passives,
352            skills: loadout.skills,
353            weapons: loadout.weapons,
354            materials: loadout.materials,
355            melee: *melee,
356        };
357        let sections = view.sections(state.tab);
358        let navigable = rows(&sections);
359        state.row = state.row.min(navigable.len().saturating_sub(1));
360        state.column = state.column.min(sections[navigable[state.row].0].row_width().saturating_sub(1));
361        let paths = sections.iter().flat_map(|section| section.cells.iter().flatten()).map(|cell| view.icon_path(cell.item));
362        library.load(&server, std::iter::once(FONT_IMAGE_PATH).chain(paths));
363        let available = grid.available_art()?;
364        let compact = available.x < WIDTH as u32;
365        let raster = if compact {
366            draw_compact_inventory(&state, &view, &sections, &unlocks, &library.art(&images), available.y)
367        } else { draw_inventory(&state, &view, &sections, &unlocks, &library.art(&images)) };
368        let focus = inventory_focus(&state, &sections, compact.then_some(available.y));
369        grid.scrollable(raster, &mut state.scroll, &keys, focus, Vec2::splat(0.5))
370    });
371    show_raster(&mut commands, &mut images, panel.single_mut().ok(), content, |image, node| {
372        (Name::new("InventoryPanel"), InventoryPanel, image, node, ZIndex(PANEL_Z_INDEX), StateDespawnMarker)
373    });
374}

376fn draw_inventory(
377    state: &InventoryState,
378    view: &View,
379    sections: &[Section],
380    unlocks: &UnlockedSkillPresentationState,
381    art: &Art,
382) -> Raster {
383    let mut canvas = Canvas::new(UVec2::new(WIDTH as u32, HEIGHT as u32), art);
384    draw_frame(&mut canvas, IVec2::new(WIDTH, HEIGHT), ACCENT, &[14, FOOTER_RULE]);
385    canvas.fill(rect(DETAIL_LEFT - 6, 15, 1, FOOTER_RULE - 15), ACCENT.dim);
386    draw_tabs(&mut canvas, IVec2::new(MARGIN, 4), &InventoryTab::LABELS, state.tab as usize, state.tabs_focused, ACCENT);
387    let selection = rows(sections).get(state.row).copied().filter(|_| !state.tabs_focused);
388    let mut top = SECTIONS_TOP;
389    for (index, section) in sections.iter().enumerate() {
390        let selected_row = selection.filter(|(section, _)| *section == index).map(|(_, row)| row);
391        draw_section(&mut canvas, view, section, top, DETAIL_LEFT - 8, selected_row.map(|row| (row, state.column)), unlocks);
392        top += 9 + section.visible_rows as i32 * STRIDE + 2;
393    }
394    let cell = selected_cell(sections, state.row, state.column).filter(|_| !state.tabs_focused);
395    if let Some(cell) = cell {
396        draw_detail(&mut canvas, view, cell, state.failure);
397    }
398    let action = cell.and_then(action_label).map(|label| (label, true));
399    draw_key_hints(&mut canvas, FOOTER_RULE + 3, (MARGIN, WIDTH - MARGIN), action, "CLOSE", ACCENT);
400    canvas.finish()
401}

403/// Narrow windows show the selected section above its details; navigation keeps the
404/// same logical rows, so changing layout never changes which item is selected.
405fn draw_compact_inventory(
406    state: &InventoryState, view: &View, sections: &[Section], unlocks: &UnlockedSkillPresentationState, art: &Art, available_height: u32,
407) -> Raster {
408    const COMPACT_WIDTH: i32 = 106;
409    const BODY_WIDTH: i32 = COMPACT_WIDTH - 2 * MARGIN;
410    let visible_rows = ((available_height as i32 - 96) / STRIDE).clamp(1, 3) as usize;
411    let detail_top = 31 + visible_rows as i32 * STRIDE;
412    let cell = selected_cell(sections, state.row, state.column).filter(|_| !state.tabs_focused);
413    let detail = cell.map(|cell| {
414        let (name, kind, description) = view.describe(cell.item);
415        let kind = match (view.count(cell.item), cell.equipped) {
416            (Some(count), _) => format!("{kind} x{}", MaterialInventory::display_count(count)),
417            (None, true) => format!("{kind} EQUIPPED"),
418            (None, false) => kind.to_owned(),
419        };
420        (name.to_ascii_uppercase(), kind, state.failure.map_or(description, str::to_owned))
421    });
422    let detail_lines = detail.as_ref().map_or(2, |(name, kind, body)| {
423        [name, kind, body].into_iter().map(|text| super::pixel_font::wrap_text(text, BODY_WIDTH as u32).len()).sum::<usize>()
424    });
425    let footer = detail_top + detail_lines as i32 * 10 + 7;
426    let height = footer + 15;
427    let mut canvas = Canvas::new(UVec2::new(COMPACT_WIDTH as u32, height as u32), art);
428    draw_frame(&mut canvas, IVec2::new(COMPACT_WIDTH, height), ACCENT, &[14, detail_top - 3, footer]);
429    draw_tabs(&mut canvas, IVec2::new(MARGIN, 4), &["SKL", "PASS", "WPN", "ITEM"], state.tab as usize, state.tabs_focused, ACCENT);
430    let (index, local) = rows(sections).get(state.row).copied().unwrap_or((0, 0));
431    let section = &sections[index];
432    let selected = (!state.tabs_focused).then_some((local, state.column));
433    let compact_section = Section {
434        label: section.label, visible_rows: if section.equipped_row { 1 } else { visible_rows },
435        equipped_row: section.equipped_row, cells: section.cells.clone(),
436    };
437    draw_section(&mut canvas, view, &compact_section, SECTIONS_TOP, COMPACT_WIDTH - MARGIN, selected, unlocks);
438    if let Some((name, kind, body)) = detail {
439        let mut y = detail_top;
440        for (text, color) in [(&name, ACCENT.bright), (&kind, MUTED), (&body, if state.failure.is_some() { BAD } else { TEXT })] {
441            y += canvas.paragraph(IVec2::new(MARGIN, y), BODY_WIDTH, 10, detail_lines, text, color) as i32 * 10;
442        }
443    } else {
444        canvas.text(IVec2::new(MARGIN, detail_top), "EMPTY SLOT", MUTED);
445    }
446    draw_key_hints(&mut canvas, footer + 3, (MARGIN, COMPACT_WIDTH - MARGIN), cell.and_then(action_label).map(|label| (label, true)), "EXIT", ACCENT);
447    canvas.finish()
448}

450fn inventory_focus(state: &InventoryState, sections: &[Section], compact_height: Option<u32>) -> Option<IRect> {
451    if state.tabs_focused {
452        let labels = if compact_height.is_some() { ["SKL", "PASS", "WPN", "ITEM"] } else { InventoryTab::LABELS };
453        let selected = state.tab as usize;
454        let left = MARGIN + labels[..selected].iter().map(|label| text_width(label) + 7).sum::<i32>();
455        return Some(rect(left, 3, text_width(labels[selected]), 10));
456    }
457    let (index, local) = rows(sections).get(state.row).copied()?;
458    let top = if compact_height.is_some() { SECTIONS_TOP } else {
459        SECTIONS_TOP + sections[..index].iter().map(|section| 11 + section.visible_rows as i32 * STRIDE).sum::<i32>()
460    };
461    let visible_rows = compact_height.map_or(sections[index].visible_rows, |height| ((height as i32 - 96) / STRIDE).clamp(1, 3) as usize);
462    let row = if sections[index].equipped_row { 0 } else { local % visible_rows };
463    Some(well_at(IVec2::new(MARGIN, top + 9), row * COLUMNS + state.column, COLUMNS))
464}

466/// A section's label and its page of wells; `selected` is the selected (row, column) within it.
467fn draw_section(
468    canvas: &mut Canvas,
469    view: &View,
470    section: &Section,
471    top: i32,
472    right: i32,
473    selected: Option<(usize, usize)>,
474    unlocks: &UnlockedSkillPresentationState,
475) {
476    canvas.text(IVec2::new(MARGIN, top), section.label, MUTED);
477    let page = selected.map_or(0, |(row, _)| row / section.visible_rows);
478    let pages = section.row_count().div_ceil(section.visible_rows);
479    if pages > 1 {
480        let indicator = format!("{}/{pages}", page + 1);
481        canvas.text(IVec2::new(right - text_width(&indicator), top), &indicator, MUTED);
482    }
483    let visible = if section.equipped_row { section.cells.len() } else { section.visible_rows * COLUMNS };
484    let first = page * section.visible_rows * COLUMNS;
485    let chosen = selected.map(|(row, column)| row * COLUMNS + column);
486    for slot in 0..visible {
487        let well = well_at(IVec2::new(MARGIN, top + 9), slot, COLUMNS);
488        let outline = (chosen == Some(first + slot)).then_some(TEXT);
489        let Some(cell) = section.cells.get(first + slot).copied().flatten() else {
490            draw_empty_well(canvas, well, outline);
491            continue;
492        };
493        draw_well(canvas, well, outline);
494        let flash = unlock_gain(cell.item, unlocks);
495        canvas.shaded_image(view.icon_path(cell.item), well, |texel| gain(texel, flash));
496        if let Some(count) = view.count(cell.item) {
497            draw_count(canvas, IRect::from_corners(well.min + 1, well.max - 1), count);
498        }
499    }
500}
501
502fn draw_empty_well(canvas: &mut Canvas, well: IRect, outline: Option<Color>) {
503    draw_well(canvas, well, outline);
504    canvas.fill(well, EMPTY_WELL);
505}

507/// Brightens a newly unlocked skill's icon.
508fn unlock_gain(item: Item, unlocks: &UnlockedSkillPresentationState) -> f32 {
509    let Item::Skill(skill) = item else { return 1.0 };
510    unlocks.entry(skill).map_or(1.0, |entry| super::skill_toolbar::unlock_gain(entry.remaining_ticks))
511}

513fn draw_detail(canvas: &mut Canvas, view: &View, cell: Cell, failure: Option<&str>) {
514    let (name, kind, description) = view.describe(cell.item);
515    let at = IVec2::new(DETAIL_LEFT, SECTIONS_TOP);
516    let lines = canvas.paragraph(at, DETAIL_WIDTH, 10, 2, &name.to_ascii_uppercase(), ACCENT.bright);
517    let tag = at + IVec2::Y * 10 * lines as i32;
518    let kind = match (view.count(cell.item), cell.equipped) {
519        (Some(count), _) => format!("{kind}  x{}", MaterialInventory::display_count(count)),
520        (None, true) => format!("{kind}  EQUIPPED"),
521        (None, false) => kind.to_owned(),
522    };
523    canvas.text(tag, &kind, MUTED);
524    let (body, color) = failure.map_or((description.as_str(), TEXT), |failure| (failure, BAD));
525    canvas.paragraph(tag + IVec2::Y * 12, DETAIL_WIDTH, 10, DETAIL_LINES - lines, body, color);
526}
