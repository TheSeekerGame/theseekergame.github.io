170/// How an equipped skill's icon reacts to use, cooldowns, and recharge.
171#[derive(Debug, Default, Clone, Copy, PartialEq)]
172struct IconState {
173    press_ticks: u32,
174    channeling: bool,
175    unavailable: bool,
176    flash_ticks: u32,
177    energy_full: Option<bool>,
178}
179
180impl IconState {
181    fn pressed(self) -> bool {
182        self.channeling || self.press_ticks > 0
183    }
184
185    /// Advances one tick. `ready_flash` is whether becoming usable again flashes the icon, and
186    /// `energy_full` whether the skill's energy is full, for skills that recharge through energy.
187    fn advance(&mut self, used: bool, channeling: bool, unavailable: bool, ready_flash: bool, energy_full: Option<bool>) {
188        if !self.channeling {
189            self.press_ticks = self.press_ticks.saturating_sub(1);
190        }
191        self.flash_ticks = self.flash_ticks.saturating_sub(1);
192        if used {
193            self.press_ticks = PRESS_TICKS;
194        }
195        if self.channeling && !channeling {
196            self.press_ticks = 0;
197        }
198        self.channeling = channeling;
199        if unavailable {
200            self.flash_ticks = 0;
201        } else if self.unavailable && ready_flash {
202            self.flash_ticks = READY_FLASH_TICKS;
203        }
204        self.unavailable = unavailable;
205        if self.energy_full == Some(false) && energy_full == Some(true) {
206            self.flash_ticks = READY_FLASH_TICKS;
207        }
208        self.energy_full = energy_full;
209    }
210
211    /// The icon's brightness: dimmed while unusable, flashing when it becomes usable.
212    fn gain(self) -> f32 {
213        if self.unavailable {
214            return UNAVAILABLE_GAIN;
215        }
216        1.0 + (READY_FLASH_GAIN - 1.0) * self.flash_ticks as f32 / READY_FLASH_TICKS as f32
217    }
218}

220/// Brightness of a newly unlocked skill's icon while it flashes.
221pub(crate) fn unlock_gain(remaining_ticks: u32) -> f32 {
222    let remaining = remaining_ticks as f32 / UnlockedSkillPresentationState::FLASH_TICKS as f32;
223    1.0 + (UNLOCK_FLASH_GAIN - 1.0) * remaining.sqrt()
224}

226/// Which slots' keys the player has used. A filled slot shows its key until then; the hints are
227/// never reset, so they stay learned across deaths and new expeditions.
228#[derive(Resource, Default, Debug, Clone, PartialEq)]
229struct KeyHints {
230    learned: [bool; SLOT_INPUTS.len()],
231}
232
233impl KeyHints {
234    /// Learns the key of every filled slot whose key was just pressed.
235    fn advance(&mut self, filled: usize, pressed: impl Fn(PlayerAction) -> bool) {
236        for input in SLOT_INPUTS.iter().take(filled) {
237            self.learned[input.index] |= pressed(input.action);
238        }
239    }
240
241    fn shown(&self, slot: usize) -> bool {
242        self.learned.get(slot).is_some_and(|learned| !learned)
243    }
244}

268impl ToolbarPlayerItem<'_, '_> {
269    /// Missing ammunition, stock, or a tether connection, which no cooldown restores.
270    fn lacks_means(&self, skill: SkillId, tether_connected: bool) -> bool {
271        match skill {
272            SkillId::BowAttack => !self.weapons.can_fire_bow(),
273            SkillId::TetherNode if !tether_connected => true,
274            _ => skill.stock_material().is_some_and(|material| self.materials.backpack_count(material) == 0),
275        }
276    }
277
278    fn energy_full(&self, skill: SkillId) -> bool {
279        match skill {
280            SkillId::Whirl => self.whirl.energy >= whirl_metadata().max_energy,
281            SkillId::FlickerStrike => self.flicker.energy >= flicker_strike_metadata().max_energy,
282            _ => false,
283        }
284    }
285
286    /// How much of the icon is still shaded: the remaining cooldown, or missing energy.
287    fn recharge(&self, skill: SkillId, cooldowns: &Cooldowns) -> f32 {
288        match skill {
289            SkillId::Whirl => 1.0 - self.whirl.energy / whirl_metadata().max_energy,
290            SkillId::FlickerStrike => 1.0 - self.flicker.energy / flicker_strike_metadata().max_energy,
291            _ => cooldowns.get(self.entity, skill).map_or(0.0, |entry| entry.remaining / entry.initial.max(1.0)),
292        }
293        .clamp(0.0, 1.0)
294    }
295
296    /// The stock a skill spends, if it counts one.
297    fn stock(&self, skill: SkillId) -> Option<u32> {
298        match skill {
299            SkillId::BowAttack => {
300                let arrows = self.weapons.equipped_ammo().filter(|ammo| *ammo == PlayerAmmo::Arrows);
301                Some(arrows.map_or(0, |ammo| self.weapons.count_ammo(ammo)))
302            },
303            _ => skill.stock_material().map(|material| self.materials.backpack_count(material)),
304        }
305    }
306}

308fn advance_toolbar_icons(
309    player: Query<ToolbarPlayer, With<Player>>,
310    channeled: Query<&ChanneledSkill, With<Player>>,
311    mut events: MessageReader<PassiveEvent>,
312    cooldowns: Res<Cooldowns>,
313    tether: Res<TetherConnectionStatus>,
314    time: Res<GameTime>,
315    mut icons: ResMut<ToolbarIcons>,
316) {
317    let Ok(player) = player.single() else { return };
318    let used: HashSet<SkillId> = events
319        .read()
320        .filter_map(|event| match *event {
321            PassiveEvent::SkillUsed { owner, skill } if owner == player.entity => Some(skill),
322            _ => None,
323        })
324        .collect();
325    let channeling: HashSet<SkillId> = channeled.iter().map(|channeled| channeled.skill).collect();
326    let ToolbarIcons { states, channeled } = &mut *icons;
327    channeled.extend(CHANNELED_SKILLS.into_iter().chain(channeling.iter().copied()));
328    states.retain(|skill, _| player.skills.is_equipped(skill));
329    for &skill in &player.skills.equipped {
330        let recharges = channeled.contains(&skill);
331        let lacks_means = player.lacks_means(skill, tether.available);
332        let cooling = !cooldowns.is_ready(player.entity, skill, time.tick()) || skill == SkillId::Stealth && player.stealthed;
333        let unavailable = lacks_means || !recharges && cooling;
334        let energy_full = recharges.then(|| player.energy_full(skill));
335        let state = states.entry(skill).or_default();
336        state.advance(used.contains(&skill), channeling.contains(&skill), unavailable, !recharges, energy_full);
337    }
338}

340fn advance_key_hints(
341    player: Query<(&SkillInventory, &ActionState<PlayerAction>), With<Player>>,
342    mut hints: ResMut<KeyHints>,

385fn present_toolbar(
386    player: Query<ToolbarPlayer, With<Player>>,
387    sources: ToolbarSources,
388    library: Res<ArtLibrary>,
389    mut images: ResMut<Assets<Image>>,
390    mut toolbar: Query<(&ImageNode, &mut Node), With<ToolbarFrame>>,
391    mut labels: Query<(&StockLabel, &mut Text, &mut Node), Without<ToolbarFrame>>,
392    grid: UiPixelGrid,
393) {
394    let (Ok(player), Ok((image, mut node))) = (player.single(), toolbar.single_mut()) else { return };
395    let skills = player.skills.equipped.iter().enumerate().map(|(index, &skill)| {
396        let state = sources.icons.states.get(&skill).copied().unwrap_or_default();
397        let flash = sources.unlocks.entry(skill).map(|entry| unlock_gain(entry.remaining_ticks));
398        Slot {
399            icon: skill.icon_path_with_melee_weapon(*sources.melee),
400            pressed: state.pressed(),
401            gain: flash.unwrap_or(state.gain()),
402            shade: (player.recharge(skill, &sources.cooldowns) * ICON as f32).ceil() as i32,
403            energy: matches!(skill, SkillId::Whirl | SkillId::FlickerStrike),
404            stock: player.stock(skill),
405            key_hint: sources.hints.shown(index),
406        }
407    });
408    let toolbar = Toolbar {
409        slots: if player.passives.contains(&Passive::BatteryPack) { 5 } else { 4 },
410        skills: skills.collect(),
411        health: player.health.current as f32 / player.health.max.max(1) as f32,
412        alignment: player.alignment.factor(),
413        elapsed_secs: sources.time.elapsed_secs(),
414        frame_pulse: sources.pulse.intensity,
415    };
416    let raster = draw_toolbar(&toolbar, &library.art(&images));
417    let Some((layout, unit)) = grid.anchored(raster.size, Vec2::new(0.5, 1.0)).zip(grid.unit()) else { return };
418    node.set_if_neq(layout);
419    for (label, mut text, mut node) in &mut labels {
420        let stock = toolbar.skills.get(label.0).filter(|_| label.0 < toolbar.slots).and_then(|slot| slot.stock);
421        let hidden = Node { display: Display::None, position_type: PositionType::Absolute, ..default() };
422        let (count, placement) = stock.map_or((String::new(), hidden), |stock| {
423            (MaterialInventory::display_count(stock), stock_label_node(label.0, raster.size, toolbar.rise(), unit))
424        });
425        if text.0 != count {
426            text.0 = count;
427        }
428        node.set_if_neq(placement);
429    }
430    raster.publish(&mut images, &image.image);
431}

433/// Pins a slot's count just inside its icon's bottom-right corner; `rise` is the space above the frame.
434fn stock_label_node(slot: usize, toolbar: UVec2, rise: i32, unit: f32) -> Node {
435    let icon_end = FIRST_SLOT + IVec2::new(SLOT_STRIDE * slot as i32 + ICON, ICON + rise);
436    let inset = (toolbar.as_ivec2() - icon_end).as_vec2() * unit;
437    Node {
438        position_type: PositionType::Absolute,
439        right: px(inset.x + 0.8 * unit),
440        bottom: px(inset.y + 0.4 * unit),
441        ..default()
442    }
443}

456fn draw_toolbar(toolbar: &Toolbar, art: &Art) -> Raster {
457    let slots = toolbar.slots as i32;
458    let bar_width = slots * SLOT_STRIDE - 2;
459    let top = IVec2::Y * toolbar.rise();
460    let mut canvas = Canvas::new(UVec2::new((slots * SLOT_STRIDE + 16) as u32, (FRAME_HEIGHT + top.y) as u32), art);
461    draw_bar(&mut canvas, HEALTH_BAR + top, IVec2::new(bar_width, 2), toolbar.health, (HEALTH_TRACK, HEALTH_FILL));
462    let alignment = alignment_colors(toolbar.alignment, toolbar.elapsed_secs);
463    draw_bar(&mut canvas, ALIGNMENT_BAR + top, IVec2::new(bar_width, 1), toolbar.alignment, alignment);
464    let frame = frame_path(toolbar.slots);
465    let pulse = ALIGNMENT_PULSE_LINEAR_GRAY * toolbar.frame_pulse;
466    let size = UVec2::new((slots * SLOT_STRIDE + 16) as u32, 30);
467    canvas.sprite(frame, URect::from_corners(UVec2::Y * FRAME_TOP as u32, size), top, |texel| lift(texel, pulse));
468    for (index, slot) in toolbar.shown_slots().enumerate() {
469        let at = FIRST_SLOT + top + IVec2::X * SLOT_STRIDE * index as i32;
470        draw_slot(&mut canvas, at, slot);
471        if slot.key_hint {
472            draw_key_hint(&mut canvas, at, index);
473        }
474    }
475    canvas.finish()
476}

478/// The slot's keycap, centered on the frame just above its icon at `icon`.
479fn draw_key_hint(canvas: &mut Canvas, icon: IVec2, slot: usize) {

484fn draw_bar(canvas: &mut Canvas, at: IVec2, size: IVec2, fraction: f32, (track, fill): (Color, Color)) {
485    let filled = (fraction.clamp(0.0, 1.0) * size.x as f32).ceil() as i32;
486    canvas.fill(rect(at.x, at.y, size.x, size.y), track);
487    canvas.fill(rect(at.x, at.y, filled, size.y), fill);
488}

490/// A pressed icon sinks into its slot: its outer ring gives way to the frame's.
491fn draw_slot(canvas: &mut Canvas, at: IVec2, slot: &Slot) {
492    let inset = u32::from(slot.pressed);
493    let source = URect::new(inset, inset, ICON as u32 - inset, ICON as u32 - inset);
494    canvas.sprite(slot.icon, source, at + inset as i32, |texel| gain(texel, slot.gain));
495    let shade = slot.shade.clamp(0, ICON);
496    let top = if slot.energy { at.y } else { at.y + ICON - shade };
497    canvas.fill(rect(at.x, top, ICON, shade), COOLDOWN_SHADE);
498}

500/// Track and fill colors; low Alignment pulses toward a warning color, and depletion pulses the track.
501fn alignment_colors(fraction: f32, elapsed_secs: f32) -> (Color, Color) {
502    let pulse = |hz: f32| 0.5 + 0.5 * (elapsed_secs * hz * TAU).sin();
503    let fraction = fraction.clamp(0.0, 1.0);
504    if fraction > ALIGNMENT_LOW_THRESHOLD {
505        return (ALIGNMENT_TRACK, ALIGNMENT_FILL);
506    }
507    if fraction <= 0.0 {
508        let strength = 0.18 + pulse(ALIGNMENT_DEPLETED_PULSE_HZ) * 0.22;
509        return (mix(ALIGNMENT_TRACK, ALIGNMENT_DEPLETED, strength), ALIGNMENT_FILL);
510    }
511    let urgency = 0.2 + 0.8 * (1.0 - fraction / ALIGNMENT_LOW_THRESHOLD);
512    let strength = urgency * pulse(ALIGNMENT_LOW_PULSE_HZ);
513    (mix(ALIGNMENT_TRACK, ALIGNMENT_WARNING, strength * 0.12), mix(ALIGNMENT_FILL, ALIGNMENT_WARNING, strength * 0.32))
514}
515
516fn mix(from: Color, to: Color, amount: f32) -> Color {
517    from.to_linear().mix(&to.to_linear(), amount).into()
518}

573/// An announcement across the top of the screen; `age` counts from when it was queued.
574struct Caption {
575    lines: Vec<String>,
576    /// How long it stays fully visible before fading.
577    hold_secs: f32,
578    age: f32,
579}
580
581impl Caption {
582    /// Fully visible after a short delay, then fading out; `None` before it appears.
583    fn opacity(&self) -> Option<f32> {
584        let shown = self.age - CAPTION_DELAY_SECS;
585        (shown >= 0.0).then(|| (1.0 - (shown - self.hold_secs).max(0.0) / CAPTION_FADE_SECS).max(0.0))
586    }
587
588    fn expired(&self) -> bool {
589        self.age >= CAPTION_DELAY_SECS + self.hold_secs + CAPTION_FADE_SECS
590    }
591}
592
593#[derive(Resource, Default)]
594struct Captions(Vec<Caption>);
595
596impl Captions {
597    fn push(&mut self, lines: impl IntoIterator<Item = impl Into<String>>, hold_secs: f32) {
598        if self.0.len() < CAPTION_LIMIT {
599            self.0.push(Caption { lines: lines.into_iter().map(Into::into).collect(), hold_secs, age: 0.0 });
600        }
601    }
602}
603
604fn age_captions(mut captions: ResMut<Captions>, time: Res<Time<Real>>) {
605    for caption in &mut captions.0 {
606        caption.age += time.delta_secs();
607    }
608    captions.0.retain(|caption| !caption.expired());
609}

623#[derive(Debug, Clone, Copy, PartialEq, Eq)]
624enum AlignmentWarning {
625    Low,
626    Critical,
627}
628
629impl AlignmentWarning {
630    fn lines(self) -> &'static [&'static str] {
631        match self {
632            Self::Low => &["ALIGNMENT LOW", "ESCAPE THE FOG TO RECHARGE"],
633            Self::Critical => &["ALIGNMENT CRITICAL", "THE FOG IS HURTING YOU", "REACH A CAPSULE OR A TETHER NODE"],
634        }
635    }
636}

638/// The Alignment warnings shown since Alignment last recovered above
639/// [`ALIGNMENT_WARNING_REARM`]. Each shows once as Alignment falls past its level, so lingering
640/// near a level never repeats it.
641#[derive(Resource, Default, Debug, Clone, Copy, PartialEq)]
642struct AlignmentWarnings {
643    low: bool,
644    critical: bool,
645}
646
647impl AlignmentWarnings {
648    /// The warning that Alignment at `fraction` raises, if it has not shown yet.
649    fn advance(&mut self, fraction: f32) -> Option<AlignmentWarning> {
650        if fraction > ALIGNMENT_WARNING_REARM {
651            *self = Self::default();
652        }
653        if fraction <= 0.0 && !self.critical {
654            // Running out supersedes the low warning, which would only repeat it.
655            *self = Self { low: true, critical: true };
656            return Some(AlignmentWarning::Critical);
657        }
658        if fraction <= ALIGNMENT_LOW_THRESHOLD && !self.low {
659            self.low = true;
660            return Some(AlignmentWarning::Low);
661        }
662        None
663    }
664}

666/// Warns out in the fog only: a sheltered player is recharging.
667fn queue_alignment_warnings(
668    player: Query<(&Alignment, Has<CapsuleOccupant>), With<Player>>,
669    toolbar: Query<&Visibility, With<ToolbarFrame>>,
670    mut warnings: ResMut<AlignmentWarnings>,
671    mut captions: ResMut<Captions>,
672) {
673    let Ok((alignment, sheltered)) = player.single() else { return };
674    if sheltered || !hud_shown(&toolbar) {
675        return;
676    }
677    if let Some(warning) = warnings.advance(alignment.factor()) {
678        captions.push(warning.lines().iter().copied(), NOTICE_HOLD_SECS);
679    }
680}

682/// Whether the player knows the inventory key: told at their first pickup, or learned by opening
683/// the inventory first. Never reset, so it is told once across deaths and new expeditions.
684#[derive(Resource, Default, Debug, Clone, Copy, PartialEq)]
685struct InventoryHint {
686    learned: bool,
687}
688
689impl InventoryHint {
690    /// Whether to tell the player the key now.
691    fn advance(&mut self, picked_up: bool, inventory_open: bool) -> bool {
692        let tell = picked_up && !inventory_open && !self.learned;
693        self.learned |= picked_up || inventory_open;
694        tell
695    }
696}

698fn queue_inventory_hint(
699    mut pickups: MessageReader<ItemPickedUp>,
700    inventory: Res<InventoryState>,
701    toolbar: Query<&Visibility, With<ToolbarFrame>>,
702    mut hint: ResMut<InventoryHint>,
703    mut captions: ResMut<Captions>,
704) {
705    let picked_up = pickups.read().count() > 0;
706    if hud_shown(&toolbar) && hint.advance(picked_up, inventory.is_open()) {
707        captions.push(INVENTORY_HINT, NOTICE_HOLD_SECS);
708    }
709}

733/// Stacked plates, each fading with its caption's `opacity`.
734fn draw_captions(captions: &[(&[String], f32)], art: &Art) -> Raster {
735    let widest = captions.iter().flat_map(|(lines, _)| lines.iter()).map(|line| text_width(line)).max().unwrap_or(0);
736    let width = widest + 12;
737    let heights: Vec<i32> = captions.iter().map(|(lines, _)| lines.len() as i32 * 10 + 5).collect();
738    let height = heights.iter().map(|height| height + 1).sum::<i32>() - 1;
739    let mut canvas = Canvas::new(UVec2::new(width as u32, height.max(1) as u32), art);
740    let mut top = 0;
741    for ((lines, opacity), plate_height) in captions.iter().zip(heights) {
742        let fill = Color::srgba(0.02, 0.03, 0.05, 0.72 * opacity);
743        canvas.plate(rect(0, top, width, plate_height), 1, fill, Color::WHITE.with_alpha(0.28 * opacity));
744        for (index, line) in lines.iter().enumerate() {
745            let at = IVec2::new((width - text_width(line)) / 2, top + 4 + 10 * index as i32);
746            canvas.text(at, line, CAPTION_COLOR.with_alpha(*opacity));
747        }
748        top += plate_height + 1;
749    }
750    canvas.finish()
751}
