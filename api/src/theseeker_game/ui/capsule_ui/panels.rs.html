9// Layout in art pixels. Text positions name the top of the capital letters.
10const WIDTH: i32 = 106;
11pub(super) const MARGIN: i32 = 6;
12pub(super) const RIGHT: i32 = WIDTH - MARGIN;
13pub(super) const INGREDIENT_ICON: i32 = 13;
14pub(super) const INGREDIENT_GAP: i32 = 3;
15const TERMINAL_GRID_TOP: i32 = 18;
16/// The rule between the terminal's entry grid and the selected entry's detail.
17const TERMINAL_DETAIL: i32 = TERMINAL_GRID_TOP + TERMINAL_ROWS as i32 * STRIDE + 1;
18const TERMINAL_HEIGHT: i32 = TERMINAL_DETAIL + 65;
19const STORAGE_HEIGHT: i32 = 108;

21const AMBER: Accent = Accent { bright: Color::srgb(0.96, 0.61, 0.12), dim: Color::srgb(0.42, 0.25, 0.07) };
22const CYAN: Accent = Accent { bright: Color::srgb(0.31, 0.91, 0.91), dim: Color::srgb(0.10, 0.36, 0.39) };

24const ESSENCE_MASK: &[&str] = &[".###.", "#####", "#####", "#####", ".###."];
25const RESEARCH_MASK: &[&str] = &[
26    "..#####..", "....#....", "....#....", "...###...", "...#.#...", "..#...#..", ".#.....#.", ".#######.",
27];
28const OWNED_MASK: &[&str] = &["....#", "...#.", "#.#..", ".#..."];

39pub(super) fn draw_panel(station: Station, state: &CapsuleUiState, ledger: &LedgerItem, art: &Art) -> Raster {
40    let height = match station {
41        Station::Terminal => TERMINAL_HEIGHT,
42        Station::Storage => STORAGE_HEIGHT,
43    };
44    let mut canvas = Canvas::new(UVec2::new(WIDTH as u32, height as u32), art);
45    match station {
46        Station::Terminal => draw_terminal(&mut canvas, state, ledger),
47        Station::Storage => draw_storage(&mut canvas, state, ledger.materials),
48    }
49    canvas.finish()
50}

52/// Tabs and essence, the entry grid, the selected entry's detail, and key hints.
53fn draw_terminal(canvas: &mut Canvas, state: &CapsuleUiState, ledger: &LedgerItem) {
54    let detail = TERMINAL_DETAIL;
55    draw_frame(canvas, IVec2::new(WIDTH, TERMINAL_HEIGHT), AMBER, &[14, detail, detail + 50]);
56    let tabs_focused = state.terminal_focus == TerminalFocus::Tabs;
57    draw_tabs(canvas, IVec2::new(MARGIN, 4), &["SHOP", "CRAFT"], state.terminal_tab as usize, tabs_focused, AMBER);
58    let balance = ledger.essence.get().to_string();
59    let balance_x = RIGHT - text_width(&balance);
60    canvas.text(IVec2::new(balance_x, 4), &balance, TEXT);
61    draw_essence_orb(canvas, IVec2::new(balance_x - 7, 4));
62
63    let entries = terminal_entries(state.terminal_tab, ledger.knowledge);
64    let page = state.terminal_selection / TERMINAL_PAGE * TERMINAL_PAGE;
65    let focused = state.terminal_focus == TerminalFocus::Items;
66    for slot in 0..TERMINAL_PAGE {
67        let index = page + slot;
68        let well = well_rect(slot, TERMINAL_GRID_TOP);
69        let selected = (index == state.terminal_selection).then_some(if focused { TEXT } else { AMBER.dim });
70        draw_well(canvas, well, selected);
71        let Some(entry) = entries.get(index).copied() else {
72            canvas.fill(well, EMPTY_WELL);
73            continue;
74        };
75        let availability = entry.availability(ledger);
76        draw_entry_icon(canvas, entry, well, availability);
77        if availability == Availability::Owned {
78            canvas.mask(IVec2::new(well.max.x - 6, well.max.y - 5), OWNED_MASK, GOOD);
79        }
80    }
81
82    let Some(entry) = entries.get(state.terminal_selection).copied() else {
83        draw_key_hints(canvas, detail + 54, (MARGIN, RIGHT), None, "EXIT", AMBER);
84        return;
85    };
86    canvas.text(IVec2::new(MARGIN, detail + 4), &entry.title().to_ascii_uppercase(), AMBER.bright);
87    let (message, color) = match state.feedback {
88        Some(Ok(message)) => (message, GOOD),
89        Some(Err(message)) => (message, BAD),
90        None => (entry.description(), MUTED),
91    };
92    canvas.paragraph(IVec2::new(MARGIN, detail + 14), RIGHT - MARGIN, 10, 2, message, color);
93    match entry {
94        TerminalEntry::Recipe(id) => draw_ingredients(canvas, id, ledger.materials, detail + 35),
95        TerminalEntry::ShopTether => draw_essence_cost(canvas, ledger.essence, detail + 35, None),
96        TerminalEntry::Research => {
97            let discoveries = discoverable_recipes(ledger.knowledge, ledger.materials).len();
98            draw_essence_cost(canvas, ledger.essence, detail + 35, Some(discoveries));
99        },
100    }
101    let ready = entry.availability(ledger) == Availability::Ready;
102    draw_key_hints(canvas, detail + 54, (MARGIN, RIGHT), Some((entry.action().0, ready)), "EXIT", AMBER);
103}

105/// Ingredient icons with owned/required counts; owned counts show at most 99 so three fit one row.
106fn draw_ingredients(canvas: &mut Canvas, id: CapsuleRecipeId, materials: &MaterialInventory, y: i32) {
107    let mut x = MARGIN;
108    for cost in recipe(id).ingredients {
109        let owned = materials.total_count(cost.material);
110        let color = if owned >= cost.amount { GOOD } else { BAD };
111        canvas.image(cost.material.icon_path(), rect(x, y, 12, 12), Color::WHITE);
112        x += INGREDIENT_ICON + canvas.text(IVec2::new(x + INGREDIENT_ICON, y + 3), &ingredient_label(owned, cost.amount), color);
113        x += INGREDIENT_GAP;
114    }
115}

117pub(super) fn ingredient_label(owned: u32, required: u32) -> String {
118    format!("{}/{required}", owned.min(99))
119}

121fn draw_essence_orb(canvas: &mut Canvas, at: IVec2) {
122    canvas.mask(at, ESSENCE_MASK, AMBER.bright);
123    canvas.fill(rect(at.x + 1, at.y + 1, 1, 1), TEXT);
124}

126fn draw_essence_cost(canvas: &mut Canvas, essence: &Essence, y: i32, discoveries: Option<usize>) {
127    let color = if essence.can_spend(CAPSULE_ACTION_ESSENCE_COST) { GOOD } else { BAD };
128    draw_essence_orb(canvas, IVec2::new(MARGIN, y + 3));
129    let cost = canvas.text(IVec2::new(MARGIN + 8, y + 3), &CAPSULE_ACTION_ESSENCE_COST.to_string(), color);
130    if let Some(discoveries) = discoveries {
131        canvas.text(IVec2::new(MARGIN + 8 + cost + 8, y + 3), &format!("{discoveries} TO DISCOVER"), MUTED);
132    }
133}

135/// Backpack and storage rows with stack counts, the selected stack's name, and key hints.
136fn draw_storage(canvas: &mut Canvas, state: &CapsuleUiState, materials: &MaterialInventory) {
137    draw_frame(canvas, IVec2::new(WIDTH, STORAGE_HEIGHT), CYAN, &[40, 80]);
138    draw_storage_pane(canvas, state, materials, StoragePane::Backpack, 0);
139    draw_storage_pane(canvas, state, materials, StoragePane::Storage, 40);
140    let selected = storage_stacks(materials, state.storage_pane).get(state.storage_selection).copied();
141    let (status, color) = match (state.feedback, selected) {
142        (Some(Ok(message)), _) => (message.to_owned(), GOOD),
143        (Some(Err(message)), _) => (message.to_owned(), BAD),
144        (None, Some(material)) => (material.name().to_ascii_uppercase(), TEXT),
145        (None, None) => ("NOTHING HERE".to_owned(), MUTED),
146    };
147    canvas.text(IVec2::new(MARGIN, 84), &status, color);
148    let verb = match state.storage_pane {
149        StoragePane::Backpack => "STORE",
150        StoragePane::Storage => "TAKE",
151    };
152    draw_key_hints(canvas, 95, (MARGIN, RIGHT), Some((verb, selected.is_some())), "EXIT", CYAN);
153}

155fn draw_storage_pane(canvas: &mut Canvas, state: &CapsuleUiState, materials: &MaterialInventory, pane: StoragePane, top: i32) {
156    let stacks = storage_stacks(materials, pane);
157    let active = state.storage_pane == pane;
158    let selection = if active { state.storage_selection } else { 0 };
159    let page = selection / STORAGE_PAGE;
160    let label = match pane {
161        StoragePane::Backpack => "BACKPACK",
162        StoragePane::Storage => "STORAGE",
163    };
164    canvas.text(IVec2::new(MARGIN, top + 4), label, if active { CYAN.bright } else { MUTED });
165    let pages = stacks.len().max(selection + 1).div_ceil(STORAGE_PAGE);
166    if pages > 1 {
167        let indicator = format!("{}/{pages}", page + 1);
168        canvas.text(IVec2::new(RIGHT - text_width(&indicator), top + 4), &indicator, MUTED);
169    }
170    for slot in 0..STORAGE_PAGE {
171        let well = well_rect(slot, top + 14);
172        draw_well(canvas, well, (active && page * STORAGE_PAGE + slot == selection).then_some(TEXT));
173        let Some(material) = stacks.get(page * STORAGE_PAGE + slot).copied() else {
174            canvas.fill(well, EMPTY_WELL);
175            continue;
176        };
177        canvas.image(material.icon_path(), well, Color::WHITE);
178        let count = MaterialInventory::display_count(pane.count(materials, material));
179        canvas.text(IVec2::new(well.min.x + (WELL + 1 - text_width(&count)) / 2, top + 30), &count, TEXT);
180    }
181}

183fn well_rect(slot: usize, top: i32) -> IRect {
184    well_at(IVec2::new(MARGIN, top), slot, COLUMNS)
185}

187fn draw_entry_icon(canvas: &mut Canvas, entry: TerminalEntry, area: IRect, availability: Availability) {
188    let ready = availability == Availability::Ready;
189    let (tint, research) = if ready { (Color::WHITE, AMBER.bright) } else { (UNAVAILABLE_ICON, DISABLED) };
190    match entry.recipe() {
191        Some(recipe) => canvas.image(recipe.icon_path, area, tint),
192        None => canvas.mask(area.min + (area.size() - IVec2::new(9, 8)) / 2, RESEARCH_MASK, research),
193    }
194}

196/// The active well or tabs guide a scroll viewport in unusually small windows.
197pub(super) fn selection_area(station: Station, state: &CapsuleUiState) -> Option<IRect> {
198    let well = match station {
199        Station::Terminal if state.terminal_focus == TerminalFocus::Tabs => terminal_tab_area(state.terminal_tab),
200        Station::Terminal => well_rect(state.terminal_selection % TERMINAL_PAGE, TERMINAL_GRID_TOP),
201        Station::Storage => {
202            let top = if state.storage_pane == StoragePane::Backpack { 14 } else { 54 };
203            well_rect(state.storage_selection % STORAGE_PAGE, top)
204        },
205    };
206    Some(well)
207}

209fn terminal_tab_area(tab: TerminalTab) -> IRect {
210    let (left, label) = match tab {
211        TerminalTab::Shop => (MARGIN, "SHOP"),
212        TerminalTab::Crafting => (MARGIN + text_width("SHOP") + 7, "CRAFT"),
213    };
214    rect(left, 3, text_width(label), 10)
