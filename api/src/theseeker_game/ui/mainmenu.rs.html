70#[derive(SystemParam)]
71pub(crate) struct WorldStarter<'w> {
72    config: ResMut<'w, LevelPngConfig>,
73    images: ResMut<'w, Assets<Image>>,
74    handle: Option<Res<'w, LevelPngHandle>>,
75    chunks: ResMut<'w, mapgen::ChunkLibrary>,
76    maps: ResMut<'w, mapgen::MapLibrary>,
77}
78
79impl WorldStarter<'_> {
80    /// Builds the request before switching screens, so a failed request leaves the current world intact.
81    pub(crate) fn generated(&mut self, commands: &mut Commands, next: &mut NextState<AppState>, current: AppState) {
82        let request = levelgen::pipeline::GenerationRequest::random(self.config.clone());
83        let result = match mapgen::generate_map(&mut self.chunks, &mut self.maps, &request) {
84            Ok(result) => result,
85            Err(err) => { error!("Failed to generate chunk map: {err}"); return; },
86        };
87        self.clear_previous_asset();
88        levelgen::diagnostics::begin(commands);
89        levelgen::reset_level_runtime(commands);
90        reset_expedition_progress(commands);
91        commands.remove_resource::<crate::game::intro::IntroRequested>();
92        if !levelgen::diagnostics::enabled() && std::env::var_os("SEEKER_SKIP_INTRO").is_none() {
93            commands.insert_resource(crate::game::intro::IntroRequested);
94        }
95        self.config.path.clone_from(&result.synthetic_path);
96        insert_generated_map_resources(commands, &result);
97        info!("generation_fixture: seed={} map={} templates={:?}", result.seed, result.map_name, result.selected_templates);
98        commands.insert_resource(request);
99        commands.insert_resource(result.compiled);
100        next.set(start_state(current));
101    }
102
103    fn custom(&mut self, commands: &mut Commands, next: &mut NextState<AppState>, current: AppState) {
104        let Some(path) = pick_custom_level_file() else { return };
105        let image = match load_custom_level_image(&path, &mut self.images) {
106            Ok(image) => image,
107            Err(err) => { error!("Failed to load custom level: {err}"); return; },
108        };
109        self.clear_previous_asset();
110        levelgen::reset_level_runtime(commands);
111        commands.remove_resource::<crate::game::GameOver>();
112        commands.remove_resource::<crate::game::intro::IntroRequested>();
113        reset_player_runtime(commands);
114        commands.insert_resource(LevelPngHandle(image));
115        self.config.path = path.to_string_lossy().into_owned();
116        commands.insert_resource(levelgen::pipeline::GenerationRequest::random(self.config.clone()));
117        next.set(start_state(current));
118    }
119
120    fn clear_previous_asset(&mut self) {
121        let handle = self.handle.as_ref().map(|handle| handle.0.clone());
122        cleanup_previous_level_asset(&mut self.images, handle, &self.config.path);
123    }
124}

126/// A newly generated expedition starts with a fresh player loadout and no owner-scoped runtime state.
127fn reset_expedition_progress(commands: &mut Commands) {
128    use crate::game::player::{spawn::PersistedPlayerProgression, weapon::PlayerMeleeWeapon};
129    commands.insert_resource(PersistedPlayerProgression::default());
130    commands.insert_resource(PlayerMeleeWeapon::default());
131    commands.insert_resource(crate::game::combat::KillCount::default());
132    reset_player_runtime(commands);
133    commands.remove_resource::<crate::game::GameOver>();
134}

136fn reset_player_runtime(commands: &mut Commands) {
137    use crate::game::player::progression::UnlockedSkillPresentationState;
138    use crate::game::player::passives::frenzied_attack::{EnergyRegenDeltas, LifeDebt};
139    commands.insert_resource(crate::game::player::skills::cooldowns::Cooldowns::default());
140    commands.insert_resource(EnergyRegenDeltas::default());
141    commands.insert_resource(LifeDebt::default());
142    commands.insert_resource(UnlockedSkillPresentationState::default());
143    commands.insert_resource(crate::game::player::skills::tether_node::TetherConnectionStatus::default());
144}

150fn insert_generated_map_resources(commands: &mut Commands, result: &mapgen::GeneratedMap) {
151    commands.insert_resource(LevelBiomeMask {
152        width: result.width, height: result.height, cells: result.biomes.clone(), tundra_inner: Vec::new(),
153    });
154    if let Some(pre_impact) = result.pre_impact.clone() { commands.insert_resource(pre_impact); }
155    commands.insert_resource(LevelChunkColorMask {
156        chunk_size: mapgen::CHUNK_IMAGE_SIZE, chunks_w: result.chunks_w, chunks_h: result.chunks_h, colors: result.chunk_colors.clone(),
157    });
158    commands.insert_resource(LevelChunkOccupancyMask {
159        chunks_w: result.chunks_w, chunks_h: result.chunks_h, occupied: result.chunk_occupied.clone(),
160    });
161}

163#[expect(clippy::too_many_arguments, reason = "The menu translates frame input into app lifecycle commands and file selection.")]
164fn handle_menu_input(
165    keys: Res<ButtonInput<KeyCode>>,
166    mouse: Res<ButtonInput<MouseButton>>,
167    app_state: Res<State<AppState>>,
168    mut menu: ResMut<MenuState>,
169    panels: (Res<super::inventory::InventoryState>, Res<super::capsule_ui::CapsuleUiState>),
170    grid: UiPixelGrid,
171    mut starter: WorldStarter,
172    mut commands: Commands,
173    mut next: ResMut<NextState<AppState>>,
174    mut exit: MessageWriter<AppExit>,
175    mut pause: ResMut<crate::gamestate::PauseRequests>,
176) {
177    menu.consumed_input = false;
178    let standalone = *app_state.get() == AppState::MainMenu;
179    if keys.just_pressed(KeyCode::Escape) && !panels.0.is_open() && !panels.1.is_open() {
180        menu.consumed_input = true;
181        menu.open = !menu.open;
182        menu.selected = 0;
183        menu.scroll = default();
184        return;
185    }
186    if !menu.open && !standalone { return; }
187    let first = usize::from(standalone);
188    menu.selected = menu.selected.saturating_add_signed(vertical_delta(&keys) as isize).clamp(first, 3);
189    let clicked = menu_pointer(&grid, &menu).filter(|_| mouse.just_pressed(MouseButton::Left));
190    let Some(index) = clicked.or_else(|| keys.any_just_pressed([KeyCode::KeyF, KeyCode::Enter, KeyCode::Space]).then_some(menu.selected)) else { return };
191    menu.consumed_input = true;
192    menu.selected = index.max(first);
193    match MenuAction::ALL[menu.selected] {
194        MenuAction::Resume => { menu.open = false; pause.resume_manual(); },
195        MenuAction::NewGame => starter.generated(&mut commands, &mut next, *app_state.get()),
196        MenuAction::LoadCustom => starter.custom(&mut commands, &mut next, *app_state.get()),
197        MenuAction::Quit => { exit.write(AppExit::Success); },
198    }
199}

201fn menu_pointer(grid: &UiPixelGrid, menu: &MenuState) -> Option<usize> {
202    let cursor = grid.cursor_art()?;
203    let available = grid.available_art()?;
204    let shown = UVec2::new(WIDTH as u32, HEIGHT as u32).min(available);
205    let origin = ((available - shown) / 2).as_vec2();
206    let point = cursor - origin + menu.scroll.offset.as_vec2();
207    if point.x < 7.0 || point.x >= (WIDTH - 7) as f32 || point.y < FIRST_BUTTON as f32 { return None; }
208    let index = (point.y as i32 - FIRST_BUTTON) / BUTTON_HEIGHT;
209    (index < 4).then_some(index as usize)
210}

212#[expect(clippy::too_many_arguments, reason = "The pixel menu publishes its raster on the current UI grid.")]
213fn present_menu(
214    mut commands: Commands,
215    mut state: ResMut<MenuState>,
216    app_state: Res<State<AppState>>,
217    keys: Res<ButtonInput<KeyCode>>,
218    mut library: ResMut<ArtLibrary>,
219    server: Res<AssetServer>,
220    mut images: ResMut<Assets<Image>>,
221    mut panel: Query<(Entity, &ImageNode, &mut Node), With<MenuPanel>>,
222    grid: UiPixelGrid,
223) {
224    let standalone = *app_state.get() == AppState::MainMenu;
225    let content = (state.open || standalone).then(|| {
226        library.load(&server, [FONT_IMAGE_PATH]);
227        let raster = draw_menu(&state, standalone, &library.art(&images));
228        let focus = menu_focus(state.selected);
229        grid.scrollable(raster, &mut state.scroll, &keys, Some(focus), Vec2::splat(0.5))
230    }).flatten();
231    show_raster(&mut commands, &mut images, panel.single_mut().ok(), content, |image, node| {
232        (Name::new("PauseMenu"), MenuPanel, image, node, GlobalZIndex(i32::MAX - 900), StateDespawnMarker)
233    });
234}

236fn menu_focus(selected: usize) -> IRect {
237    let width = text_width(MenuAction::ALL[selected].label());
238    rect((WIDTH - width) / 2 - 3, FIRST_BUTTON + selected as i32 * BUTTON_HEIGHT + 2, width + 6, 12)
239}

241fn draw_menu(state: &MenuState, standalone: bool, art: &Art) -> Raster {
242    let mut canvas = Canvas::new(UVec2::new(WIDTH as u32, HEIGHT as u32), art);
243    draw_frame(&mut canvas, IVec2::new(WIDTH, HEIGHT), ACCENT, &[14, 93]);
244    let title = if standalone { "THE SEEKER" } else { "PAUSED" };
245    canvas.text(IVec2::new((WIDTH - text_width(title)) / 2, 4), title, ACCENT.bright);
246    for (index, action) in MenuAction::ALL.into_iter().enumerate().skip(usize::from(standalone)) {
247        let top = FIRST_BUTTON + index as i32 * BUTTON_HEIGHT;
248        let selected = index == state.selected;
249        canvas.plate(rect(7, top, WIDTH - 14, 15), 1, if selected { ACCENT.dim } else { super::pixel_panel::WELL_FILL }, ACCENT.dim);
250        let label = action.label();
251        canvas.text(IVec2::new((WIDTH - text_width(label)) / 2, top + 4), label, if selected { TEXT } else { MUTED });
252    }
253    key_hint(&mut canvas, IVec2::new(7, 98), "F", "SELECT", ACCENT.bright);
254    key_hint(&mut canvas, IVec2::new(7, 110), "ESC", "RESUME", MUTED);
255    key_hint(&mut canvas, IVec2::new(7, 122), "PGUP/DN", "SCROLL", MUTED);
256    canvas.finish()
257}

259fn load_custom_level_image(
260    path: &Path,
261    images: &mut Assets<Image>,
262) -> Result<Handle<Image>, String> {
263    let bytes = std::fs::read(path).map_err(|err| {
264        format!(
265            "could not read {}: {err}",
266            path.display()
267        )
268    })?;
269    let image = Image::from_buffer(
270        &bytes,
271        ImageType::Extension("png"),
272        CompressedImageFormats::default(),
273        true,
274        ImageSampler::Default,
275        RenderAssetUsages::default(),
276    )
277    .map_err(|err| {
278        format!(
279            "could not decode {}: {err}",
280            path.display()
281        )
282    })?;
283    Ok(images.add(image))
284}

286fn default_levels_directory() -> Option<PathBuf> {
287    let mut dir = std::env::current_dir().ok()?;
288    dir.push("assets");
289    dir.push("levels");
290    Some(dir)

293fn cleanup_previous_level_asset(
294    images: &mut Assets<Image>,
295    previous_handle: Option<Handle<Image>>,
296    previous_path: &str,
297) {
298    let is_custom_file = Path::new(previous_path).is_absolute();
299    let is_generated =
300        previous_path.starts_with(mapgen::GENERATED_LEVEL_PREFIX);
301    if !(is_custom_file || is_generated) {
302        return;
303    }
304
305    if let Some(handle) = previous_handle {
306        images.remove(handle.id());
307    }
308}

310#[cfg(not(target_arch = "wasm32"))]
311fn pick_custom_level_file() -> Option<PathBuf> {
312    let mut dialog = rfd::FileDialog::new();
313    dialog = dialog.add_filter("PNG Level", &["png"]);
314    if let Some(default_dir) = default_levels_directory() {
315        dialog = dialog.set_directory(default_dir);
316    }
317    dialog.pick_file()
318}

338    fn new_expedition_clears_persisted_loadout_storage_and_run_counters() {
339        use bevy::ecs::system::RunSystemOnce;
340        use crate::game::player::{CraftMaterial, MaterialInventory, SkillInventory};
341        use crate::game::player::spawn::PersistedPlayerProgression;
342        use crate::game::player::skills::types::SkillId;
343        use crate::game::player::weapon::PlayerMeleeWeapon;
344        let mut world = World::new();
345        let mut materials = MaterialInventory::empty();
346        materials.add_to_storage(CraftMaterial::BatWing, 7);
347        let mut skills = SkillInventory::default();
348        skills.unlock_skill(SkillId::Dash);
349        world.insert_resource(PersistedPlayerProgression { materials: Some(materials), skills: Some(skills), ..default() });
350        world.insert_resource(PlayerMeleeWeapon::Hammer);
351        world.insert_resource(crate::game::combat::KillCount(17));
352        world.insert_resource(crate::game::GameOver);
353        world.run_system_once(|mut commands: Commands| reset_expedition_progress(&mut commands)).unwrap();
354        let progress = world.resource::<PersistedPlayerProgression>();
355        assert!(progress.materials.is_none(), "a new expedition cannot inherit protected storage");
356        assert!(progress.skills.is_none(), "unlocked skills belong to the preceding expedition");
357        assert_eq!(*world.resource::<PlayerMeleeWeapon>(), PlayerMeleeWeapon::Sword);
358        assert_eq!(world.resource::<crate::game::combat::KillCount>().0, 0);
359        assert!(!world.contains_resource::<crate::game::GameOver>());
360    }
