1//! Damage numbers: pixel digits that rise from a damaged actor and fade, pausing with the game.
2use bevy::color::palettes::css;
3use theseeker_engine::animation::SpriteAnimationBundle;
4use theseeker_engine::audio::transport::TICK_RATE_HZ;
5use theseeker_engine::physics::Collider;
6use theseeker_engine::script::ScriptPlayer;

8use crate::game::combat::damage_source::apply_damage;
9use crate::game::combat::DamageInfo;
10use crate::game::player::Player;
11use crate::prelude::*;

13/// Damage numbers stay hidden while this is off.
14const DAMAGE_NUMBERS_ENABLED: bool = false;
15/// How fast a number rises, in world units per second of game time, when it starts fading,
16/// and when it is gone.
17const RISE_SPEED: f32 = 6.0;
18const FADE_START: f32 = 0.25;
19const LIFETIME: f32 = 3.0;
20/// How far above its target's collider a number appears.
21const ABOVE_COLLIDER: f32 = 15.0;
22/// A digit's width, and the step from one digit to the next.
23const DIGIT_WIDTH: f32 = 3.0;
24const DIGIT_STEP: f32 = 4.0;
25/// Each number lies this much above the one before, until a second passes without one.
26const STACK_STEP: f32 = 0.000_000_1;
27
28pub struct DmgNumbersPlugin;
29
30impl Plugin for DmgNumbersPlugin {
31    fn build(&self, app: &mut App) {
32        app.init_resource::<DamageNumberStack>()
33            .add_systems(GameTickUpdate, spawn_damage_numbers.after(apply_damage))
34            .add_systems(Update, rise_and_fade);
35    }
36}

38/// Marks an entity whose damage shows no numbers.
39#[derive(Component)]
40pub struct NoDamageNumbers;

42/// How far above the world the next number lies, so later numbers cover earlier ones.
43#[derive(Resource, Default)]
44struct DamageNumberStack {
45    z: f32,
46    last_spawn_tick: u64,
47}
48
49#[derive(Component)]
50struct DamageDigit {
51    start_y: f32,
52    /// When the digit appeared, in game seconds.
53    spawned_at: f64,
54}
55
56fn spawn_damage_numbers(
57    mut commands: Commands,
58    mut damage: MessageReader<DamageInfo>,
59    targets: Query<(&GlobalTransform, Option<&Collider>, Has<Player>, Has<NoDamageNumbers>)>,
60    time: Res<GameTime>,
61    mut stack: ResMut<DamageNumberStack>,
62) {
63    if !DAMAGE_NUMBERS_ENABLED {
64        return;
65    }
66    for info in damage.read() {
67        let Ok((transform, collider, player, silent)) = targets.get(info.target) else {
68            continue;
69        };
70        if silent {
71            continue;
72        }
73        let color = match (player, info.crit, info.stealthed) {
74            (true, ..) => css::RED.into(),
75            (false, true, true) => css::PURPLE.into(),
76            (false, true, false) => (css::YELLOW * 1.1).into(),
77            (false, false, true) => css::PINK.into(),
78            (false, false, false) => Color::WHITE,
79        };
80        let above = collider.map_or(Vec3::ZERO, |collider| Vec3::new(1.0, collider.half_extents().y + ABOVE_COLLIDER, 0.0));
81        let position = transform.translation() + above;
82        if time.tick() > stack.last_spawn_tick + u64::from(TICK_RATE_HZ) {
83            stack.z = 0.0;
84        }
85        stack.z += STACK_STEP;
86        stack.last_spawn_tick = time.tick();
87        let text = format!("{:.0}", info.amount);
88        let left = position.x - (text.len() as f32 * DIGIT_STEP - (DIGIT_STEP - DIGIT_WIDTH)) / 2.0;
89        for (i, digit) in text.chars().enumerate() {
90            let mut player = ScriptPlayer::default();
91            player.play_key(&format!("anim.numbers.{digit}"));
92            commands.spawn((
93                DamageDigit { start_y: position.y, spawned_at: time.time_in_seconds() },
94                SpriteAnimationBundle { player },
95                Sprite { texture_atlas: Some(default()), color, ..default() },
96                Transform::from_xyz(left + i as f32 * DIGIT_STEP, position.y, position.z + 0.1 + stack.z),
97                StateDespawnMarker,
98            ));
99        }
100    }
101}

103fn rise_and_fade(mut commands: Commands, time: Res<GameTime>, mut digits: Query<(Entity, &DamageDigit, &mut Transform, &mut Sprite)>) {
104    for (entity, digit, mut transform, mut sprite) in &mut digits {
105        let age = (time.time_in_seconds() - digit.spawned_at) as f32;
106        transform.translation.y = digit.start_y + RISE_SPEED * age;
107        sprite.color.set_alpha(1.0 - ((age - FADE_START) / (LIFETIME - FADE_START)).clamp(0.0, 1.0));
108        if age > LIFETIME {
109            commands.entity(entity).despawn();
110        }
111    }
112}
