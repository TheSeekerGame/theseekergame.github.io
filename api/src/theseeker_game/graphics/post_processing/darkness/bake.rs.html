1//! The startup lightmap bake. Light reaches the cave where daylight can: a lightmap pixel
2//! is a source where a camera looking at it sees through the ground and every background
3//! layer to the sky, and every pixel takes the light of its nearest source, full at the
4//! source and fading to nothing at the light's radius.
5
6use super::*;
7use crate::levelgen::distance::DistanceField;
8
9/// Lightmap pixels per level tile.
10const LIGHTMAP_SCALE: u32 = 2;
11/// How far a source lights, in world units.
12const SOURCE_LIGHT_RADIUS_WORLD: f32 = 24.0;
13/// How quickly a source's light fades toward its radius (1 is linear).
14const SOURCE_LIGHT_FALLOFF_EXPONENT: f32 = 2.0;
15
16const MASK_CHUNK_EDGE: u32 = mapgen::CHUNK_IMAGE_SIZE * crate::levelgen::GROUND_TILE_ART_SIZE_PX;
17
18/// One layer's coverage, which parallax shifts against the camera by its depth.
19#[derive(Clone, Default)]
20pub struct DarknessBakeMaskLayer {
21    chunks: Arc<HashMap<UVec2, MaskChunk>>,
22    size_px: UVec2,
23    world_center: Vec2,
24    world_size: Vec2,
25    parallax_depth: f32,
26}
27
28#[derive(Clone)]
29struct MaskChunk { size: UVec2, alpha: Vec<u8> }
30
31impl DarknessBakeMaskLayer {
32    pub(crate) fn set_chunks(&mut self, size: UVec2,
33        chunks: Vec<(UVec2, u32, u32, Vec<u8>)>, center: Vec2, world_size: Vec2, depth: f32) {
34        let chunks = chunks.into_iter().filter(|(_, _, _, alpha)| alpha.contains(&1))
35            .map(|(origin, width, height, alpha)| (origin / MASK_CHUNK_EDGE, MaskChunk { size: UVec2::new(width, height), alpha }));
36        *self = Self { chunks: Arc::new(chunks.collect()), size_px: size, world_center: center, world_size, parallax_depth: depth };
37    }
38
39    fn is_empty(&self) -> bool {
40        self.chunks.is_empty() || self.size_px.min_element() == 0 || self.world_size.min_element() <= 0.0
41    }
42
43    /// Whether the layer covers `world` for a camera centered on it. Parallax shows a layer
44    /// its depth times nearer its own center there.
45    fn covers(&self, world: Vec2) -> bool {
46        let depth = if self.parallax_depth.is_finite() && self.parallax_depth > 0.0 { self.parallax_depth } else { 1.0 };
47        let offset = (world - self.world_center) / depth / self.world_size;
48        let uv = Vec2::new(0.5 + offset.x, 0.5 - offset.y);
49        if uv.cmplt(Vec2::ZERO).any() || uv.cmpge(Vec2::ONE).any() {
50            return false;
51        }
52        let pixel = (uv * self.size_px.as_vec2()).as_uvec2().min(self.size_px - 1);
53        self.chunks.get(&(pixel / MASK_CHUNK_EDGE)).is_some_and(|chunk| {
54            let local = pixel % MASK_CHUNK_EDGE;
55            local.cmplt(chunk.size).all() && chunk.alpha[(local.y * chunk.size.x + local.x) as usize] != 0
56        })
57    }
58}

60/// The layers the sky is seen through.
61#[derive(Resource, Clone, Default)]
62pub struct DarknessBakeMaskAtlas {
63    pub near: DarknessBakeMaskLayer,
64    pub middle: DarknessBakeMaskLayer,
65    pub far: DarknessBakeMaskLayer,
66    pub ground: DarknessBakeMaskLayer,
67}
68
69impl DarknessBakeMaskAtlas {
70    pub fn clear_background_layers(&mut self) {
71        let ground = std::mem::take(&mut self.ground);
72        *self = Self { ground, ..default() };
73    }
74
75    /// The layers holding coverage, the ground (most often covering) first.
76    fn layers(&self) -> Vec<&DarknessBakeMaskLayer> {
77        [&self.ground, &self.near, &self.middle, &self.far].into_iter().filter(|layer| !layer.is_empty()).collect()
78    }
79}

81pub(super) fn reset_bake(mut target: ResMut<DarknessLightMapTarget>, mut masks: ResMut<DarknessBakeMaskAtlas>) {
82    target.baked = false;
83    *masks = default();
84}

86pub(super) fn lightmap_size(level: &CompiledLevel) -> UVec2 {
87    UVec2::new(level.width, level.height).saturating_mul(UVec2::splat(LIGHTMAP_SCALE)).max(UVec2::ONE)
88}

90/// Bakes the lightmap once the level's surfaces, and with them all of its coverage, are ready.
91pub(super) fn bake_lightmap(
92    build: Option<ResMut<GenerationBuild>>,
93    level: Option<Res<CompiledLevel>>,
94    occupancy: Option<Res<LevelChunkOccupancyMask>>,
95    masks: Res<DarknessBakeMaskAtlas>,
96    mut target: ResMut<DarknessLightMapTarget>,
97    mut images: ResMut<Assets<Image>>,
98    device: Res<RenderDevice>,
99) {
100    let (Some(mut build), Some(level)) = (build, level) else { return; };
101    if target.baked || !build.surfaces_ready() {
102        return;
103    }
104    let pages = bake(&level, occupancy.as_deref(), &masks).ok_or_else(|| "Darkness coverage could not be baked".to_owned())
105        .and_then(|light| atlas::pack(lightmap_size(&level), &light, device.limits().max_texture_array_layers));
106    match pages {
107        Ok((pages, table)) => {
108            images.insert(&target.handle, pages).expect("lightmap handle remains reserved");
109            images.insert(&target.table, table).expect("lightmap page-table handle remains reserved");
110            target.baked = true;
111        },
112        Err(error) => build.fail(error),
113    }
114}

116/// Bakes the lightmap, or `None` for a level without size or coverage. Chunks `occupancy`
117/// leaves out of the map hold no sources.
118fn bake(level: &CompiledLevel, occupancy: Option<&LevelChunkOccupancyMask>, masks: &DarknessBakeMaskAtlas) -> Option<Vec<u8>> {
119    let _timer = crate::levelgen::diagnostics::StageTimer::new("darkness_lightmap_bake");
120    let layers = masks.layers();
121    let size = lightmap_size(level);
122    let pixel_world = level.bounds.size() / size.as_vec2();
123    if level.width == 0 || level.height == 0 || pixel_world.min_element() <= 0.0 || layers.is_empty() {
124        return None;
125    }
126    let chunk = mapgen::CHUNK_IMAGE_SIZE * LIGHTMAP_SCALE;
127    let occupancy = occupancy.filter(|mask| {
128        let matches = mask.matches_chunks(size.x.div_ceil(chunk), size.y.div_ceil(chunk));
129        if !matches {
130            warn!("darkness: chunk occupancy mask does not match the level; every chunk may hold sources");
131        }
132        matches
133    });
134    let mut nearest = DistanceField::default();
135    nearest.compute(size.x as usize, size.y as usize, |x, y| {
136        let pixel = UVec2::new(x as u32, y as u32);
137        let world = Vec2::new(level.bounds.min.x, level.bounds.max.y) + (pixel.as_vec2() + 0.5) * pixel_world * Vec2::new(1.0, -1.0);
138        occupancy.is_none_or(|mask| mask.is_occupied(pixel.x / chunk, pixel.y / chunk))
139            && !layers.iter().any(|layer| layer.covers(world))
140    });
141    let light = light_by_distance(pixel_world.min_element());
142    Some(nearest.squared.iter().map(|&distance2| light.get(distance2 as usize).copied().unwrap_or(0)).collect())
143}
144
145/// Light by squared distance from the nearest source, in lightmap pixels `pixel_world` world
146/// units wide, as 8-bit coverage.
147fn light_by_distance(pixel_world: f32) -> Vec<u8> {
148    let reach = (SOURCE_LIGHT_RADIUS_WORLD / pixel_world).ceil() as usize;
149    (0..=reach * reach).map(|distance2| {
150        let t = ((distance2 as f32).sqrt() * pixel_world / SOURCE_LIGHT_RADIUS_WORLD).min(1.0);
151        ((1.0 - t).powf(SOURCE_LIGHT_FALLOFF_EXPONENT) * 255.0 + 0.5) as u8
152    }).collect()
153}
