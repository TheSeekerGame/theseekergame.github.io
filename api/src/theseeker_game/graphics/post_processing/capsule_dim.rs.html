20const CAPSULE_DIM_FADE_SPEED: f32 = 1.0;
21const CAPSULE_DIM_MAX_ALPHA: f32 = 0.2;
22pub(crate) const CAPSULE_DIM_RECT_LIMIT: usize = 64;

24/// What the capsule dim pass reads, laid out as `CapsuleDimSettings` in `capsule_dim.wgsl`.
25/// Positions and sizes are in world units. The default dims nothing.
26#[derive(Component, Clone, Copy, ExtractComponent, ShaderType, Reflect)]
27#[extract_component_sync_target((Self, bevy::render::extract_component::DynamicUniformIndex<Self>))]
28pub struct CapsuleDimSettings {
29    /// The bottom-left corner of the camera's view.
30    view_min: Vec2,
31    view_size: Vec2,
32    /// The share of light the view loses outside the lit rectangles.
33    dim: f32,
34    /// How many of `lit_rects` the capsule lights.
35    lit_count: u32,
36    /// The rooms the capsule lights, as minimum and maximum corners.
37    lit_rects: [Vec4; CAPSULE_DIM_RECT_LIMIT],
38}
39
40impl Default for CapsuleDimSettings {
41    fn default() -> Self {
42        Self { view_min: Vec2::ZERO, view_size: Vec2::ZERO, dim: 0.0, lit_count: 0, lit_rects: [Vec4::ZERO; CAPSULE_DIM_RECT_LIMIT] }
43    }
44}

46impl Plugin for CapsuleDimPlugin {
47    fn build(&self, app: &mut App) {
48        app.add_plugins((
49            ExtractComponentPlugin::<CapsuleDimSettings>::default(),
50            UniformComponentPlugin::<CapsuleDimSettings>::default(),
51        ));
52        app.register_type::<CapsuleDimSettings>();
53        app.register_required_components::<MainCamera, CapsuleDimSettings>();
54        app.add_systems(
55            Update,
56            sync_capsule_dim_settings
57                .after(update_camera)
58                .after(update_pixel_view)
59                .in_set(GameTickSet::Post)
60                .run_if(in_state(AppState::InGame)),
61        );
62    }
63
64    fn finish(&self, app: &mut App) {
65        let Some(render_app) = app.get_sub_app_mut(RenderApp) else {
66            return;
67        };
68        render_app.add_systems(bevy::render::RenderStartup, |world: &mut World| { world.init_resource::<CapsuleDimPipeline>(); });
69    }
70}

72/// Fades the view's dimming in while the player shelters in a capsule, sparing the rooms the
73/// capsule lights, and drops it when the player leaves.
74fn sync_capsule_dim_settings(
75    time: Res<PresentationTime>,
76    view: Res<PixelView>,
77    player: Query<&CapsuleOccupant, With<Player>>,
78    capsules: Query<(&Transform, &CapsuleLightMask), With<Capsule>>,
79    mut progress: Local<f32>,
80    mut cameras: Query<(&mut CapsuleDimSettings, &Projection), With<MainCamera>>,
81) {
82    let occupant = player.single().ok();
83    *progress = occupant.map_or(0.0, |_| (*progress + time.delta_secs() * CAPSULE_DIM_FADE_SPEED).min(1.0));
84    let Some(view) = view.0 else { return; };
85    let lit = occupant.and_then(|occupant| capsules.get(occupant.shelter).ok())
86        .map(|(transform, mask)| mask.world_rects(transform.translation.truncate()))
87        .unwrap_or_default();
88    let mut lit_rects = [Vec4::ZERO; CAPSULE_DIM_RECT_LIMIT];
89    for (slot, rect) in lit_rects.iter_mut().zip(&lit) {
90        *slot = Vec4::from((rect.min, rect.max));
91    }
92    for (mut settings, projection) in &mut cameras {
93        let Projection::Orthographic(ortho) = projection else { continue; };
94        let view_size = ortho.area.size();
95        *settings = CapsuleDimSettings {
96            view_min: view.displayed_center() - 0.5 * view_size,
97            view_size,
98            dim: *progress * CAPSULE_DIM_MAX_ALPHA,
99            lit_count: lit.len().min(CAPSULE_DIM_RECT_LIMIT) as u32,
100            lit_rects,
101        };
102    }
103}

105pub(crate) fn capsule_dim_pass(
106    world: &World,
107    view: ViewQuery<(FullscreenView, &CapsuleDimSettings, &DynamicUniformIndex<CapsuleDimSettings>)>,
108    mut render_context: RenderContext,
109) {
110    let (view, settings, settings_index) = view.into_inner();
111    // Undimmed, the pass would only copy the view.
112    if settings.dim <= 0.0 {
113        return;
114    }
115    let Some(settings) = world.resource::<ComponentUniforms<CapsuleDimSettings>>().uniforms().binding() else {
116        return;
117    };
118    world.resource::<CapsuleDimPipeline>().0.draw(world, &mut render_context, view,
119        |image| BindGroupEntries::sequential((image, settings)), &[settings_index.index()]);
120}

122#[derive(Resource)]
123struct CapsuleDimPipeline(FullscreenPass);

125impl FromWorld for CapsuleDimPipeline {
126    fn from_world(world: &mut World) -> Self {

128            (fullscreen::texture(), uniform_buffer::<CapsuleDimSettings>(true)), None))
129    }
