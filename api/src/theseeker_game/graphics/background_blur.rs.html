14/// Per-parallax blur radii expressed in screen pixels.
15#[derive(Resource, Clone)]
16pub struct BackgroundBlurSettings {
17    /// Blur radius in pixels for each background parallax group.
18    /// Index 0: bg1-3, 1: bg4-6, 2: bg7-8.
19    pub group_blur_pixels: [f32; 3],
20}

22impl Default for BackgroundBlurSettings {
23    fn default() -> Self {
24        Self {
25            group_blur_pixels: [2.0, 3.0, 5.0],
26        }
27    }
28}

36#[derive(Default)]
37struct BlurScratch { linear: Vec<Vec4>, temp: Vec<Vec4> }
38thread_local! { static SCRATCH: std::cell::RefCell<BlurScratch> = std::cell::RefCell::default(); }

40/// Blurs linear premultiplied pixels; conversion occurs only at the image boundary.
41pub fn apply_background_blur(image: &mut Image, radius_px: f32) {
42    SCRATCH.with_borrow_mut(|scratch| blur_with_scratch(image, radius_px, scratch));
43}
44
45fn blur_with_scratch(image: &mut Image, radius_px: f32, scratch: &mut BlurScratch) {
46    let _timer = crate::levelgen::diagnostics::StageTimer::new("apply_background_blur");
47    if radius_px <= 0.0001 { return; }
48    if !matches!(image.texture_descriptor.format, TextureFormat::Rgba8UnormSrgb | TextureFormat::Rgba8Unorm) { return; }
49    let srgb = image.texture_descriptor.format == TextureFormat::Rgba8UnormSrgb;
50    let width = image.width() as usize;
51    let height = image.height() as usize;
52    let Some(data) = image.data.as_mut() else { return; };
53    if width == 0 || height == 0 || data.len() != width * height * 4 { return; }
54    let kernel = gaussian_kernel(radius_px);
55    let BlurScratch { linear, temp } = scratch;
56    linear.clear();
57    linear.extend(data.as_chunks::<4>().0.iter().map(|px| decode(px, srgb)));
58    temp.resize(linear.len(), Vec4::ZERO);
59    horizontal(linear, temp, width, height, &kernel);
60    vertical(temp, linear, width, height, &kernel);
61    for (pixel, &value) in data.as_chunks_mut::<4>().0.iter_mut().zip(linear.iter()) {
62        encode(pixel, value, srgb);
63    }
64    image.texture_descriptor.mip_level_count = 1;
65}

67pub(crate) fn support(radius_px: f32) -> u32 {
68    (radius_px.max(0.5) * 1.5).ceil() as u32
69}
70
71fn gaussian_kernel(radius: f32) -> Vec<f32> {
72    let sigma = radius.max(0.5) * 0.5;
73    let mut kernel: Vec<_> = (0..=support(radius)).map(|i| {
74        (-((i * i) as f32) / (2.0 * sigma * sigma)).exp()
75    }).collect();
76    let sum = kernel[0] + 2.0 * kernel.iter().skip(1).sum::<f32>();
77    for weight in &mut kernel { *weight /= sum; }
78    kernel
79}

81fn horizontal(src: &[Vec4], dst: &mut [Vec4], width: usize, height: usize, kernel: &[f32]) {
82    for y in 0..height {
83        let row = &src[y * width..(y + 1) * width];
84        let out = &mut dst[y * width..(y + 1) * width];
85        for (x, pixel) in out.iter_mut().enumerate() {
86            *pixel = horizontal_pixel(row, x, kernel);
87        }
88    }
89}

91fn vertical(src: &[Vec4], dst: &mut [Vec4], width: usize, height: usize, kernel: &[f32]) {
92    for y in 0..height {
93        let out = &mut dst[y * width..(y + 1) * width];
94        for (x, pixel) in out.iter_mut().enumerate() {
95            *pixel = src[y * width + x] * kernel[0];
96        }
97        for (offset, &weight) in kernel.iter().enumerate().skip(1) {
98            let top = y.saturating_sub(offset) * width;
99            let bottom = (y + offset).min(height - 1) * width;
100            add_rows(out, &src[top..top + width], &src[bottom..bottom + width], weight);
101        }
102    }
103}

105fn decode(pixel: &[u8], srgb: bool) -> Vec4 {
106    static LINEAR: std::sync::OnceLock<[f32; 256]> = std::sync::OnceLock::new();
107    let lut = LINEAR.get_or_init(|| std::array::from_fn(|i| {
108        let value = i as f32 / 255.0;
109        if value <= 0.04045 { value / 12.92 } else { ((value + 0.055) / 1.055).powf(2.4) }
110    }));
111    let alpha = pixel[3] as f32 / 255.0;
112    let channel = |i: usize| if srgb { lut[pixel[i] as usize] } else { pixel[i] as f32 / 255.0 };
113    Vec4::new(channel(0) * alpha, channel(1) * alpha, channel(2) * alpha, alpha)
114}

116fn encode(pixel: &mut [u8], value: Vec4, srgb: bool) {
117    static SRGB: std::sync::OnceLock<[u8; 4097]> = std::sync::OnceLock::new();
118    let lut = SRGB.get_or_init(|| std::array::from_fn(|i| {
119        let value = i as f32 / 4096.0;
120        let encoded = if value <= 0.003_130_8 { value * 12.92 } else { 1.055 * value.powf(1.0 / 2.4) - 0.055 };
121        (encoded * 255.0 + 0.5) as u8
122    }));
123    let color = if value.w > 0.0001 { value / value.w } else { Vec4::ZERO };
124    for channel in 0..3 {
125        let value = color[channel].clamp(0.0, 1.0);
126        pixel[channel] = if srgb { lut[(value * 4096.0 + 0.5) as usize] } else { (value * 255.0 + 0.5) as u8 };
127    }
128    pixel[3] = (value.w.clamp(0.0, 1.0) * 255.0 + 0.5) as u8;
129}

131fn horizontal_pixel(row: &[Vec4], x: usize, kernel: &[f32]) -> Vec4 {
132    let mut value = row[x] * kernel[0];
133    for (offset, &weight) in kernel.iter().enumerate().skip(1) {
134        value += (row[x.saturating_sub(offset)] + row[(x + offset).min(row.len() - 1)]) * weight;
135    }
136    value
137}
138
139fn add_rows(out: &mut [Vec4], top: &[Vec4], bottom: &[Vec4], weight: f32) {
140    for ((pixel, &a), &b) in out.iter_mut().zip(top).zip(bottom) {
141        *pixel += (a + b) * weight;
142    }
