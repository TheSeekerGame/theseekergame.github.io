1//! Fullscreen effects. Each owns its shader, uniforms, and targets, and draws through a
2//! `FullscreenPass` from the view's current image into its next, within the camera's viewport.
3//! The image spans the whole render target, so shaders read it at `in.position`; `in.uv`
4//! spans the viewport, which is what the camera shows.
5use bevy::prelude::*;
6use bevy::core_pipeline::FullscreenShader;
7use bevy::render::camera::ExtractedCamera;
8use bevy::render::render_resource::binding_types::{sampler as sampler_binding, texture_2d};
9use bevy::render::render_resource::*;
10use bevy::render::renderer::RenderContext;
11use bevy::render::view::ViewTarget;
12
13/// A filterable 2D color texture, such as the view's image.
14pub fn texture() -> BindGroupLayoutEntryBuilder {
15    texture_2d(TextureSampleType::Float { filterable: true })
16}
17
18/// A filtering sampler.
19pub fn sampler() -> BindGroupLayoutEntryBuilder {
20    sampler_binding(SamplerBindingType::Filtering)
21}
22
23/// What a fullscreen pass draws on: the view's images, and the camera whose viewport bounds it.
24pub type FullscreenView = (&'static ViewTarget, &'static ExtractedCamera);
25
26/// A fullscreen effect's pipeline, which reads the view's current image and writes its next.
27pub struct FullscreenPass {
28    label: &'static str,
29    layout: BindGroupLayoutDescriptor,
30    pipeline: CachedRenderPipelineId,
31    /// Whether the effect blends onto the view's next image, which it must then keep.
32    blends: bool,
33}
34
35impl FullscreenPass {
36    /// Queues the pass drawing the fragment shader at `shader` with `bindings`, the first of
37    /// them the view's current image. With `blend`, it blends onto the view's next image.
38    pub fn new<const N: usize>(world: &World, label: &'static str, shader: &'static str,
39        bindings: impl IntoBindGroupLayoutEntryBuilderArray<N>, blend: Option<BlendState>,
40    ) -> Self {
41        let layout = BindGroupLayoutDescriptor::new(label, &BindGroupLayoutEntries::sequential(ShaderStages::FRAGMENT, bindings));
42        let pipeline = world.resource::<PipelineCache>().queue_render_pipeline(RenderPipelineDescriptor {
43            label: Some(label.into()),
44            layout: vec![layout.clone()],
45            vertex: world.resource::<FullscreenShader>().to_vertex_state(),
46            fragment: Some(FragmentState {
47                shader: world.resource::<AssetServer>().load(shader),
48                shader_defs: vec![],
49                entry_point: Some("fragment".into()),
50                targets: vec![Some(ColorTargetState { format: TextureFormat::Rgba8UnormSrgb, blend, write_mask: ColorWrites::ALL })],
51            }),
52            primitive: PrimitiveState::default(),
53            depth_stencil: None,
54            multisample: MultisampleState::default(),
55            immediate_size: 0,
56            zero_initialize_workgroup_memory: false,
57        });
58        Self { label, layout, pipeline, blends: blend.is_some() }
59    }
60
61    /// Draws the pass over `view` once its pipeline has compiled. `bindings` is given the
62    /// view's current image and binds it first, then the effect's own resources.
63    pub fn draw<'a, const N: usize>(&self, world: &'a World, render_context: &mut RenderContext,
64        (view, camera): (&'a ViewTarget, &ExtractedCamera),
65        bindings: impl FnOnce(&'a TextureView) -> BindGroupEntries<'a, N>, dynamic_offsets: &[u32],
66    ) {
67        let cache = world.resource::<PipelineCache>();
68        let Some(pipeline) = cache.get_render_pipeline(self.pipeline) else {
69            return;
70        };
71        // Taking the next image swaps the view's images, so nothing may skip the draw from here.
72        let images = view.post_process_write();
73        let layout = cache.get_bind_group_layout(&self.layout);
74        let bind_group = render_context.render_device().create_bind_group(self.label, &layout, &bindings(images.source));
75        let load = if self.blends { LoadOp::Load } else { LoadOp::Clear(default()) };
76        let mut pass = render_context.begin_tracked_render_pass(RenderPassDescriptor {
77            label: Some(self.label),
78            color_attachments: &[Some(RenderPassColorAttachment {
79                view: images.destination,
80                resolve_target: None,
81                depth_slice: None,
82                ops: Operations { load, store: StoreOp::Store },
83            })],
84            depth_stencil_attachment: None,
85            timestamp_writes: None,
86            occlusion_query_set: None,
87            multiview_mask: None,
88        });
89        if let Some(viewport) = &camera.viewport {
90            pass.set_camera_viewport(viewport);
91        }
92        pass.set_render_pipeline(pipeline);
93        pass.set_bind_group(0, &bind_group, dynamic_offsets);
94        pass.draw(0..3, 0..1);
95    }
96}
