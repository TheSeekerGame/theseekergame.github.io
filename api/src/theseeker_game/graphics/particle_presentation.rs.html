1//! Gameplay owns emitter descriptors; Hanabi owns delayed render instances.
2use bevy::camera::visibility::RenderLayers;
3use bevy_hanabi::{EffectProperties, EffectSimulation, ParticleEffect};
4use theseeker_engine::presentation::{PresentationSet, TickHistory, interpolate_transform};
5use crate::prelude::*;

7#[derive(Component)]
8pub struct PresentedEffect(pub ParticleEffect);
9
10#[derive(Clone)]
11struct EmitterPose {
12    transform: Transform,
13    properties: EffectProperties,
14    layers: RenderLayers,
15    visible: bool,
16}
17
18struct EmitterTrack {
19    effect: ParticleEffect,
20    proxy: Option<Entity>,
21    history: TickHistory<EmitterPose>,
22    dead_at: Option<u64>,
23}
24
25#[derive(Resource, Default)]
26struct Emitters(HashMap<Entity, EmitterTrack>);
27
28pub struct ParticlePresentationPlugin;
29impl Plugin for ParticlePresentationPlugin {
30    fn build(&self, app: &mut App) {
31        app.init_resource::<Emitters>();
32        app.add_systems(GameTickPost, capture_emitters.in_set(PresentationSet::Capture));
33        app.add_systems(Update, sync_particle_time.in_set(GameTickSet::Post));
34        app.add_systems(PostUpdate, present_emitters.in_set(PresentationSet::Present));
35        app.add_systems(OnExit(AppState::InGame), clear_emitters);
36    }
37}

39fn sync_particle_time(
40    presentation: Res<PresentationTime>, mut effect: ResMut<Time<EffectSimulation>>,
41    mut clock: Local<Time<EffectSimulation>>,
42) {
43    clock.advance_to(presentation.elapsed());
44    *effect = *clock;
45}

47type EmitterQuery<'w, 's> = Query<'w, 's, (Entity, &'static PresentedEffect,
48    Option<&'static EffectProperties>, Option<&'static RenderLayers>, Option<&'static Visibility>)>;
49
50fn capture_emitters(
51    time: Res<GameTime>, query: EmitterQuery, transforms: TransformHelper,
52    mut emitters: ResMut<Emitters>,
53) {
54    for track in emitters.0.values_mut() {
55        if track.dead_at.is_none() { track.dead_at = Some(time.tick()); }
56    }
57    for (entity, effect, properties, layers, visibility) in &query {
58        let Ok(transform) = transforms.compute_global_transform(entity) else { continue; };
59        let track = emitters.0.entry(entity).or_insert_with(|| EmitterTrack {
60            effect: effect.0.clone(), proxy: None, history: TickHistory::default(), dead_at: None,
61        });
62        track.dead_at = None;
63        track.history.push(time.tick(), EmitterPose {
64            transform: transform.compute_transform(), properties: properties.cloned().unwrap_or_default(),
65            layers: layers.cloned().unwrap_or_default(), visible: visibility != Some(&Visibility::Hidden),
66        });
67    }
68}
69
70fn present_emitters(mut commands: Commands, cursor: Res<RenderInterpolation>, mut emitters: ResMut<Emitters>) {
71    emitters.0.retain(|_, track| {
72        if track.dead_at.is_none_or(|tick| cursor.position() < tick as f64) { return true; }
73        if let Some(proxy) = track.proxy { commands.entity(proxy).try_despawn(); }
74        false
75    });
76    for track in emitters.0.values_mut() {
77        let Some((previous, next, alpha)) = track.history.pair(cursor.position()) else { continue; };
78        let proxy = *track.proxy.get_or_insert_with(|| commands.spawn(track.effect.clone()).id());
79        commands.entity(proxy).insert((interpolate_transform(&previous.transform, &next.transform, alpha),
80            previous.properties.clone(), previous.layers.clone(),
81            if previous.visible { Visibility::Visible } else { Visibility::Hidden }));
82        track.history.discard_before(cursor.position());
83    }
84}
85
86fn clear_emitters(mut commands: Commands, mut emitters: ResMut<Emitters>) {
87    for (_, track) in emitters.0.drain() {
88        if let Some(proxy) = track.proxy { commands.entity(proxy).try_despawn(); }
89    }
90}

92#[cfg(test)]
93mod tests {
94    use super::*;
95    use theseeker_engine::time::{MasterClock, GameTickPre, run_gametickupdate_schedule};
96    #[test]
97    fn parent_death_removes_the_source_and_proxy_death_waits_for_the_presentation_cursor() {
98        let mut world = World::new();
99        world.init_resource::<Emitters>();
100        world.init_resource::<GameTime>();
101        world.insert_resource(RenderInterpolation::at(1.0));
102        let parent = world.spawn(Transform::from_xyz(10.0, 20.0, 0.0)).id();
103        let source = world.spawn((PresentedEffect(ParticleEffect::new(Handle::default())),
104            Transform::from_xyz(1.0, 2.0, 0.0), ChildOf(parent))).id();
105        world.add_schedule(Schedule::new(GameTickPre));
106        world.add_schedule(Schedule::new(GameTickUpdate));
107        let mut capture = Schedule::new(GameTickPost);
108        capture.add_systems(capture_emitters);
109        world.add_schedule(capture);
110        let mut present = Schedule::default();
111        present.add_systems(present_emitters);
112        for tick in 1..=2 {
113            world.resource_mut::<GameTime>().update_from_master(&MasterClock { tick, ..default() });
114            run_gametickupdate_schedule(&mut world);
115        }
116        present.run(&mut world);
117        let proxy = world.resource::<Emitters>().0[&source].proxy.unwrap();
118        assert_eq!(world.get::<Transform>(proxy).unwrap().translation, Vec3::new(11.0, 22.0, 0.0));
119        world.despawn(parent);
120        assert!(world.get_entity(source).is_err());
121        world.resource_mut::<GameTime>().update_from_master(&MasterClock { tick: 3, ..default() });
122        run_gametickupdate_schedule(&mut world);
123        world.insert_resource(RenderInterpolation::at(1.99));
124        present.run(&mut world);
125        assert!(world.get_entity(proxy).is_ok());
126        world.insert_resource(RenderInterpolation::at(2.0));
127        present.run(&mut world);
128        assert!(world.get_entity(proxy).is_err());
129        assert!(world.resource::<Emitters>().0.is_empty());
130    }
131}
