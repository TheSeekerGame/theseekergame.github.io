12    pub radius: f32,

17impl CircleParticles {
18    pub fn effect(self, capacity: u32, spawner: SpawnerSettings) -> EffectAsset {
19        let writer = ExprWriter::new();
20        let age = SetAttributeModifier::new(Attribute::AGE, writer.lit(0.0).expr());
21        let lifetime = SetAttributeModifier::new(Attribute::LIFETIME, writer.lit(self.lifetime).expr());
22        let position = SetPositionCircleModifier {
23            center: writer.lit(Vec3::ZERO).expr(), axis: writer.lit(Vec3::Z).expr(),
24            radius: writer.lit(self.radius).expr(), dimension: ShapeDimension::Surface,
25        };
26        let velocity = SetVelocityCircleModifier {
27            center: writer.lit(Vec3::ZERO).expr(), axis: writer.lit(Vec3::Z).expr(),
28            speed: writer.lit(self.speed).expr(),
29        };
30        let mut module = writer.finish();
31        let square_pixels = RoundModifier::constant(&mut module, 0.0);
32        EffectAsset::new(capacity, spawner, module)
33            .init(age).init(lifetime).init(position).init(velocity)
34            .render(SizeOverLifetimeModifier { gradient: Gradient::constant(Vec3::splat(self.size)), screen_space_size: false })
35            .render(ColorOverLifetimeModifier::new(self.colors)).render(square_pixels)
36    }
