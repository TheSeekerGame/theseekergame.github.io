8#[derive(Debug, Clone, Deserialize, TypePath)]
9#[serde(deny_unknown_fields)]
10pub struct MusicTrack {
11    pub audio_asset: String,
12    pub sample_rate: u32,
13    pub first_downbeat: u64,
14    pub loop_start: u64,
15    pub loop_end: u64,
16    pub gain: f32,
17    pub beats_per_bar: u32,
18}
19
20impl MusicTrack {
21    fn valid(&self) -> bool {
22        self.sample_rate > 0 && self.loop_start <= self.first_downbeat
23            && self.first_downbeat < self.loop_end && self.gain.is_finite() && self.gain >= 0.0
24            && self.beats_per_bar > 0 && self.loop_end > self.loop_start
25            && (u128::from(self.loop_end - self.loop_start) * 3).is_multiple_of(u128::from(self.sample_rate) * 2)
26    }
27}

80fn start_game_music(mut commands: MessageWriter<MusicCommand>) {
81    commands.write(MusicCommand::Play { track: "main".to_owned(), boundary: MusicBoundary::NextBeat });
82}
83fn stop_game_music(mut state: ResMut<MusicState>, mut engine: ResMut<AudioEngine>) {
84    state.pending = None;
85    state.current_track = None;
86    engine.set_command_time(AudioStartTime::Immediate);
87    if let Some(current) = state.current.take() { engine.stop_one(current.id); }
88    for fading in state.fading.drain(..) { engine.stop_one(fading.id); }

91#[derive(bevy::ecs::system::SystemParam)]
92struct MusicAssets<'w> {
93    library: Res<'w, MusicLibraryHandle>,
94    libraries: Res<'w, Assets<MusicLibrary>>,
95    preloaded: Res<'w, PreloadedAssets>,
96}
97
98fn music_controller(
99    mut commands: MessageReader<MusicCommand>, mut state: ResMut<MusicState>,
100    mut engine: ResMut<AudioEngine>, assets: MusicAssets, time: Res<GameTime>,
101) {
102    let tick = time.total_ticks();
103    for command in commands.read() { apply_music_command(command, &mut state, &mut engine, tick); }
104    recover_music(&mut state, &engine);
105    state.fading.retain(|music| engine.contains(music.id));
106    let Some(library) = assets.libraries.get(&assets.library.0) else { return; };
107    if state.current.is_none() { prepare_default_music(library, &assets.preloaded, &mut engine); }
108    try_start_music(&mut state, &mut engine, library, &assets.preloaded, tick);
109    let handles: Vec<_> = state.current.iter().chain(&state.fading).map(|music| music.handle.clone()).collect();
110    engine.retain_music(&handles);
111}

113fn prepare_default_music(library: &MusicLibrary, assets: &PreloadedAssets, engine: &mut AudioEngine) {
114    let Some(track) = library.tracks.get("main") else { return; };
115    let Some(handle) = music_handle(assets, track) else { return; };
116    engine.prepare(&handle);
117}

119fn music_handle(assets: &PreloadedAssets, track: &MusicTrack) -> Option<Handle<AudioSource>> {
120    assets.get_multi_asset(&track.audio_asset)?.first().map(|handle| handle.clone().typed::<AudioSource>())
121}

123fn boundary_tick(tick: u64, boundary: MusicBoundary, beats_per_bar: u32) -> u64 {
124    match boundary {
125        MusicBoundary::Immediate => tick,
126        MusicBoundary::NextBeat => next_boundary(tick, TICKS_PER_BEAT),
127        MusicBoundary::NextBar => next_boundary(tick, TICKS_PER_BEAT * u64::from(beats_per_bar)),
128    }
129}
130
131fn apply_music_command(command: &MusicCommand, state: &mut MusicState, engine: &mut AudioEngine, tick: u64) {
132    match command {
133        MusicCommand::Play { track, boundary } => state.pending = Some(PendingMusic { key: track.clone(), boundary: *boundary, fade_bars: 1, origin: None }),
134        MusicCommand::Transition { track, boundary, fade_bars } => state.pending = Some(PendingMusic { key: track.clone(), boundary: *boundary, fade_bars: *fade_bars, origin: None }),
135        MusicCommand::Stop { boundary } => stop_music(state, engine, tick, *boundary),
136        MusicCommand::SetVolume(volume) => set_music_volume(state, engine, *volume),
137    }
138}
139
140fn stop_music(state: &mut MusicState, engine: &mut AudioEngine, tick: u64, boundary: MusicBoundary) {
141    state.pending = None;
142    state.current_track = None;
143    let beats = state.current.as_ref().map_or(4, |music| music.track.beats_per_bar);
144    let at = boundary_tick(tick, boundary, beats);
145    if let Some(current) = state.current.take() { state.fading.push(current); }
146    for music in &state.fading {
147        engine.fade(music.id, 0.0, AudioStartTime::Tick(at), Duration::from_millis(10), true);
148    }
149}
150
151fn set_music_volume(state: &mut MusicState, engine: &mut AudioEngine, volume: f32) {
152    if !volume.is_finite() { return; }
153    state.volume = volume.clamp(0.0, 2.0);
154    engine.set_bus_volume(AudioBus::Music, state.volume);
155}
156
157fn recover_music(state: &mut MusicState, engine: &AudioEngine) {
158    if state.pending.is_some() || state.current.as_ref().is_none_or(|music| engine.contains(music.id)) { return; }
159    let Some(current) = state.current.take() else { return; };
160    state.pending = Some(PendingMusic { key: current.key, boundary: MusicBoundary::NextBeat, fade_bars: 0, origin: Some(current.origin) });
161}
162
163fn try_start_music(state: &mut MusicState, engine: &mut AudioEngine, library: &MusicLibrary, assets: &PreloadedAssets, tick: u64) {
164    let Some(pending) = &state.pending else { return; };
165    let Some(track) = library.tracks.get(&pending.key).filter(|track| track.valid()) else {
166        warn!("Music track '{}' is missing or has invalid timing metadata", pending.key);
167        state.pending = None;
168        return;
169    };
170    let Some(handle) = music_handle(assets, track) else { return; };
171    engine.prepare(&handle);
172    if engine.preparation_failed(&handle) { state.pending = None; return; }
173    if !engine.is_prepared(&handle) || !engine.running() { return; }
174    let at = boundary_tick(tick, pending.boundary, track.beats_per_bar);
175    let fade = Duration::from_secs_f64(f64::from(pending.fade_bars) * f64::from(track.beats_per_bar) * 2.0 / 3.0);
176    let origin = pending.origin.unwrap_or(at);
177    let config = music_playback(track, at, origin, fade);
178    engine.set_bus_volume(AudioBus::Music, state.volume);
179    let Some(id) = engine.play_music(&handle, config) else { return; };
180    if let Some(current) = state.current.take() {
181        engine.fade(current.id, 0.0, AudioStartTime::Tick(at), fade, true);
182        state.fading.push(current);
183    }
184    let key = pending.key.clone();
185    state.current_track = Some(key.clone());
186    state.current = Some(PlayingMusic { id, key, handle, track: track.clone(), origin });
187    state.pending = None;
188}
189
190fn music_playback(track: &MusicTrack, at: u64, origin: u64, fade: Duration) -> AudioPlaybackConfig {
191    let elapsed = at.saturating_sub(origin) as f64 / f64::from(theseeker_engine::audio::transport::TICK_RATE_HZ);
192    AudioPlaybackConfig {
193        timing: AudioStartTime::Tick(at), volume: track.gain, looped: true,
194        loop_region: Some((track.loop_start, track.loop_end, track.sample_rate)),
195        position: Duration::from_secs_f64(track.first_downbeat as f64 / f64::from(track.sample_rate) + elapsed),
196        fade_in: fade, ..Default::default()
197    }
198}

204    #[test]
205    fn configured_soundtrack_is_an_exact_480_beat_loop() {
206        let library: MusicLibrary = toml::from_str(include_str!("../../assets/audio/music/tracks.music.toml")).unwrap();
207        let track = &library.tracks["main"];
208        assert!(track.valid());
209        assert_eq!((track.loop_end - track.loop_start) * 3 / (u64::from(track.sample_rate) * 2), 480);
210        let recovered = music_playback(track, 64 + 96 * 321, 64, Duration::ZERO);
211        assert_eq!(recovered.position, Duration::from_secs(321));
212        assert_eq!(recovered.loop_region, Some((0, 15_360_000, 48_000)));
213        assert_eq!(recovered.volume, 0.04);
214    }

216    #[test]
217    fn beat_and_bar_requests_always_select_a_strictly_later_boundary() {
218        assert_eq!(boundary_tick(64, MusicBoundary::NextBeat, 4), 128);
219        assert_eq!(boundary_tick(255, MusicBoundary::NextBar, 4), 256);
220        assert_eq!(boundary_tick(256, MusicBoundary::NextBar, 4), 512);
221        assert_eq!(boundary_tick(192, MusicBoundary::NextBar, 3), 384);
222        assert_eq!(boundary_tick(65, MusicBoundary::Immediate, 4), 65);
