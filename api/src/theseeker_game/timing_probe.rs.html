21pub struct TimingProbePlugin;
22impl Plugin for TimingProbePlugin {
23    fn build(&self, app: &mut App) {
24        app.insert_resource(Probe { visible: std::env::var_os("SEEKER_TIMING_PROBE").is_some(), clicks: 0, frame_latency: crate::frame_metrics::DEFAULT_FRAME_LATENCY });
25        app.init_resource::<TimingLog>().add_plugins(crate::frame_metrics::FrameMetricsPlugin);
26        app.add_systems(Startup, (spawn_probe, open_timing_log));
27        app.add_systems(Update, (probe_controls, update_probe).chain().in_set(GameTickSet::Post));
28        app.add_systems(Last, write_timing_log.after(crate::frame_metrics::finish_frame));
29        app.add_systems(GameTickUpdate, gameplay_click);
30        app.add_systems(GameTickPost, player_presentation_override.in_set(theseeker_engine::presentation::PresentationSet::Prepare));
31    }
32}

34fn open_timing_log(mut log: ResMut<TimingLog>) {
35    let path = std::env::var_os("SEEKER_TIMING_LOG").map(std::path::PathBuf::from)
36        .unwrap_or_else(|| std::path::PathBuf::from("target/timing-diagnostics.log"));
37    match std::fs::File::create(&path) {
38        Ok(file) => {
39            let mut file = std::io::BufWriter::new(file);
40            let _ = writeln!(file, "The Seeker timing diagnostics — new session {:?}\n\
41                One row per second and on pause/resynchronization. Sample frames include all channels.\n\
42                click mode: 0=off, 1=callback reference, 2=gameplay. Callback gaps are suspected underruns, not confirmed device losses.", std::time::SystemTime::now());
43            let _ = file.flush();
44            log.file = Some(file);
45            info!("Timing diagnostics: {}", path.display());
46        },
47        Err(error) => warn!("Cannot create timing log {}: {error}", path.display()),
48    }
49}

51fn write_timing_log(
52    mut log: ResMut<TimingLog>, time: Res<Time<Real>>,
53    (probe, settings, music, mut metrics): (Res<Probe>, Res<TimingSettings>, Res<crate::music::MusicState>, ResMut<crate::frame_metrics::FrameMetrics>), engine: Res<AudioEngine>,
54    (transport, render_metrics): (Res<GameTransport>, Res<crate::frame_metrics::RenderMetrics>), cursor: Res<RenderInterpolation>, debug: Res<PresentationDebug>,
55) {
56    let changed = log.paused != transport.paused || log.resynchronizations != transport.resynchronizations;
57    if !changed && time.elapsed().saturating_sub(log.last) < Duration::from_secs(1) { return; }
58    if log.lines >= 86_400 { return; }
59    log.last = time.elapsed();
60    log.paused = transport.paused;
61    log.resynchronizations = transport.resynchronizations;
62    log.lines += 1;
63    let Some(file) = &mut log.file else { return; };
64    let stats = engine.diagnostics;
65    let render = render_metrics.take();
66    let result = writeln!(file, "t={:.3}s frame_ms={:.3} tick={:.3} paused={} clicks={} rate={} buffer={} lead_ms={:.3} latency_ms={:.3} healthy={} epoch={} resync={} late_plays={} late_controls={} max_late_frames={} queue_pressure={} rejected={} stolen={} callbacks={} callback_max_us={} gaps={} timestamp_fallbacks={} clipped={} history={} proxies={} pixels={} interpolate={} av_ms={:.1} display_ms={:.3} music={:?} music_volume={:.3} frame_latency={} panel={} fps={:.2} frame_avg_ms={:.3} frame_max_ms={:.3} main_avg_ms={:.3} main_max_ms={:.3} sim_avg_ms={:.3} sim_max_ms={:.3} sim_ticks={} focused_frames={}/{} window={}x{} render_frames={} render_prepare_ms={:.3} render_submit_ms={:.3} render_max_ms={:.3}",
67        time.elapsed_secs_f64(), time.delta_secs_f64() * 1000.0, cursor.position(), transport.paused, probe.clicks,
68        engine.sample_rate(), engine.block_frames(), transport.lead_seconds * 1000.0,
69        engine.clock().latency().as_secs_f64() * 1000.0, engine.clock().healthy(Instant::now()), transport.generation,
70        transport.resynchronizations, stats.late_plays, stats.late_controls, stats.max_lateness_frames,
71        engine.dropped_commands, stats.rejected_events, stats.stolen_voices, stats.callbacks,
72        stats.max_callback_micros, stats.delayed_callbacks, stats.timestamp_fallbacks, stats.clipped_samples,
73        debug.retained_samples, debug.proxies, debug.player_pixels.unwrap_or(true), !debug.interpolation_disabled,
74        settings.av_offset_ms, settings.display_latency_ms, music.current_track, music.volume, probe.frame_latency, probe.visible,
75        if metrics.frame_seconds > 0.0 { f64::from(metrics.frames) / metrics.frame_seconds } else { 0.0 },
76        metrics.frame_seconds * 1000.0 / f64::from(metrics.frames.max(1)), metrics.max_frame_seconds * 1000.0,
77        metrics.main_seconds * 1000.0 / f64::from(metrics.frames.max(1)), metrics.max_main_seconds * 1000.0,
78        metrics.simulation_seconds * 1000.0 / f64::from(metrics.frames.max(1)), metrics.max_simulation_seconds * 1000.0, metrics.ticks,
79        metrics.focused_frames, metrics.frames, metrics.window_size.x, metrics.window_size.y, render.frames,
80        render.prepare_seconds * 1000.0 / f64::from(render.frames.max(1)),
81        render.submit_seconds * 1000.0 / f64::from(render.frames.max(1)), render.max_seconds * 1000.0)
82        .and_then(|()| file.flush());
83    *metrics = crate::frame_metrics::FrameMetrics::default();
84    if let Err(error) = result { warn!("Timing log write failed: {error}"); log.file = None; }
85}

87fn spawn_probe(mut commands: Commands) {
88    commands.spawn((ProbeText, Text::default(), TextFont { font_size: FontSize::Px(14.0), ..Default::default() },
89        TextColor(Color::WHITE), BackgroundColor(Color::srgba(0.02, 0.02, 0.03, 0.92)),
90        Node { position_type: PositionType::Absolute, left: Val::Px(12.0), top: Val::Px(12.0), max_width: Val::Percent(96.0),
91            padding: UiRect::all(Val::Px(12.0)), ..Default::default() }, GlobalZIndex(10000)));

94fn probe_controls(
95    keys: Res<ButtonInput<KeyCode>>, mut probe: ResMut<Probe>, mut engine: ResMut<AudioEngine>,
96    mut debug: ResMut<PresentationDebug>, mut settings: ResMut<TimingSettings>,
97    mut windows: Query<&mut Window, With<bevy::window::PrimaryWindow>>,
98) {
99    if keys.just_pressed(KeyCode::F9) { probe.visible = !probe.visible; }
100    if keys.just_pressed(KeyCode::F8) {
101        probe.frame_latency = if probe.frame_latency == 1 { 2 } else { 1 };
102        if let Ok(mut window) = windows.single_mut() {
103            window.desired_maximum_frame_latency = std::num::NonZeroU32::new(probe.frame_latency);
104        }
105    }
106    if !probe.visible { return; }
107    if keys.just_pressed(KeyCode::F10) {
108        probe.clicks = (probe.clicks + 1) % 3;
109        engine.reference_clicks(probe.clicks == 1);
110    }
111    if keys.just_pressed(KeyCode::F11) { debug.player_pixels = Some(!debug.player_pixels.unwrap_or(true)); }
112    if keys.just_pressed(KeyCode::F12) { debug.interpolation_disabled = !debug.interpolation_disabled; }
113    if keys.just_pressed(KeyCode::BracketLeft) { settings.av_offset_ms = (settings.av_offset_ms - 1.0).max(-100.0); }
114    if keys.just_pressed(KeyCode::BracketRight) { settings.av_offset_ms = (settings.av_offset_ms + 1.0).min(100.0); }
115}

130fn update_probe(
131    probe: Res<Probe>, engine: Res<AudioEngine>, transport: Res<GameTransport>, cursor: Res<RenderInterpolation>,
132    (settings, mut last): (Res<TimingSettings>, Local<Option<ProbeDisplayKey>>), debug: Res<PresentationDebug>,
133    mut query: Query<(&mut Text, &mut Visibility, &mut BackgroundColor), With<ProbeText>>,
134) {
135    let Ok((mut text, mut visible, mut background)) = query.single_mut() else { return; };
136    *visible = if probe.visible { Visibility::Visible } else { Visibility::Hidden };
137    if !probe.visible { *last = None; return; }
138    let tick = cursor.tick();
139    let key = (tick / 8, probe.clicks, debug.player_pixels.unwrap_or(true), debug.interpolation_disabled, settings.av_offset_ms.to_bits(), probe.frame_latency);
140    if *last == Some(key) { return; }
141    *last = Some(key);
142    let stats = engine.diagnostics;
143    let mode = ["off", "callback reference", "gameplay scheduling"][usize::from(probe.clicks)];
144    background.0 = if (tick / 8).is_multiple_of(2) { Color::srgba(0.04, 0.08, 0.18, 0.95) }
145        else { Color::srgba(0.12, 0.04, 0.07, 0.95) };
146    text.0 = format!("TIMING PROBE   12 Hz frame {}   beat {} / 4   bar {}\n\
147        tick {}   clicks: {}   {} Hz / {} sample frames\n\
148        lead {:.2} ms   AV offset {:+.1} ms   output latency ~{:.2} ms\n\
149        late {} (plays {} / controls {}, max {:.2} ms)   queue retries/drops {}   rejected {}   stolen {}\n\
150        callback max {} us   gaps {}   timestamp fallbacks {}\n\
151        history {} samples / {} sprites   resync {}\n\
152        F9 panel   F10 clicks   F11 pixels ({})   F12 interpolation ({})\n\
153        [ / ] AV offset   P pause   F8 queued frames ({})",
154        tick / 8, tick / 64 % 4 + 1, tick / 256 + 1, tick, mode, engine.sample_rate(), engine.block_frames(),
155        transport.lead_seconds * 1000.0, settings.av_offset_ms, engine.clock().latency().as_secs_f64() * 1000.0,
156        stats.late_events, stats.late_plays, stats.late_controls, stats.max_lateness_frames as f64 * 1000.0 / f64::from(engine.sample_rate()), engine.dropped_commands, stats.rejected_events, stats.stolen_voices,
157        stats.max_callback_micros, stats.delayed_callbacks, stats.timestamp_fallbacks, debug.retained_samples, debug.proxies, transport.resynchronizations,
158        debug.player_pixels.unwrap_or(true), !debug.interpolation_disabled, probe.frame_latency);
159}
