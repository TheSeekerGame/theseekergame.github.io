31impl ResolvedActor {
32    /// Publication requires stats, so every resolved actor has them.
33    pub fn stats(&self) -> &ArchetypeStats {
34        self.archetype.stats.as_ref().expect("published actors have stats")

37    /// Structural body and lifecycle settings belong to the spawned actor.
38    pub fn adopt_actions(&self, latest: &Self) -> Self {
39        if self.body.controller != latest.body.controller { return self.clone(); }
40        let mut archetype = self.archetype.clone();
41        archetype.anim.clone_from(&latest.archetype.anim);
42        archetype.actions.clone_from(&latest.archetype.actions);
43        Self {
44            archetype, body: self.body.clone(), behavior: latest.behavior.clone(), walking: latest.walking.clone(),
45            actions: latest.actions.clone(), clips: latest.clips.clone(),
46        }
47    }
48}

50/// The body an actor spawns with: its authored setup with defaults applied.
51#[derive(Clone)]
52pub struct ActorBody {
53    pub name: String,
54    pub controller: EnemySetupController,
55    pub collider_half_extents: Vec2,
56    pub spawn_offset: Vec3,
57    pub facing: EnemyFacingSetup,
58    /// Playback slots for facing right and left.
59    pub facing_slots: Option<(String, String)>,
60    pub gfx: EnemyGfxKind,
61    pub gfx_offset: Vec3,
62    pub pixel_aligned: bool,
63    /// Animation bindings tried in order for the first clip.
64    pub animation_keys: Vec<String>,
65    pub needs_line_of_sight: bool,
66    /// Floors for the vision and melee ranges after per-spawn jitter.
67    pub min_vision_range: f32,
68    pub min_melee_range: f32,
69    pub trap_prey: bool,
70}

72impl ActorBody {
73    fn resolve(actor: &EnemyArchetype) -> Result<Self, String> {
74        let setup = actor.setup.as_ref().ok_or_else(|| format!("{}: missing body setup", actor.id))?;
75        let controller = setup.controller.ok_or_else(|| format!("{}: missing body controller", actor.id))?;
76        let collider = setup.collider.ok_or_else(|| format!("{}: missing body collider", actor.id))?;
77        if collider.half_extents.iter().any(|value| !value.is_finite() || *value <= 0.0) {
78            return Err(format!("{}: invalid body collider", actor.id));
79        }
80        let gfx = setup.gfx.clone().unwrap_or_default();
81        let minimums = setup.stat_minimums;
82        Ok(Self {
83            name: setup.name.clone().unwrap_or_else(|| actor.id.clone()),
84            controller,
85            collider_half_extents: Vec2::from_array(collider.half_extents),
86            spawn_offset: Vec3::from_array(setup.spawn_offset.unwrap_or_default()),
87            facing: setup.facing.unwrap_or(EnemyFacingSetup::Right),
88            facing_slots: setup.facing_slots.as_ref().map(|slots| (slots.right.clone(), slots.left.clone())),
89            gfx: gfx.kind.unwrap_or(EnemyGfxKind::PixelPerfect),
90            gfx_offset: gfx.offset.map_or(Vec3::ZERO, Vec3::from_array),
91            pixel_aligned: gfx.pixel_aligned.unwrap_or(true),
92            animation_keys: gfx.keys.unwrap_or_default(),
93            needs_line_of_sight: setup.needs_line_of_sight.unwrap_or(false),
94            min_vision_range: minimums.map_or(0.0, |minimums| minimums.vision_range),
95            min_melee_range: minimums.map_or(0.0, |minimums| minimums.melee_range),
96            trap_prey: setup.trap_prey.unwrap_or(false),
97        })
98    }
99}

108#[derive(SystemParam)]
109pub struct DefinitionAssets<'w> {
110    sources: Res<'w, Assets<AuthoredActor>>,
111    actions: Res<'w, Assets<ActionDefinition>>,
112    clips: Res<'w, Assets<SpriteAnimation>>,
113    projectiles: Res<'w, Assets<ProjectileDescriptor>>,
114    attacks: Res<'w, Assets<AttackDescriptor>>,
115}

117pub fn resolve_catalog(assets: DefinitionAssets, preloaded: Res<PreloadedAssets>, mut catalog: ResMut<ActorCatalog>) {
118    if !assets.changed() && catalog.generation != 0 {
119        return;
120    }
121    match assets.resolve(&preloaded) {
122        Ok(mut resolved) => {
123            resolved.generation = catalog.generation.wrapping_add(1);
124            *catalog = resolved;
125        },
126        Err(error) => error!("Actor definitions rejected: {error}"),
127    }
128}

130impl DefinitionAssets<'_> {
131    fn changed(&self) -> bool {
132        self.sources.is_changed() || self.actions.is_changed() || self.clips.is_changed()
133            || self.projectiles.is_changed() || self.attacks.is_changed()
134    }
135
136    pub(super) fn resolve(&self, preloaded: &PreloadedAssets) -> Result<ActorCatalog, String> {
137        let sources: HashMap<_, _> = self.sources.iter()
138            .map(|(_, source)| (source.0.id.clone(), source.0.clone())).collect();
139        let definitions: HashMap<_, _> = self.actions.iter()
140            .map(|(_, action)| (action.id.clone(), action.clone())).collect();
141        if sources.len() != self.sources.len() { return Err("duplicate actor id".to_owned()); }
142        if definitions.len() != self.actions.len() { return Err("duplicate action id".to_owned()); }
143        let mut catalog = ActorCatalog::default();
144        for (id, source) in &sources {
145            let actor = resolve_actor_source(source, &sources)?;
146            let resolved = self.resolve_actor(actor, &definitions, preloaded)?;
147            catalog.actors.insert(id.clone(), Arc::new(resolved));
148        }
149        for definition in definitions.values() {
150            self.resolve_standalone_action(definition, &definitions, preloaded, &mut catalog)?;
151        }
152        Ok(catalog)
153    }
154
155    fn resolve_standalone_action(
156        &self,
157        source: &ActionDefinition,
158        definitions: &HashMap<String, ActionDefinition>,
159        preloaded: &PreloadedAssets,
160        catalog: &mut ActorCatalog,
161    ) -> Result<(), String> {
162        let definition = resolve_action_source(source, definitions)?;
163        if definition.presentation.as_ref().is_some_and(|binding| !binding.animation.starts_with("anim.")) {
164            return Ok(());
165        }
166        let resolved = self.resolve_action(definition, None, preloaded)?;
167        catalog.actions.insert(source.id.clone(), Arc::new(resolved));
168        Ok(())
169    }
170
171    fn resolve_actor(
172        &self,
173        actor: EnemyArchetype,
174        definitions: &HashMap<String, ActionDefinition>,
175        preloaded: &PreloadedAssets,
176    ) -> Result<ResolvedActor, String> {
177        actor.validate().map_err(|error| error.to_string())?;
178        actor.validate_animations(preloaded, &actor.anim.values().cloned().collect::<Vec<_>>())
179            .map_err(|error| error.to_string())?;
180        let behavior = actor.behavior.clone().ok_or_else(|| format!("{}: missing behavior", actor.id))?;
181        let mut actions = HashMap::new();
182        for (slot, id) in &actor.actions {
183            let source = definitions.get(id).ok_or_else(|| format!("{}: missing action {id}", actor.id))?;
184            let definition = resolve_action_source(source, definitions)?;
185            let resolved = self.resolve_action(definition, Some(&actor), preloaded)?;
186            actions.insert(slot.clone(), Arc::new(resolved));
187        }
188        let body = ActorBody::resolve(&actor)?;
189        validate_actor_body(&actor, body.controller, &behavior, &actions)?;
190        let walking = self.resolve_walking(&actor, preloaded)?;
191        let clips = self.resolve_actor_clips(&actor, preloaded)?;
192        Ok(ResolvedActor { archetype: actor, body, behavior, walking, actions, clips })
193    }
194
195    fn resolve_actor_clips(&self, actor: &EnemyArchetype, preloaded: &PreloadedAssets) -> Result<HashMap<String, Arc<SpriteAnimation>>, String> {
196        let mut clips = HashMap::new();
197        for (slot, key) in &actor.anim {
198            let clip = preloaded.get_single_asset::<SpriteAnimation>(key).and_then(|handle| self.clips.get(&handle))
199                .ok_or_else(|| format!("{}: unloaded animation {key}", actor.id))?;
200            validate_clip_sections(&actor.id, None, clip)?;
201            validate_clip_cues(&actor.id, clip)?;
202            clips.insert(slot.clone(), Arc::new(clip.clone()));
203        }
204        Ok(clips)
205    }
206
207    fn resolve_walking(&self, actor: &EnemyArchetype, preloaded: &PreloadedAssets) -> Result<super::motion::ResolvedWalkMotion, String> {
208        use super::motion::{WalkMotion, ResolvedWalkMotion, FrameStep};
209        let Some(WalkMotion::FrameSteps { track, scale, steps }) = &actor.walking else { return Ok(ResolvedWalkMotion::Continuous); };
210        if !scale.is_finite() { return Err(format!("{}: walking scale must be finite", actor.id)); }
211        let key = actor.anim.get(track).unwrap_or(track);
212        let handle = preloaded.get_single_asset::<SpriteAnimation>(key)
213            .ok_or_else(|| format!("{}: missing movement track {key}", actor.id))?;
214        let clip = self.clips.get(&handle).ok_or_else(|| format!("{}: unloaded movement track {key}", actor.id))?;
215        let mut steps = steps.clone().unwrap_or_else(|| clip.frame_impulse.iter()
216            .map(|step| FrameStep { frame: step.frame, delta: [step.x, step.y] }).collect());
217        for step in &mut steps {
218            validate_movement_step(step, *scale, &actor.id)?;
219        }
220        Ok(ResolvedWalkMotion::FrameSteps(steps))
221    }

223    fn resolve_action(
224        &self,
225        source: ActionDefinition,
226        actor: Option<&EnemyArchetype>,
227        preloaded: &PreloadedAssets,
228    ) -> Result<ResolvedAction, String> {
229        let id = source.id;
230        let mechanic = source.mechanic.ok_or_else(|| format!("{id}: missing mechanic"))?;
231        let mut presentation = source.presentation;
232        let clip = self.resolve_clip(&id, &mut presentation, actor, preloaded)?;
233        let mut timing = source.timing.unwrap_or_default();
234        if let Mechanic::TentacleStrike { phases } = &mechanic {
235            phases.validate().map_err(|error| format!("{id}: {error}"))?;
236            timing.duration_ticks = Some(phases.duration());
237        }
238        validate_action_timing(&id, &timing, clip.as_deref())?;
239        let projectile = resolve_projectile(&mechanic, preloaded, &self.projectiles)?;
240        let projectile_clips = self.resolve_projectile_clips(&id, projectile.as_ref(), preloaded)?;
241        let attack = resolve_attack(&mechanic, preloaded, &self.attacks)?;
242        if source.damage.is_some_and(|damage| !damage.is_finite() || damage < 0.0) {
243            return Err(format!("{id}: damage must be finite and nonnegative"));
244        }
245        if source.motion.is_some_and(|motion| !motion.speed.is_finite() || motion.speed < 0.0) {
246            return Err(format!("{id}: motion speed must be finite and nonnegative"));
247        }
248        let motion = source.motion.or_else(|| resolve_lunge_motion(&mechanic, actor));
249        let damage = source.damage.or_else(|| resolve_action_damage(&mechanic, actor));
250        validate_action_motion(&id, &mechanic, motion, source.hits)?;
251        Ok(ResolvedAction {
252            cues: (&timing).into(),
253            id, mechanic, presentation, clip, timing, projectile, projectile_clips, attack,
254            damage, motion, controls: source.controls.unwrap_or_default(),
255            hits: source.hits,
256        })
257    }

259    fn resolve_projectile_clips(&self, id: &str, projectile: Option<&ProjectileDescriptor>, preloaded: &PreloadedAssets) -> Result<Option<super::action::ProjectileClips>, String> {
260        let Some(sprite) = projectile.and_then(|projectile| projectile.sprite.as_ref()) else { return Ok(None); };
261        let resolve = |key: &str| -> Result<Arc<SpriteAnimation>, String> {
262            let clip = preloaded.get_single_asset::<SpriteAnimation>(key).and_then(|handle| self.clips.get(&handle))
263                .ok_or_else(|| format!("{id}: missing projectile animation {key}"))?;
264            validate_clip_sections(id, None, clip)?;
265            validate_clip_cues(id, clip)?;
266            Ok(Arc::new(clip.clone()))
267        };
268        Ok(Some(super::action::ProjectileClips { idle: resolve(&sprite.idle_anim_key)?, pop: resolve(&sprite.pop_anim_key)? }))
269    }
270
271    fn resolve_clip(
272        &self,
273        id: &str,
274        binding: &mut Option<super::action::PresentationBinding>,
275        actor: Option<&EnemyArchetype>,
276        preloaded: &PreloadedAssets,
277    ) -> Result<Option<Arc<SpriteAnimation>>, String> {
278        let Some(binding) = binding else { return Ok(None); };
279        if !binding.animation.starts_with("anim.") {
280            binding.animation = actor.and_then(|actor| actor.anim.get(&binding.animation)).cloned()
281                .ok_or_else(|| format!("{id}: missing animation binding {}", binding.animation))?;
282        }
283        let handle = preloaded.get_single_asset::<SpriteAnimation>(&binding.animation)
284            .ok_or_else(|| format!("{id}: missing animation {}", binding.animation))?;
285        let clip = self.clips.get(&handle).ok_or_else(|| format!("{id}: animation is not loaded"))?;
286        if binding.offset.iter().any(|value| !value.is_finite()) { return Err(format!("{id}: presentation offset must be finite")); }
287        validate_clip_cues(id, clip)?;
288        validate_clip_sections(id, binding.section.as_deref(), clip)?;
289        validate_clip_sections(id, binding.impact_section.as_deref(), clip)?;
290        Ok(Some(Arc::new(clip.clone())))
291    }
292}

294fn resolve_actor_source(source: &EnemyArchetype, sources: &HashMap<String, EnemyArchetype>) -> Result<EnemyArchetype, String> {
295    let mut actor = source.clone();
296    if let Some(base) = &source.base {
297        let parent = sources.get(base).ok_or_else(|| format!("{}: missing base {base}", source.id))?;
298        if parent.base.is_some() {
299            return Err(format!("{}: actor variants require a concrete base", source.id));
300        }
301        actor.merge_with_base(parent);
302    }
303    actor.base = None;
304    Ok(actor)
305}

307pub fn resolve_action_source(source: &ActionDefinition, sources: &HashMap<String, ActionDefinition>) -> Result<ActionDefinition, String> {
308    let Some(base) = &source.base else { return Ok(source.clone()); };
309    let parent = sources.get(base).ok_or_else(|| format!("{}: missing base action {base}", source.id))?;
310    if parent.base.is_some() {
311        return Err(format!("{}: action variants require a concrete base", source.id));
312    }
313    Ok(ActionDefinition {
314        id: source.id.clone(), base: None,
315        mechanic: source.mechanic.clone().or_else(|| parent.mechanic.clone()),
316        presentation: source.presentation.clone().or_else(|| parent.presentation.clone()),
317        timing: source.timing.clone().or_else(|| parent.timing.clone()),
318        damage: source.damage.or(parent.damage), motion: source.motion.or(parent.motion),
319        controls: source.controls.or(parent.controls), hits: source.hits.or(parent.hits),
320    })
321}

323fn validate_clip_sections(id: &str, section: Option<&str>, clip: &SpriteAnimation) -> Result<(), String> {
324    let settings = &clip.settings.extended;
325    if settings.ticks_per_frame == 0 || settings.frame_min.0 == 0 || settings.frame_min > settings.frame_max || settings.frame_start < settings.frame_min || settings.frame_start > settings.frame_max { return Err(format!("{id}: invalid clip frame range or tick duration")); }
326    for section in clip.playback.iter().chain(clip.sections.values()) {
327        section.validate().map_err(|error| format!("{id}: {error}"))?;
328        if section.end > settings.frame_max || section.start < settings.frame_min {
329            return Err(format!("{id}: clip section lies outside the animation frame range"));
330        }
331    }
332    if section.is_some_and(|name| !clip.sections.contains_key(name)) {
333        return Err(format!("{id}: unknown clip section {section:?}"));
334    }
335    Ok(())
336}

338fn validate_action_timing(id: &str, timing: &super::action::ActionTiming, clip: Option<&SpriteAnimation>) -> Result<(), String> {
339    for cue in [&timing.release, &timing.cancel, &timing.complete].into_iter().flatten() {
340        let Some(clip) = clip else { return Err(format!("{id}: cue {cue} requires a clip")); };
341        if !clip.events.iter().any(|event| &event.emit == cue) && !clip.frame_bookmarks.contains_key(cue) {
342            return Err(format!("{id}: missing cue {cue}"));
343        }
344    }
345    if timing.duration_ticks == Some(0) {
346        return Err(format!("{id}: action duration must be positive"));
347    }
348    Ok(())
349}

369fn validate_movement_step(step: &mut super::motion::FrameStep, scale: f32, actor: &str) -> Result<(), String> {
370    if step.frame.0 == 0 || step.delta.iter().any(|value| !value.is_finite()) {
371        return Err(format!("{actor}: invalid frame step at {}", step.frame.0));
372    }
373    step.delta = [step.delta[0] * scale, step.delta[1] * scale];
374    Ok(())
375}

377fn resolve_lunge_motion(mechanic: &Mechanic, actor: Option<&EnemyArchetype>) -> Option<super::action::LungeMotion> {
378    if !matches!(mechanic, Mechanic::Lunge) { return None; }
379    let actor = actor?;
380    let speed = actor.stats.as_ref()?.chase_speed;
381    let multiplier = actor.flying.and_then(|flying| flying.attack).and_then(|attack| attack.speed_multiplier).unwrap_or(2.0);
382    Some(super::action::LungeMotion { speed: speed * multiplier, direction: None, impact_ticks: None })
383}

385fn resolve_action_damage(mechanic: &Mechanic, actor: Option<&EnemyArchetype>) -> Option<f32> {
386    let stats = actor?.stats.as_ref()?;
387    Some(match mechanic { Mechanic::Projectile { .. } => stats.dmg_ranged as f32, _ => stats.dmg_melee as f32 })
388}

390fn validate_action_motion(id: &str, mechanic: &Mechanic, motion: Option<super::action::LungeMotion>, hits: Option<super::lunge::LungeHits>) -> Result<(), String> {
391    if matches!(mechanic, Mechanic::Lunge) && motion.is_none() { return Err(format!("{id}: lunge requires motion")); }
392    if let Some(motion) = motion {
393        if !motion.speed.is_finite() || motion.speed < 0.0 || motion.impact_ticks == Some(0)
394            || motion.direction.is_some_and(|direction| direction.iter().any(|value| !value.is_finite()) || direction == [0.0, 0.0])
395        { return Err(format!("{id}: invalid lunge motion")); }
396    }
397    if let Some(super::lunge::LungeHits::Sweep { radius, impact_radius, impact_max_targets }) = hits {
398        if !radius.is_finite() || radius <= 0.0 || !impact_radius.is_finite() || impact_radius < 0.0 || impact_max_targets == 0 {
399            return Err(format!("{id}: invalid sweep hit policy"));
400        }
401    }
402    Ok(())
403}

405fn validate_clip_cues(id: &str, clip: &SpriteAnimation) -> Result<(), String> {
406    let max = clip.settings.extended.frame_max;
407    for event in &clip.events {
408        let on = &event.on;
409        let count = usize::from(on.frame_enter.is_some()) + usize::from(on.frame_exit.is_some())
410            + usize::from(on.bookmark.is_some()) + usize::from(on.anim_finished);
411        let invalid_frame = on.frame_enter.into_iter().chain(on.frame_exit).any(|frame| frame.0 == 0 || frame > max);
412        if event.emit.is_empty() || count != 1 || invalid_frame || on.bookmark.as_ref().is_some_and(|name| !clip.frame_bookmarks.contains_key(name)) {
413            return Err(format!("{id}: invalid cue {}", event.emit));
414        }
415    }
416    Ok(())
417}

419fn validate_actor_body(
420    actor: &EnemyArchetype,
421    body: EnemySetupController,
422    behavior: &ControllerSpec,
423    actions: &HashMap<String, Arc<ResolvedAction>>,
424) -> Result<(), String> {
425    let compatible = match behavior {
426        ControllerSpec::Combat { approach: super::controller::Approach::Flying, .. } | ControllerSpec::Pursuit => body == EnemySetupController::Flying,
427        ControllerSpec::Combat { .. } => body == EnemySetupController::Ground,
428        ControllerSpec::Hopper => body == EnemySetupController::Jump,
429        ControllerSpec::Ambush => body == EnemySetupController::StationaryTrap,
430        ControllerSpec::Wander => matches!(body, EnemySetupController::Ground | EnemySetupController::Flying),
431    };
432    if !compatible { return Err(format!("{}: controller is incompatible with body {body:?}", actor.id)); }
433    validate_controller_bindings(actor, behavior, body)?;
434    for (slot, action) in actions {
435        validate_action_body(&actor.id, slot, &action.mechanic, body)?;
436    }
437    Ok(())
438}

440fn validate_controller_bindings(actor: &EnemyArchetype, behavior: &ControllerSpec, body: EnemySetupController) -> Result<(), String> {
441    use super::controller::Approach;
442    let mut clips = vec!["Idle"];
443    let mut actions = Vec::new();
444    match behavior {
445        ControllerSpec::Combat { approach, defensive, .. } => {
446            actions.push("Attack");
447            if *defensive { actions.push("Defense"); }
448            clips.extend(match approach { Approach::Ground => Some("Chase"), Approach::Roll => Some("Roll"), Approach::Flying => Some("Fly"), Approach::Stationary => None });
449        },
450        ControllerSpec::Hopper => { actions.extend(["Jump", "AirJump"]); clips.push("Fall"); },
451        ControllerSpec::Ambush => { actions.extend(["AttackLeft", "AttackRight"]); clips.extend(["Prepare", "EatLeft", "EatRight"]); },
452        ControllerSpec::Wander => {},
453        ControllerSpec::Pursuit => { clips.push("Fly"); },
454    }
455    if body == EnemySetupController::Ground { clips.push("Walk"); }
456    for slot in clips {
457        if !actor.anim.contains_key(slot) { return Err(format!("{}: controller requires animation binding {slot}", actor.id)); }
458    }
459    for slot in actions {
460        if !actor.actions.contains_key(slot) { return Err(format!("{}: controller requires action binding {slot}", actor.id)); }
461    }
462    Ok(())
463}

465fn validate_action_body(id: &str, slot: &str, mechanic: &Mechanic, body: EnemySetupController) -> Result<(), String> {
466    let compatible = match mechanic {
467        Mechanic::Lunge => body == EnemySetupController::Flying,
468        Mechanic::Jump => body == EnemySetupController::Jump,
469        Mechanic::TrapStrike => body == EnemySetupController::StationaryTrap,
470        Mechanic::TentacleStrike { .. } => false,
471        Mechanic::Projectile { .. } | Mechanic::Contact { .. } | Mechanic::Guard => true,
472    };
473    if compatible { Ok(()) } else { Err(format!("{id}: action {slot} is incompatible with body {body:?}")) }
474}
