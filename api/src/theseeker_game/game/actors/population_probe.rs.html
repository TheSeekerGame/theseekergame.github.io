19/// Presence disables the ordinary population director for a controlled workload.
20#[derive(Resource)]
21pub struct PopulationProbe {
22    requested: usize,
23    seeded: bool,
24    warmup: u32,
25    tick_start: Instant,
26    sample_ticks: u32,
27    sample_time: Duration,
28    samples: Vec<f64>,
29    minimum_actors: usize,
30    maximum_actors: usize,
31    minimum_active: usize,
32    maximum_active: usize,
33}
34
35impl Plugin for PopulationProbePlugin {
36    fn build(&self, app: &mut App) {
37        if !cfg!(debug_assertions) { return; }
38        let count = std::env::args().find_map(|arg| arg.strip_prefix("--actor-benchmark=").and_then(|count| count.parse::<usize>().ok()));
39        let Some(count @ (500 | 1_000)) = count else { return; };
40        app.insert_resource(PopulationProbe {
41            requested: count, seeded: false, warmup: 192, tick_start: Instant::now(),
42            sample_ticks: 0, sample_time: Duration::ZERO, samples: Vec::new(),
43            minimum_actors: usize::MAX, maximum_actors: 0,
44            minimum_active: usize::MAX, maximum_active: 0,
45        });
46        app.add_systems(GameTickPre, begin_tick);
47        app.add_systems(GameTickUpdate, seed_population.before(EnemyStateSet::Sensors).run_if(in_state(AppState::InGame)));
48        app.add_systems(GameTickUpdate, protect_population.before(EnemyStateSet::Actuator));
49        app.add_systems(GameTickPost, finish_tick);
50    }
51}

53fn protect_population(
54    mut commands: Commands,
55    mut actors: Query<(Entity, Option<&mut Invulnerable>), (Or<(With<Player>, With<Enemy>)>, Without<ProbeProtected>)>,
56) {
57    for (entity, protection) in &mut actors {
58        if let Some(mut protection) = protection {
59            protection.add_source(InvulnerabilitySource::PopulationProbe);
60        } else {
61            commands.entity(entity).insert(Invulnerable::from_source(InvulnerabilitySource::PopulationProbe));
62        }
63        commands.entity(entity).insert(ProbeProtected);
64    }
65}

67fn begin_tick(mut probe: ResMut<PopulationProbe>) {
68    probe.tick_start = Instant::now();
69}
70
71fn seed_population(
72    mut commands: Commands,
73    mut probe: ResMut<PopulationProbe>,
74    catalog: Res<ActorCatalog>,
75    players: Query<(Entity, &Transform), With<Player>>,
76    enemies: Query<(Entity, Option<&Gent>), With<Enemy>>,
77) {
78    if probe.seeded { return; }
79    let Some((player, transform)) = players.iter().next() else { return; };
80    if catalog.actors.is_empty() { return; }
81    for (entity, gent) in &enemies {
82        commands.entity(entity).try_despawn();
83        if let Some(gent) = gent {
84            commands.entity(gent.e_gfx).try_despawn();
85            commands.entity(gent.e_effects_gfx).try_despawn();
86        }
87    }
88    commands.entity(player).insert(Health { current: u32::MAX, max: u32::MAX });
89    let mut ids: Vec<_> = catalog.actors.keys().collect();
90    ids.sort();
91    for index in 0..probe.requested {
92        let offset = Vec3::new((index % 20) as f32 * 12.0 - 120.0, (index / 20) as f32 * 8.0 + 16.0, 0.0);
93        spawn_enemy_actor_blueprint(&mut commands, ids[index % ids.len()],
94            Transform::from_translation(transform.translation + offset));
95    }
96    probe.seeded = true;
97}

99fn finish_tick(mut probe: ResMut<PopulationProbe>, enemies: Query<Has<crate::game::gentstate::Dead>, With<Enemy>>, mut exit: MessageWriter<AppExit>) {
100    if !probe.seeded { return; }
101    if probe.warmup > 0 { probe.warmup -= 1; return; }
102    let duration = probe.tick_start.elapsed();
103    let count = enemies.iter().len();
104    let active = enemies.iter().filter(|dead| !*dead).count();
105    probe.minimum_actors = probe.minimum_actors.min(count);
106    probe.maximum_actors = probe.maximum_actors.max(count);
107    probe.minimum_active = probe.minimum_active.min(active);
108    probe.maximum_active = probe.maximum_active.max(active);
109    probe.sample_time += duration;
110    probe.sample_ticks += 1;
111    if probe.sample_ticks < 96 { return; }
112    let mean_us = probe.sample_time.as_secs_f64() * 1_000_000.0 / 96.0;
113    probe.samples.push(mean_us);
114    probe.sample_time = Duration::ZERO;
115    probe.sample_ticks = 0;
116    println!("Actor population probe: requested={} sample={} mean_us={mean_us:.3} actors={}..{}",
117        probe.requested, probe.samples.len(), probe.minimum_actors, probe.maximum_actors);
118    if probe.samples.len() < 7 { return; }
119    probe.samples.sort_by(f64::total_cmp);
120    println!("ACTOR_POPULATION_RESULT requested={} entities_min={} entities_max={} active_min={} active_max={} samples=7 ticks_per_sample=96 median_us={:.3} min_us={:.3} max_us={:.3}",
121        probe.requested, probe.minimum_actors, probe.maximum_actors, probe.minimum_active, probe.maximum_active,
122        probe.samples[3], probe.samples[0], probe.samples[6]);
123    exit.write(AppExit::Success);
124}
