76/// An item the capsule gives the player.
77#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
78pub(crate) enum PlayerItemGrant {
79    BowBundle { arrows: u32 },
80    Arrows { amount: u32 },
81    SkillStock { skill: SkillId, material: CraftMaterial, amount: u32 },
82    Passive(Passive),
83}
84
85impl PlayerItemGrant {
86    /// Whether the player already owns this one-time item; stock can always be granted again.
87    pub(crate) fn is_owned(self, passives: &Passives, weapons: &WeaponInventory) -> bool {
88        match self {
89            Self::BowBundle { .. } => weapons.has_ranged(PlayerRangedWeapon::Bow),
90            Self::Passive(passive) => passives.is_unlocked(passive),
91            Self::Arrows { .. } | Self::SkillStock { .. } => false,
92        }
93    }
94
95    /// Adds the item, unlocking and equipping the skills it provides.
96    pub(crate) fn grant(
97        self,
98        materials: &mut MaterialInventory,
99        skills: &mut SkillInventory,
100        passives: &mut Passives,
101        weapons: &mut WeaponInventory,
102    ) {
103        let max_slots = SkillInventory::max_slots_for(passives);
104        match self {
105            Self::BowBundle { arrows } => {
106                weapons.grant_bow_bundle(arrows);
107                skills.unlock_skill_with_capacity(SkillId::BowAttack, max_slots);
108            },
109            Self::Arrows { amount } => weapons.grant_arrows(amount),
110            Self::SkillStock { skill, material, amount } => {
111                materials.add_to_backpack(material, amount);
112                grant_skill(skills, skill, max_slots);
113            },
114            Self::Passive(passive) => passives.add_passive(passive),
115        }
116    }
117}
118
119/// Unlocks `skill` and equips it when a slot is free.
120fn grant_skill(skills: &mut SkillInventory, skill: SkillId, max_slots: usize) {
121    if !skills.is_unlocked(&skill) {
122        skills.unlock_skill(skill);
123    }
124    if !skills.is_equipped(&skill) {
125        skills.auto_equip_skill(skill, max_slots);
126    }
127}

340#[derive(Debug, Clone, Copy, PartialEq, Eq)]
341pub(crate) enum CapsuleActionResult {
342    Completed,
343    MissingEssence,
344    MissingIngredients,
345    NotDiscovered,
346    AlreadyOwned,
347    NothingToDiscover,
348}

350pub(crate) fn recipe(id: CapsuleRecipeId) -> &'static CapsuleRecipe {
351    CAPSULE_RECIPES
352        .iter()
353        .find(|recipe| recipe.id == id)
354        .expect("all CapsuleRecipeId values have definitions")
355}
356
357pub(crate) fn discoverable_recipes(
358    knowledge: &CraftingKnowledge,
359    materials: &MaterialInventory,
360) -> Vec<CapsuleRecipeId> {
361    CAPSULE_RECIPES
362        .iter()
363        .filter(|recipe| {
364            !knowledge.is_discovered(recipe.id)
365                && recipe.ingredients.iter().all(|ingredient| {
366                    materials.total_count(ingredient.material) > 0
367                })
368        })
369        .map(|recipe| recipe.id)
370        .collect()
371}

373pub(crate) fn research_recipes(
374    knowledge: &mut CraftingKnowledge,
375    materials: &MaterialInventory,
376    essence: &mut Essence,
377) -> CapsuleActionResult {
378    let discoverable = discoverable_recipes(knowledge, materials);
379    if discoverable.is_empty() {
380        return CapsuleActionResult::NothingToDiscover;
381    }
382    if !essence.try_spend(CAPSULE_ACTION_ESSENCE_COST) {
383        return CapsuleActionResult::MissingEssence;
384    }
385    knowledge
386        .discovered
387        .extend(discoverable);
388    CapsuleActionResult::Completed
389}

391pub(crate) fn purchase_tether(
392    essence: &mut Essence,
393    materials: &mut MaterialInventory,
394    skills: &mut SkillInventory,
395    passives: &mut Passives,
396    weapons: &mut WeaponInventory,
397) -> CapsuleActionResult {
398    if !essence.try_spend(CAPSULE_ACTION_ESSENCE_COST) {
399        return CapsuleActionResult::MissingEssence;
400    }
401    recipe(CapsuleRecipeId::TetherNode).output.grant(materials, skills, passives, weapons);
402    CapsuleActionResult::Completed
403}

405pub(crate) fn craft_recipe(
406    id: CapsuleRecipeId,
407    knowledge: &CraftingKnowledge,
408    materials: &mut MaterialInventory,
409    skills: &mut SkillInventory,
410    passives: &mut Passives,
411    weapons: &mut WeaponInventory,
412) -> CapsuleActionResult {
413    if !knowledge.is_discovered(id) {
414        return CapsuleActionResult::NotDiscovered;
415    }
416    let definition = recipe(id);
417    if definition.output.is_owned(passives, weapons) {
418        return CapsuleActionResult::AlreadyOwned;
419    }
420    if !materials.try_pay(definition.ingredients.iter().map(MaterialCost::pair)) {
421        return CapsuleActionResult::MissingIngredients;
422    }
423    definition.output.grant(materials, skills, passives, weapons);
424    CapsuleActionResult::Completed
425}
