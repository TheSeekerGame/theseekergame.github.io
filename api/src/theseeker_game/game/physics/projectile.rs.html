10/// Gravity for projectiles in px/tick^2 at 96Hz (independent of player fall accel)
11const PROJECTILE_GRAVITY: f32 = 4.5;
12
13/// Component for entities with ballistic physics (initial velocity in px/s)
14#[derive(Component, Debug)]
15pub struct Projectile {
16    pub vel: LinearVelocity,
17}
18
19impl Projectile {
20    /// Create a projectile with a ballistic trajectory toward the target.
21    /// Gravity is in px/s² (e.g., 432 px/s² = 4.5 px/tick² at 96 Hz)
22    pub fn with_vel(
23        target: Vec2,
24        start: Vec2,
25        max_speed: f32,
26        gravity: f32,
27    ) -> Option<Self> {
28        let result = solve_ballistic_arc(start, max_speed, target, gravity);
29        if result.2 == 0 {
30            return None;
31        }
32
33        Some(Self {
34            vel: LinearVelocity(higher_ballistic_arc(result.0, result.1)),
35        })
36    }
37}
38
39fn higher_ballistic_arc(first: Vec2, second: Vec2) -> Vec2 {
40    if first.y > second.y {
41        first
42    } else {
43        second
44    }
45}

74/// Applies gravity to projectiles (except player arrows and grounded projectiles)
75pub fn arc_projectile(
76    mut query: Query<
77        (
78            &mut Transform,
79            &mut Projectile,
80            Option<&mut PreviousPosition>,
81            Has<Arrow>,
82        ),
83        (
84            With<DamageSource>,
85            Without<GroundedProjectile>,
86        ),
87    >,
88    time: Res<theseeker_engine::time::GameTime>,
89) {
90    for (mut transform, mut projectile, prev_pos, arrow) in query.iter_mut() {
91        if let Some(mut prev_pos) = prev_pos {
92            prev_pos.0 = transform.translation.xy();
93        }
94
95        if !arrow {
96            projectile.vel.0.y -= PROJECTILE_GRAVITY;
97        }
98        let z = transform.translation.z;
99        transform.translation = (transform.translation.xy()
100            + projectile.vel.0 * (1.0 / time.hz as f32))
101            .extend(z);
102    }
103}

105/// Continuous collision detection for arrows to prevent tunneling
106pub fn arrow_continuous_collision_detection(
107    mut arrow_query: Query<
108        (
109            Entity,
110            &Transform,
111            &PreviousPosition,
112            &mut DamageSource,
113            &Collider,
114            Has<WallPiercingProjectile>,
115        ),
116        With<Arrow>,
117    >,
118    spatial_query: PhysicsWorld,
119    collision_world: Option<Res<TileCollisionWorld>>,
120    mut commands: Commands,
121) {
122    for (
123        arrow_entity,
124        arrow_transform,
125        prev_pos,
126        mut attack,
127        arrow_collider,
128        wall_piercing,
129    ) in arrow_query.iter_mut()
130    {
131        let current_pos = arrow_transform.translation.xy();
132        let previous_pos = prev_pos.0;
133
134        if current_pos.distance_squared(previous_pos) < 1.0 {
135            continue;
136        }
137
138        let direction = current_pos - previous_pos;
139        let max_distance = direction.length();
140        if max_distance < 0.001 {
141            continue;
142        }
143        let normalized_dir = direction / max_distance;
144
145        let wall_hit = if wall_piercing {
146            None
147        } else {
148            ray_cast_ready_tile_world(
149                collision_world.as_deref(),
150                previous_pos,
151                normalized_dir,
152                max_distance,
153            )
154        };
155
156        let enemy_hit = spatial_query.cast_shape(ShapeCast {
157            origin: previous_pos,
158            dir: normalized_dir,
159            shape: arrow_collider,
160            max_toi: max_distance,
161            interaction: CollisionGroups::new(PLAYER_ATTACK, ENEMY),
162            exclude: Some(arrow_entity),
163        }, |_| true);
164
165        let wall_distance = wall_hit
166            .map(|hit| hit.time_of_impact)
167            .unwrap_or(f32::MAX);
168
169        if record_arrow_enemy_hit_before_wall(
170            &mut attack,
171            enemy_hit,
172            wall_distance,
173        ) {
174            continue;
175        }
176
177        if !wall_piercing && wall_distance <= max_distance {
178            commands.entity(arrow_entity).try_despawn();
179        }
180    }
181}

183pub fn despawn_projectile(
184    query: Query<(Entity, &DamageSource), (With<Projectile>, With<Hit>)>,
185    mut commands: Commands,
186) {
187    for (entity, damage_source) in query.iter() {
188        if damage_source.damaged_set.len() == damage_source.max_targets as usize
189        {
190            commands.entity(entity).try_despawn();
191        }
192    }
193}

195/// Detects ground collision for arc projectiles and stops them.
196/// Uses surface normals to distinguish ground (upward-facing) from ceilings/walls.
197/// A surface is considered "ground" if its normal.y > 0.7 (roughly horizontal, facing up).
198pub fn arc_projectile_ground_collision(
199    mut query: Query<
200        (Entity, &Transform, &mut Projectile),
201        (
202            With<GroundDespawnProjectile>,
203            Without<GroundedProjectile>,
204        ),
205    >,
206    collision_world: Option<Res<TileCollisionWorld>>,
207    mut commands: Commands,
208) {
209    // Threshold for "upward-facing" surface (cos(45°) ≈ 0.707)
210    // Surfaces tilted less than ~45° from horizontal are considered ground-like.
211    const GROUND_NORMAL_THRESHOLD: f32 = 0.7;
212
213    for (entity, transform, mut projectile) in query.iter_mut() {
214        // Only check when falling - optimization to avoid unnecessary raycasts when rising.
215        // The normal check is the authoritative ground detection; velocity is just a filter.
216        if projectile.vel.0.y >= 0.0 {
217            continue;
218        }
219
220        // Cast distance proportional to fall speed to prevent tunneling.
221        let fall_speed = -projectile.vel.0.y;
222        let check_distance = (fall_speed / 96.0).max(2.0);
223
224        let hit = ray_cast_ready_tile_world(
225            collision_world.as_deref(),
226            transform.translation.xy(),
227            Vec2::NEG_Y,
228            check_distance,
229        );
230
231        // If we hit ground (upward-facing surface), stop the projectile and start despawn timer.
232        let Some(intersection) = hit else {
233            continue;
234        };
235        if intersection.normal.y <= GROUND_NORMAL_THRESHOLD {
236            continue;
237        }
238
239        ground_projectile(entity, &mut projectile, &mut commands);
240    }
241}

243fn ray_cast_ready_tile_world(
244    collision_world: Option<&TileCollisionWorld>,
245    origin: Vec2,
246    direction: Vec2,
247    max_distance: f32,
248) -> Option<CastHit> {
249    let world = collision_world?;
250    if !world.is_ready() {
251        return None;
252    }
253
254    world.ray_cast_solid(origin, direction, max_distance)
255}

257fn record_arrow_enemy_hit_before_wall(
258    attack: &mut DamageSource,
259    enemy_hit: Option<(Entity, CastHit)>,
260    wall_distance: f32,
261) -> bool {
262    let Some((hit_entity, hit_info)) = enemy_hit else {
263        return false;
264    };
265    if hit_info.time_of_impact >= wall_distance {
266        return false;
267    }
268
269    if damage_source_has_target_capacity(attack) {
270        attack.target_set.insert(hit_entity);
271    }
272    true
273}
274
275fn damage_source_has_target_capacity(attack: &DamageSource) -> bool {
276    let total_targets = attack.damaged_set.len() + attack.target_set.len();
277    total_targets < attack.max_targets as usize
278}

280fn ground_projectile(
281    entity: Entity,
282    projectile: &mut Projectile,
283    commands: &mut Commands,
284) {
285    projectile.vel.0 = Vec2::ZERO;
286    commands.entity(entity).insert(GroundedProjectile {
287        ticks_remaining: GroundedProjectile::DESPAWN_DELAY_TICKS,
288    });
289}

291/// Counts down grounded projectiles and despawns them when the timer expires.
292pub fn grounded_projectile_despawn(
293    mut query: Query<(Entity, &mut GroundedProjectile)>,
294    mut commands: Commands,
295) {
296    for (entity, mut grounded) in query.iter_mut() {
297        if grounded.ticks_remaining == 0 {
298            commands.entity(entity).try_despawn();
299        } else {
300            grounded.ticks_remaining -= 1;
301        }
302    }
