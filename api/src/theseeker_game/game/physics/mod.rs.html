62/// Knockback applied to a gent. Initial velocity is set once, then horizontal movement is blocked until expiry.
63#[derive(Component, Default, Clone, Copy, Debug)]
64pub struct Knockback {
65    pub ticks: u32,
66    pub max_ticks: u32,
67    pub strength: Vec2,
68}
69
70impl Knockback {
71    pub fn new(strength: Vec2, max_ticks: u32) -> Self {
72        Self {
73            ticks: 0,
74            max_ticks,
75            strength,
76        }
77    }
78}
79
80pub fn knockback(
81    mut query: Query<
82        (Entity, &mut Knockback, &mut LinearVelocity, Has<Defense>),
83        Without<Frozen>,
84    >,
85    mut commands: Commands,
86) {
87    for (entity, mut knockback, mut velocity, is_defending) in query.iter_mut() {
88        knockback.ticks += 1;
89        if knockback.is_added() && !is_defending {
90            velocity.0 = knockback.strength;
91        }
92        if knockback.ticks > knockback.max_ticks {
93            velocity.0.x = 0.;
94            commands.entity(entity).remove::<Knockback>();
95        }
96    }
97}
