8pub struct YakPlugin;

10impl Plugin for YakPlugin {
11    fn build(&self, app: &mut App) {
12        app.add_systems(
13            GameTickUpdate,
14            (setup_yak.run_if(in_state(GameState::Playing)))
15                .run_if(in_state(AppState::InGame)),
16        );
17    }
18}

44pub fn setup_yak(
45    mut q: Query<(&mut Transform, Entity), Added<YakBlueprint>>,
46    mut commands: Commands,
47) {
48    for (mut xf_gent, e_gent) in q.iter_mut() {
49        xf_gent.translation.z = 12.0 * 0.000_001;
50        xf_gent.translation.y += 5.0;
51        let e_gfx = commands.spawn(()).id();
52        commands
53            .entity(e_gent)
54            .insert(Name::new("Yak"))
55            .remove::<ChildOf>();
56        let mut player = ScriptPlayer::<SpriteAnimation>::default();
57        player.play_key("anim.yak.Idle");
58        commands.entity(e_gfx).insert((
59            YakGfxBundle {
60                marker: YakGfx,
61                gent2gfx: TransformGfxFromGent {
62                    pixel_aligned: false,
63                    gent: e_gent,
64                    offset: None,
65                },
66                sprite: Sprite {
67                    texture_atlas: Some(TextureAtlas::default()),
68                    ..default()
69                },
70                transform: *xf_gent,
71                global_transform: GlobalTransform::default(),
72                visibility: Visibility::Visible,
73                inherited_visibility: InheritedVisibility::VISIBLE,
74                view_visibility: ViewVisibility::default(),
75                animation: SpriteAnimationBundle { player },
76            },
77            StateDespawnMarker,
78        ));
79    }
