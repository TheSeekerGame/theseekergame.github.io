14pub struct ElevatorStatePlugin;

16const ELEVATOR_ANIMATION_KEY: &str = "anim.player.Elevator";
17const PLAYER_WORLD_RENDER_LAYER: usize = 0;
18const PLAYER_OVERLAY_RENDER_LAYER: usize = 2;
19
20impl Plugin for ElevatorStatePlugin {
21    fn build(&self, app: &mut App) {
22        app.add_systems(
23            GameTickUpdate,
24            elevator_update_system
25                .in_set(crate::game::player::PlayerBehaviorSet::Execution),
26        );
27        app.add_systems(
28            Update,
29            sync_elevator_render_layers.in_set(GameTickSet::Post),
30        );
31    }
32}

34fn elevator_update_system(
35    mut query: Query<
36        (ActionMut<Elevator>, &mut LinearVelocity, &mut Transform, &Gent),
37        With<Player>,
38    >,
39    capsule_q: Query<&Transform, (With<Capsule>, Without<Player>)>,
40    mut gfx_query: Query<&mut ScriptPlayer<SpriteAnimation>, With<PlayerGfx>>,
41) {
42    for (mut elevator, mut velocity, mut transform, gent) in query.iter_mut() {
43        velocity.0 = elevator.velocity();
44        if let Ok(capsule_transform) = capsule_q.get(elevator.shelter()) {
45            transform.translation.x = capsule_transform.translation.x;
46        }
47
48        play_elevator_animation(gent, &mut gfx_query, elevator.execution_mut());
49    }
50}
51
52fn play_elevator_animation(
53    gent: &Gent,
54    gfx_query: &mut Query<&mut ScriptPlayer<SpriteAnimation>, With<PlayerGfx>>,
55    execution: &mut crate::game::player::foreground::ForegroundAction,
56) {
57    let Ok(mut animation) = gfx_query.get_mut(gent.e_gfx) else {
58        return;
59    };
60
61    if execution.playback == theseeker_engine::animation::PlaybackId::default()
62        || animation.current_key() != Some(ELEVATOR_ANIMATION_KEY) {
63        execution.play(&mut animation, ELEVATOR_ANIMATION_KEY);
64    }
65}
66
67fn sync_elevator_render_layers(
68    player_query: Query<(&Gent, Has<ActionTag<Elevator>>), With<Player>>,
69    mut gfx_query: Query<&mut RenderLayers, With<PlayerGfx>>,
70) {
71    let world_layer = RenderLayers::layer(PLAYER_WORLD_RENDER_LAYER);
72    let overlay_layer = RenderLayers::layer(PLAYER_OVERLAY_RENDER_LAYER);
73
74    for (gent, is_elevator) in player_query.iter() {
75        let Ok(mut render_layers) = gfx_query.get_mut(gent.e_gfx) else {
76            continue;
77        };
78        let target_layers = if is_elevator {
79            &world_layer
80        } else {
81            &overlay_layer
82        };
83
84        if *render_layers != *target_layers {
85            *render_layers = target_layers.clone();
86        }
87    }
