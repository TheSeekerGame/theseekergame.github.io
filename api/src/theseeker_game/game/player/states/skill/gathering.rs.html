14pub const GATHERING_TICKS: u32 = 36;
15const GATHERING_ANIMATION_KEY: &str = "anim.player.Gather";
16
17pub struct GatheringStatePlugin;
18
19impl Plugin for GatheringStatePlugin {
20    fn build(&self, app: &mut App) {
21        app.add_systems(
22            GameTickUpdate,
23            (gathering_enter_system, gathering_update_system)
24                .chain()
25                .in_set(PlayerStateSet::Behavior)
26                .in_set(crate::game::player::PlayerBehaviorSet::Execution),
27        );
28    }
29}
30
31fn gathering_enter_system(
32    mut query: Query<
33        (
34            ActionMut<Gathering>,
35            &mut LinearVelocity,
36            &crate::game::gentstate::Facing,
37            &Gent,
38        ),
39        (With<Player>, Added<ActionTag<Gathering>>),
40    >,
41    mut gfx_query: Query<&mut ScriptPlayer<SpriteAnimation>, With<PlayerGfx>>,
42) {
43    for (mut gathering, mut velocity, facing, gent) in query.iter_mut() {
44        velocity.0 = Vec2::ZERO;
45
46        if let Ok(mut script_player) = gfx_query.get_mut(gent.e_gfx) {
47            set_direction_slots(&mut script_player, facing);
48            gathering.execution_mut().play(&mut script_player, GATHERING_ANIMATION_KEY);
49        }
50    }
51}
52
53fn gathering_update_system(
54    mut commands: Commands,
55    mut query: Query<
56        (
57            Entity,
58            ActionMut<Gathering>,
59            &mut LinearVelocity,
60            &GroundSensor,
61        ),
62        With<Player>,
63    >,
64) {
65    for (entity, mut gathering, mut velocity, ground_sensor) in query.iter_mut()
66    {
67        if !ground_sensor.is_grounded {
68            gathering.finish(&mut commands, entity);
69            continue;
70        }
71
72        velocity.0 = Vec2::ZERO;
73        gathering.tick = gathering.tick.saturating_add(1);
74        if gathering.tick >= GATHERING_TICKS {
75            gathering.finish(&mut commands, entity);
76        }
77    }
78}
