1//! Shared velocity evolution; pickup, damage, and target ownership stay with each effect.
2use crate::prelude::*;
3
4#[derive(Clone, Copy)]
5pub struct ScatterHomingProfile {
6    pub ticks: u32,
7    pub damping_start: f32,
8    pub damping_end: f32,
9    pub max_speed: f32,
10    pub steering: f32,
11}
12#[derive(Debug, Clone, Copy)]
13pub struct ScatterHoming {
14    ticks_left: u32,
15    peak_speed: f32,
16}
17impl ScatterHoming {
18    pub fn new(profile: ScatterHomingProfile) -> Self {
19        Self { ticks_left: profile.ticks, peak_speed: 0.0 }
20    }
21    /// None after scattering; Some marks whether this step completed it.
22    pub fn scatter(&mut self, velocity: &mut Vec2, profile: ScatterHomingProfile) -> Option<bool> {
23        if self.ticks_left == 0 { return None; }
24        let t = 1.0 - self.ticks_left as f32 / profile.ticks as f32;
25        *velocity *= profile.damping_start + (profile.damping_end - profile.damping_start) * t;
26        self.ticks_left -= 1;
27        Some(self.ticks_left == 0)
28    }
29    pub fn steer(&mut self, velocity: &mut Vec2, direction: Vec2, distance: f32, profile: ScatterHomingProfile) {
30        self.peak_speed = self.peak_speed.max((distance * 0.9).min(profile.max_speed));
31        let target_velocity = direction * self.peak_speed;
32        let t = 1.0 - self.ticks_left as f32 / profile.ticks as f32;
33        let homing_scale = t * t * (3.0 - 2.0 * t);
34        *velocity = velocity.lerp(target_velocity, profile.steering * homing_scale);
35    }
36}
