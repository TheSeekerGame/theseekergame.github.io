26impl Plugin for SensorPlugin {

28        app.add_systems(
29            GameTickUpdate,

31                update_ground_sensor,

34                update_enemy_proximity_sensor,

93pub(crate) struct RecoveryCollisionWorld<'w, 's> {
94    terrain: Option<Res<'w, TileCollisionWorld>>,
95    interiors: Option<Res<'w, TentInteriorAssets>>,
96    capsules: CapsuleCollisionQuery<'w, 's, (With<Capsule>, Without<Player>)>,
97    spatial: PhysicsWorld<'w>,
98    spikes: Query<'w, 's, (), With<crate::game::spikes::SpikeHazard>>,
99}

101impl RecoveryCollisionWorld<'_, '_> {
102    pub fn capsule_occupant(
103        &self,
104        position: Vec2,
105        occupant: Option<&TentInteriorOccupant>,
106    ) -> Option<crate::game::world_objects::CapsuleOccupant> {
107        if occupant.is_some() { return None; }
108        crate::game::world_objects::capsule_occupant_at_world_position(position, &self.capsules)
109    }
110
111    pub fn is_safe(
112        &self,
113        position: Vec2,
114        collider: &Collider,
115        occupant: Option<&TentInteriorOccupant>,
116        platforms: &CapsuleDropPlatformState,
117    ) -> bool {
118        let Some(half) = collider.cuboid_half_extents() else { return false; };
119        let Some(view) = PlayerCollisionView::new(occupant, self.interiors.as_deref(),
120            self.terrain.as_deref(), &self.capsules) else { return false; };
121        view.distance(ProbeDirection::Down, position, half,
122            MAX_GROUND_CHECK_DISTANCE, Some(platforms)) <= GROUND_CONTACT_EPS
123            && !view.overlaps_solid(position, half)
124            && !self.spatial.intersect(position, collider,
125                theseeker_engine::physics::groups::player_body(), None)
126                .into_iter().any(|entity| self.spikes.contains(entity))
127    }
128
129    /// Resolve a safe supported fallback near an invalidated anchor or initial spawn.
130    pub fn supported_destination(
131        &self,
132        reference: Vec2,
133        collider: &Collider,
134        occupant: Option<&TentInteriorOccupant>,
135        platforms: &CapsuleDropPlatformState,
136    ) -> Option<Vec2> {
137        if self.is_safe(reference, collider, occupant, platforms) { return Some(reference); }
138        let half = collider.cuboid_half_extents()?;
139        let view = PlayerCollisionView::new(occupant, self.interiors.as_deref(),
140            self.terrain.as_deref(), &self.capsules)?;
141        let tile_size = view.tile_size();
142        let reach = crate::levelgen::mapgen::CHUNK_IMAGE_SIZE as f32 * tile_size;
143        let radius = crate::levelgen::mapgen::CHUNK_IMAGE_SIZE as i32;
144        (-radius..=radius).filter_map(|x| {
145            let start = reference + Vec2::new(x as f32 * tile_size, half.y * 2.0);
146            let distance = view.distance(ProbeDirection::Down, start, half, reach, Some(platforms));
147            let candidate = start - Vec2::Y * distance;
148            (distance < reach && self.is_safe(candidate, collider, occupant, platforms)).then_some(candidate)
149        }).min_by(|a, b| a.distance_squared(reference).total_cmp(&b.distance_squared(reference)))
150    }
151}

153fn record_safe_grounded_position(
154    mut players: Query<(Entity, &Transform, &Collider, Option<&TentInteriorOccupant>,
155        &CapsuleDropPlatformState, &super::foreground::ForegroundAction,
156        Option<&mut SafeGroundedPosition>),
157        (With<Player>, Without<Dead>, Without<PendingTentEntry>,
158            Without<crate::game::spikes::SpikeRespawnPending>)>,
159    world: RecoveryCollisionWorld,
160    mut commands: Commands,
161) {
162    for (entity, transform, collider, occupant, platforms, foreground, checkpoint) in &mut players {
163        if foreground.owns_motion() { continue; }
164        let position = transform.translation.truncate();
165        if !world.is_safe(position, collider, occupant, platforms) { continue; }
166        let safe = SafeGroundedPosition { position, space: occupant.map(|occupant| occupant.kind) };
167        if let Some(mut checkpoint) = checkpoint { *checkpoint = safe; }
168        else { commands.entity(entity).insert(safe); }
169    }
170}

172const WALL_CAST_DISTANCE: f32 = 8.0;
173const CEILING_CAST_DISTANCE: f32 = 8.0;
174const MELEE_RANGE: f32 = 32.0;
175const RANGED_RANGE: f32 = 200.0;
176const MAX_GROUND_CHECK_DISTANCE: f32 = 50.0; // Cap for distance when no hit is found
177const GROUND_CONTACT_EPS: f32 = 0.01;
178
179fn update_ground_sensor(
180    mut query: Query<
181        (
182            Entity,
183            &Transform,
184            &mut GroundSensor,
185            &Collider,
186            Option<&mut LastGroundedPosition>,
187            Has<crate::game::player::states::InAir>,
188            Option<&TentInteriorOccupant>,
189            &CapsuleDropPlatformState,
190        ),
191        With<Player>,
192    >,
193    collision_world: Option<Res<TileCollisionWorld>>,
194    tent_interior: Option<Res<TentInteriorAssets>>,
195    capsule_q: CapsuleCollisionQuery<With<Capsule>>,
196    _time: Res<GameTime>,
197    mut commands: Commands,
198) {
199    for (
200        entity,
201        transform,
202        mut sensor,
203        collider,
204        mut last_grounded_pos,
205        is_in_air,
206        interior_occupant,
207        drop_platform_state,
208    ) in query.iter_mut()
209    {
210        let Some(view) = PlayerCollisionView::new(interior_occupant,
211            tent_interior.as_deref(), collision_world.as_deref(), &capsule_q) else {
212            sensor.is_grounded = false;
213            sensor.distance = MAX_GROUND_CHECK_DISTANCE;
214            continue;
215        };
216
217        let Some(half_extents) = collider.cuboid_half_extents() else {
218            sensor.is_grounded = false;
219            sensor.distance = MAX_GROUND_CHECK_DISTANCE;
220            continue;
221        };
222        let distance = view.distance(
223            ProbeDirection::Down,
224            transform.translation.truncate(),
225            half_extents,
226            MAX_GROUND_CHECK_DISTANCE,
227            Some(drop_platform_state),
228        );
229        sensor.distance = distance;
230        sensor.is_grounded = distance <= GROUND_CONTACT_EPS;
231
232        if !is_in_air {
233            update_last_grounded_position(
234                &mut commands,
235                entity,
236                transform.translation.truncate(),
237                last_grounded_pos.as_deref_mut(),
238            );
239        }
240    }
241}
242
243fn update_last_grounded_position(
244    commands: &mut Commands,
245    entity: Entity,
246    position: Vec2,
247    last_grounded_pos: Option<&mut LastGroundedPosition>,
248) {
249    let Some(last_pos) = last_grounded_pos else {
250        commands.entity(entity).insert(LastGroundedPosition { position });
251        return;
252    };
253
254    last_pos.position = position;
255}

258fn update_wall_sensor(
259    mut query: Query<
260        (
261            &Transform,
262            &mut WallSensor,
263            &Collider,
264            Option<&TentInteriorOccupant>,
265        ),
266        With<Player>,
267    >,
268    collision_world: Option<Res<TileCollisionWorld>>,
269    tent_interior: Option<Res<TentInteriorAssets>>,
270    capsule_q: CapsuleCollisionQuery<With<Capsule>>,
271) {
272    for (transform, mut sensor, collider, interior_occupant) in query.iter_mut() {
273        let Some(view) = PlayerCollisionView::new(interior_occupant,
274            tent_interior.as_deref(), collision_world.as_deref(), &capsule_q) else {
275            sensor.left_contact = false;
276            sensor.right_contact = false;
277            sensor.left_distance = WALL_CAST_DISTANCE;
278            sensor.right_distance = WALL_CAST_DISTANCE;
279            continue;
280        };
281
282        let Some(half_extents) = collider.cuboid_half_extents() else {
283            sensor.left_contact = false;
284            sensor.right_contact = false;
285            sensor.left_distance = WALL_CAST_DISTANCE;
286            sensor.right_distance = WALL_CAST_DISTANCE;
287            continue;
288        };
289
290        let left_distance = view.distance(
291            ProbeDirection::Left,
292            transform.translation.truncate(),
293            half_extents,
294            WALL_CAST_DISTANCE,
295            None,
296        );
297        sensor.left_distance = left_distance;
298        sensor.left_contact = left_distance <= half_extents.x + 0.05;
299
300        let right_distance = view.distance(
301            ProbeDirection::Right,
302            transform.translation.truncate(),
303            half_extents,
304            WALL_CAST_DISTANCE,
305            None,
306        );
307        sensor.right_distance = right_distance;
308        sensor.right_contact = right_distance <= half_extents.x + 0.05;
309    }
310}

312fn update_ceiling_sensor(
313    mut query: Query<
314        (
315            &Transform,
316            &mut CeilingSensor,
317            &Collider,
318            Option<&TentInteriorOccupant>,
319        ),
320        With<Player>,
321    >,
322    collision_world: Option<Res<TileCollisionWorld>>,
323    tent_interior: Option<Res<TentInteriorAssets>>,
324    capsule_q: CapsuleCollisionQuery<With<Capsule>>,
325) {
326    for (transform, mut sensor, collider, interior_occupant) in query.iter_mut() {
327        let Some(view) = PlayerCollisionView::new(interior_occupant,
328            tent_interior.as_deref(), collision_world.as_deref(), &capsule_q) else {
329            sensor.is_touching_ceiling = false;
330            sensor.distance = CEILING_CAST_DISTANCE;
331            continue;
332        };
333
334        let Some(half_extents) = collider.cuboid_half_extents() else {
335            sensor.is_touching_ceiling = false;
336            sensor.distance = CEILING_CAST_DISTANCE;
337            continue;
338        };
339
340        let distance = view.distance(
341            ProbeDirection::Up,
342            transform.translation.truncate(),
343            half_extents,
344            CEILING_CAST_DISTANCE,
345            None,
346        );
347        sensor.distance = distance;
348        sensor.is_touching_ceiling = distance <= half_extents.y + 2.0;
349    }
350}

352fn update_enemy_proximity_sensor(
353    mut query: Query<
354        (
355            Entity,
356            &Transform,
357            &Facing,
358            &Collider,
359            &mut EnemyProximitySensor,
360        ),
361        With<Player>,
362    >,
363    enemy_query: Query<(Entity, &Transform), (With<Enemy>, Without<Dead>)>,
364    spatial_query: PhysicsWorld,
365) {
366    for (
367        player_entity,
368        player_transform,
369        facing,
370        player_collider,
371        mut sensor,
372    ) in query.iter_mut()
373    {
374        let player_pos = player_transform.translation.truncate();
375
376        // Reset previous frame's data
377        sensor.enemies_in_melee_range.clear();
378        sensor.enemies_in_ranged_range.clear();
379        sensor.closest_enemy = None;
380        sensor.closest_distance = f32::MAX;
381        sensor.has_forward_enemy = false;
382        sensor.has_enemy_below = false;
383        sensor.has_blocking_enemy_ahead = false;
384
385        // Check all enemies
386        for (enemy_entity, enemy_transform) in enemy_query.iter() {
387            let enemy_pos = enemy_transform.translation.truncate();
388            record_enemy_sensor_target(
389                &mut sensor,
390                enemy_entity,
391                player_pos,
392                enemy_pos,
393                facing,
394            );
395        }
396
397        let sweep_shape = Collider::cuboid(60.0, 1.0);
398        sensor.has_enemy_below = spatial_query
399            .cast_shape(ShapeCast {
400                origin: player_pos,
401                dir: Vec2::NEG_Y,
402                shape: &sweep_shape,
403                max_toi: 180.0,
404                interaction: CollisionGroups::new(PLAYER, ENEMY),
405                exclude: Some(player_entity),
406            }, |_| true)
407            .is_some();
408
409        let dir = facing.direction();
410        let cast_dir = if dir >= 0.0 { Vec2::X } else { Vec2::NEG_X };
411        sensor.has_blocking_enemy_ahead = spatial_query
412            .cast_shape(ShapeCast {
413                origin: player_pos,
414                dir: cast_dir,
415                shape: player_collider,
416                max_toi: 2.0,
417                interaction: CollisionGroups::new(PLAYER, ENEMY),
418                exclude: Some(player_entity),
419            }, |_| true)
420            .is_some();
421    }
422}

424fn record_enemy_sensor_target(
425    sensor: &mut EnemyProximitySensor,
426    enemy_entity: Entity,
427    player_pos: Vec2,
428    enemy_pos: Vec2,
429    facing: &Facing,
430) {
431    let distance = player_pos.distance(enemy_pos);
432    if distance < sensor.closest_distance {
433        sensor.closest_enemy = Some(enemy_entity);
434        sensor.closest_distance = distance;
435    }
436
437    if distance <= MELEE_RANGE {
438        sensor.enemies_in_melee_range.push(enemy_entity);
439    }
440    if distance <= RANGED_RANGE {
441        sensor.enemies_in_ranged_range.push(enemy_entity);
442    }
443
444    sensor.has_forward_enemy |=
445        enemy_in_forward_sensor_window(player_pos, enemy_pos, facing);
446}
447
448fn enemy_in_forward_sensor_window(
449    player_pos: Vec2,
450    enemy_pos: Vec2,
451    facing: &Facing,
452) -> bool {
453    let forward_len = 60.0;
454    let half_width = 10.0;
455    let dir = facing.direction();
456    let start_x = player_pos.x;
457    let end_x = player_pos.x + forward_len * dir;
458    let within_x = if dir > 0.0 {
459        enemy_pos.x >= start_x && enemy_pos.x <= end_x
460    } else {
461        enemy_pos.x <= start_x && enemy_pos.x >= end_x
462    };
463    let within_y = (enemy_pos.y - player_pos.y).abs() <= half_width;
464    within_x && within_y
465}

467/// Reset jump count when player touches a wall - enables wall jumping
468fn reset_jump_count_on_wall_touch(
469    mut query: Query<
470        (
471            &mut JumpCount,
472            &WallSensor,
473            &Collider,
474            Option<&mut Falling>,
475            Option<&mut Jumping>,
476        ),
477        With<Player>,
478    >,
479) {
480    for (mut jump_count, wall_sensor, collider, mut falling, mut jumping) in
481        query.iter_mut()
482    {
483        // Player half-width for accurate distance calculation
484        let half_width = if let Some(cuboid_half) = collider.cuboid_half_extents() {
485            cuboid_half.x
486        } else {
487            2.0 // Default fallback
488        };
489
490        // Forgiving wall detection within 2 px of the wall
491        const WALL_FORGIVENESS_DISTANCE: f32 = 2.0;
492        let near_left_wall =
493            wall_sensor.left_distance <= half_width + WALL_FORGIVENESS_DISTANCE;
494        let near_right_wall = wall_sensor.right_distance
495            <= half_width + WALL_FORGIVENESS_DISTANCE;
496        let is_near_wall = near_left_wall || near_right_wall;
497
498        // Reset jump count near a wall in Falling or Jumping states
499        if !is_near_wall {
500            continue;
501        }
502
503        reset_wall_touch_jump_count(
504            &mut jump_count,
505            falling.as_deref_mut(),
506            jumping.as_deref_mut(),
507        );
508    }
509}
510
511fn reset_wall_touch_jump_count(
512    jump_count: &mut JumpCount,
513    falling: Option<&mut Falling>,
514    jumping: Option<&mut Jumping>,
515) {
516    if let Some(falling_state) = falling {
517        // Treat wall jumps as fresh jumps for forgiveness.
518        falling_state.jump_count = 0;
519        jump_count.reset();
520    }
521
522    if let Some(jumping_state) = jumping {
523        // Maintain consistency when transitioning from jump to wall.
524        jumping_state.jump_count = 0;
525        jump_count.reset();
526    }
527}

529#[cfg(test)]
530mod recovery_tests {
531    use super::*;
532    use crate::game::player::foreground::start_action;
533    use theseeker_engine::gent::Gent;
534    use theseeker_engine::physics::{EntityCollisionWorld, Group, StaticCollider,
535        groups, rebuild_entity_collision_world};
536
537    #[test]
538    fn recovery_anchor_requires_support_and_survives_air_and_activated_spikes() {
539        let mut world = World::new();
540        world.insert_resource(GameTime::default());
541        world.insert_resource(EntityCollisionWorld::default());
542        let mut terrain = vec![false; 32 * 16];
543        terrain[32 * 15..].fill(true);
544        let mut tiles = TileCollisionWorld::default();
545        tiles.rebuild(32, 16, 1.0, &terrain);
546        world.insert_resource(tiles);
547        let gfx = world.spawn_empty().id();
548        let player = world.spawn((Player, Gent { e_gfx: gfx, e_effects_gfx: gfx },
549            Transform::from_xyz(6.0, 5.0, 0.0), Collider::cuboid(1.0, 2.0),
550            CapsuleDropPlatformState::default())).id();
551        start_action(&mut world, player, crate::game::player::Ready);
552        let mut record = Schedule::default();
553        record.add_systems(record_safe_grounded_position);
554        record.run(&mut world);
555        assert!(world.get::<SafeGroundedPosition>(player).is_none());
556        world.get_mut::<Transform>(player).unwrap().translation.y = 3.0;
557        record.run(&mut world);
558        assert_eq!(world.get::<SafeGroundedPosition>(player).unwrap().position, Vec2::new(6.0, 3.0));
559        world.get_mut::<Transform>(player).unwrap().translation.y = 5.0;
560        record.run(&mut world);
561        assert_eq!(world.get::<SafeGroundedPosition>(player).unwrap().position, Vec2::new(6.0, 3.0));
562        let spike = world.spawn((crate::game::spikes::SpikeHazard, StaticCollider,
563            GlobalTransform::from_xyz(12.0, 1.5, 0.0), Collider::cuboid(0.5, 0.5),
564            CollisionGroups::new(Group::NONE, Group::NONE))).id();
565        let mut index = Schedule::default();
566        index.add_systems(rebuild_entity_collision_world);
567        index.run(&mut world);
568        world.get_mut::<Transform>(player).unwrap().translation = Vec3::new(12.0, 3.0, 0.0);
569        record.run(&mut world);
570        assert_eq!(world.get::<SafeGroundedPosition>(player).unwrap().position, Vec2::new(12.0, 3.0));
571        world.entity_mut(spike).insert(groups::enemy_attack());
572        index.run(&mut world);
573        world.get_mut::<SafeGroundedPosition>(player).unwrap().position = Vec2::new(6.0, 3.0);
574        record.run(&mut world);
575        assert_eq!(world.get::<SafeGroundedPosition>(player).unwrap().position, Vec2::new(6.0, 3.0));
576    }
