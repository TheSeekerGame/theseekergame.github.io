39impl Plugin for PlayerCollisionPlugin {

42            GameTickUpdate,

45                .run_if(in_state(AppState::InGame)),
46        );

55pub fn player_collisions(
56    collision_world: Option<Res<TileCollisionWorld>>,
57    spatial_query: PhysicsWorld,
58    mut q_gent: Query<
59        (
60            Entity,
61            &mut Transform,
62            &mut LinearVelocity,
63            &Collider,
64            Option<&mut WallSlideTime>,
65            Option<ActionRef<Dashing>>,
66            Has<ActionTag<DashStrike>>,
67            Has<ActionTag<Whirling>>,
68            Has<ActionTag<BurningDashing>>,
69            Has<ActionTag<crate::game::player::states::FlickerStriking>>,
70            Has<crate::game::player::states::locomotion::running::AutoDodgeRolling>,
71            Option<&Falling>,
72            Has<StealthEffect>,
73            Option<&TentInteriorOccupant>,
74            &mut CapsuleDropPlatformState,
75        ),
76        (With<Player>, Without<Dead>, Without<crate::game::spikes::SpikeRespawnPending>),
77    >,
78    q_enemy: Query<(), With<Enemy>>,
79    q_capsule: CapsuleCollisionQuery<(With<Capsule>, Without<Player>)>,
80    q_elevator: Query<Option<ActionRef<Elevator>>, With<Player>>,
81    mut commands: Commands,
82    time: Res<GameTime>,
83    tent_interior: Option<Res<TentInteriorAssets>>,
84    mut profiler: TickProfiler,
85) {
86    let _profile = profiler.scope("player_collisions");
87    for (
88        entity,
89        mut pos,
90        mut linear_velocity,
91        collider,
92        slide,
93        dashing,
94        is_dash_strike,
95        is_whirling,
96        is_burning_dashing,
97        is_flicker_striking,
98        is_auto_dodge_rolling,
99        falling_state,
100        is_stealthed,
101        interior_occupant,
102        mut drop_platform_state,
103    ) in q_gent.iter_mut()
104    {
105        let Some(view) = PlayerCollisionView::new(interior_occupant,
106            tent_interior.as_deref(), collision_world.as_deref(), &q_capsule) else {
107            continue;
108        };
109        let origin = view.origin;
110
111        let Some(half_extents) = collider.cuboid_half_extents() else {
112            continue;
113        };
114        let original_pos = pos.translation.xy() - origin;
115        let z = pos.translation.z;
116        let mut velocity = linear_velocity.0;
117        let wall_side = falling_state.and_then(|f| f.wall_slide);
118        let is_wall_sliding = wall_side.is_some();
119
120        if is_wall_sliding && velocity.y > 0.0 {
121            velocity.y = 0.0;
122        }
123
124        let elevator = q_elevator.get(entity).ok().flatten();
125        let active_elevator = elevator
126            .as_deref().copied()
127            .filter(|elevator| {
128                interior_occupant.is_none()
129                    && q_capsule.get(elevator.shelter()).is_ok()
130            });
131        if elevator.is_some() && active_elevator.is_none() {
132            finish_optional_action(elevator, &mut commands, entity);
133        }
134        if let Some(elevator) = active_elevator {
135            velocity = elevator.velocity();
136        }
137        let elevator_stop_y =
138            active_elevator.and_then(|elevator| {
139                capsule_elevator_stop_y(
140                    pos.translation.xy(),
141                    velocity.y,
142                    half_extents,
143                    elevator,
144                    &q_capsule,
145                )
146            });
147        if let Some(stop_y) = elevator_stop_y {
148            velocity.y = stop_y - pos.translation.y;
149        }
150
151        let (resolved_y, blocked_y) = view.resolve_axis(Axis::Y, original_pos,
152            velocity.y, half_extents, Some(&drop_platform_state));
153        if blocked_y {
154            velocity.y = 0.0;
155        }
156        let pos_after_y = Vec2::new(original_pos.x, resolved_y);
157        let reached_elevator_stop =
158            elevator_stop_y.is_some_and(|stop_y| {
159                ((pos_after_y.y + origin.y) - stop_y).abs()
160                    <= CAPSULE_TRANSIT_STOP_EPS
161            });
162        let elevator_blocked_vertically = active_elevator.is_some() && blocked_y;
163
164        let (mut final_x, blocked_x) = view.resolve_axis(Axis::X, pos_after_y,
165            velocity.x, half_extents, Some(&drop_platform_state));
166        if blocked_x {
167            velocity.x = 0.0;
168        }
169
170        let allow_enemy_queries = !(is_whirling
171            || is_burning_dashing
172            || is_dash_strike
173            || is_auto_dodge_rolling
174            || is_stealthed
175            || interior_occupant.is_some());
176        let allow_enemy_block =
177            allow_enemy_queries && dashing.is_none() && !is_flicker_striking;
178        let allowed_dx = final_x - pos_after_y.x;
179        if let Some(enemy_blocked_x) = enemy_blocked_player_x(
180            allow_enemy_block,
181            &spatial_query,
182            entity,
183            pos_after_y,
184            collider,
185            allowed_dx,
186            &q_enemy,
187        ) {
188            final_x = enemy_blocked_x;
189            velocity.x = 0.0;
190        }
191
192        let final_pos = Vec2::new(final_x, pos_after_y.y);
193
194        if allow_enemy_queries && !is_flicker_striking {
195            record_player_enemy_inside(
196                &spatial_query,
197                final_pos,
198                collider,
199                entity,
200                &q_enemy,
201                &mut commands,
202            );
203        }
204
205        if velocity.x.abs() < 0.00001
206            && dashing.is_some() {
207                velocity = Vec2::ZERO;
208                finish_optional_action(dashing, &mut commands, entity);
209            }
210
211        pos.translation = (final_pos + origin).extend(z);
212        if reached_elevator_stop || elevator_blocked_vertically {
213            velocity = Vec2::ZERO;
214            finish_optional_action(elevator, &mut commands, entity);
215        }
216        linear_velocity.0 = velocity;
217        if interior_occupant.is_none() {
218            sync_capsule_drop_platform_state(
219                original_pos + origin,
220                pos.translation.xy(),
221                half_extents,
222                &mut drop_platform_state,
223                &q_capsule,
224            );
225        }
226
227        const WALL_SLIDE_CHECK_DISTANCE: f32 = 8.0;
228        const WALL_SLIDE_TOLERANCE: f32 = 0.1;
229        let direction = wall_side.map(|side| match side {
230            WallSide::Left => ProbeDirection::Left, WallSide::Right => ProbeDirection::Right,
231        });
232        let wall_slide = direction.is_some_and(|direction| {
233            view.distance_local(direction, final_pos, half_extents, WALL_SLIDE_CHECK_DISTANCE)
234                <= half_extents.x + WALL_SLIDE_TOLERANCE
235        });
236
237        update_wall_slide_timer(slide, wall_slide, time.hz);
238    }
239}

244fn enemy_blocked_player_x(
245    allow_enemy_block: bool,
246    spatial_query: &PhysicsWorld,
247    entity: Entity,
248    pos_after_y: Vec2,
249    collider: &Collider,
250    allowed_dx: f32,
251    q_enemy: &Query<(), With<Enemy>>,
252) -> Option<f32> {
253    if !allow_enemy_block || allowed_dx.abs() <= 0.0 {
254        return None;
255    }
256
257    let dir = allowed_dx.signum();
258    let max_toi = allowed_dx.abs();
259    let (_enemy, hit) = spatial_query.cast_shape(
260        ShapeCast {
261            origin: pos_after_y,
262            dir: Vec2::new(dir, 0.0),
263            shape: collider,
264            max_toi,
265            interaction: CollisionGroups::new(PLAYER, ENEMY),
266            exclude: Some(entity),
267        },
268        |other| q_enemy.contains(other),
269    )?;
270    Some(pos_after_y.x + dir * hit.time_of_impact)
271}
272
273fn record_player_enemy_inside(
274    spatial_query: &PhysicsWorld,
275    final_pos: Vec2,
276    collider: &Collider,
277    entity: Entity,
278    q_enemy: &Query<(), With<Enemy>>,
279    commands: &mut Commands,
280) {
281    for other in spatial_query.intersect(
282        final_pos,
283        collider,
284        CollisionGroups::new(PLAYER, ENEMY),
285        Some(entity),
286    ) {
287        if !q_enemy.contains(other) {
288            continue;
289        }
290
291        theseeker_engine::physics::inside::set(commands, entity, other);
292    }
293}

298fn update_wall_slide_timer(
299    slide: Option<Mut<'_, WallSlideTime>>,
300    wall_slide: bool,
301    tick_hz: f64,
302) {
303    let Some(mut slide) = slide else {
304        return;
305    };
306    if wall_slide {
307        slide.0 = 0.0;
308    } else {
309        slide.0 += 1.0 / tick_hz as f32;
310    }
311}

313fn finish_optional_action<T: crate::game::player::foreground::ActionState>(
314    action: Option<crate::game::player::foreground::ActionRefItem<'_, '_, T>>,
