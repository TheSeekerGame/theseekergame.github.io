25pub struct SpitPlugin;
26impl Plugin for SpitPlugin {
27    fn build(&self, app: &mut App) {
28        app.add_systems(GameTickUpdate, tick_spitting
29            .run_if(in_state(AppState::InGame))

31            .after(theseeker_engine::animation::AnimationSet::Signals)
32            .before(crate::game::combat::damage_source::determine_damage_targets));

47fn tick_spitting(
48    mut players: Query<(Entity, ActionMut<Spitting>, &Transform, &Gent, &mut LinearVelocity,
49        &mut Facing, &HorizontalInputIntent, &Passives), (With<Player>, Without<Dead>, Without<PlayerGfx>)>,
50    mut context: SpitContext,
51) {
52    for (entity, mut spitting, transform, gent, mut velocity, mut facing, input, passives) in &mut players {
53        let position = transform.translation.xy();
54        if spitting.action.is_none() {
55            let execution = spitting.execution().id;
56            begin_spit(execution, entity, &mut spitting, &mut facing, gent, position, &mut context);
57        }
58        let Some(action) = &mut spitting.action else { continue; };
59        action.motion_intent(TickDisplacement::default()).apply(action.id, &mut velocity);
60        let effects = action.advance(context.time.tick(), context.signals.get(gent.e_gfx).ok(), input.movement != 0.0);
61        if effects.releases > 0 {
62            release_spit(action, entity, position, passives, &mut context);
63        }
64        if effects.outcome.is_some() {
65            crate::game::player::foreground::finish_action(&mut context.commands, entity, action.id);
66        }
67    }
68}

70fn begin_spit(execution: u64, entity: Entity, spitting: &mut Spitting, facing: &mut Facing, gent: &Gent, position: Vec2, context: &mut SpitContext) {
71    let definition = context.catalog.actions.get("player_spit").expect("validated player_spit action").clone();
72    let binding = definition.presentation.as_ref().expect("Spit presentation");
73    let nearest = nearest_enemy_facing(position, spitting.facing, &context.enemies);
74    let Ok((mut player, mut offset)) = context.gfx.get_mut(gent.e_gfx) else { return; };
75    let mut action = ActionRequest { forward: nearest.direction(), actor: entity, target: None, damage: definition.damage.expect("Spit damage"), definition: definition.clone() }.activate(PlaybackId::default());
76    action.id = execution;
77    crate::game::player::player_anim::set_direction_slots(&mut player, &nearest);
78    action.present(&mut player, ActionPresentation::Execute);
79    action.claim_facing(&mut context.commands);
80    offset.offset = Some(Vec2::from_array(binding.offset).extend(0.0));
81    spitting.facing = nearest;
82    *facing = nearest;
83    spitting.action = Some(action);
84}

86fn release_spit(action: &ActiveAction, owner: Entity, position: Vec2, passives: &Passives, context: &mut SpitContext) {
87    emit_projectile_shot(&mut context.commands, ProjectileShot {
88        owner, start: position, target: find_nearest_enemy_position(position, &context.enemies, SPIT_TARGET_RANGE),
89        forward: action.forward, descriptor: action.definition.projectile.as_ref().expect("Spit projectile"),
90        damage: action.damage,
91        particle_effect: context.particles.as_ref().map(|effect| effect.0.clone()),
92        split_angle: passives.contains(&Passive::Prism).then_some(5.0_f32.to_radians()),
93        policy: ProjectilePolicy {
94            groups: groups::player_attack(), skill: Some(SkillId::Spit), max_targets: Some(1),
95            descriptor_hit_effects: false, z: 0.000_025, layers: Some(RenderLayers::layer(2)), state_lifetime: true,
96        },
97    }, None);
98}

100fn nearest_enemy_facing(
101    player_pos: Vec2,
102    fallback: Facing,
103    enemy_query: &LiveEnemyQuery,
104) -> Facing {
105    let Some(enemy_pos) = find_nearest_enemy_position_any_range(
106        player_pos,
107        enemy_query,
108    ) else {
109        return fallback;
110    };
111
112    if enemy_pos.x > player_pos.x {
113        Facing::Right
114    } else {
115        Facing::Left
116    }
117}

119/// Find the position of the nearest enemy within range
120fn find_nearest_enemy_position(
121    from: Vec2,
122    enemy_query: &LiveEnemyQuery,
123    max_range: f32,
124) -> Option<Vec2> {
125    let max_range_squared = max_range * max_range;
126
127    nearest_enemy_position_by_distance(from, enemy_query)
128        .filter(|(dist_squared, _pos)| *dist_squared <= max_range_squared)
129        .map(|(_dist_squared, pos)| pos)
130}
131
132fn find_nearest_enemy_position_any_range(
133    from: Vec2,
134    enemy_query: &LiveEnemyQuery,
135) -> Option<Vec2> {
136    nearest_enemy_position_by_distance(from, enemy_query)
137        .map(|(_dist_squared, pos)| pos)
138}
139
140fn nearest_enemy_position_by_distance(
141    from: Vec2,
142    enemy_query: &LiveEnemyQuery,
143) -> Option<(f32, Vec2)> {
144    enemy_query
145        .iter()
146        .map(|(_entity, enemy_tf)| {
147            let enemy_pos = enemy_tf.translation().truncate();
148            (from.distance_squared(enemy_pos), enemy_pos)
149        })
150        .min_by(|(dist_a, _pos_a), (dist_b, _pos_b)| {
151            dist_a.total_cmp(dist_b)
152        })
153}
