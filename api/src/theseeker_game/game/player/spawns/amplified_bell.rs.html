57impl Plugin for AmplifiedBellPlugin {
58    fn build(&self, app: &mut App) {
59        // Core bell tick and state changes: ensure ring spawns happen before damage target computation
60        app.add_systems(
61            GameTickUpdate,
62            (
63                on_bell_alive_enter,
64                // Spawn rings before transform propagation so GlobalTransform is valid
65                // for the damage query on the same tick. This also guarantees we run
66                // before determine_damage_targets (which is after Propagate).
67                tick_bell_alive
68                    .before(bevy::transform::TransformSystems::Propagate)
69                    .before(crate::game::combat::damage_source::determine_damage_targets),
70                tick_bell_ending,
71            )
72                .chain(),
73        );
74        // React to damage after it has been applied (schedule new rings and play visuals)
75        app.add_systems(
76            GameTickUpdate,
77            bell_damage_event_listener
78                .after(crate::game::combat::damage_source::apply_damage),
79        );
80        // Prune AoE targets between determine_targets and damage modifications
81        app.add_systems(
82            GameTickUpdate,
83            prune_bell_aoe_targets
84                .after(crate::game::combat::damage_source::determine_damage_targets)
85                .before(crate::game::combat::damage_source::apply_damage_modifications),
86        );
87        // Spark lifetime management
88        app.add_systems(GameTickUpdate, tick_ring_sparks);
89
90        // Ensure arrows are not consumed by hitting the bell: run after damage is applied
91        // (so the bell can schedule its ring) but before projectile despawn logic.
92        app.add_systems(
93            GameTickUpdate,
94            prune_arrow_bell_hits
95                .after(crate::game::combat::damage_source::apply_damage)
96                .before(crate::game::physics::projectile::despawn_projectile),
97        );
98    }
99}

103const BELL_HALF_HEIGHT: f32 = 8.0;

112/// Public spawn helper called by the skill driver
113pub fn spawn_bell(
114    commands: &mut Commands,
115    owner: Entity,
116    center: Vec2,
117    _facing_dir: f32,
118) {
119    // Spawn logic entity (gent)
120    // Use a Z below enemies (0.000014) and player (0.000015)
121    let base_z = 13.0 * 0.000_001;
122    let gfx = commands
123        .spawn((
124            BellGfx,
125            SpriteAnimationBundle::new_play_key(
126                "anim.player.AmplifiedBellDeploy",
127            ),
128            Sprite {
129                texture_atlas: Some(TextureAtlas::default()),
130                ..Default::default()
131            },
132            Transform::from_translation(center.extend(base_z)),
133            GlobalTransform::default(),
134            Visibility::Visible,
135            InheritedVisibility::default(),
136            ViewVisibility::default(),
137            // Render on world layer (0) so player (layer 2) renders on top
138            RenderLayers::layer(0),
139            StateDespawnMarker,
140        ))
141        .id();
142
143    let bell_entity = commands
144        .spawn((
145            Bell,
146            Gent {
147                e_gfx: gfx,
148                e_effects_gfx: gfx,
149            },
150            Transform::from_translation(center.extend(base_z)),
151            GlobalTransform::default(),
152            Collider::cuboid(BELL_HALF_WIDTH, BELL_HALF_HEIGHT),
153            Sensor,               // Do not block player movement
154            groups::enemy_body(), // So player attacks (PLAYER_ATTACK) can intersect this
155            NoDamageNumbers,      // Don't spawn damage numbers on bell hits
156            // Provide Facing so damage system queries match and emit DamageInfo
157            Facing::Right,
158            crate::game::combat::Health {
159                current: u32::MAX,
160                max: u32::MAX,
161            },
162            BellAlive {
163                owner,
164                tick: 0,
165                max_ticks: BELL_LIFETIME_TICKS,
166                hide_ticks_remaining: 0,
167                ring_z_counter: 0.0,
168            },
169            BellRingSchedule::default(),
170            StateDespawnMarker,
171        ))
172        .id();
173
174    // Attach gfx to gent transform
175    commands.entity(gfx).insert((
176        TransformGfxFromGent {
177            pixel_aligned: false,
178            gent: bell_entity,
179            offset: None,
180        },
181        Name::new("BellGfx"),
182    ));
183
184    // Optionally set direction slots if needed by your animation runtime here (omitted for simplicity)
185}

198fn tick_bell_alive(
199    mut commands: Commands,
200    mut query: Query<(
201        Entity,
202        &mut BellAlive,
203        &Gent,
204        &mut BellRingSchedule,
205        &GlobalTransform,
206    )>,
207    mut gfx_query: Query<
208        (
209            &mut Visibility,
210            &mut ScriptPlayer<SpriteAnimation>,
211        ),
212        With<BellGfx>,
213    >,
214    time: Res<GameTime>,
215    stealthed_owner: Query<(), With<StealthEffect>>,
216) {
217    let now = time.tick();
218    for (entity, mut alive, gent, mut schedule, bell_tf) in query.iter_mut() {
219        // Lifetime
220        alive.tick = alive.tick.saturating_add(1);
221        if alive.tick >= alive.max_ticks {
222            // Transition to ending
223            commands.entity(entity).remove::<BellAlive>();
224            commands.entity(entity).insert(BellEnding {
225                ticks_left: END_ANIM_TICKS,
226            });
227            play_bell_end_animation(gent, &mut gfx_query);
228            continue;
229        }
230
231        // Handle visibility during spark
232        update_bell_spark_visibility(&mut alive, gent, &mut gfx_query);
233
234        // Fire due rings (spawn AoE damage)
235        // Drain triggers <= now and aggregate by scheduled tick so multiple hits in
236        // the same originating tick produce a single ring with multiplied damage.
237        let due_counts = drain_due_bell_ring_counts(&mut schedule, now);
238        spawn_due_bell_rings(
239            &mut commands,
240            &stealthed_owner,
241            bell_tf,
242            &alive,
243            due_counts,
244        );
245    }
246}

265fn update_bell_spark_visibility(
266    alive: &mut BellAlive,
267    gent: &Gent,
268    gfx_query: &mut Query<
269        (
270            &mut Visibility,
271            &mut ScriptPlayer<SpriteAnimation>,
272        ),
273        With<BellGfx>,
274    >,
275) {
276    let target_visibility = if alive.hide_ticks_remaining > 0 {
277        alive.hide_ticks_remaining -= 1;
278        Visibility::Hidden
279    } else {
280        Visibility::Visible
281    };
282
283    let Ok((mut vis, _anim)) = gfx_query.get_mut(gent.e_gfx) else {
284        return;
285    };
286
287    *vis = target_visibility;
288}

290fn drain_due_bell_ring_counts(
291    schedule: &mut BellRingSchedule,
292    now: u64,
293) -> BTreeMap<u64, u32> {
294    let mut remaining = Vec::with_capacity(schedule.triggers.len());
295    let mut due_counts = BTreeMap::new();
296    for &trigger_tick in schedule.triggers.iter() {
297        record_bell_ring_trigger(
298            trigger_tick,
299            now,
300            &mut due_counts,
301            &mut remaining,
302        );
303    }
304    schedule.triggers = remaining;
305
306    due_counts
307}
308
309fn record_bell_ring_trigger(
310    trigger_tick: u64,
311    now: u64,
312    due_counts: &mut BTreeMap<u64, u32>,
313    remaining: &mut Vec<u64>,
314) {
315    if trigger_tick > now {
316        remaining.push(trigger_tick);
317        return;
318    }
319
320    *due_counts.entry(trigger_tick).or_insert(0) += 1;
321}

323fn spawn_due_bell_rings(
324    commands: &mut Commands,
325    stealthed_owner: &Query<(), With<StealthEffect>>,
326    bell_tf: &GlobalTransform,
327    alive: &BellAlive,
328    due_counts: BTreeMap<u64, u32>,
329) {
330    if due_counts.is_empty() {
331        return;
332    }
333
334    let world_pos = bell_tf.translation();
335    let owner_is_stealthed = stealthed_owner.get(alive.owner).is_ok();
336    for (_scheduled_tick, count) in due_counts {
337        let scaled_damage = RING_DAMAGE * (count as f32);
338        spawn_ring_damage(
339            commands,
340            world_pos,
341            alive.owner,
342            owner_is_stealthed,
343            scaled_damage,
344        );
345    }
346}

348fn tick_bell_ending(
349    mut commands: Commands,
350    mut query: Query<(Entity, &mut BellEnding, &Gent)>,
351) {
352    for (entity, mut ending, gent) in query.iter_mut() {
353        if ending.ticks_left > 0 {
354            ending.ticks_left -= 1;
355        }
356        if ending.ticks_left == 0 {
357            commands.entity(gent.e_gfx).despawn();
358            commands.entity(entity).despawn();
359        }
360    }
361}

363fn bell_damage_event_listener(
364    mut commands: Commands,
365    mut damage_events: MessageReader<crate::game::combat::DamageInfo>,
366    mut query: Query<(
367        Entity,
368        &mut BellAlive,
369        &mut BellRingSchedule,
370    )>,
371    time: Res<GameTime>,
372    mut ripple_events: MessageWriter<RippleEvent>,
373    mut foliage_impulses: MessageWriter<FoliageImpulseEvent>,
374    bell_tf_query: Query<&GlobalTransform>,
375) {
376    let now = time.tick();
377    for dmg in damage_events.read() {
378        if let Ok((bell_entity, mut alive, mut schedule)) =
379            query.get_mut(dmg.target)
380        {
381            // Schedule ring with delay
382            schedule.triggers.push(now + RING_DELAY_TICKS as u64);
383            // Spawn ring visual spark immediately (as independent entity followed to gent)
384            spawn_ring_visual(
385                &mut commands,
386                bell_entity,
387                alive.ring_z_counter,
388            );
389            // Keep Z deltas tiny to remain behind player (player ~15e-6)
390            alive.ring_z_counter += 0.000_000_01; // 1e-8 per ring
391            alive.hide_ticks_remaining = RING_SPARK_TICKS;
392            write_bell_ring_ripple(
393                bell_entity,
394                &bell_tf_query,
395                &mut ripple_events,
396                &mut foliage_impulses,
397            );
398        }
399    }
400}

402fn write_bell_ring_ripple(
403    bell_entity: Entity,
404    bell_tf_query: &Query<&GlobalTransform>,
405    ripple_events: &mut MessageWriter<RippleEvent>,
406    foliage_impulses: &mut MessageWriter<FoliageImpulseEvent>,
407) {
408    let Ok(tf) = bell_tf_query.get(bell_entity) else {
409        return;
410    };
411
412    let descriptor = RippleDescriptor::shockwave();
413    ripple_events.write(RippleEvent {
414        origin: tf.translation(),
415        descriptor,
416    });
417    foliage_impulses.write(FoliageImpulseEvent::ring(
418        tf.translation().truncate(),
419        descriptor.start_radius,
420        descriptor.max_radius,
421        descriptor.speed_per_tick,
422        descriptor.width,
423        4.0,
424        descriptor.fade_power,
425    ));
426}

463fn spawn_ring_damage(
464    commands: &mut Commands,
465    world_pos: Vec3,
466    owner: Entity,
467    stealthed: bool,
468    damage: f32,
469) {
470    let mut e = commands.spawn((
471        BellDamageSource,
472        DamageSource::new(1, owner, damage).with_max_targets(16),
473        Collider::ball(RING_RADIUS),
474        // Use canonical group helper for player attacks
475        groups::player_attack(),
476        Transform::from_translation(world_pos),
477        GlobalTransform::from_translation(world_pos),
478        StateDespawnMarker,
479    ));
480    if stealthed {
481        e.insert(Stealthed);
482    }
483}

485/// Allow player arrows to trigger the bell without being consumed.
486/// After damage has been applied (and DamageInfo emitted for the bell),
487/// remove bell targets from the arrow's damaged_set so the arrow can still
488/// hit an enemy behind the bell before despawning.
489fn prune_arrow_bell_hits(
490    mut query: Query<
491        &mut DamageSource,
492        With<crate::game::physics::projectile::Arrow>,
493    >,
494    is_bell: Query<(), With<Bell>>,
495) {
496    for mut src in query.iter_mut() {
497        if src.damaged_set.is_empty() {
498            continue;
499        }
500        let to_remove: Vec<_> = src
501            .damaged_set
502            .iter()
503            .filter(|e| is_bell.get(**e).is_ok())
504            .copied()
505            .collect();
506        for e in to_remove {
507            src.damaged_set.remove(&e);
508        }
509    }
510}

512/// Tracks lifetime of a bell ring spark animation entity
513#[derive(Component)]
514struct RingSpark {
515    lifetime_ticks: u32,
516}
517
518/// Ticks ring spark lifetimes and despawns them when done
519fn tick_ring_sparks(
520    mut commands: Commands,
521    mut query: Query<(Entity, &mut RingSpark)>,
522) {
523    for (entity, mut spark) in query.iter_mut() {
524        if spark.lifetime_ticks > 0 {
525            spark.lifetime_ticks -= 1;
526        }
527        if spark.lifetime_ticks == 0 {
528            commands.entity(entity).despawn();
529        }
530    }
531}

533/// After targets are computed, prune self and downward hemisphere
534fn prune_bell_aoe_targets(
535    mut query: Query<
536        (&GlobalTransform, &mut DamageSource),
537        With<BellDamageSource>,
538    >,
539    target_tf: Query<&GlobalTransform>,
540    is_bell: Query<(), With<Bell>>,
541) {
542    for (src_tf, mut src) in query.iter_mut() {
543        let src_pos = src_tf.translation();
544        let mut new_set = src.target_set.clone();
545        new_set.retain(|target| {
546            !should_prune_bell_aoe_target(
547                *target, src_pos, &target_tf, &is_bell,
548            )
549        });
550        src.target_set = new_set;
551    }
552}
553
554fn should_prune_bell_aoe_target(
555    target: Entity,
556    src_pos: Vec3,
557    target_tf: &Query<&GlobalTransform>,
558    is_bell: &Query<(), With<Bell>>,
559) -> bool {
560    if is_bell.get(target).is_ok() {
561        return true;
562    }
563
564    let Ok(tf) = target_tf.get(target) else {
565        return false;
566    };
567
568    let bell_bottom = src_pos.y - BELL_HALF_HEIGHT - 1.0;
569    tf.translation().y < bell_bottom
570}
