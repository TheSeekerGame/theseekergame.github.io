59fn spawn_orbs_on_death(
60    enemy_q: Query<
61        (
62            &GlobalTransform,
63            Option<&crate::game::enemy::NoXpDrops>,
64        ),
65        (With<Enemy>, Added<Dead>),
66    >,
67    player_q: Query<&Passives, With<Player>>,
68    mut commands: Commands,
69) {
70    for (tr, skip_xp) in enemy_q.iter() {
71        if skip_xp.is_some() {
72            continue;
73        }
74        let enemy_pos = tr.translation().truncate();
75
76        let mut rng = rand::rng();
77
78        // Initial scatter radius and tick-based explosion speed (units per tick)
79        const POS_RADIUS: f32 = 3.0;
80        // Previously 120..220 units/sec at 96 Hz -> ≈1.25..2.30 units/tick
81        const EXPLOSION_SPEED_PER_TICK_MIN: f32 = 1.25;
82        const EXPLOSION_SPEED_PER_TICK_MAX: f32 = 2.30;
83
84        let use_red = player_q
85            .single()
86            .map(|passives| passives.contains(&Passive::Bloodstone))
87            .unwrap_or(false);
88
89        // Spawn a fixed number of orbs per enemy
90        for _ in 0..7 {
91            let pos = Vec2::new(
92                rng.random_range(-POS_RADIUS..POS_RADIUS),
93                rng.random_range(-POS_RADIUS..POS_RADIUS),
94            )
95            .clamp_length_max(POS_RADIUS);
96            let orb_position = enemy_pos + pos;
97
98            let dir = scatter_direction(pos, &mut rng);
99            let speed = rng.random_range(
100                EXPLOSION_SPEED_PER_TICK_MIN..EXPLOSION_SPEED_PER_TICK_MAX,
101            );
102            let vel = dir * speed;
103
104            let player = xp_orb_animation_player(use_red);
105
106            commands.spawn((
107                // Animation + sprite
108                player,
109                Sprite {
110                    texture_atlas: Some(TextureAtlas::default()),
111                    ..Default::default()
112                },
113                // Kinematics
114                theseeker_engine::physics::LinearVelocity(vel),
115                XpOrb {
116                    motion: ScatterHoming::new(MOTION),
117                    pickup_active: false,
118
119                },
120                // Visuals & transforms
121                Transform::from_translation(
122                    orb_position.extend(
123                        15.0 * 0.000_001 - 0.000_000_1,
124                    ),
125                ),
126                GlobalTransform::default(),
127                Visibility::Visible,
128                InheritedVisibility::VISIBLE,
129                ViewVisibility::default(),
130                // Render orb on pixel-perfect layer
131                pixel_perfect_layers(),
132                // Simple ghosting trail
133                GhostingSource {
134                    spawn_interval_ticks: 1,
135                    ghost_lifetime_ticks: 20,
136                    initial_alpha: 1.0,
137                    fade_curve: FadeCurve::Constant,
138                    scale_over_time: ScaleCurve::Axes { x: 1.0, y: 0.5 },
139                    render_layers: Some(RenderLayers::layer(0)),
140                    ..Default::default()
141                },
142                StateDespawnMarker,
143                Name::new("xp_orb"),
144            ));
145        }
146    }
147}

149fn scatter_direction<R: Rng + ?Sized>(pos: Vec2, rng: &mut R) -> Vec2 {
150    if pos.length_squared() > 1e-3 {
151        return pos.normalize();
152    }
153
154    let angle = rng.random_range(0.0..std::f32::consts::TAU);
155    Vec2::new(angle.cos(), angle.sin())
156}

169// Tick-based tuning

171const DIST_THRESHOLD: f32 = 2.0;

173fn update_orbs_pos(
174    mut query: Query<(
175        &mut Transform,
176        &mut theseeker_engine::physics::LinearVelocity,
177        &XpOrb,
178    )>,
179) {
180    for (mut tr, vel, _orb) in query.iter_mut() {
181        tr.translation += vel.0.extend(0.0);
182    }
183}

185fn update_orbs_vel_and_pickup(
186    mut commands: Commands,
187    mut query: Query<(
188        Entity,
189        &GlobalTransform,
190        &mut theseeker_engine::physics::LinearVelocity,
191        &mut XpOrb,
192    )>,
193    p_query: Query<&GlobalTransform, With<Player>>,
194    mut xp_event: MessageWriter<XpOrbPickup>,
195) {
196    let Ok(p) = p_query.single() else {
197        return;
198    };
199    let p_pos = p.translation().truncate();
200
201    for (entity, tr, mut vel, mut xp_orb) in query.iter_mut() {
202        let pos = tr.translation().truncate();
203        let dist = p_pos.distance(pos);
204        let dir = (p_pos - pos).normalize_or_zero();
205
206        tick_explosion_phase(&mut xp_orb, &mut vel);
207
208        // Pickup detection during chasing phase via transform distance only
209        if xp_orb.pickup_active && dist < DIST_THRESHOLD {
210            commands.entity(entity).despawn();
211            xp_event.write(XpOrbPickup {
212                amount: AdaptationProgress::XP_PER_ORB,
213            });
214            continue;
215        }
216
217        if dist >= DIST_THRESHOLD {
218            xp_orb.motion.steer(&mut vel.0, dir, dist, MOTION);
219        }
220    }
221}
222
223fn tick_explosion_phase(xp_orb: &mut XpOrb, vel: &mut LinearVelocity) {
224    if let Some(finished) = xp_orb.motion.scatter(&mut vel.0, MOTION) {
225        xp_orb.pickup_active = finished;
226    }
227}
228
229const MOTION: ScatterHomingProfile = ScatterHomingProfile {
230    ticks: 22, damping_start: 0.98, damping_end: 0.92, max_speed: 2.0, steering: 0.10,
231};
