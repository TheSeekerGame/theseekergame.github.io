20/// Player animation helper systems for direction slots and event-based animation triggers
21pub struct PlayerAnimationPlugin;
22
23impl Plugin for PlayerAnimationPlugin {
24    fn build(&self, app: &mut App) {
25        app.add_systems(
26            GameTickUpdate,
27            (
28                // Direction slots must reflect `Facing` before picking animation keys/frames
29                sprite_flip,
30                // Event-based animation slot handlers
31                xp_orb_animation_handler,
32                player_damage_animation_handler,
33            )
34                .in_set(PlayerStateSet::Animation)
35                .run_if(in_state(AppState::InGame)),
36        );
37    }
38}

40/// Sets DirectionRight/DirectionLeft slots based on `Facing`.
41/// Centralized to avoid duplicated branching across states.
42pub fn set_direction_slots(
43    anim: &mut ScriptPlayer<SpriteAnimation>,
44    facing: &Facing,
45) {
46    match facing {
47        Facing::Right => {
48            anim.set_slot("DirectionRight", true);
49            anim.set_slot("DirectionLeft", false);
50        },
51        Facing::Left => {
52            anim.set_slot("DirectionRight", false);
53            anim.set_slot("DirectionLeft", true);
54        },
55    }
56}

58fn sprite_flip(
59    query: Query<(&Facing, &Gent, Option<&Falling>)>,
60    mut gfx_query: Query<&mut ScriptPlayer<SpriteAnimation>, With<PlayerGfx>>,
61    mut current_direction: Local<bool>,
62    mut old_direction: Local<bool>,
63) {
64    for (facing, gent, maybe_falling) in query.iter() {
65        let Ok(mut player) = gfx_query.get_mut(gent.e_gfx) else {
66            continue;
67        };
68
69        update_player_direction_slots(
70            &mut player,
71            facing,
72            maybe_falling,
73            &mut current_direction,
74            &mut old_direction,
75        );
76    }
77}

79fn update_player_direction_slots(
80    player: &mut ScriptPlayer<SpriteAnimation>,
81    facing: &Facing,
82    maybe_falling: Option<&Falling>,
83    current_direction: &mut Local<bool>,
84    old_direction: &mut Local<bool>,
85) {
86    **old_direction = **current_direction;
87
88    // If wall sliding, defer flip control to wall-slide logic.
89    if is_wall_sliding(maybe_falling) {
90        set_direction_changed_slot(player, **old_direction, **current_direction);
91        return;
92    }
93
94    // Use the facing direction as-is for non-wall-slide cases.
95    set_direction_slots(player, facing);
96    **current_direction = matches!(facing, Facing::Right);
97
98    // Simple change detection for DirectionChanged slot.
99    set_direction_changed_slot(player, **old_direction, **current_direction);
100}
101
102fn is_wall_sliding(maybe_falling: Option<&Falling>) -> bool {
103    maybe_falling
104        .map(|falling| falling.wall_slide.is_some())
105        .unwrap_or(false)
106}
107
108fn set_direction_changed_slot(
109    player: &mut ScriptPlayer<SpriteAnimation>,
110    old_direction: bool,
111    current_direction: bool,
112) {
113    player.set_slot("DirectionChanged", old_direction != current_direction);
114}

116fn xp_orb_animation_handler(
117    xp_events: MessageReader<XpOrbPickup>,
118    mut gfx_query: Query<&mut ScriptPlayer<SpriteAnimation>, With<PlayerGfx>>,
119) {
120    let xp_event_occurred = !xp_events.is_empty();
121    for mut anim in gfx_query.iter_mut() {
122        if xp_event_occurred {
123            anim.set_slot("XpOrb", true);
124        } else {
125            anim.set_slot("XpOrb", false);
126        }
127    }
128}
129
130fn player_damage_animation_handler(
131    mut damage_events: MessageReader<DamageInfo>,
132    mut gfx_query: Query<&mut ScriptPlayer<SpriteAnimation>, With<PlayerGfx>>,
133    player_query: Query<(), With<crate::game::player::Player>>,
134) {
135    // Only set the Damaged slot when the PLAYER entity is the damage TARGET.
136    let mut player_was_damaged = false;
137    for dmg in damage_events.read() {
138        if player_query.get(dmg.target).is_ok() {
139            player_was_damaged = true;
140            break;
141        }
142    }
143
144    for mut anim in gfx_query.iter_mut() {
145        if player_was_damaged {
146            anim.set_slot("Damaged", true);
147        } else {
148            anim.set_slot("Damaged", false);
149        }
150    }
