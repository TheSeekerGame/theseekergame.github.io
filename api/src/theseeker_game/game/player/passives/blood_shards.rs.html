21/// A heal was applied; Blood Shards turn its overheal into projectiles.
22#[derive(Message, Clone, Copy)]
23pub struct HealApplied {
24    pub target: Entity,
25    /// Healing that would have been applied beyond max health
26    pub overheal: u32,
27    /// Max health at the time of the heal
28    pub max_health: u32,
29}
30
31impl HealApplied {
32    /// Heals `health` by `amount`, capped at max health.
33    pub fn heal(target: Entity, health: &mut Health, amount: u32) -> Self {
34        let before = health.current;
35        health.current = health.current.saturating_add(amount).min(health.max);
36        let healed = health.current.saturating_sub(before);
37        Self { target, overheal: amount.saturating_sub(healed), max_health: health.max }
38    }
39}

48// Shard size tiers for visual representation
49#[derive(Clone, Copy, PartialEq)]
50enum ShardSizeTier {
51    Tiny,
52    Small,
53    Medium,
54    Large,
55}
56
57impl ShardSizeTier {
58    /// Determine size tier based on heal percentage of max health
59    fn from_heal_percent(heal_pct: f32) -> Self {
60        if heal_pct <= 0.03 {
61            // Up to 3% (e.g., bloodstone 2 HP on 120 max HP = 1.67%)
62            ShardSizeTier::Small
63        } else if heal_pct <= 0.10 {
64            ShardSizeTier::Medium
65        } else {
66            ShardSizeTier::Large
67        }
68    }
69
70    /// Demote tier by one (for Prism effect)
71    fn demoted(self) -> Self {
72        match self {
73            ShardSizeTier::Large => ShardSizeTier::Medium,
74            ShardSizeTier::Medium => ShardSizeTier::Small,
75            ShardSizeTier::Small | ShardSizeTier::Tiny => ShardSizeTier::Tiny,
76        }
77    }
78
79    /// Animation slot name for this tier
80    fn slot_name(self) -> &'static str {
81        match self {
82            ShardSizeTier::Tiny => "ShardTiny",
83            ShardSizeTier::Small => "ShardSmall",
84            ShardSizeTier::Medium => "ShardMedium",
85            ShardSizeTier::Large => "ShardLarge",
86        }
87    }
88}

90// Tick-based tuning (matching xp_orbs pattern)
91const SHARD_LIFETIME_TICKS: u32 = 240;
92
93const EXPLOSION_SPEED_PER_TICK_MIN: f32 = 1.25;
94const EXPLOSION_SPEED_PER_TICK_MAX: f32 = 2.30;
95const POS_RADIUS: f32 = 3.0;
96
97const DIST_THRESHOLD: f32 = 2.0;
98const SHARD_COLLIDER_RADIUS: f32 = 3.0;
99const SHARD_MAX_TARGET_DISTANCE: f32 = 180.0;
100
101// Damage scaling constants
102const BASE_DAMAGE_PER_OVERHEAL_DEFAULT: f32 = 1.5;

115pub fn spawn_blood_shards_on_heal(
116    mut commands: Commands,
117    mut heal_events: MessageReader<HealApplied>,
118    player_q: Query<(Entity, &GlobalTransform, &Passives), With<Player>>,
119    targets_q: BloodShardTargetQuery<'_, '_>,
120) {
121    let Ok((player, player_tr, passives)) = player_q.single() else {
122        return;
123    };
124    if !passives.contains(&Passive::BloodShards) {
125        return;
126    }
127
128    let has_prism = passives.contains(&Passive::Prism);
129    let damage_per_overheal = passives
130        .passive_effective_value(Passive::BloodShards)
131        .unwrap_or(BASE_DAMAGE_PER_OVERHEAL_DEFAULT);
132    let player_pos = player_tr.translation().truncate();
133    let mut rng = rand::rng();
134
135    for evt in heal_events.read() {
136        // Only trigger on overheal (healing while at full health)
137        if evt.target != player || evt.overheal == 0 {
138            continue;
139        }
140
141        // Calculate heal percentage and size tier
142        let heal_pct = evt.overheal as f32 / evt.max_health.max(1) as f32;
143        let base_tier = ShardSizeTier::from_heal_percent(heal_pct);
144
145        // Prism: triple shards but demote tier
146        let (shard_count, size_tier) = if has_prism {
147            (3, base_tier.demoted())
148        } else {
149            (1, base_tier)
150        };
151
152        // Damage scales smoothly with overheal (not tied to visual tier)
153        // Prism keeps full damage per shard, effectively tripling total damage
154        let per_shard_damage =
155            (evt.overheal as f32 * damage_per_overheal).max(1.0);
156
157        // Gather candidate targets and shuffle for uniqueness
158        let mut candidates: Vec<(Entity, Vec2)> = targets_q
159            .iter()
160            .filter_map(|(e, tr)| blood_shard_candidate(e, tr, player_pos))
161            .collect();
162        candidates.shuffle(&mut rng);
163
164        for _ in 0..shard_count {
165            // Random offset from player position
166            let offset = Vec2::new(
167                rng.random_range(-POS_RADIUS..POS_RADIUS),
168                rng.random_range(-POS_RADIUS..POS_RADIUS),
169            )
170            .clamp_length_max(POS_RADIUS);
171            let spawn_pos = player_pos + offset;
172
173            // Random explosion direction
174            let dir = blood_shard_explosion_direction(offset, &mut rng);
175            let speed = rng.random_range(
176                EXPLOSION_SPEED_PER_TICK_MIN..EXPLOSION_SPEED_PER_TICK_MAX,
177            );
178            let vel = dir * speed;
179
180            // Assign target from candidates (each shard gets a unique target if available)
181            let target = candidates.pop().map(|(e, _)| e);
182
183            // Set up animation with correct size tier
184            let mut anim = ScriptPlayer::<SpriteAnimation>::default();
185            anim.play_key("anim.particles.BloodShards");
186            anim.set_slot(size_tier.slot_name(), true);
187
188            commands.spawn((
189                // Core gameplay components
190                (
191                    DamageSource::new(
192                        SHARD_LIFETIME_TICKS,
193                        player,
194                        per_shard_damage,
195                    )
196                    .with_max_targets(1),
197                    BloodShard {
198                        target,
199                        motion: ScatterHoming::new(MOTION),
200                        pickup_active: false,
201
202                    },
203                    LinearVelocity(vel),
204                    Collider::ball(SHARD_COLLIDER_RADIUS),
205                    groups::player_attack(),
206                ),
207                // Animation + sprite
208                (
209                    anim,
210                    Sprite {
211                        texture_atlas: Some(TextureAtlas::default()),
212                        ..Default::default()
213                    },
214                ),
215                // Transforms and visibility
216                (
217                    Transform::from_translation(
218                        spawn_pos.extend(15.0 * 0.000_001 - 0.000_000_1),
219                    ),
220                    GlobalTransform::default(),
221                    Visibility::Visible,
222                    InheritedVisibility::VISIBLE,
223                    ViewVisibility::default(),
224                ),
225                // Rendering and effects
226                (
227                    pixel_perfect_layers(),
228                    GhostingSource {
229                        spawn_interval_ticks: 1,
230                        ghost_lifetime_ticks: 20,
231                        initial_alpha: 1.0,
232                        fade_curve: FadeCurve::Constant,
233                        scale_over_time: ScaleCurve::Axes { x: 1.0, y: 0.5 },
234                        render_layers: Some(RenderLayers::layer(0)),
235                        ..Default::default()
236                    },
237                    StateDespawnMarker,
238                    Name::new("BloodShard"),
239                ),
240            ));
241        }
242    }
243}

245fn blood_shard_candidate(
246    entity: Entity,
247    transform: &GlobalTransform,
248    player_pos: Vec2,
249) -> Option<(Entity, Vec2)> {
250    let pos = transform.translation().truncate();
251    let max_dist2 = SHARD_MAX_TARGET_DISTANCE * SHARD_MAX_TARGET_DISTANCE;
252    (pos.distance_squared(player_pos) <= max_dist2).then_some((entity, pos))
253}

255fn blood_shard_explosion_direction(offset: Vec2, rng: &mut impl Rng) -> Vec2 {
256    if offset.length_squared() > 1e-3 {
257        return offset.normalize();
258    }
259
260    let angle = rng.random_range(0.0..std::f32::consts::TAU);
261    Vec2::new(angle.cos(), angle.sin())
262}

276pub fn update_blood_shards_vel_and_hit(
277    mut commands: Commands,
278    mut query: Query<(
279        Entity,
280        &GlobalTransform,
281        &mut LinearVelocity,
282        &mut BloodShard,
283    )>,
284    target_q: BloodShardTargetQuery<'_, '_>,
285) {
286    for (entity, tr, mut vel, mut shard) in query.iter_mut() {
287        let pos = tr.translation().truncate();
288
289        tick_blood_shard_explosion(&mut shard, &mut vel);
290
291        clear_invalid_blood_shard_target(&mut shard, &target_q);
292        retarget_blood_shard(&mut shard, pos, &target_q);
293
294        if update_blood_shard_homing(pos, &mut vel, &mut shard, &target_q) {
295            // Shard reached target - damage is applied via DamageSource collision.
296            commands.entity(entity).despawn();
297            continue;
298        }
299
300        // Despawn if velocity collapsed and no target to chase
301        if shard.pickup_active
302            && shard.target.is_none()
303            && vel.0.length_squared() < 0.01
304        {
305            commands.entity(entity).despawn();
306        }
307    }
308}

310fn tick_blood_shard_explosion(shard: &mut BloodShard, vel: &mut LinearVelocity) {
311    if let Some(finished) = shard.motion.scatter(&mut vel.0, MOTION) {
312        shard.pickup_active |= finished;
313    }
314}

316fn clear_invalid_blood_shard_target(

329fn retarget_blood_shard(
330    shard: &mut BloodShard,
331    pos: Vec2,
332    target_q: &BloodShardTargetQuery<'_, '_>,
333) {
334    if !shard.pickup_active || shard.target.is_some() {
335        return;
336    }
337
338    shard.target = closest_blood_shard_target(pos, target_q);
339}

341fn closest_blood_shard_target(
342    pos: Vec2,
343    target_q: &BloodShardTargetQuery<'_, '_>,
344) -> Option<Entity> {
345    target_q
346        .iter()
347        .min_by(|a, b| compare_blood_shard_target_distance(pos, a.1, b.1))
348        .map(|(entity, _)| entity)
349}
350
351fn compare_blood_shard_target_distance(
352    pos: Vec2,
353    a: &GlobalTransform,
354    b: &GlobalTransform,
355) -> std::cmp::Ordering {
356    let dist_a = a.translation().truncate().distance_squared(pos);
357    let dist_b = b.translation().truncate().distance_squared(pos);
358    dist_a
359        .partial_cmp(&dist_b)
360        .unwrap_or(std::cmp::Ordering::Equal)
361}

363fn update_blood_shard_homing(
364    pos: Vec2,
365    vel: &mut LinearVelocity,
366    shard: &mut BloodShard,
367    target_q: &BloodShardTargetQuery<'_, '_>,
368) -> bool {
369    let Some(target_entity) = shard.target else {
370        return false;
371    };
372    let Ok((_, target_tr)) = target_q.get(target_entity) else {
373        return false;
374    };
375
376    let target_pos = target_tr.translation().truncate();
377    let dist = pos.distance(target_pos);
378    if shard.pickup_active && dist < DIST_THRESHOLD {
379        return true;
380    }
381    if dist < DIST_THRESHOLD {
382        return false;
383    }
384
385    steer_blood_shard_toward_target(pos, target_pos, dist, vel, shard);
386    false
387}
388
389fn steer_blood_shard_toward_target(
390    pos: Vec2, target_pos: Vec2, dist: f32, vel: &mut LinearVelocity, shard: &mut BloodShard,
391) {
392    shard.motion.steer(&mut vel.0, (target_pos - pos).normalize_or_zero(), dist, MOTION);
393}

395/// Cleanup on hit - removes shards immediately after registering a hit.
