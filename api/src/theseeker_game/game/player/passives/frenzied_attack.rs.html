12// Frenzied Attack design:
13// - Convert a fraction of cooldown progress (and inverse-cooldown energy regen) into extra advance each tick
14// - Convert that extra advance into HP cost; never kills the player (floored and capped)
15
16// Fraction of progress to convert into HP cost
17pub const FA_FRACTION: f32 = 1.0;
18// Default health cost per cooldown tick-equivalent of extra progress
19pub const FA_HP_PER_TICK_DEFAULT: f32 = 0.1;
20
21// Tracks per-entity energy regeneration deltas this tick for inverse-cooldown skills
22#[derive(Resource, Default)]
23pub struct EnergyRegenDeltas {
24    pub whirl: HashMap<Entity, f32>,
25    pub flicker: HashMap<Entity, f32>,
26}
27
28// Accumulates fractional life cost per entity so we can apply whole-health damage without precision loss
29#[derive(Resource, Default)]
30pub struct LifeDebt {
31    pub acc: HashMap<Entity, f32>,
32}

34/// Runtime system that applies Frenzied Attack effects after cooldowns/energy update.
35/// - Reads natural cooldown tick deltas and adds 50% extra progress
36/// - Reads energy regen deltas and adds 50% extra regen
37/// - Converts the extra progress into health damage without killing the player
38pub fn frenzied_attack_runtime(
39    mut cooldowns: ResMut<Cooldowns>,
40    mut query: Query<
41        (
42            Entity,
43            &Passives,
44            &mut Health,
45            Option<&mut WhirlAbility>,
46            Option<&mut FlickerAbility>,
47        ),
48        With<Player>,
49    >,
50    mut energy_deltas: ResMut<EnergyRegenDeltas>,
51    mut life_debt: ResMut<LifeDebt>,
52    _time: Res<theseeker_engine::time::GameTime>,
53) {
54    for (
55        entity,
56        passives,
57        mut health,
58        mut maybe_whirl,
59        mut maybe_flicker,
60    ) in query.iter_mut()
61    {
62        if !passives.contains(&Passive::FrenziedAttack) {
63            continue;
64        }
65
66        let extra_ticks_equivalent =
67            apply_frenzied_cooldown_extra(entity, &mut cooldowns)
68                + apply_frenzied_whirl_extra(
69                    entity,
70                    &energy_deltas,
71                    maybe_whirl.as_deref_mut(),
72                )
73                + apply_frenzied_flicker_extra(
74                    entity,
75                    &energy_deltas,
76                    maybe_flicker.as_deref_mut(),
77                );
78        apply_frenzied_life_debt(
79            entity,
80            passives,
81            &mut health,
82            &mut life_debt,
83            extra_ticks_equivalent,
84        );
85    }
86
87    // Clear energy deltas after consumption to avoid stale accumulation next tick
88    energy_deltas.whirl.clear();
89    energy_deltas.flicker.clear();
90}

92fn apply_frenzied_cooldown_extra(
93    entity: Entity,
94    cooldowns: &mut Cooldowns,
95) -> f32 {
96    let keys: Vec<(Entity, SkillId)> = cooldowns
97        .tick_deltas
98        .keys()
99        .filter(|(e, _)| *e == entity)
100        .copied()
101        .collect();
102
103    keys.into_iter()
104        .map(|(e, skill)| apply_frenzied_cooldown_key(e, skill, cooldowns))
105        .sum()
106}
107
108fn apply_frenzied_cooldown_key(
109    entity: Entity,
110    skill: SkillId,
111    cooldowns: &mut Cooldowns,
112) -> f32 {
113    let Some(natural_delta) = cooldowns.tick_deltas.get(&(entity, skill)).copied()
114    else {
115        return 0.0;
116    };
117    if natural_delta <= 0.0 {
118        return 0.0;
119    }
120
121    let extra = natural_delta * FA_FRACTION;
122    cooldowns.reduce_specific(entity, skill, extra);
123    extra
124}

126fn apply_frenzied_whirl_extra(
127    entity: Entity,
128    energy_deltas: &EnergyRegenDeltas,
129    maybe_whirl: Option<&mut WhirlAbility>,
130) -> f32 {
131    let Some(whirl) = maybe_whirl else {
132        return 0.0;
133    };
134    let Some(regen) = energy_deltas.whirl.get(&entity).copied() else {
135        return 0.0;
136    };
137
138    apply_frenzied_energy_extra(regen, whirl_metadata().max_energy, &mut whirl.energy)
139}

141fn apply_frenzied_flicker_extra(
142    entity: Entity,
143    energy_deltas: &EnergyRegenDeltas,
144    maybe_flicker: Option<&mut FlickerAbility>,
145) -> f32 {
146    let Some(flicker) = maybe_flicker else {
147        return 0.0;
148    };
149    let Some(regen) = energy_deltas.flicker.get(&entity).copied() else {
150        return 0.0;
151    };
152
153    apply_frenzied_energy_extra(
154        regen,
155        flicker_strike_metadata().max_energy,
156        &mut flicker.energy,
157    )
158}

160fn apply_frenzied_energy_extra(
161    regen: f32,
162    max_energy: f32,
163    energy: &mut f32,
164) -> f32 {
165    if regen <= 0.0 {
166        return 0.0;
167    }
168
169    let extra = regen * FA_FRACTION;
170    let cap = (max_energy - *energy).max(0.0);
171    let applied_extra = extra.min(cap);
172    if applied_extra <= 0.0 {
173        return 0.0;
174    }
175
176    *energy += applied_extra;
177    // Tick-equivalent: applied_extra / regen == FA_FRACTION of a tick when not truncated.
178    applied_extra / regen
179}

181fn apply_frenzied_life_debt(
182    entity: Entity,
183    passives: &Passives,
184    health: &mut Health,
185    life_debt: &mut LifeDebt,
186    extra_ticks_equivalent: f32,
187) {
188    if extra_ticks_equivalent <= 0.0 || health.current <= 1 {
189        return;
190    }
191
192    let hp_per_tick = passives
193        .passive_effective_value(Passive::FrenziedAttack)
194        .unwrap_or(FA_HP_PER_TICK_DEFAULT);
195    let entry = life_debt.acc.entry(entity).or_insert(0.0);
196    *entry += extra_ticks_equivalent * hp_per_tick;
197
198    let damage = pending_frenzied_life_debt_damage(*entry, health.current);
199    if damage == 0 {
200        return;
201    }
202
203    health.current = health.current.saturating_sub(damage).max(1);
204    *entry -= damage as f32;
205}
206
207fn pending_frenzied_life_debt_damage(entry: f32, current_health: u32) -> u32 {
208    if entry < 1.0 {
209        return 0;
210    }
211
212    let whole = entry.floor() as u32;
213    let max_damage = current_health.saturating_sub(1);
214    whole.min(max_damage)
215}
