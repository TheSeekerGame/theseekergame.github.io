6/// Resolved for one evaluation, so rolled values and contextual inputs remain fresh.
7#[derive(Clone, Copy)]
8pub struct PassiveEvaluation {
9    pub kind: Passive,
10    value: f32,
11}

13pub fn ordered_passives(passives: &Passives) -> [Option<PassiveEvaluation>; Passives::MAX_EQUIPPED] {
14    let mut ordered = std::array::from_fn(|index| {
15        passives.equipped.get(index).map(|&kind| PassiveEvaluation {
16            kind,
17            value: passives.passive_effective_value(kind)
18                .or_else(|| kind.roll_spec().map(|spec| spec.default)).unwrap_or(0.0),
19        })
20    });
21    // Stable sorting preserves equipment order at equal priority; at most four entries.
22    ordered.sort_by_key(|entry| entry.map_or(1, |entry| -entry.priority()));
23    ordered
24}

26impl PassiveEvaluation {
27    fn priority(self) -> i32 { self.kind.definition().priority }

29    pub fn modify_stats(self, stats: &mut PlayerStatMod, context: &PassiveContext) {
30        use Passive::*;
31        match self.kind {
32            GlowingShard => stats.defense *= 1.0 + context.enemies_nearby as f32 * self.value,
33            SerpentRing => {
34                stats.speed *= 1.2;
35                stats.cdr *= 1.33;
36            }
37            RabbitsFoot => {
38                stats.speed *= 1.0 + self.value;
39                stats.extra_jumps = stats.extra_jumps.saturating_add(1);
40            }
41            HeavyBoots => {
42                let multiplier = heavy_boots_multiplier(context, self.value);
43                stats.damage *= multiplier;
44                stats.defense *= multiplier;
45            }
46            DeadlyFeather if context.in_air => stats.cdr *= 1.0 + self.value,
47            DeadlyFeather => stats.defense *= 0.5,
48            Permutator if context.rotation_active => stats.cdr *= self.value,
49            _ => {}
50        }
51    }

53    pub fn modify_damage(self, modifiers: &mut DamageModifiers, context: &PassiveContext) {
54        use Passive::*;
55        match self.kind {
56            IceDagger => {
57                modifiers.can_backstab = true;
58                modifiers.backstab_multiplier = self.value;
59            }
60            PackKiller if context.target_position.is_some() => {
61                modifiers.damage_multiplier *= pack_multiplier(context.enemies_near_target, self.value);
62            }
63            PrimeSygil if context.target_health_pct.is_some_and(|health| health > self.value) => {
64                modifiers.damage_multiplier *= 2.0;
65            }
66            Sharpshooter => modifiers.damage_multiplier *= distance_multiplier(context, self.value),
67            VitalityOverclock if context.target_position.is_none() => {
68                modifiers.damage_multiplier *= vitality_multiplier(context, self.value);
69            }
70            _ => {}
71        }
72    }

74    pub fn animation_slots(self) -> &'static [(&'static str, bool)] {
75        match self.kind {
76            Passive::SerpentRing => &[("SerpentRing", true)],
77            Passive::FrenziedAttack => &[("FrenziedAttack", true)],
78            _ => &[],
79        }
80    }

82    pub fn on_tick(self, context: &PassiveContext, tick: u64, actions: &mut Vec<PassiveAction>) {
83        if self.kind != Passive::DeathShroud || context.health.max == 0 {
84            return;
85        }
86        let interval = self.value.round().max(1.0) as u64;
87        let ratio = context.health.current as f32 / context.health.max as f32;
88        if tick.is_multiple_of(interval) && ratio < 0.5 {
89            actions.push(PassiveAction::Heal(1));
90        }
91    }
92
93    pub fn on_message(self, event: &PassiveEvent, context: &PassiveContext, actions: &mut Vec<PassiveAction>) {
94        use Passive::*;
95        use PassiveAction::*;
96        match (self.kind, *event) {
97            (Bloodstone, PassiveEvent::XpOrbPickup)
98            | (CriticalRegeneration, PassiveEvent::CriticalHit { .. }) => actions.push(Heal(self.value.round() as u32)),
99            (ObsidianNecklace, PassiveEvent::CriticalHit { .. }) => {
100                actions.extend([ReduceCooldowns(self.value), AddEnergy(self.value)]);
101            }
102            (IceDagger, PassiveEvent::BackstabKill { .. }) => actions.extend([ResetCooldowns, RefillEnergyFull]),
103            (ProtectiveSpirit, PassiveEvent::DamageTaken(info)) => {
104                protective_heal(info.amount, context, self.value, actions);
105            }
106            (PulseDrive, PassiveEvent::CriticalHit { damage_source, source_skill: Some(SkillId::DashStrike) }) => {
107                pulse_drive(self.value, damage_source, actions);
108            }
109            (_, PassiveEvent::HitCountAdvanced { hit_count, .. }) if self.schedules_crit(hit_count, context) => {
110                actions.push(ScheduleNextCrit);
111            }
112            _ => {}
113        }
114    }
115
116    fn schedules_crit(self, count: u32, context: &PassiveContext) -> bool {
117        let divisors: &[u32] = match self.kind {
118            Passive::FlamingHeart if (context.health.current as f32) < context.health.max as f32 * self.value => &[2, 3],
119            Passive::DeadlyFeather if context.in_air => &[5, 7, 11, 13],
120            Passive::ObsidianNecklace => &[23, 29, 31],
121            Passive::RabbitsFoot => &[37, 41, 43, 47],
122            _ => &[],
123        };
124        divisors.iter().any(|&divisor| count.is_multiple_of(divisor))
125    }
126}

128fn heavy_boots_multiplier(context: &PassiveContext, penalty: f32) -> f32 {
129    let stationary = context.velocity.length() < 0.01 || context.movement_input.abs() < 0.01;
130    if stationary && !context.jump_pressed { 2.0 } else { penalty }
131}
132
133fn pack_multiplier(enemies: u32, bonus: f32) -> f32 {
134    if enemies == 0 { 0.5 } else { 1.0 + enemies as f32 * bonus }
135}

137fn pulse_drive(chance: f32, source: bevy::prelude::Entity, actions: &mut Vec<PassiveAction>) {
138    // Even a guaranteed trigger consumes its random sample.
139    if rand::rng().random::<f32>() > chance {
140        return;
141    }
142    actions.push(PassiveAction::TriggerInstantSkills { source_damage: Some(source) });
143}
144
145fn protective_heal(amount: f32, context: &PassiveContext, fraction: f32, actions: &mut Vec<PassiveAction>) {
146    let max_allowed = context.health.max as f32 * fraction;
147    if amount <= max_allowed {
148        return;
149    }
150    let excess = (amount - max_allowed.max(0.0)).round() as u32;
151    if excess > 0 {
152        actions.push(PassiveAction::Heal(excess));
153    }
154}

156fn distance_multiplier(context: &PassiveContext, maximum: f32) -> f32 {
157    let (Some(target), Some(grounded)) = (context.target_position, context.last_grounded_position) else {
158        return 1.0;
159    };
160    let scale = ((maximum - 1.0) / 3.0).max(0.0);
161    let remap = |baseline: f32| 1.0 + (baseline - 1.0) * scale;
162    match target.distance(grounded) {
163        distance if distance >= 250.0 => maximum,
164        distance if distance >= 200.0 => remap(3.5),
165        distance if distance >= 150.0 => remap(3.0),
166        distance if distance >= 120.0 => remap(2.5),
167        distance if distance >= 90.0 => remap(2.0),
168        distance if distance >= 60.0 => remap(1.5),
169        distance if distance >= 30.0 => remap(1.25),
170        _ => 1.0,
171    }
172}
173
174fn vitality_multiplier(context: &PassiveContext, maximum: f32) -> f32 {
175    let scale = ((maximum - 1.0) / 2.0).max(0.0);
176    let remap = |baseline: f32| 1.0 + (baseline - 1.0) * scale;
177    match context.health.current as f32 / context.health.max as f32 {
178        ratio if ratio >= 0.9 => maximum,
179        ratio if ratio >= 0.7 => remap(2.0),
180        ratio if ratio >= 0.5 => remap(1.5),
181        ratio if ratio >= 0.25 => remap(1.25),
182        _ => 1.0,
183    }
184}
