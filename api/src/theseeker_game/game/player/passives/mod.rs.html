21/// Messages that passives can respond to.
22#[derive(Message, Clone, Copy)]
23pub enum PassiveEvent {
24    /// Active skill was used by the player (emitted on activation)
25    SkillUsed {
26        owner: Entity,
27        skill: types::SkillId,
28    },
29    /// Enemy was killed by the player
30    EnemyKilled { owner: Entity },
31    /// Enemy was killed by a backstab from the player
32    BackstabKill { owner: Entity },
33    /// XP orb was picked up
34    XpOrbPickup,
35    /// Player dealt damage
36    DamageDealt(DamageInfo),
37    /// Player took damage
38    DamageTaken(DamageInfo),
39    /// Damage entity hit a target (first hit only)
40    DamageHit { damage_source: Entity },
41    /// Critical hit occurred (source skill is provided when known)
42    CriticalHit {
43        damage_source: Entity,
44        source_skill: Option<types::SkillId>,
45    },
46    /// Player's deterministic hit counter advanced (after the first successful hit of a damage source)
47    HitCountAdvanced { owner: Entity, hit_count: u32 },
48}

50/// Actions that passives can trigger.
51#[derive(Debug, Clone)]
52pub enum PassiveAction {
53    /// Heal the player by a specified amount
54    Heal(u32),
55    /// Reduce all cooldowns by the given seconds
56    ReduceCooldowns(f32),
57    /// Reset all cooldowns
58    ResetCooldowns,
59    /// Add energy to all channelled, energy-based skills (e.g., Whirl, Flicker)
60    AddEnergy(f32),
61    /// Fully refill energy for all channeled skills (e.g., Whirl, Flicker)
62    RefillEnergyFull,
63    /// Schedule the next hit to be a critical strike for this player
64    ScheduleNextCrit,
65    /// Trigger all instant-cast skills currently equipped (ignores cooldowns)
66    TriggerInstantSkills {
67        /// Damage source that triggered the instant skills (if relevant).
68        source_damage: Option<Entity>,
69    },
70}

72/// Context provided to passives for decision making.
73pub struct PassiveContext<'a> {
74    pub health: &'a Health,
75    pub velocity: Vec2,
76    pub enemies_nearby: u32,
77    /// True if player has InAir marker (more truthful than !grounded due to predictive grounding)
78    pub in_air: bool,
79    /// Optional target position for per-target damage calculations
80    pub target_position: Option<Vec2>,
81    /// Player's last grounded position for distance calculations
82    pub last_grounded_position: Option<Vec2>,
83    /// Movement input from player (-1.0 = left, 0.0 = none/both, 1.0 = right)
84    pub movement_input: f32,
85    /// Whether jump key is currently pressed
86    pub jump_pressed: bool,
87    /// Number of enemies near the current damage target (for Pack Killer)
88    pub enemies_near_target: u32,
89    /// Target's current health percentage (0.0..=1.0) if known (per-target phase only)
90    pub target_health_pct: Option<f32>,
91    /// Whether a non-repeating skill sequence is active (last two used skills differ)
92    pub rotation_active: bool,
93}

95/// Health used when the owner has none.
96static NO_HEALTH: Health = Health { current: 0, max: 0 };

98impl<'a> PassiveContext<'a> {
99    /// A context that knows only the owner's health.
100    pub fn with_health(health: &'a Health) -> Self {
101        Self {
102            health,
103            velocity: Vec2::ZERO,
104            enemies_nearby: 0,
105            in_air: false,
106            target_position: None,
107            last_grounded_position: None,
108            movement_input: 0.0,
109            jump_pressed: false,
110            enemies_near_target: 0,
111            target_health_pct: None,
112            rotation_active: false,
113        }
114    }
115
116    /// This context aimed at a damage target.
117    pub fn at_target(self, position: Vec2, health_pct: Option<f32>, enemies_near_target: u32) -> Self {
118        Self {
119            target_position: Some(position),
120            target_health_pct: health_pct,
121            enemies_near_target,
122            ..self
123        }
124    }
125}

127/// What passives read about their owner, queried alike wherever passives run. Health is
128/// passed in separately because some of those systems also heal the owner.
129#[derive(QueryData)]
130pub struct PassiveOwner {
131    velocity: Option<&'static LinearVelocity>,
132    action_state: Option<&'static ActionState<PlayerAction>>,
133    horizontal_intent: Option<&'static HorizontalInputIntent>,
134    enemies_nearby: Option<&'static EnemiesNearby>,
135    in_air: Has<InAir>,
136    enemy_proximity: Option<&'static EnemyProximitySensor>,
137    last_grounded: Option<&'static LastGroundedPosition>,
138    permutator: Option<&'static permutator::PermutatorState>,
139}
140
141impl PassiveOwnerItem<'_, '_> {
142    pub fn context<'a>(&self, health: Option<&'a Health>) -> PassiveContext<'a> {
143        // The melee-range sensor is more precise than the coarse count when present.
144        let enemies_nearby = self.enemy_proximity.map_or_else(
145            || self.enemies_nearby.map_or(0, |enemies| enemies.0),
146            |sensor| sensor.enemies_in_melee_range.len() as u32,
147        );
148        PassiveContext {
149            velocity: self.velocity.map_or(Vec2::ZERO, |velocity| velocity.0),
150            enemies_nearby,
151            in_air: self.in_air,
152            last_grounded_position: self.last_grounded.map(|grounded| grounded.position),
153            movement_input: self.horizontal_intent.map_or(0.0, |intent| intent.movement),
154            jump_pressed: self.action_state.is_some_and(|input| input.pressed(&PlayerAction::Jump)),
155            rotation_active: self.permutator.is_some_and(|state| state.active),
156            ..PassiveContext::with_health(health.unwrap_or(&NO_HEALTH))
157        }
158    }
159}

179/// Returns the set of projectile directions after applying passive modifiers.
180/// The returned vectors are normalized; caller is responsible for scaling by projectile speed.
181pub fn projectile_directions(
182    passives: &super::Passives,
183    base_direction: Vec2,
184) -> Vec<Vec2> {
185    let len = base_direction.length();
186    if len <= f32::EPSILON {
187        return Vec::new();
188    }
189    let dir = base_direction / len;
190    if passives.contains(&super::Passive::Prism) {
191        let angle = dir.y.atan2(dir.x);
192        let prism_angle_deg = passives
193            .passive_effective_value(super::Passive::Prism)
194            .unwrap_or(5.0);
195        let offset = prism_angle_deg.to_radians();
196        return vec![
197            Vec2::from_angle(angle),
198            Vec2::from_angle(angle + offset),
199            Vec2::from_angle(angle - offset),
200        ];
201    }
202    vec![dir]
203}
