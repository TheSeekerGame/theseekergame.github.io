2use super::types::{BurningDashMetadata, CooldownMode, CooldownSpec, SkillId};
3use crate::game::player::{
4    states::{transition_action, BurningDashing},
5};

7pub(crate) const BURNING_DASH_METADATA: BurningDashMetadata =
8    BurningDashMetadata {
9        min_health_to_start: 10,
10        delayed_cooldown: CooldownSpec {
11            min_ticks: 0,
12            max_ticks: 48,
13            mode: CooldownMode::RateBased,
14        },
15        animation_key: "anim.player.BurningDash",
16    };
17
18pub(super) fn try_start_burning_dash_slot(
19    ctx: &mut super::registry::SkillActivationContext,
20    world: &mut super::activation::ActivationWorld,
21) -> bool {
22    let commands = &mut world.instant.commands;
23    let cooldowns = &mut world.instant.cooldowns;
24    // Check if the slot action is pressed (this is what makes it channeled)
25    if !ctx.action_state.pressed(&ctx.slot) {
26        return false;
27    }
28    // Gate by delayed cooldown if present
29    if !cooldowns.is_ready(ctx.entity, SkillId::BurningDash, ctx.current_tick) {
30        return false;
31    }
32
33    // Check minimum health requirement
34    if ctx.health.current < BURNING_DASH_METADATA.min_health_to_start {
35        return false;
36    }
37
38    // Get speed modifier from stats
39    let speed_mod = ctx.stats.map(|s| s.speed).unwrap_or(1.0);
40
41    // Transition to BurningDashing state
42    let mut burning_dashing = BurningDashing::new().with_speed_mod(speed_mod);
43    burning_dashing.slot_action = Some(ctx.slot);
44
45    transition_action(commands, ctx.entity, burning_dashing);
46
47    true
48}
