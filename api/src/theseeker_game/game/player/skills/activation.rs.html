14#[derive(Debug, Clone, Copy, PartialEq, Eq)]
15pub enum ActivationMode { Instant, Foreground }
16
17#[derive(Default)]
18struct ActivationBudget { foreground_started: bool }
19impl ActivationBudget {
20    fn allows(&self, mode: ActivationMode) -> bool { mode == ActivationMode::Instant || !self.foreground_started }
21    fn accept(&mut self, mode: ActivationMode) { self.foreground_started |= mode == ActivationMode::Foreground; }
22}

63fn skill_dispatch_system(
64    mut players: Query<SkillUser, (With<Player>, Without<Dead>,
65        Without<crate::game::spikes::SpikeRespawnPending>)>,
66    mut world: ActivationWorld,
67) {
68    for mut player in &mut players {
69        if player.health.current == 0 { continue; }
70        if player.foreground.state::<Elevator>().is_some() {
71            clear_skill_input_buffer(&mut player.buffer);
72            continue;
73        }
74        dispatch_slots(&mut player, &mut world);
75    }
76}

78fn dispatch_slots(player: &mut SkillUserItem<'_, '_>, world: &mut ActivationWorld) {
79    let tick = world.time.tick();
80    let facing = effective_facing(player.facing, player.facing_lock, tick);
81    let melee_weapon = *world.melee_weapon;
82    let mut budget = ActivationBudget::default();
83    for slot in SLOT_INPUTS {
84        let Some(skill) = requested_skill(player, slot.index, slot.action, tick) else { continue; };
85        let mode = skill.activation_mode();
86        if !budget.allows(mode) { continue; }
87        let mut context = SkillActivationContext {
88            entity: player.entity, slot: slot.action,
89            action_state: player.input, buffer: &mut player.buffer, melee_weapon: &melee_weapon,
90            whirl_ability: player.whirl, flicker_ability: player.flicker, transform: player.transform,
91            facing: &facing, last_move_dir: player.last_move.map_or(0.0, |direction| direction.0),
92            health: player.health, current_tick: tick, move_dir: player.horizontal.movement,
93            stats: player.stats, has_burning_dashing: player.foreground.state::<BurningDashing>().is_some(),
94            ground: player.ground, materials: &mut player.materials, weapons: &mut player.weapons,
95        };
96        let accepted = activate_skill(skill, world, &mut context);
97        if !accepted { continue; }
98        budget.accept(mode);
99        world.passive_events.write(super::super::passives::PassiveEvent::SkillUsed { owner: player.entity, skill });
100    }
101}

103fn requested_skill(player: &SkillUserItem<'_, '_>, index: usize, action: PlayerAction, tick: u64) -> Option<SkillId> {
104    if !player.input.pressed(&action) && player.buffer.check_buffered(action, tick).is_none() { return None; }
105    let skill = player.skills.get_skill_at_slot(index)?;
106    // Placement-preview tools commit their requests on release in their own input systems.
107    if matches!(skill, SkillId::PortableTent | SkillId::TetherNode) { return None; }
108    if player.foreground.state::<Gathering>().is_some() && !skill.is_instant() { return None; }
109    Some(skill)
110}
111
112fn clear_skill_input_buffer(buffer: &mut InputBuffer) {
113    for slot in SLOT_INPUTS { buffer.clear_action(slot.action); }
114}
