11/// Withdrawn stock cannot also be spent by crafting or another skill.
12#[derive(Debug)]
13pub struct ItemReservation { key: CraftMaterial, amount: u32 }
14
15impl ItemReservation {
16    pub fn reserve(inventory: &mut MaterialInventory, material: CraftMaterial, amount: u32) -> Option<Self> {
17        let key = material;
18        (amount > 0 && inventory.remove_from_backpack(key, amount)).then_some(Self { key, amount })
19    }
20
21    pub fn refund(self, inventory: &mut MaterialInventory) { inventory.add_to_backpack(self.key, self.amount); }
22    pub fn consume(self) {}
23}

25/// The same spawn function can be called immediately or at a deployment cue.
26pub type ItemSpawner = fn(&mut Commands, Entity, Vec2);
27pub type PlacementCheck = fn(&World, Entity, Vec2) -> bool;

29#[derive(Debug)]
30pub struct DeploymentDefinition {
31    pub material: CraftMaterial,
32    pub amount: u32,
33    pub animation_key: String,
34    pub clip: Arc<SpriteAnimation>,
35    pub release: String,
36    pub spawn: ItemSpawner,
37    pub validate: PlacementCheck,
38}
39
40impl DeploymentDefinition {
41    pub fn validate(&self) -> Result<(), String> {
42        if self.amount == 0 { return Err("deployment requires a positive item quantity".into()); }
43        if self.animation_key.is_empty() { return Err("deployment requires an animation key".into()); }
44        let section = self.clip.playback.ok_or("deployment requires an explicit once clip")?;
45        section.validate()?;
46        if section.mode != theseeker_engine::assets::animation::ClipMode::Once {
47            return Err("deployment playback must complete".into());
48        }
49        let settings = &self.clip.settings.extended;
50        if section.start < settings.frame_min || section.end > settings.frame_max {
51            return Err("deployment section lies outside the clip's frames".into());
52        }
53        if self.release.is_empty() || !self.clip.events.iter().any(|event| {
54            let frame = event.on.frame_enter.or(event.on.frame_exit).or_else(||
55                event.on.bookmark.as_ref().and_then(|name| self.clip.frame_bookmarks.get(name).copied()));
56            event.emit == self.release && (event.on.anim_finished
57                || frame.is_some_and(|frame| frame >= section.start && frame <= section.end))
58        }) {
59            return Err(format!("deployment clip '{}' has no reachable '{}' cue", self.animation_key, self.release));
60        }
61        Ok(())
62    }
63}

75/// Validation and stock reservation happen together at the foreground commit boundary.
76#[expect(dead_code, reason = "Animated deployment is designed in docs/architecture/actors.md, but no item uses it yet.")]
77pub fn request_deployment(commands: &mut Commands, owner: Entity, definition: Arc<DeploymentDefinition>, position: Vec2) {
78    commands.queue(move |world: &mut World| { start_deployment(world, owner, definition, position); });
79}

81pub fn start_deployment(world: &mut World, owner: Entity, definition: Arc<DeploymentDefinition>, position: Vec2) -> bool {
82    if let Err(error) = definition.validate() {
83        warn!("Cannot activate item deployment: {error}");
84        return false;
85    }
86    if !position.is_finite() || !can_deploy(world, owner) { return false; }
87    if !(definition.validate)(world, owner, position) { return false; }
88    let Some(mut inventory) = world.get_mut::<MaterialInventory>(owner) else { return false; };
89    let Some(reservation) = ItemReservation::reserve(&mut inventory, definition.material, definition.amount) else { return false; };
90    let release = cue_id(&definition.release);
91    foreground::start_action(world, owner, Deploying { definition, position,
92        reservation: Some(reservation), started: false, release_queued: false, release });
93    true
94}
95
96fn can_deploy(world: &World, owner: Entity) -> bool {
97    if world.get::<Player>(owner).is_none() || world.get::<LinearVelocity>(owner).is_none() { return false; }
98    if world.get::<Dead>(owner).is_some() || world.get::<Health>(owner).is_some_and(|health| health.current == 0) { return false; }
99    if world.get::<ForegroundAction>(owner).is_some_and(|action| action.state::<super::super::states::Elevator>().is_some()) { return false; }
100    world.get::<Gent>(owner).is_some_and(|gent| world.get::<ScriptPlayer<SpriteAnimation>>(gent.e_gfx).is_some())
101}

103pub struct ItemUsePlugin;
104impl Plugin for ItemUsePlugin {
105    fn build(&self, app: &mut App) {
106        app.add_systems(GameTickUpdate, tick_deployment.in_set(PlayerStateSet::Behavior)
107            .in_set(super::super::PlayerBehaviorSet::Execution)
108            .after(theseeker_engine::animation::AnimationSet::Signals));
109    }
110}

112fn tick_deployment(
113    mut query: Query<(Entity, ActionMut<Deploying>, &Gent, &mut LinearVelocity), (With<Player>, Without<Dead>)>,
114    mut gfx: Query<&mut ScriptPlayer<SpriteAnimation>>,
115    playback: Query<&PlaybackState>,
116    mut commands: Commands,
117) {
118    for (entity, mut deployment, gent, mut velocity) in &mut query {
119        velocity.0 = Vec2::ZERO;
120        if !deployment.started && !start_deployment_playback(&mut deployment, gent, &mut gfx) {
121            deployment.finish(&mut commands, entity);
122            continue;
123        }
124        let Some(events) = deployment.execution().playback(playback.get(gent.e_gfx).ok()) else { continue; };
125        let finished = events.finished;
126        let release = events.ordered.iter().any(|event| event.playback == events.playback
127            && matches!(event.kind, theseeker_engine::animation::PlaybackEventKind::Cue(cue) if cue == deployment.release));
128        if release && !deployment.release_queued {
129            deployment.release_queued = true;
130            let execution = deployment.execution().id;
131            commands.queue(move |world: &mut World| release_deployment(world, entity, execution));
132        }
133        if finished { deployment.finish(&mut commands, entity); }
134    }
135}
136
137fn start_deployment_playback(
138    deployment: &mut foreground::ActionMutItem<'_, '_, Deploying>,
139    gent: &Gent,
140    gfx: &mut Query<&mut ScriptPlayer<SpriteAnimation>>,
141) -> bool {
142    let Ok(mut player) = gfx.get_mut(gent.e_gfx) else { return false; };
143    let definition = deployment.definition.clone();
144    player.acquire_presentation(deployment.execution().id);
145    player.play_snapshot(&definition.animation_key, definition.clip.clone(), None);
146    deployment.execution_mut().playback = theseeker_engine::animation::PlaybackId(player.execution_id());
147    deployment.started = true;
148    true
149}
150
151fn release_deployment(world: &mut World, owner: Entity, execution: u64) {
152    if !can_deploy(world, owner) { return; }
153    let Some(action) = world.get::<ForegroundAction>(owner).filter(|action| action.id == execution) else { return; };
154    let Some(deployment) = action.state::<Deploying>() else { return; };
155    let definition = deployment.definition.clone();
156    let position = deployment.position;
157    if !(definition.validate)(world, owner, position) {
158        foreground::start_action(world, owner, super::super::states::Ready);
159        return;
160    }
161    let reservation = world.get_mut::<ForegroundAction>(owner)
162        .and_then(|mut action| action.state_mut::<Deploying>().and_then(|deployment| deployment.reservation.take()));
163    let Some(reservation) = reservation else { return; };
164    reservation.consume();
165    (definition.spawn)(&mut world.commands(), owner, position);
166}
