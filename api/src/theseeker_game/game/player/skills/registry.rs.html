68pub(super) fn activate_skill(
69    skill: SkillId,
70    world: &mut super::activation::ActivationWorld,
71    ctx: &mut SkillActivationContext<'_>,
72) -> bool {
73    match skill {
74        SkillId::QuarkPylon | SkillId::Stealth | SkillId::AmplifiedBell
75        | SkillId::Spinner | SkillId::ExplosiveMine | SkillId::IceNova => activate_instant_slot(skill, world, ctx),
76        SkillId::Attack => attack::try_start_melee_attack_slot(ctx, world),
77        SkillId::BowAttack => attack::try_start_bow_attack_slot(ctx, world),
78        SkillId::Dash => dash::try_start_dash_slot(ctx, world),
79        SkillId::DashStrike => dash_strike::try_start_dash_strike_slot(ctx, world),
80        SkillId::Whirl => whirl::try_start_whirl_slot(ctx, world),
81        SkillId::BurningDash if !ctx.has_burning_dashing => burning_dash::try_start_burning_dash_slot(ctx, world),
82        SkillId::FlickerStrike => flicker_strike::try_start_flicker_strike_slot(ctx, world),
83        SkillId::Spit => spit::try_start_spit_slot(ctx, world),
84        SkillId::RedBerry | SkillId::BlueBerry | SkillId::MossBerry => consumables::try_start_consumable_slot(skill, ctx, world),
85        SkillId::BurningDash | SkillId::PortableTent | SkillId::TetherNode => false,
86    }
87}

89fn activate_instant_slot(
90    skill: SkillId,
91    world: &mut super::activation::ActivationWorld,
92    ctx: &mut SkillActivationContext,
93) -> bool {
94    use super::instant::{ActivationOrigin, InstantResult, InstantUser};
95    let slot = ctx.slot;
96    let entity = ctx.entity;
97    let buffered = matches!(skill, SkillId::Spinner | SkillId::IceNova | SkillId::AmplifiedBell | SkillId::QuarkPylon);
98    if !(ctx.action_state.just_pressed(&slot)
99        || buffered && ctx.buffer.check_buffered(slot, ctx.current_tick).is_some()) {
100        return false;
101    }
102    let user = InstantUser { entity, transform: ctx.transform, facing: ctx.facing, ground: ctx.ground,
103        tick: ctx.current_tick, cdr: ctx.stats.map_or(1.0, |stats| stats.cdr), origin: ActivationOrigin::Input };
104    let result = world.instant.activate(skill, &user, Some(ctx.materials));
105    let clear_invalid = result == InstantResult::InvalidPlacement
106        && matches!(skill, SkillId::AmplifiedBell | SkillId::QuarkPylon);
107    if buffered && result == InstantResult::Used || clear_invalid {
108        ctx.buffer.clear_action(slot);
109    }
110    result == InstantResult::Used
111}
