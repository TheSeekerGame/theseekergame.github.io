51pub(crate) fn update_placement_previews(
52    mut players: Query<PlacementUser, (With<Player>, Without<PlacementPreview>)>,
53    mut world: PreviewWorld,
54) {
55    world.connection.available = false;
56    let Ok(mut player) = players.single_mut() else { return; };
57    // Matches the tool order in the gameplay schedule.
58    for tool in [PlacementTool::Tent, PlacementTool::Tether] {
59        update_tool(tool, &mut player, &mut world);
60    }
61}

63fn update_tool(tool: PlacementTool, player: &mut PlacementUserItem, world: &mut PreviewWorld) {
64    if tool == PlacementTool::Tether && (player.collider.is_none() || player.ground.is_none()) {
65        return;
66    }
67    let mut existing = world.previews.iter().filter(|(_, preview)| preview.owner == player.entity && preview.tool == tool);
68    let active = existing.next().map(|(entity, _)| entity);
69    for (duplicate, _) in existing {
70        world.commands.entity(duplicate).despawn();
71    }
72    let slot = SLOT_INPUTS.iter().find(|slot| player.skills.get_skill_at_slot(slot.index) == Some(tool.skill()));
73    let eligible = !player.recovering && !(tool == PlacementTool::Tent && player.elevator)
74        && player.materials.backpack_count(tool.material()) > 0;
75    let Some(slot) = slot.filter(|_| eligible) else {
76        despawn_preview(&mut world.commands, active);
77        return;
78    };
79    let placement = placement(tool, player, world);
80    if player.input.just_released(&slot.action) {
81        despawn_preview(&mut world.commands, active);
82        commit_placement(tool, placement, player, world);
83        return;
84    }
85    if !player.input.pressed(&slot.action) {
86        despawn_preview(&mut world.commands, active);
87        return;
88    }
89    sync_preview(tool, placement, active, player, world);
90}

92fn placement(tool: PlacementTool, player: &PlacementUserItem, world: &mut PreviewWorld) -> Placement {
93    match tool {
94        PlacementTool::Tent => portable_tent::preview_placement(player.transform, player.facing,
95            player.occupant, world.tiles.as_deref(), world.interiors.as_deref()),
96        PlacementTool::Tether => {
97            let placement = tether_node::compute_tether_placement(player.transform, player.facing,
98                player.collider.expect("tether player has collider"), player.ground.expect("tether player has sensor"), world.tiles.as_deref());
99            world.connection.available = tether_connector_in_range(placement.center,
100                world.tethers.iter().filter(|(_, tether)| tether.is_network_active())
101                    .map(|(transform, _)| transform.translation.truncate())
102                    .chain(world.capsules.iter().map(|transform| transform.translation.truncate())));
103            placement
104        }
105    }
106}

108fn commit_placement(tool: PlacementTool, placement: Placement, player: &mut PlacementUserItem, world: &mut PreviewWorld) {
109    if !placement.valid || !player.materials.remove_from_backpack(tool.material(), 1) {
110        return;
111    }
112    match tool {
113        PlacementTool::Tent => portable_tent::spawn_portable_tent(&mut world.commands, placement.center),
114        PlacementTool::Tether => { spawn_tether_node(&mut world.commands, placement.center); }
115    }
116    world.events.write(PassiveEvent::SkillUsed { owner: player.entity, skill: tool.skill() });
117}

119fn sync_preview(tool: PlacementTool, placement: Placement, active: Option<Entity>, player: &PlacementUserItem, world: &mut PreviewWorld) {
120    let color = if placement.valid { Color::srgba(0.6, 1.0, 0.6, 0.55) } else { Color::srgba(1.0, 0.45, 0.45, 0.55) };
121    let position = placement.center.extend(tool.z());
122    let layers = tool.layers(player.occupant.is_some());
123    if let Some(entity) = active {
124        let Ok((mut sprite, mut transform, mut render_layers)) = world.visuals.get_mut(entity) else { return; };
125        sprite.color = color;
126        transform.translation = position;
127        *render_layers = layers;
128        return;
129    }
130    world.commands.spawn((
131        PlacementPreview { owner: player.entity, tool },
132        SpriteAnimationBundle::new_play_key(tool.animation()),
133        Sprite { color, texture_atlas: Some(TextureAtlas::default()), ..default() },
134        Transform::from_translation(position), Visibility::Visible, InheritedVisibility::VISIBLE,
135        layers, StateDespawnMarker, Name::new(tool.name()),
136    ));
137}
138
139fn despawn_preview(commands: &mut Commands, entity: Option<Entity>) {
140    if let Some(entity) = entity { commands.entity(entity).despawn(); }
141}

143pub(crate) fn cleanup_orphaned_previews(
144    mut commands: Commands,
145    previews: Query<(Entity, &PlacementPreview)>,
146    players: Query<(), With<Player>>,
147) {
148    for (entity, preview) in &previews {
149        if players.get(preview.owner).is_err() { commands.entity(entity).despawn(); }
150    }
151}

153impl PlacementTool {
154    fn skill(self) -> SkillId { match self { Self::Tent => SkillId::PortableTent, Self::Tether => SkillId::TetherNode } }
155    fn material(self) -> CraftMaterial { match self { Self::Tent => CraftMaterial::Tent, Self::Tether => CraftMaterial::Tether } }
156    fn z(self) -> f32 { match self { Self::Tent => portable_tent::TENT_Z, Self::Tether => TETHER_VISUAL_Z } }
157    fn animation(self) -> &'static str { match self { Self::Tent => "anim.objects.TentPreview", Self::Tether => "anim.objects.TetherPreview" } }
158    fn name(self) -> &'static str { match self { Self::Tent => "PortableTentPreview", Self::Tether => "TetherNodePreview" } }
159    fn layers(self, inside: bool) -> RenderLayers {
160        match self {
161            Self::Tent if inside => RenderLayers::layer(2),
162            Self::Tent => RenderLayers::layer(0),
163            Self::Tether => crate::graphics::pixel_perfect::pixel_perfect_layers(),
164        }
165    }
166}
