13const BELL_HALF_WIDTH: f32 = 6.0;
14const BELL_HALF_HEIGHT: f32 = 8.0;

16pub(crate) const AMPLIFIED_BELL_METADATA: AmplifiedBellMetadata =
17    AmplifiedBellMetadata {
18        cooldown: CooldownSpec {
19            min_ticks: 0,

25/// Determine a valid placement point for the amplified bell given the current player pose.
26pub fn find_amplified_bell_placement(
27    player_transform: &Transform,
28    facing: &Facing,
29    spatial_query: &PhysicsWorld,
30    collision_world: Option<&TileCollisionWorld>,
31    owner: Entity,
32) -> Option<Vec2> {
33    let collision_world = collision_world.filter(|world| world.is_ready())?;
34    let player_pos = player_transform.translation.truncate();
35    let dir = facing.direction();
36    let candidate_offsets = [16.0, 12.0, 8.0, 4.0, 0.0];
37
38    for offset in candidate_offsets.into_iter() {
39        if let Some(candidate) = amplified_bell_candidate(
40            player_pos,
41            dir,
42            offset,
43            spatial_query,
44            collision_world,
45            owner,
46        ) {
47            return Some(candidate);
48        }
49    }
50
51    None
52}
53
54fn amplified_bell_candidate(
55    player_pos: Vec2,
56    dir: f32,
57    offset: f32,
58    spatial_query: &PhysicsWorld,
59    collision_world: &TileCollisionWorld,
60    owner: Entity,
61) -> Option<Vec2> {
62    let x = player_pos.x + dir * offset;
63    let start_y = player_pos.y + 4.0;
64    let distance =
65        collision_world.distance(ProbeDirection::Down, Vec2::new(x, start_y), Vec2::ZERO, 64.0);
66    if distance >= 64.0 {
67        return None;
68    }
69
70    let ground_y = start_y - distance;
71    if player_pos.y - ground_y > 50.0 {
72        return None;
73    }
74
75    let clearance = GROUNDED_THRESHOLD + 0.1;
76    let bell_center_y = ground_y + BELL_HALF_HEIGHT + clearance;
77    let candidate_center = Vec2::new(x, bell_center_y);
78    if collision_world.overlaps_solid(
79        candidate_center,
80        Vec2::new(BELL_HALF_WIDTH, BELL_HALF_HEIGHT),
81    ) {
82        return None;
83    }
84
85    let overlaps_enemy = spatial_query.intersect(
86        candidate_center,
87        &Collider::cuboid(BELL_HALF_WIDTH, BELL_HALF_HEIGHT),
88        CollisionGroups::new(Group::all(), ENEMY),
89        Some(owner),
90    );
91    overlaps_enemy.is_empty().then_some(candidate_center)
92}
