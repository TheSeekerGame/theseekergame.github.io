14pub(crate) const FLICKER_STRIKE_METADATA: FlickerStrikeMetadata =
15    FlickerStrikeMetadata {
16        max_energy: 3.0,
17        chunk_cost: 0.25,
18        regen_per_second: 0.075,
19        range: 150.0,
20    };
21
22pub(super) fn try_start_flicker_strike_slot(
23    ctx: &mut super::registry::SkillActivationContext,
24    world: &mut super::activation::ActivationWorld,
25) -> bool {
26    let requested = ctx.action_state.just_pressed(&ctx.slot)
27        || ctx.buffer.check_buffered(ctx.slot, ctx.current_tick).is_some();
28    // Needs energy for one hit and an enemy in range.
29    if !requested || ctx.flicker_ability.energy < FLICKER_STRIKE_METADATA.chunk_cost {
30        return false;
31    }
32    let player_pos = ctx.transform.translation.truncate();
33    let enemy_in_range = world.enemies.iter().any(|(_, enemy)| {
34        player_pos.distance(enemy.translation.truncate()) <= FLICKER_STRIKE_METADATA.range
35    });
36    if !enemy_in_range {
37        return false;
38    }
39
40    ctx.buffer.clear_action(ctx.slot);
41    let mut flickering = FlickerStriking::new(*ctx.melee_weapon);
42    flickering.slot_action = Some(ctx.slot);
43    transition_action(&mut world.instant.commands, ctx.entity, flickering);
44    true
45}

47/// Regenerate flicker energy while not flickering
48pub fn flicker_energy_regen(
49    mut query: Query<
50        (
51            Entity,
52            &mut FlickerAbility,
53            Option<&PlayerStatMod>,
54        ),
55        (With<Player>, Without<ActionTag<FlickerStriking>>),
56    >,
57    mut energy_deltas: ResMut<EnergyRegenDeltas>,
58    time: Res<GameTime>,
59) {
60    // Clear previous tick values once per tick
61    energy_deltas.flicker.clear();
62    for (entity, mut flicker_ability, stat_mod) in query.iter_mut() {
63        let cdr = stat_mod.map(|s| s.cdr).unwrap_or(1.0);
64        let regen_per_tick =
65            FLICKER_STRIKE_METADATA.regen_per_second * cdr / time.hz as f32;
66        let capacity = (FLICKER_STRIKE_METADATA.max_energy
67            - flicker_ability.energy)
68            .max(0.0);
69        let applied = regen_per_tick.min(capacity);
70        if applied > 0.0 {
71            flicker_ability.energy += applied;
72            energy_deltas.flicker.insert(entity, applied);
73        }
74    }
75}

77/// Ends the flicker strike once its button is released; the outro always plays out.
78pub fn flicker_active_tick(
79    mut commands: Commands,
80    mut query: Query<
81        (
82            Entity,
83            ActionMut<FlickerStriking>,
84            &leafwing_input_manager::action_state::ActionState<PlayerAction>,
85        ),
86        With<Player>,
87    >,
88) {
89    for (entity, mut flickering, input) in &mut query {
90        let held = flickering.slot_action.is_some_and(|slot| input.pressed(&slot));
91        if flickering.phase != FlickerPhase::Outro && !held {
92            request_flicker_outro(entity, &mut commands, &mut flickering);
93        }
94    }
95}

97fn request_flicker_outro(
98    entity: Entity,
99    commands: &mut Commands,
100    flickering: &mut FlickerStriking,
101) {
102    // Do not cancel during dash or intro; defer outro until safe.
103    if matches!(
104        flickering.phase,
105        FlickerPhase::Dashing | FlickerPhase::Intro
106    ) {
107        flickering.pending_outro = true;
108        return;
109    }
110
111    // Not mid-dash/intro: safe to transition immediately.
112    commands.entity(entity).insert(FlickerOutroTransition);
113}
