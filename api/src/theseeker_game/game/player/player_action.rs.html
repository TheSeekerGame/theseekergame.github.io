113/// Resolve movement and facing so horizontal motion can stay independent from facing
114/// when the player presses the opposite direction while continuing to hold the original.
115fn resolve_horizontal_input_intent(
116    keyboard: Res<ButtonInput<KeyCode>>,
117    panels: crate::ui::OpenPanels,
118    time: Res<GameTime>,
119    mut query: Query<
120        (
121            &mut ActionState<PlayerAction>,
122            &mut HorizontalInputIntent,
123            &mut super::states::LastMoveDir,
124            &mut Facing,
125            Option<&super::autoaim::RangedAttackFacingLock>,
126            Option<&crate::game::actors::action::ActionFacing>,
127        ),
128        With<super::Player>,
129    >,
130) {
131    let modal_ui_open = panels.any();
132
133    for (
134        mut action_state,
135        mut intent,
136        mut last_move_dir,
137        mut facing,
138        facing_lock,
139        action_facing,
140    ) in query.iter_mut() {
141        let locked_facing = action_facing.map(crate::game::actors::action::ActionFacing::facing)
142            .or_else(|| facing_lock.and_then(|lock| lock.active_facing(time.tick())));
143
144        if modal_ui_open {
145            action_state.reset(&PlayerAction::Jump);
146            action_state.reset(&PlayerAction::Fall);
147            action_state.reset(&PlayerAction::Move);
148            action_state.reset(&PlayerAction::Interact);
149            *intent = HorizontalInputIntent {
150                movement: 0.0,
151                facing: None,
152            };
153            apply_resolved_facing(&mut facing, locked_facing, None);
154            continue;
155        }
156
157        let axis_movement = action_state.clamped_value(&PlayerAction::Move);
158        let left_held = is_pressed(
159            &keyboard,
160            &[KeyCode::KeyA, KeyCode::ArrowLeft],
161        );
162        let right_held = is_pressed(
163            &keyboard,
164            &[KeyCode::KeyD, KeyCode::ArrowRight],
165        );
166        let left_just_pressed = is_just_pressed(
167            &keyboard,
168            &[KeyCode::KeyA, KeyCode::ArrowLeft],
169        );
170        let right_just_pressed = is_just_pressed(
171            &keyboard,
172            &[KeyCode::KeyD, KeyCode::ArrowRight],
173        );
174
175        let previous_intent = *intent;
176        let fallback_movement = fallback_horizontal_movement(
177            previous_intent,
178            last_move_dir.0,
179            axis_movement,
180            *facing,
181        );
182        let fallback_facing = previous_intent.facing.unwrap_or(*facing);
183
184        *intent = match (left_held, right_held) {
185            (true, false) => HorizontalInputIntent {
186                movement: -1.0,
187                facing: Some(Facing::Left),
188            },
189            (false, true) => HorizontalInputIntent {
190                movement: 1.0,
191                facing: Some(Facing::Right),
192            },
193            (true, true) => both_directions_intent(
194                left_just_pressed,
195                right_just_pressed,
196                fallback_movement,
197                fallback_facing,
198            ),
199            (false, false) => axis_horizontal_intent(axis_movement),
200        };
201
202        if intent.movement != 0.0 {
203            last_move_dir.0 = intent.movement.signum();
204        }
205        apply_resolved_facing(&mut facing, locked_facing, intent.facing);
206    }
207}

209fn apply_resolved_facing(
210    facing: &mut Facing,
211    locked_facing: Option<Facing>,
212    intent_facing: Option<Facing>,
213) {
214    if let Some(locked_facing) = locked_facing {
215        *facing = locked_facing;
216        return;
217    }
218
219    if let Some(intent_facing) = intent_facing {
220        *facing = intent_facing;
221    }
222}

224fn fallback_horizontal_movement(
225    previous_intent: HorizontalInputIntent,
226    last_move_dir: f32,
227    axis_movement: f32,
228    facing: Facing,
229) -> f32 {
230    [previous_intent.movement, last_move_dir, axis_movement]
231        .into_iter()
232        .find(|movement| *movement != 0.0)
233        .map(f32::signum)
234        .unwrap_or_else(|| facing.direction())
235}
236
237fn both_directions_intent(
238    left_just_pressed: bool,
239    right_just_pressed: bool,
240    fallback_movement: f32,
241    fallback_facing: Facing,
242) -> HorizontalInputIntent {
243    match (left_just_pressed, right_just_pressed) {
244        (true, false) => HorizontalInputIntent {
245            movement: 1.0,
246            facing: Some(Facing::Left),
247        },
248        (false, true) => HorizontalInputIntent {
249            movement: -1.0,
250            facing: Some(Facing::Right),
251        },
252        _ => HorizontalInputIntent {
253            movement: fallback_movement,
254            facing: Some(fallback_facing),
255        },
256    }
257}
258
259fn axis_horizontal_intent(axis_movement: f32) -> HorizontalInputIntent {
260    HorizontalInputIntent {
261        movement: axis_movement,
262        facing: facing_from_axis(axis_movement),
263    }
264}
265
266fn facing_from_axis(axis_movement: f32) -> Option<Facing> {
267    match axis_movement.total_cmp(&0.0) {
268        std::cmp::Ordering::Greater => Some(Facing::Right),
269        std::cmp::Ordering::Less => Some(Facing::Left),
270        std::cmp::Ordering::Equal => None,
271    }
272}

274fn is_pressed(keyboard: &ButtonInput<KeyCode>, keys: &[KeyCode]) -> bool {
275    keys.iter().copied().any(|key| keyboard.pressed(key))
276}
277
278fn is_just_pressed(keyboard: &ButtonInput<KeyCode>, keys: &[KeyCode]) -> bool {
279    keys.iter().copied().any(|key| keyboard.just_pressed(key))
