17const WIDTH: i32 = 142;
18const MARGIN: i32 = 7;
19const FIRST_STAT: i32 = 19;
20const STAT_ROW: i32 = 11;
21const FIRST_BUTTON: i32 = 46;
22const BUTTON_HEIGHT: i32 = 17;
23const HINT_TOP: i32 = FIRST_BUTTON + GameOverAction::ALL.len() as i32 * BUTTON_HEIGHT + 9;
24const HEIGHT: i32 = HINT_TOP + 13;
25const ACCENT: Accent = Accent { bright: Color::srgb(0.86, 0.53, 0.50), dim: Color::srgb(0.34, 0.17, 0.19) };
26/// The screen behind the menu darkens to this opacity over two seconds.
27const FADE_OPACITY: f32 = 0.77;
28const FADE_SECONDS: f32 = 2.0;

30/// A plugin that handles when the player has a game over
31pub struct GameOverPlugin;
32
33impl Plugin for GameOverPlugin {
34    fn build(&self, app: &mut App) {
35        app.add_systems(
36            GameTickUpdate,
37            on_game_over
38                .run_if(in_state(GameState::Playing))
39                .after(PlayerStateSet::Animation)
40                .run_if(in_state(AppState::InGame))
41                .run_if(resource_added::<GameOver>),
42        );
43        app.add_systems(
44            PreUpdate,
45            handle_game_over_input
46                .after(UiInputSet::Menu)
47                .run_if(in_state(AppState::InGame).and_then(resource_exists::<GameOver>).and_then(resource_exists::<GameOverMenu>)),
48        );
49        app.add_systems(
50            Update,
51            (
52                update_fade_in,
53                present_game_over_menu
54                    .after(crate::ui::sync_ui_scale)
55                    .run_if(in_state(AppState::InGame).and_then(resource_exists::<GameOver>).and_then(resource_exists::<GameOverMenu>)),
56            ),
57        );
58    }
59}

61/// Fades the screen to a target opacity using master clock seconds.
62#[derive(Component)]
63#[component(storage = "SparseSet")]
64pub struct FadeIn {
65    progress: f32,
66}
67
68pub fn update_fade_in(
69    mut commands: Commands,
70    time: Res<Time<Real>>,
71    mut query: Query<(
72        Entity,
73        &mut BackgroundColor,
74        &mut FadeIn,
75    )>,
76) {
77    for (entity, mut bg_color, mut fade_in) in query.iter_mut() {
78        fade_in.progress += time.delta_secs() / FADE_SECONDS;
79        fade_in.progress = fade_in.progress.clamp(0.0, 1.0);
80        bg_color.0.set_alpha(fade_in.progress * FADE_OPACITY);
81        if fade_in.progress >= 1.0 {
82            commands.entity(entity).remove::<FadeIn>();
83        }
84    }
85}

104#[derive(Copy, Clone, Debug, Eq, PartialEq)]
105enum GameOverAction {
106    NewExpedition,
107    AbandonPlanet,
108}
109
110impl GameOverAction {
111    const ALL: [Self; 2] = [Self::NewExpedition, Self::AbandonPlanet];
112
113    fn label(self) -> &'static str {
114        match self {
115            Self::NewExpedition => "NEW EXPEDITION",
116            Self::AbandonPlanet => "ABANDON PLANET",
117        }
118    }
119}

121pub fn on_game_over(
122    q_cam: Query<Entity, With<MainCamera>>,
123    mut commands: Commands,
124    mut kill_count: ResMut<KillCount>,
125    time: Res<GameTime>,
126) {
127    let Ok(cam_e) = q_cam.single() else {
128        return;
129    };
130
131    commands.spawn((
132        Name::new("GameOverFade"),
133        Node {
134            width: Val::Percent(100.0),
135            height: Val::Percent(100.0),
136            ..default()
137        },
138        BackgroundColor(Color::srgba_u8(0, 0, 0, 0)),
139        // This ensures we draw the ui above all other uis
140        GlobalZIndex(i32::MAX - 1000),
141        FadeIn { progress: 0.0 },
142        TargetCamera(cam_e),
143        StateDespawnMarker,
144        GameOverUi,
145    ));
146
147    let score = (f64::from(kill_count.0) / time.time_in_seconds().max(f64::EPSILON)) * 100.0;
148    commands.insert_resource(GameOverMenu { kills: kill_count.0, score: score as u32, ..default() });
149    kill_count.0 = 0;
150}

152#[expect(clippy::too_many_arguments, reason = "The death menu translates frame input into app lifecycle commands.")]
153fn handle_game_over_input(
154    mut commands: Commands,
155    keys: Res<ButtonInput<KeyCode>>,
156    mouse: Res<ButtonInput<MouseButton>>,
157    pause_menu: Res<MenuState>,
158    mut menu: ResMut<GameOverMenu>,
159    grid: UiPixelGrid,
160    q_ui: Query<Entity, With<GameOverUi>>,
161    mut next_app_state: ResMut<NextState<AppState>>,
162    mut next_game_state: ResMut<NextState<GameState>>,
163) {
164    if pause_menu.blocks_panel_input() {
165        return;
166    }
167    let last = GameOverAction::ALL.len() - 1;
168    menu.selected = menu.selected.saturating_add_signed(vertical_delta(&keys) as isize).min(last);
169    let clicked = game_over_pointer(&grid, &menu).filter(|_| mouse.just_pressed(MouseButton::Left));
170    let confirmed = keys.any_just_pressed([KeyCode::KeyF, KeyCode::Enter, KeyCode::Space]).then_some(menu.selected);
171    let Some(index) = clicked.or(confirmed) else { return };
172
173    for entity in &q_ui {
174        commands.entity(entity).despawn();
175    }
176    commands.remove_resource::<GameOver>();
177    commands.remove_resource::<GameOverMenu>();
178    match GameOverAction::ALL[index] {
179        GameOverAction::NewExpedition => {
180            commands.insert_resource(SnapCameraNext);
181            next_game_state.set(GameState::Playing);
182        },
183        GameOverAction::AbandonPlanet => next_app_state.set(AppState::MainMenu),
184    }
185}

187/// The button under the cursor, if any.
188fn game_over_pointer(grid: &UiPixelGrid, menu: &GameOverMenu) -> Option<usize> {
189    let cursor = grid.cursor_art()?;
190    let available = grid.available_art()?;
191    let shown = UVec2::new(WIDTH as u32, HEIGHT as u32).min(available);
192    let origin = ((available - shown) / 2).as_vec2();
193    let point = cursor - origin + menu.scroll.offset.as_vec2();
194    if point.x < MARGIN as f32 || point.x >= (WIDTH - MARGIN) as f32 || point.y < FIRST_BUTTON as f32 {
195        return None;
196    }
197    let index = (point.y as i32 - FIRST_BUTTON) / BUTTON_HEIGHT;
198    (index < GameOverAction::ALL.len() as i32).then_some(index as usize)
199}

201#[expect(clippy::too_many_arguments, reason = "The death menu publishes its raster on the current UI grid.")]
202fn present_game_over_menu(
203    mut commands: Commands,
204    mut menu: ResMut<GameOverMenu>,
205    keys: Res<ButtonInput<KeyCode>>,
206    mut library: ResMut<ArtLibrary>,
207    server: Res<AssetServer>,
208    mut images: ResMut<Assets<Image>>,
209    mut panel: Query<(Entity, &ImageNode, &mut Node), With<GameOverUi>>,
210    grid: UiPixelGrid,
211) {
212    library.load(&server, [FONT_IMAGE_PATH]);
213    let raster = draw_game_over(&menu, &library.art(&images));
214    let focus = button_rect(menu.selected);
215    let content = grid.scrollable(raster, &mut menu.scroll, &keys, Some(focus), Vec2::splat(0.5));
216    show_raster(&mut commands, &mut images, panel.single_mut().ok(), content, |image, node| {
217        (Name::new("GameOverMenu"), GameOverUi, image, node, GlobalZIndex(i32::MAX - 999), StateDespawnMarker)
218    });
219}

221fn button_rect(index: usize) -> IRect {
222    rect(MARGIN, FIRST_BUTTON + index as i32 * BUTTON_HEIGHT, WIDTH - 2 * MARGIN, BUTTON_HEIGHT - 2)
223}
224
225fn draw_game_over(menu: &GameOverMenu, art: &Art) -> Raster {
226    let mut canvas = Canvas::new(UVec2::new(WIDTH as u32, HEIGHT as u32), art);
227    draw_frame(&mut canvas, IVec2::new(WIDTH, HEIGHT), ACCENT, &[14, FIRST_BUTTON - 5, HINT_TOP - 5]);
228    let title = "GAME OVER";
229    canvas.text(IVec2::new((WIDTH - text_width(title)) / 2, 4), title, ACCENT.bright);
230    for (index, (label, value)) in [("KILLS", menu.kills), ("SCORE", menu.score)].into_iter().enumerate() {
231        let y = FIRST_STAT + index as i32 * STAT_ROW;
232        canvas.text(IVec2::new(MARGIN, y), label, MUTED);
233        let value = value.to_string();
234        canvas.text(IVec2::new(WIDTH - MARGIN - text_width(&value), y), &value, TEXT);
235    }
236    for (index, action) in GameOverAction::ALL.into_iter().enumerate() {
237        let button = button_rect(index);
238        let selected = index == menu.selected;
239        canvas.plate(button, 1, if selected { ACCENT.dim } else { WELL_FILL }, ACCENT.dim);
240        let label = action.label();
241        canvas.text(IVec2::new((WIDTH - text_width(label)) / 2, button.min.y + 4), label, if selected { TEXT } else { MUTED });
242    }
243    key_hint(&mut canvas, IVec2::new(MARGIN, HINT_TOP), "F", "SELECT", ACCENT.bright);
244    canvas.finish()
245}

258    fn confirm(selected: usize) -> World {
259        use bevy::ecs::system::RunSystemOnce;
260        let mut world = World::new();
261        let mut keys = ButtonInput::<KeyCode>::default();
262        keys.press(KeyCode::KeyF);
263        world.insert_resource(keys);
264        world.init_resource::<ButtonInput<MouseButton>>();
265        world.init_resource::<MenuState>();
266        world.init_resource::<crate::camera::GameplayPixelScale>();
267        world.init_resource::<bevy::ui::UiScale>();
268        world.init_resource::<NextState<AppState>>();
269        world.init_resource::<NextState<GameState>>();
270        world.insert_resource(GameOver);
271        world.insert_resource(GameOverMenu { selected, ..default() });
272        world.spawn(GameOverUi);
273        world.run_system_once(handle_game_over_input).unwrap();
274        world
275    }

277    #[test]
278    fn a_new_expedition_resumes_play_and_closes_the_menu() {
279        let mut world = confirm(0);

281        assert!(!world.contains_resource::<GameOverMenu>());

284        assert!(matches!(world.resource::<NextState<AppState>>(), NextState::Unchanged));
285        assert_eq!(world.query_filtered::<(), With<GameOverUi>>().iter(&world).count(), 0);
286    }
