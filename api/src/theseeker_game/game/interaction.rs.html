26impl Plugin for InteractionPlugin {
27    fn build(&self, app: &mut App) {

29        app.add_systems(GameTickUpdate, (

37fn update_proximity(
38    mut commands: Commands,
39    players: Query<(&GlobalTransform, &Collider), With<Player>>,
40    objects: Query<(Entity, &GlobalTransform, &Collider, Option<&NpcKind>, Option<&Gent>),
41        Or<(With<NpcKind>, With<WorldObjectKind>)>>,
42    mut animations: Query<&mut ScriptPlayer<SpriteAnimation>>,
43) {
44    let player = players.single().ok();
45    for (entity, transform, collider, npc, gent) in &objects {
46        let nearby = player.is_some_and(|(player_transform, player_collider)| {
47            colliders_in_range(player_transform.translation().truncate(), player_collider,
48                transform.translation().truncate(), collider)
49        });
50        if nearby {
51            commands.entity(entity).insert(InInteractionRange);
52        } else {
53            commands.entity(entity).remove::<InInteractionRange>();
54        }
55        if npc != Some(&NpcKind::Vagrant) { continue; }
56        let Some(gent) = gent else { continue; };
57        let Ok(mut animation) = animations.get_mut(gent.e_gfx) else { continue; };
58        animation.set_slot("PlayerNearby", nearby);
59    }
60}

62fn colliders_in_range(player: Vec2, player_collider: &Collider, object: Vec2, collider: &Collider) -> bool {
63    let half_extents = |collider: &Collider| collider.cuboid_half_extents()
64        .unwrap_or_else(|| collider.half_extents());
65    let delta = (player - object).abs();
66    let extent = half_extents(player_collider) + half_extents(collider);
67    delta.x <= extent.x && delta.y <= extent.y
68}

70fn select_focus(
71    players: Query<&GlobalTransform, With<Player>>,
72    objects: Query<(Entity, &GlobalTransform, &WorldObjectKind), With<InInteractionRange>>,
73    mut focus: ResMut<InteractionFocus>,
74) {
75    let player = players.single().ok().map(|player| player.translation().truncate());
76    focus.0 = player.and_then(|player| {
77        objects
78            .iter()
79            .filter(|(_, _, kind)| kind.interior_kind().is_some())
80            .map(|(entity, transform, kind)| (entity, *kind, transform.translation().truncate().distance_squared(player)))
81            .min_by(|(.., a), (.., b)| a.total_cmp(b))
82            .map(|(entity, kind, _)| (entity, kind))
83    });
84}

86fn activate_interaction(
87    mut commands: Commands,
88    players: Query<(Entity, &ActionState<PlayerAction>, &GlobalTransform),
89        (With<Player>, Without<crate::game::spikes::SpikeRespawnPending>)>,
90    focus: Res<InteractionFocus>,
91) {
92    let Ok((player, actions, transform)) = players.single() else { return; };
93    let Some((shelter, kind)) = focus.0.filter(|_| actions.just_pressed(&PlayerAction::Interact)) else { return; };
94    let Some(kind) = kind.interior_kind() else { return; };
95    commands.entity(player).insert(PendingTentEntry { shelter, kind, return_position: transform.translation().truncate() });
96}

103    fn proximity_is_inclusive_rectangular_overlap() {
104        let player = Collider::cuboid(2.0, 3.0);
105        let npc = Collider::cuboid(14.0, 14.0);
106        assert!(colliders_in_range(Vec2::ZERO, &player, Vec2::new(16.0, 17.0), &npc));
107        assert!(!colliders_in_range(Vec2::ZERO, &player, Vec2::new(16.01, 17.0), &npc));
