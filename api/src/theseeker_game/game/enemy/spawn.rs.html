170struct SummedAreaTable {
171    width: usize,
172    height: usize,
173    sums: Vec<u32>,
174}
175
176impl SummedAreaTable {
177    #[expect(
178        clippy::excessive_nesting,
179        reason = "A summed-area table is constructed by a fixed two-dimensional scan."
180    )]
181    fn new(width: usize, height: usize, cells: &[bool]) -> Self {
182        let stride = width + 1;
183        let mut sums = vec![0; stride * (height + 1)];
184        for y in 0..height {
185            let mut row_sum = 0u32;
186            for x in 0..width {
187                row_sum += u32::from(cells[y * width + x]);
188                let dst = (y + 1) * stride + x + 1;
189                sums[dst] = sums[y * stride + x + 1] + row_sum;
190            }
191        }
192        Self {
193            width,
194            height,
195            sums,
196        }
197    }
198
199    fn count_inclusive(
200        &self,
201        min_x: i32,
202        min_y: i32,
203        max_x: i32,
204        max_y: i32,
205    ) -> Option<u32> {
206        if min_x < 0
207            || min_y < 0
208            || max_x < min_x
209            || max_y < min_y
210            || max_x >= self.width as i32
211            || max_y >= self.height as i32
212        {
213            return None;
214        }
215
216        let stride = self.width + 1;
217        let x0 = min_x as usize;
218        let y0 = min_y as usize;
219        let x1 = max_x as usize + 1;
220        let y1 = max_y as usize + 1;
221        Some(
222            self.sums[y1 * stride + x1]
223                + self.sums[y0 * stride + x0]
224                - self.sums[y0 * stride + x1]
225                - self.sums[y1 * stride + x0],
226        )
227    }
228
229    fn rect_is_empty(
230        &self,
231        min_x: i32,
232        min_y: i32,
233        max_x: i32,
234        max_y: i32,
235    ) -> bool {
236        self.count_inclusive(min_x, min_y, max_x, max_y) == Some(0)
237    }
238}

240struct SpawnCatalogGrid {
241    width: usize,
242    height: usize,
243    tile_size: f32,
244    blocked: SummedAreaTable,
245    safe_ground: SummedAreaTable,
246    protected: Vec<bool>,
247    biomes: Vec<BiomeKind>,
248}
249
250impl SpawnCatalogGrid {
251    fn index(&self, x: i32, y: i32) -> Option<usize> {
252        if x < 0 || y < 0 || x >= self.width as i32 || y >= self.height as i32
253        {
254            return None;
255        }
256        Some(y as usize * self.width + x as usize)
257    }
258
259    fn is_protected(&self, x: i32, y: i32) -> bool {
260        self.index(x, y)
261            .and_then(|index| self.protected.get(index))
262            .copied()
263            .unwrap_or(true)
264    }
265
266    fn biome_at(&self, x: i32, y: i32) -> BiomeKind {
267        self.index(x, y)
268            .and_then(|index| self.biomes.get(index))
269            .copied()
270            .unwrap_or_default()
271    }
272
273    fn world_position(&self, x: i32, y: i32) -> Vec2 {
274        Vec2::new(
275            (x as f32 + 0.5) * self.tile_size,
276            (self.height as f32 - y as f32 - 0.5) * self.tile_size,
277        )
278    }
279}

288#[expect(
289    clippy::too_many_arguments,
290    reason = "Catalog construction reads independent finalized level resources."
291)]
292pub(super) fn build_enemy_spawn_catalog(
293    mut commands: Commands,
294    existing: Option<Res<EnemySpawnCatalog>>,
295    compiled: Option<Res<CompiledLevel>>,
296    biomes: Option<Res<LevelBiomeMask>>,
297    occupancy: Option<Res<LevelChunkOccupancyMask>>,
298    ground_ready: Option<Res<GenerationBuild>>,
299    collision_world: Res<TileCollisionWorld>,
300    level_seed: Res<crate::game::actors::LevelSeed>,
301) {
302    if existing.is_some()
303        || !ground_ready.as_ref().is_some_and(|ready| ready.ground_ready())
304        || !collision_world.is_ready()
305    {
306        return;
307    }
308    let (Some(compiled), Some(biomes)) = (compiled, biomes) else {
309        return;
310    };
311    if !catalog_inputs_match(&compiled, &biomes, &collision_world) {
312        return;
313    }
314
315    let Some(grid) = build_catalog_grid(
316        &compiled,
317        &biomes,
318        occupancy.as_deref(),
319        &collision_world,
320    ) else {
321        return;
322    };
323    let Some(player_chunk) = player_spawn_chunk(&compiled) else {
324        warn!("Enemy spawn catalog requires a player spawn marker");
325        return;
326    };
327
328    let raw_candidates = collect_raw_candidates(&grid, player_chunk, level_seed.0);
329    let candidates = thin_candidates(raw_candidates);
330    let catalog = group_candidates_by_region(
331        candidates,
332        &grid,
333        player_chunk,
334        compiled.width,
335        compiled.height,
336    );
337    let candidate_count = catalog
338        .regions
339        .values()
340        .map(|region| region.candidates.len())
341        .sum::<usize>();
342    info!(
343        "Built enemy spawn catalog: regions={}, candidates={candidate_count}",
344        catalog.regions.len()
345    );
346    commands.insert_resource(catalog);
347}

360fn build_catalog_grid(
361    compiled: &CompiledLevel,
362    biomes: &LevelBiomeMask,
363    occupancy: Option<&LevelChunkOccupancyMask>,
364    collision_world: &TileCollisionWorld,
365) -> Option<SpawnCatalogGrid> {
366    let width = compiled.width as usize;
367    let height = compiled.height as usize;
368    let len = width.checked_mul(height)?;
369    if compiled.spike_tiles.len() != len {
370        warn!("Enemy spawn catalog terrain masks have mismatched dimensions");
371        return None;
372    }
373
374    let occupancy = match occupancy {
375        Some(mask) if occupancy_mask_matches(mask, compiled.width, compiled.height) => {
376            Some(mask)
377        },
378        Some(_) => {
379            warn!("Enemy spawn catalog chunk occupancy dimensions do not match the level");
380            return None;
381        },
382        None => None,
383    };
384    let mut blocked = vec![false; len];
385    let mut safe_ground = vec![false; len];
386    let accessible = crate::levelgen::placement_area(compiled, occupancy);
387    for y in 0..height {
388        for x in 0..width {
389            let index = y * width + x;
390            let occupied = tile_is_in_occupied_chunk(occupancy, x as u32, y as u32);
391            let solid = collision_world.tile_solid_info(x as i32, y as i32);
392            let has_solid = solid.is_some();
393            let terrain = solid.is_some_and(|info| info.terrain);
394            blocked[index] = !occupied || !accessible[index] || has_solid || compiled.spike_tiles[index];
395            safe_ground[index] = occupied
396                && accessible[index]
397                && terrain
398                && !compiled.spike_tiles[index];
399        }
400    }
401
402    let mut protected = vec![false; len];
403    for spawn in &compiled.spawns {
404        mark_protected_world_position(
405            &mut protected,
406            width,
407            height,
408            collision_world.tile_size,
409            spawn.position,
410        );
411    }
412
413    Some(SpawnCatalogGrid {
414        width,
415        height,
416        tile_size: collision_world.tile_size,
417        blocked: SummedAreaTable::new(width, height, &blocked),
418        safe_ground: SummedAreaTable::new(width, height, &safe_ground),
419        protected,
420        biomes: biomes.cells.clone(),
421    })
422}

434fn tile_is_in_occupied_chunk(
435    occupancy: Option<&LevelChunkOccupancyMask>,
436    tile_x: u32,
437    tile_y: u32,
438) -> bool {
439    let Some(mask) = occupancy else {
440        return true;
441    };
442    let chunk_x = tile_x / mapgen::CHUNK_IMAGE_SIZE;
443    let chunk_y = tile_y / mapgen::CHUNK_IMAGE_SIZE;
444    if chunk_x >= mask.chunks_w || chunk_y >= mask.chunks_h {
445        return false;
446    }
447    mask.occupied[(chunk_y * mask.chunks_w + chunk_x) as usize]
448}

450#[expect(
451    clippy::excessive_nesting,
452    reason = "Protected anchor masks are filled by a bounded square tile scan."
453)]
454fn mark_protected_world_position(
455    protected: &mut [bool],
456    width: usize,
457    height: usize,
458    tile_size: f32,
459    position: Vec2,
460) {
461    let center_x = (position.x / tile_size).floor() as i32;
462    let center_y = (height as f32 - 1.0 - (position.y / tile_size).floor()) as i32;
463    for y in (center_y - PROTECTED_ANCHOR_RADIUS_TILES)
464        ..=(center_y + PROTECTED_ANCHOR_RADIUS_TILES)
465    {
466        for x in (center_x - PROTECTED_ANCHOR_RADIUS_TILES)
467            ..=(center_x + PROTECTED_ANCHOR_RADIUS_TILES)
468        {
469            if x < 0 || y < 0 || x >= width as i32 || y >= height as i32 {
470                continue;
471            }
472            protected[y as usize * width + x as usize] = true;
473        }
474    }
475}

489fn collect_raw_candidates(
490    grid: &SpawnCatalogGrid,
491    player_chunk: IVec2,
492    level_seed: u32,
493) -> Vec<RawCandidate> {
494    let mut candidates = Vec::new();
495    for y in 0..grid.height as i32 {
496        for x in 0..grid.width as i32 {
497            let chunk = tile_chunk(x, y);
498            if chunk == player_chunk || grid.is_protected(x, y) {
499                continue;
500            }
501            let habitats = candidate_habitats(grid, x, y);
502            if habitats.is_empty() {
503                continue;
504            }
505            candidates.push(RawCandidate {
506                tile: IVec2::new(x, y),
507                habitats,
508                hash: position_hash(level_seed, x, y, 0),
509            });
510        }
511    }
512    candidates
513}

515fn candidate_habitats(grid: &SpawnCatalogGrid, x: i32, y: i32) -> HabitatMask {
516    let mut habitats = HabitatMask::default();
517    let radius = air_clearance_radius(grid.tile_size);
518    if grid.blocked.rect_is_empty(x - radius, y - radius, x + radius, y + radius) {
519        habitats.insert(SpawnHabitat::Air);
520    }
521    if ground_candidate_is_valid(grid, x, y, GROUND_SMALL_PROFILE) {
522        habitats.insert(SpawnHabitat::GroundSmall);
523    }
524    if ground_candidate_is_valid(grid, x, y, GROUND_LARGE_PROFILE) {
525        habitats.insert(SpawnHabitat::GroundLarge);
526    }
527    if ground_candidate_is_valid(grid, x, y, GROUND_TALL_PROFILE) {
528        habitats.insert(SpawnHabitat::GroundTall);
529    }
530    habitats
531}

537fn ground_candidate_is_valid(
538    grid: &SpawnCatalogGrid,
539    x: i32,
540    y: i32,
541    profile: GroundProfile,
542) -> bool {
543    if !grid.blocked.rect_is_empty(
544        x - profile.side_clearance,
545        y - profile.headroom,
546        x + profile.side_clearance,
547        y,
548    ) {
549        return false;
550    }
551
552    let support_y = y + 1;
553    let support_left = (profile.support_width - 1) / 2;
554    let support_right = profile.support_width - 1 - support_left;
555    grid.safe_ground.count_inclusive(
556        x - support_left,
557        support_y,
558        x + support_right,
559        support_y,
560    ) == Some(profile.support_width as u32)
561}

563fn thin_candidates(mut raw: Vec<RawCandidate>) -> Vec<RawCandidate> {
564    raw.sort_unstable_by_key(|candidate| {
565        (candidate.habitats.contains(SpawnHabitat::Air), candidate.hash, candidate.tile.y, candidate.tile.x)
566    });
567    let mut occupied_profile_cells: [HashSet<IVec2>; 4] =
568        array::from_fn(|_| HashSet::new());
569    let mut proximity_cells: HashMap<IVec2, Vec<IVec2>> = HashMap::new();
570    let mut accepted = Vec::new();
571
572    for mut candidate in raw {
573        candidate.habitats = available_habitats(candidate, &occupied_profile_cells);
574        if candidate.habitats.is_empty()
575            || candidate_is_too_close(candidate.tile, &proximity_cells)
576        {
577            continue;
578        }
579
580        mark_profile_cells(candidate, &mut occupied_profile_cells);
581        proximity_cells
582            .entry(proximity_cell(candidate.tile))
583            .or_default()
584            .push(candidate.tile);
585        accepted.push(candidate);
586    }
587    accepted
588}

590fn mark_profile_cells(
591    candidate: RawCandidate,
592    occupied_profile_cells: &mut [HashSet<IVec2>; 4],
593) {
594    for habitat in SpawnHabitat::ALL {
595        if candidate.habitats.contains(habitat) {
596            occupied_profile_cells[habitat.index()]
597                .insert(profile_cell(candidate.tile, habitat));
598        }
599    }
600}

602fn available_habitats(
603    candidate: RawCandidate,
604    occupied_profile_cells: &[HashSet<IVec2>; 4],
605) -> HabitatMask {
606    let mut available = HabitatMask::default();
607    for habitat in SpawnHabitat::ALL {
608        if candidate.habitats.contains(habitat)
609            && !occupied_profile_cells[habitat.index()]
610                .contains(&profile_cell(candidate.tile, habitat))
611        {
612            available.insert(habitat);
613        }
614    }
615    available
616}

630fn candidate_is_too_close(
631    tile: IVec2,
632    proximity_cells: &HashMap<IVec2, Vec<IVec2>>,
633) -> bool {
634    let center = proximity_cell(tile);
635    (-1..=1)
636        .flat_map(|offset_y| {
637            (-1..=1).map(move |offset_x| {
638                center + IVec2::new(offset_x, offset_y)
639            })
640        })
641        .filter_map(|cell| proximity_cells.get(&cell))
642        .flatten()
643        .any(|neighbor| {
644            tile.distance_squared(*neighbor)
645                < MIN_CANDIDATE_DISTANCE_TILES.pow(2)
646        })
647}

649fn group_candidates_by_region(
650    candidates: Vec<RawCandidate>,
651    grid: &SpawnCatalogGrid,
652    player_chunk: IVec2,
653    width: u32,
654    height: u32,
655) -> EnemySpawnCatalog {
656    let mut regions: HashMap<IVec2, RegionCatalog> = HashMap::new();
657    for (id, raw) in candidates.into_iter().enumerate() {
658        let region_key = tile_chunk(raw.tile.x, raw.tile.y);
659        let biome_y = if raw.habitats.contains(SpawnHabitat::GroundSmall)
660            || raw.habitats.contains(SpawnHabitat::GroundLarge)
661            || raw.habitats.contains(SpawnHabitat::GroundTall)
662        {
663            raw.tile.y + 1
664        } else {
665            raw.tile.y
666        };
667        let region = regions.entry(region_key).or_insert_with(|| RegionCatalog {
668            depth: region_key.y.saturating_sub(player_chunk.y).max(0) as u32,
669            candidates: Vec::new(),
670        });
671        region.candidates.push(SpawnCandidate {
672            id: u32::try_from(id).unwrap_or(u32::MAX),
673            tile: raw.tile,
674            world_position: grid.world_position(raw.tile.x, raw.tile.y),
675            biome: grid.biome_at(raw.tile.x, biome_y),
676            habitats: raw.habitats,
677        });
678    }
679    EnemySpawnCatalog {
680        tile_size: grid.tile_size,
681        width,
682        height,
683        regions,
684    }
685}

687fn tile_chunk(tile_x: i32, tile_y: i32) -> IVec2 {
688    let chunk_size = mapgen::CHUNK_IMAGE_SIZE as i32;
689    IVec2::new(
690        tile_x.div_euclid(chunk_size),
691        tile_y.div_euclid(chunk_size),
692    )
693}

695fn position_hash(seed: u32, x: i32, y: i32, salt: u32) -> u64 {
696    let mut value = u64::from(seed)
697        ^ (x as u32 as u64).wrapping_mul(0x9E37_79B1)
698        ^ (y as u32 as u64).wrapping_mul(0x85EB_CA77)
699        ^ u64::from(salt).wrapping_mul(0xC2B2_AE3D);
700    value ^= value >> 30;
701    value = value.wrapping_mul(0xBF58_476D_1CE4_E5B9);
702    value ^= value >> 27;
703    value = value.wrapping_mul(0x94D0_49BB_1331_11EB);
704    value ^ (value >> 31)
705}

707#[expect(
708    clippy::too_many_arguments,
709    reason = "The director system consumes independent world, progression, and camera state."
710)]
711pub(super) fn run_enemy_spawn_director(
712    mut commands: Commands,
713    catalog: Option<Res<EnemySpawnCatalog>>,
714    compiled: Option<Res<CompiledLevel>>,
715    collision: Res<TileCollisionWorld>,
716    actors: Res<ActorCatalog>,
717    mut runtime: ResMut<EnemySpawnRuntime>,
718    game_time: Res<GameTime>,
719    level_seed: Res<crate::game::actors::LevelSeed>,
720    kill_count: Res<KillCount>,
721    pixel_scale: Res<GameplayPixelScale>,
722    player_q: Query<&Transform, (With<Player>, Without<Dead>)>,
723    camera_q: Query<&GlobalTransform, With<MainCamera>>,
724    actor_q: Query<
725        (&Transform, Option<&Gent>, Option<&Dead>),
726        With<SpawnDirectorOwned>,
727    >,
728) {
729    if !game_time.tick().is_multiple_of(DIRECTOR_INTERVAL_TICKS) {
730        return;
731    }
732    let (Some(catalog), Some(compiled)) = (catalog, compiled) else {
733        return;
734    };
735    let Ok(player_transform) = player_q.single() else {
736        return;
737    };
738    let Ok(camera_transform) = camera_q.single() else {
739        return;
740    };
741    let visible_size = pixel_scale.visible_game_size();
742    if visible_size.x <= 0.0 || visible_size.y <= 0.0 {
743        return;
744    }
745
746    let player_position = player_transform.translation.xy();
747    let player_region = world_region(&catalog, player_position);
748    let initial_player_region =
749        *runtime.initial_player_region.get_or_insert(player_region);
750    let camera_position = camera_transform.translation().xy();
751    let view_half_size = visible_size * 0.5;
752    update_existing_spawn_slots(
753        &mut commands,
754        &mut runtime,
755        &actor_q,
756        player_region,
757        player_position,
758        camera_position,
759        view_half_size,
760    );
761
762    let immunity = PlanetImmunityLevel::from_kills(**kill_count).get();
763    let active_regions = catalog
764        .regions
765        .iter()
766        .filter(|(region, _)| {
767            chunk_distance(**region, player_region) <= ACTIVE_CHUNK_RADIUS
768        })
769        .map(|(region, data)| (*region, data.depth))
770        .collect::<Vec<_>>();
771
772    for (region_key, depth) in active_regions {
773        let target = target_population(immunity, depth);
774        let initial_region =
775            chunk_distance(region_key, initial_player_region) <= ACTIVE_CHUNK_RADIUS;
776        let slots = prepare_region_for_spawn(&mut runtime, region_key, target,
777            initial_region, immunity == PlanetImmunityLevel::MAX);
778        let region_families = runtime.regions[&region_key].slots.iter()
779            .filter(|slot| slot.actor.is_some()).filter_map(|slot| slot.family).collect::<Vec<_>>();
780        let context = SpawnContext {
781            catalog: &catalog, compiled: &compiled, collision: &collision, actors: &actors,
782            leased_candidates: &runtime.leased_candidates, region_families: &region_families,
783            level_seed: level_seed.0, player_position, camera_position, view_half_size,
784        };
785        let Some((slot_index, selection)) = select_ready_spawn(&context, &runtime, region_key, slots) else { continue; };
786        let entity = spawn_selection(&mut commands, selection, depth, level_seed.0);
787        commands.entity(entity).insert(SpawnDirectorOwned {
788            region: region_key,
789            slot: slot_index,
790            candidate_id: selection.candidate.id,
791        });
792        runtime.leased_candidates.insert(selection.candidate.id);
793        let region = runtime.regions.get_mut(&region_key).expect("region exists");
794        let slot = &mut region.slots[slot_index];
795        slot.actor = Some(entity);
796        slot.family = Some(selection.family);
797        slot.candidate_id = Some(selection.candidate.id);
798        slot.generation = slot.generation.wrapping_add(1);
799    }
800}

861fn update_spawn_slot(
862    commands: &mut Commands,
863    slot: &mut DirectorSlot,
864    leased_candidates: &mut HashSet<u32>,
865    actor_q: &Query<
866        (&Transform, Option<&Gent>, Option<&Dead>),
867        With<SpawnDirectorOwned>,
868    >,
869    stream: StreamContext,
870) {
871    slot.cooldown_ticks = slot
872        .cooldown_ticks
873        .saturating_sub(DIRECTOR_INTERVAL_TICKS as u32);
874    let Some(actor) = slot.actor else { return; };
875    let Ok((transform, gent, dead)) = actor_q.get(actor) else {
876        release_slot(slot, leased_candidates, 0);
877        return;
878    };
879    if dead.is_some() {
880        return;
881    }
882    let actor_position = transform.translation.xy();
883    let may_stream_out = stream.outside_stream
884        && actor_position.distance(stream.player_position)
885            >= STREAM_DESPAWN_DISTANCE
886        && position_is_offscreen(
887            actor_position,
888            stream.camera_position,
889            stream.view_half_size,
890        );
891    if !may_stream_out {
892        return;
893    }
894
895    if let Some(gent) = gent {
896        despawn_enemy_with_gfx(commands, actor, gent);
897    } else {
898        despawn_entity_if_exists(commands, actor);
899    }
900    release_slot(slot, leased_candidates, 0);
901}

903fn release_slot(
904    slot: &mut DirectorSlot,
905    leased_candidates: &mut HashSet<u32>,

907) {

909        leased_candidates.remove(&candidate_id);
910    }
911    slot.actor = None;
912    slot.family = None;
913    slot.cooldown_ticks = cooldown_ticks;
914}

916fn prepare_region_for_spawn(
917    runtime: &mut EnemySpawnRuntime,
918    region_key: IVec2,
919    target: usize,
920    initial_region: bool,
921    escalation: bool,
922) -> [Option<(usize, bool)>; 2] {
923    let region = runtime.regions.entry(region_key).or_default();
924    let initializing_region = !region.initialized;
925    region.initialized = true;
926    // Population contraction cannot discard living actors' leases or their cleanup indexes.
927    let required = (target + usize::from(!escalation)).max(region.slots.len());
928    region.slots.resize_with(required, || DirectorSlot {
929        initial_distance_exemption: initializing_region && initial_region, ..default()
930    });
931    region.slots[VOID_SLOT].initial_distance_exemption = false;
932    if region.slots.iter().filter(|slot| slot.actor.is_some()).count() >= target {
933        return [None, None];
934    }
935    let void = &region.slots[VOID_SLOT];
936    let void_slot = (escalation && void.actor.is_none() && void.cooldown_ticks == 0)
937        .then_some((VOID_SLOT, false));
938    let ordinary = first_ready_ordinary_slot(region, target - usize::from(escalation)).map(|slot| {
939        let distance_exempt = std::mem::take(&mut region.slots[slot].initial_distance_exemption);
940        (slot, distance_exempt)
941    });
942    [void_slot, ordinary]
943}
944
945fn first_ready_ordinary_slot(region: &DirectorRegion, limit: usize) -> Option<usize> {
946    let ready = |(_, slot): &(usize, &DirectorSlot)| slot.actor.is_none() && slot.cooldown_ticks == 0;
947    region.slots.iter().enumerate().skip(1).take(limit)
948        .find(|slot| ready(slot) && slot.1.initial_distance_exemption)
949        .or_else(|| region.slots.iter().enumerate().skip(1).take(limit).find(ready))
950        .map(|(index, _)| index)
951}

953fn world_region(catalog: &EnemySpawnCatalog, position: Vec2) -> IVec2 {
954    let tile_x = (position.x / catalog.tile_size).floor() as i32;
955    let tile_y = (catalog.height as f32
956        - 1.0
957        - (position.y / catalog.tile_size).floor()) as i32;
958    let bounded_x = tile_x.clamp(0, catalog.width.saturating_sub(1) as i32);
959    let bounded_y = tile_y.clamp(0, catalog.height.saturating_sub(1) as i32);
960    tile_chunk(bounded_x, bounded_y)
961}

967fn target_population(immunity: u8, depth: u32) -> usize {
968    let immunity_increase = u32::from(immunity.saturating_sub(1)) / 2;
969    let depth_increase = (depth / 2).min(2);

972}

1057) -> Option<(usize, SpawnSelection)> {
1058    for (slot_index, distance_exempt) in slots.into_iter().flatten() {
1059        let generation = runtime.regions[&region_key].slots[slot_index].generation;
1060        if let Some(selection) = select_spawn(context, region_key, slot_index, generation, distance_exempt) {

1067fn select_spawn(
1068    context: &SpawnContext,
1069    region_key: IVec2,
1070    slot_index: usize,
1071    generation: u32,
1072    distance_exempt: bool,
1073) -> Option<SpawnSelection> {
1074    let region = context.catalog.regions.get(&region_key)?;
1075    let salt = (slot_index as u32).wrapping_mul(0x9E37_79B9).wrapping_add(generation);
1076    let available = roster(region.depth, slot_index == VOID_SLOT)
1077        .filter(|entry| family_fits_region(context, region_key, *entry))
1078        .filter(|entry| region.candidates.iter().any(|candidate| {
1079            spawn_candidate(context, *entry, *candidate, region.depth, distance_exempt).is_some()
1080        })).collect::<Vec<_>>();
1081    let total_weight = available.iter().map(|entry| entry.weight).sum::<u32>();
1082    if total_weight == 0 { return None; }
1083    let roll = position_hash(context.level_seed, region_key.x, region_key.y, salt) % u64::from(total_weight);
1084    let family = weighted_family_at_roll(&available, roll as u32)?;
1085    let entry = available.into_iter().find(|entry| entry.family == family)?;
1086    region.candidates.iter().filter_map(|candidate| {
1087        spawn_candidate(context, entry, *candidate, region.depth, distance_exempt)
1088    }).min_by_key(|selection| position_hash(context.level_seed, selection.candidate.tile.x,
1089        selection.candidate.tile.y, salt ^ selection.candidate.id))
1090}

1092/// A family joins a region only while under its cap, and only where the region is one of its nests.
1093fn family_fits_region(context: &SpawnContext, region_key: IVec2, entry: WeightedFamily) -> bool {
1094    let living = context.region_families.iter().filter(|&&family| family == entry.family).count();
1095    living < entry.region_cap
1096        && (entry.nest_percent >= 100 || region_is_nest(context.level_seed, region_key, entry.family, entry.nest_percent))
1097}

1099fn region_is_nest(seed: u32, region_key: IVec2, family: EnemyFamily, nest_percent: u32) -> bool {
1100    let salt = NEST_SALT ^ (family as u32).wrapping_mul(0x9E37_79B9);
1101    position_hash(seed, region_key.x, region_key.y, salt) % 100 < u64::from(nest_percent)
1102}

1104fn spawn_candidate(
1105    context: &SpawnContext,
1106    entry: WeightedFamily,
1107    candidate: SpawnCandidate,
1108    depth: u32,
1109    distance_exempt: bool,
1110) -> Option<SpawnSelection> {
1111    if depth < entry.min_depth || !candidate.habitats.contains(entry.family.habitat())
1112        || (entry.tundra_only && !candidate.biome.is_tundra())
1113        || context.leased_candidates.contains(&candidate.id)
1114        || !dynamic_candidate_habitat_is_valid(context.compiled, candidate.tile,
1115            entry.family.habitat(), context.catalog.tile_size) { return None; }
1116    let archetype = family_archetype(entry.family, candidate, depth, context.level_seed);
1117    let actor = context.actors.actors.get(archetype)?;
1118    let center = candidate_actor_position(candidate, entry.family.habitat(),
1119        context.catalog.tile_size, actor);
1120    let minimum = if distance_exempt { 0.0 } else {
1121        context.catalog.tile_size * mapgen::CHUNK_IMAGE_SIZE as f32 * MIN_SPAWN_DISTANCE_CHUNKS
1122    };
1123    if !position_is_spawnable(center, context.player_position, minimum * minimum,
1124        context.camera_position, context.view_half_size)
1125        || !spawn_body_is_clear(context, center, actor.body.collider_half_extents)
1126        || (entry.family.habitat() != SpawnHabitat::Air
1127            && !body_has_safe_support(context, center, actor.body.collider_half_extents)) { return None; }
1128    Some(SpawnSelection { family: entry.family, candidate, archetype,
1129        blueprint_position: center - actor.body.spawn_offset.truncate() })
1130}

1132fn candidate_actor_position(candidate: SpawnCandidate, habitat: SpawnHabitat,
1133    tile_size: f32, actor: &ResolvedActor,
1134) -> Vec2 {
1135    if habitat == SpawnHabitat::Air { return candidate.world_position; }
1136    candidate.world_position + Vec2::Y * (actor.body.collider_half_extents.y - tile_size * 0.5)
1137}

1139fn spawn_body_is_clear(context: &SpawnContext, center: Vec2, half: Vec2) -> bool {
1140    let bounds = Rect::from_corners(center - half, center + half);
1141    let world_max = Vec2::new(context.catalog.width as f32, context.catalog.height as f32) * context.catalog.tile_size;
1142    if bounds.min.min_element() < 0.0 || bounds.max.x > world_max.x || bounds.max.y > world_max.y
1143        || context.collision.overlaps_solid(center, half) { return false; }
1144    let Some(range) = context.collision.tile_range_for_rect(bounds) else { return false; };
1145    range.tiles().all(|tile| !context.compiled.spike_tiles[(tile.y as u32 * context.compiled.width + tile.x as u32) as usize])
1146}

1148fn body_has_safe_support(context: &SpawnContext, center: Vec2, half: Vec2) -> bool {
1149    let tile_size = context.catalog.tile_size;
1150    let left = ((center.x - half.x + 0.001) / tile_size).floor() as i32;
1151    let right = ((center.x + half.x - 0.001) / tile_size).floor() as i32;
1152    let row = context.catalog.height as i32 - 1 - ((center.y - half.y - 0.001) / tile_size).floor() as i32;
1153    (left..=right).all(|x| compiled_tile_is_safe_ground(context.compiled, x, row))
1154}

1175fn dynamic_candidate_habitat_is_valid(
1176    compiled: &CompiledLevel,
1177    tile: IVec2,
1178    habitat: SpawnHabitat,
1179    tile_size: f32,
1180) -> bool {
1181    let radius = air_clearance_radius(tile_size);
1182    match habitat {
1183        SpawnHabitat::Air => compiled_rect_is_clear(
1184            compiled,
1185            tile.x - radius,

1189        ),

1192        },
1193        SpawnHabitat::GroundLarge => {
1194            compiled_ground_candidate_is_valid(compiled, tile, GROUND_LARGE_PROFILE)

1202fn compiled_ground_candidate_is_valid(
1203    compiled: &CompiledLevel,
1204    tile: IVec2,
1205    profile: GroundProfile,
1206) -> bool {
1207    if !compiled_rect_is_clear(
1208        compiled,
1209        tile.x - profile.side_clearance,
1210        tile.y - profile.headroom,
1211        tile.x + profile.side_clearance,
1212        tile.y,
1213    ) {
1214        return false;
1215    }
1216
1217    let support_y = tile.y + 1;
1218    let support_left = (profile.support_width - 1) / 2;
1219    let support_right = profile.support_width - 1 - support_left;
1220    (tile.x - support_left..=tile.x + support_right)
1221        .all(|x| compiled_tile_is_safe_ground(compiled, x, support_y))
1222}

1224fn compiled_rect_is_clear(
1225    compiled: &CompiledLevel,
1226    min_x: i32,
1227    min_y: i32,
1228    max_x: i32,
1229    max_y: i32,
1230) -> bool {
1231    if min_x < 0
1232        || min_y < 0
1233        || max_x >= compiled.width as i32
1234        || max_y >= compiled.height as i32
1235    {
1236        return false;
1237    }
1238    (min_y..=max_y).all(|y| {
1239        (min_x..=max_x).all(|x| {
1240            let index = (y as u32 * compiled.width + x as u32) as usize;
1241            !compiled.terrain.get(index).copied().unwrap_or(true)
1242                && !compiled.spike_tiles.get(index).copied().unwrap_or(true)
1243        })
1244    })
1245}

1247fn compiled_tile_is_safe_ground(compiled: &CompiledLevel, x: i32, y: i32) -> bool {
1248    if x < 0 || y < 0 || x >= compiled.width as i32 || y >= compiled.height as i32 {
1249        return false;
1250    }
1251    let index = (y as u32 * compiled.width + x as u32) as usize;
1252    compiled.terrain.get(index).copied().unwrap_or(false)
1253        && !compiled.spike_tiles.get(index).copied().unwrap_or(true)
1254}

1390pub(super) fn release_dead_spawn_leases(
1391    mut commands: Commands,
1392    mut runtime: ResMut<EnemySpawnRuntime>,
1393    game_time: Res<GameTime>,
1394    dead_q: Query<(Entity, &SpawnDirectorOwned), Added<Dead>>,
1395) {

1397    for (entity, lease) in dead_q.iter() {
1398        let Some(region) = runtime.regions.get_mut(&lease.region) else {
1399            continue;
1400        };
1401        let Some(slot) = region.slots.get_mut(lease.slot) else {
1402            continue;
1403        };

1405            || slot.candidate_id != Some(lease.candidate_id)
1406        {
1407            continue;
1408        }
1409        slot.actor = None;
1410        slot.candidate_id = None;

1414    }
