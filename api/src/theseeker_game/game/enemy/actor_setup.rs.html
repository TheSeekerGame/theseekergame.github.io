58/// Builds each new blueprint from its catalog actor.
59pub(super) fn setup_enemy_actor(
60    mut blueprints: Query<(Entity, &mut Transform, &EnemyActorBlueprint), Added<EnemyActorBlueprint>>,
61    mut commands: Commands,
62    catalog: Res<ActorCatalog>,
63    level_seed: Res<LevelSeed>,
64) {
65    for (entity, mut transform, blueprint) in &mut blueprints {
66        let Some(actor) = catalog.actors.get(&blueprint.arch_id) else {
67            error!("Failed to resolve actor definition for archetype: {}", blueprint.arch_id);
68            commands.entity(entity).despawn();
69            continue;
70        };
71        let mut offset = actor.body.spawn_offset;
72        if let Some(spider) = blueprint.spider {
73            offset.z = spider.spawn_z;
74        }
75        transform.translation += offset;
76        assemble_enemy(&mut commands, entity, *transform, blueprint, actor, level_seed.0);
77        commands.entity(entity).remove::<EnemyActorBlueprint>();
78    }
79}

81fn assemble_enemy(
82    commands: &mut Commands,
83    entity: Entity,
84    transform: Transform,
85    blueprint: &EnemyActorBlueprint,
86    actor: &Arc<ResolvedActor>,
87    level_seed: u32,
88) {
89    let graphics = (commands.spawn(()).id(), commands.spawn(()).id());
90    let (body, stats, archetype) = (&actor.body, actor.stats(), &actor.archetype);
91    let collider = Collider::cuboid(body.collider_half_extents.x, body.collider_half_extents.y);
92    commands.entity(entity).insert((
93        Name::new(body.name.clone()),
94        Enemy,
95        Gent { e_gfx: graphics.0, e_effects_gfx: graphics.1 },
96        FoliageInteractor::body(&collider),
97        GentPhysicsBundle { collider, linear_velocity: LinearVelocity(Vec2::ZERO) },
98        theseeker_engine::physics::groups::enemy_body(),
99        Health { current: stats.spawn_hp, max: stats.spawn_hp },
100        spawn_facing(body.facing),
101        StateDespawnMarker,
102        DeathPolicy::from_setup(archetype.death.as_ref()),
103    ));
104    if let Some((right, left)) = &body.facing_slots {
105        commands.entity(entity).insert(FacingSlots::new(right.clone(), left.clone()));
106    }
107    insert_body_runtime(commands, entity, actor, transform.translation.xy());
108    let spider = blueprint.spider.map(|spider| (spider, SpiderJitter::new(entity, level_seed, spider.small)));
109    let jitter = spider.as_ref().map(|(_, jitter)| jitter);
110    commands.entity(entity).insert((ActorBundle::new(actor.clone(), entity, level_seed), cached_stats(actor, jitter)));
111    if stats.dmg_ranged > 0 {
112        commands.entity(entity).insert(ProjectileCache::new());
113    }
114    if let Some(from_frame) = archetype.defense.and_then(|defense| defense.invulnerable_from_frame) {
115        let anim_key = archetype.anim.get("Defense").cloned();
116        commands.entity(entity).insert(DefenseInvulnerabilityTiming { from_frame, anim_key });
117    }
118    if let Some((spider, jitter)) = &spider {
119        insert_spider_traits(commands, entity, spider, jitter, stats.melee_range);
120    }
121    insert_graphics(commands, entity, graphics, transform, actor);
122}

124fn spawn_facing(setup: EnemyFacingSetup) -> Facing {
125    match setup {
126        EnemyFacingSetup::Right => Facing::Right,
127        EnemyFacingSetup::Left => Facing::Left,
128        EnemyFacingSetup::RandomHorizontal => {
129            if rand::rng().random_bool(0.5) { Facing::Right } else { Facing::Left }
130        },
131    }
132}

134/// The body's motor and its standing effects.
135fn insert_body_runtime(commands: &mut Commands, entity: Entity, actor: &ResolvedActor, home: Vec2) {
136    let (archetype, body) = (&actor.archetype, &actor.body);
137    let mut entity = commands.entity(entity);
138    if body.trap_prey {
139        entity.insert(TrapPrey);
140    }
141    match body.controller {
142        EnemySetupController::Flying => {
143            entity.insert(FlyingMotor::new(home, FlyingMotorConfig::from_setup(archetype.flying, actor.stats().walk_speed)))
144        },
145        EnemySetupController::Ground => entity.insert((Navigation::Grounded, GroundMotor, MovementState::default())),
146        EnemySetupController::Jump => {
147            entity.insert((Navigation::Grounded, JumpMotor::new(archetype.jump), MovementState::default()))
148        },
149        EnemySetupController::StationaryTrap => entity.insert((
150            DamageWindow { open: false },
151            StationaryTrapController::new(body.collider_half_extents, archetype.trap),
152        )),
153    };
154    if let Some(regen) = archetype.health_regen {
155        entity.insert(HealthRegen::new(regen.amount, regen.interval_ticks));
156    }
157    if let Some(contact) = archetype.contact_damage {
158        let pushback = contact.pushback.map(|pushback| ContactPushback {
159            horizontal: pushback.horizontal,
160            vertical: pushback.vertical,
161            ticks: pushback.ticks,
162        });
163        entity.insert(ContactDamage::new(contact.damage, contact.cooldown_ticks, pushback));
164    }
165}

167/// Perception stats after a spider's jitter, raised to the body's floors.
168fn cached_stats(actor: &ResolvedActor, jitter: Option<&SpiderJitter>) -> CachedArchetypeStats {
169    let (stats, body) = (actor.stats(), &actor.body);
170    let vision = jitter.map_or(stats.vision_range, |jitter| jitter.vision_range(stats.vision_range));
171    let melee = jitter.map_or(stats.melee_range, |jitter| jitter.melee_range(stats.melee_range));
172    CachedArchetypeStats {
173        vision_range: vision.max(body.min_vision_range),
174        melee_range: melee.max(body.min_melee_range),
175        walk_speed: stats.walk_speed,
176        chase_speed: stats.chase_speed,
177        needs_line_of_sight: body.needs_line_of_sight,
178        reaction_time: jitter.map_or(stats.reaction_time, |jitter| jitter.reaction_time(stats.reaction_time)),
179    }
180}

182fn insert_spider_traits(
183    commands: &mut Commands,
184    entity: Entity,
185    spider: &SpiderActorSetup,
186    jitter: &SpiderJitter,
187    melee_range: f32,
188) {
189    let mut entity = commands.entity(entity);
190    entity.insert((
191        spider.role,
192        spider.tier,
193        spider.biome,
194        DistanceJitter { melee: jitter.melee_scale(melee_range), vision: jitter.vision_scale },
195        PatrolTimeScale(jitter.patrol_scale),
196    ));
197    if spider.large_freeze {
198        entity.insert(LargeFreezeTarget);
199    }
200}

202/// The body sprite and the effects sprite. A split Void shows its face on the effects sprite, in
203/// front of its ghosting body.
204fn insert_graphics(
205    commands: &mut Commands,
206    entity: Entity,
207    (gfx, effects_gfx): (Entity, Entity),
208    transform: Transform,
209    actor: &ResolvedActor,
210) {
211    let (archetype, body) = (&actor.archetype, &actor.body);
212    let clip = first_clip(archetype, &body.animation_keys);
213    commands.entity(effects_gfx).insert(StateDespawnMarker);
214    match body.gfx {
215        EnemyGfxKind::PixelPerfect => {
216            let sprite = enemy_sprite(entity, transform, body.gfx_offset, body.pixel_aligned, clip);
217            commands.entity(gfx).insert((sprite, pixel_perfect_layers()));
218        },
219        EnemyGfxKind::World => {
220            let sprite = enemy_sprite(entity, transform, body.gfx_offset, body.pixel_aligned, clip);
221            commands.entity(gfx).insert((sprite, RenderLayers::layer(0)));
222        },
223        EnemyGfxKind::VoidSplit => {
224            let ghosting = GhostingSource {
225                spawn_interval_ticks: VOID_GHOST_INTERVAL_TICKS,
226                ghost_lifetime_ticks: VOID_GHOST_LIFETIME_TICKS,
227                initial_alpha: 1.0,
228                fade_curve: FadeCurve::Constant,
229                color_mode: GhostColorMode::Original,
230                scale_over_time: ScaleCurve::Axes { x: VOID_GHOST_MIN_SCALE, y: VOID_GHOST_MIN_SCALE },
231                movement: GhostMovement::Static,
232                render_layers: Some(pixel_perfect_layers()),
233                ..default()
234            };
235            let face = Vec3::new(0.0, 0.0, VOID_FACE_Z_OFFSET);
236            let face_transform = Transform::from_translation(transform.translation + face);
237            let face_sprite = enemy_sprite(entity, face_transform, face, true, first_clip(archetype, &["Face"]));
238            commands.entity(entity).insert(OrientToVelocity);
239            commands.entity(gfx).insert((enemy_sprite(entity, transform, Vec3::ZERO, true, clip), ghosting, pixel_perfect_layers()));
240            commands.entity(effects_gfx).insert((face_sprite, OrientToVelocity, pixel_perfect_layers()));
241        },
242    }
243}

248    transform: Transform,
249    offset: Vec3,
250    pixel_aligned: bool,
251    player: ScriptPlayer<SpriteAnimation>,

254        EnemyGfx,

267/// A playback of the first of `slots` the archetype binds.
268fn first_clip(archetype: &EnemyArchetype, slots: &[impl AsRef<str>]) -> ScriptPlayer<SpriteAnimation> {
269    let mut player = ScriptPlayer::default();
270    if let Some(key) = slots.iter().find_map(|slot| archetype.anim.get(slot.as_ref())) {
271        player.play_key(key);
272    }
273    player
274}

276/// Per-spider variation of patrol pace, vision, reaction, and (for small spiders) melee reach,
277/// seeded by the entity and the level.
278struct SpiderJitter {
279    patrol_scale: f32,
280    vision_scale: f32,
281    reaction_scale: f32,
282    melee_offset: f32,
283}
284
285impl SpiderJitter {
286    fn new(entity: Entity, level_seed: u32, small: bool) -> Self {
287        let bits = entity.to_bits();
288        let mut state = ((bits ^ (bits >> 32)) as u32) ^ level_seed ^ 0x9E37_79B9;
289        let mut next = move || {
290            state = state.wrapping_mul(1_664_525).wrapping_add(1_013_904_223);
291            state
292        };
293        // A magnitude from `least` to `least + spread`, with a random sign.
294        let mut signed = |least: f32, spread: f32| {
295            let magnitude = least + next() as f32 / u32::MAX as f32 * spread;
296            let sign = (next() & 1) as f32 * 2.0 - 1.0;
297            sign * magnitude
298        };
299        Self {
300            patrol_scale: 1.0 + signed(0.05, 0.05),
301            vision_scale: 1.0 + signed(0.05, 0.05),
302            reaction_scale: 1.0 + signed(0.05, 0.05),
303            melee_offset: if small { signed(0.0, 2.0) } else { 0.0 },
304        }
305    }
306
307    fn melee_range(&self, base: f32) -> f32 {
308        if base <= 0.0 { base } else { (base + self.melee_offset).max(1.0) }
309    }
310
311    fn melee_scale(&self, base: f32) -> f32 {
312        if base <= 0.0 { 1.0 } else { self.melee_range(base) / base }
313    }
314
315    fn vision_range(&self, base: f32) -> f32 {
316        if base <= 0.0 { base } else { base * self.vision_scale }
317    }
318
319    fn reaction_time(&self, base: u8) -> u8 {
320        if base == 0 { 0 } else { ((base as f32 * self.reaction_scale).round() as i32).clamp(1, 32) as u8 }
321    }
