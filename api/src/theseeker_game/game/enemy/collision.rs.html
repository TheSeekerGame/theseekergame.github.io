3pub(super) fn enemy_gravity(
4    mut query: Query<
5        (&mut LinearVelocity, &mut Navigation),
6        (
7            With<Enemy>,
8            With<ActorController>,
9            With<GroundMotor>,
10            Without<crate::game::effects::frozen::Frozen>,
11        ),
12    >,
13) {
14    for (mut velocity, mut nav) in query.iter_mut() {
15        match *nav {
16            Navigation::Falling {
17                mut fall_ticks,
18                jumping,
19            } => {
20                velocity.0.y =
21                    curve_velocity(ENEMY_FALL_VELOCITIES, fall_ticks);
22                fall_ticks += 1;
23                *nav = Navigation::Falling {
24                    fall_ticks,
25                    jumping,
26                };
27            },
28            Navigation::Grounded | Navigation::Blocked => {},
29        }
30    }
31}

33pub(super) fn move_collide(
34    mut commands: Commands,
35    collision_world: Option<Res<TileCollisionWorld>>,
36    mut query: Query<
37        (
38            Entity,
39            &mut LinearVelocity,
40            &mut Transform,
41            &mut Navigation,
42            &Collider,
43            Has<Knockback>,
44            &mut MovementState,
45        ),
46        (
47            With<Enemy>,
48            With<GroundMotor>,
49            Without<crate::game::effects::frozen::Frozen>,
50        ),
51    >,
52    time: Res<GameTime>,
53    mut profiler: TickProfiler,
54) {
55    let _profile = profiler.scope("enemy_move_collide");
56    let collision_world =
57        collision_world.as_deref().filter(|world| world.is_ready());
58
59    for (
60        entity,
61        mut linear_velocity,
62        mut transform,
63        mut nav,
64        collider,
65        is_knocked,
66        mut movement,
67    ) in query.iter_mut()
68    {
69        let half_extents = collider.half_extents();
70        let half_extents = Vec2::new(half_extents.x, half_extents.y);
71        if !movement.frame_steps.is_empty() {
72            commands.entity(entity).insert(theseeker_engine::presentation::PresentationJump(time.tick()));
73        }
74        for step in movement.frame_steps.drain(..) {
75            resolve_ground_displacement(collision_world, &mut transform, &mut nav, half_extents, step.0, is_knocked);
76        }
77        linear_velocity.0 = resolve_ground_displacement(
78            collision_world, &mut transform, &mut nav, half_extents, linear_velocity.0, is_knocked,
79        );
80    }
81}

83fn resolve_ground_displacement(
84    collision_world: Option<&TileCollisionWorld>,
85    transform: &mut Transform,
86    nav: &mut Navigation,
87    half_extents: Vec2,
88    mut displacement: Vec2,
89    is_knocked: bool,
90) -> Vec2 {
91    let z = transform.translation.z;
92    let Some(collision_world) = collision_world.filter(|_| half_extents.x > 0.0 && half_extents.y > 0.0) else {
93        transform.translation += displacement.extend(0.0);
94        return displacement;
95    };
96    let was_falling = matches!(*nav, Navigation::Falling { .. });
97    let blocked = stop_at_ground_edge(collision_world, &mut displacement, transform, half_extents, nav);
98    let result = collision_world.resolve_movement(transform.translation.xy(), displacement, half_extents);
99    if (blocked || result.hit_wall) && !is_knocked && !was_falling {
100        *nav = Navigation::Blocked;
101    }
102    if result.grounded && !matches!(*nav, Navigation::Blocked) { *nav = Navigation::Grounded; }
103    let mut resolved = result.velocity;
104    apply_falling_transition(collision_world, nav, result.position, result.grounded, half_extents, &mut resolved);
105    transform.translation = result.position.extend(z);
106    resolved
107}

109pub(super) fn remove_inside(
110    mut enemies: Query<
111        (Entity, &GlobalTransform, &Collider),
112        (With<Inside>, With<Enemy>),
113    >,
114    players_q: Query<
115        (Entity, &GlobalTransform, &Collider),
116        With<theseeker_engine::physics::inside::PlayerInsideEnemy>,
117    >,
118    mut commands: Commands,
119    spatial_query: PhysicsWorld,
120) {
121    for (enemy_entity, transform, collider) in enemies.iter_mut() {
122        let intersections = spatial_query.intersect(
123            transform.translation().xy(),
124            collider,
125            theseeker_engine::physics::groups::groups(ENEMY_INSIDE),
126            Some(enemy_entity),
127        );
128        let still_inside: std::collections::HashSet<Entity> =
129            intersections.into_iter().collect();
130
131        for player in players_q.iter() {
132            clear_player_inside_if_needed(
133                enemy_entity,
134                player,
135                &still_inside,
136                &spatial_query,
137                &mut commands,
138            );
139        }
140    }
141}

143fn stop_at_ground_edge(
144    collision_world: &TileCollisionWorld,
145    velocity: &mut Vec2,
146    transform: &Transform,
147    half_extents: Vec2,
148    nav: &Navigation,
149) -> bool {
150    if !matches!(*nav, Navigation::Grounded) || velocity.x.abs() <= f32::EPSILON
151    {
152        return false;
153    }
154
155    let dir = velocity.x.signum();
156    let ahead_offset = half_extents.x + 2.0;
157    let foot_offset = half_extents.y + 1.0;
158    let start = Vec2::new(
159        transform.translation.x + ahead_offset * dir,
160        transform.translation.y - half_extents.y + 0.5,
161    );
162    let ray_length = foot_offset + 4.0;
163    let probe_half_extents = Vec2::new(1.0, 0.0);
164    let distance =
165        collision_world.distance(ProbeDirection::Down, start, probe_half_extents, ray_length);
166    let blocked = distance >= ray_length;
167    if blocked {
168        velocity.x = 0.0;
169    }
170    blocked
171}

173fn apply_falling_transition(
174    collision_world: &TileCollisionWorld,
175    nav: &mut Navigation,
176    position: Vec2,
177    grounded: bool,
178    half_extents: Vec2,
179    final_velocity: &mut Vec2,
180) {
181    if !matches!(*nav, Navigation::Grounded) || grounded {
182        return;
183    }
184
185    let ground_dist =
186        collision_world.distance(ProbeDirection::Down, position, half_extents, GROUNDED_THRESHOLD);
187    if ground_dist < GROUNDED_THRESHOLD {
188        return;
189    }
190
191    *nav = Navigation::Falling {
192        fall_ticks: 0,
193        jumping: false,
194    };
195    final_velocity.y = 0.0;
196}

198fn clear_player_inside_if_needed(
199    enemy_entity: Entity,
200    player: (Entity, &GlobalTransform, &Collider),
201    still_inside: &std::collections::HashSet<Entity>,
202    spatial_query: &PhysicsWorld,
203    commands: &mut Commands,
204) {
205    let (player_entity, transform, collider) = player;
206    if still_inside.contains(&player_entity) {
207        return;
208    }
209
210    let overlap = spatial_query.intersect(
211        transform.translation().xy(),
212        collider,
213        theseeker_engine::physics::groups::groups(ENEMY_INSIDE),
214        Some(player_entity),
215    );
216    if overlap.is_empty() {
217        theseeker_engine::physics::inside::clear(
218            commands,
219            enemy_entity,
220            player_entity,
221        );
222    }
