1//! Parallax: entities at a depth other than 1 shift against the camera as if nearer or farther.
2use bevy::transform::TransformSystems::Propagate;
3
4use crate::camera::MainCamera;
5use crate::prelude::*;
6
7/// Places parallaxed entities around the camera each frame, once it has moved.
8pub struct ParallaxPlugin;
9
10impl Plugin for ParallaxPlugin {
11    fn build(&self, app: &mut App) {
12        app.add_systems(PostUpdate, (init_parallax, apply_parallax).chain().before(Propagate));
13    }
14}
15
16/// How far from the camera an entity lies: 1 is the world's own distance, less is nearer and
17/// more is farther.
18#[derive(Clone, PartialEq, Debug, Default, Component)]
19pub struct Parallax {
20    pub depth: f32,
21}
22
23/// Where a parallaxed entity lies at depth 1, taken from its first translation. Move the entity
24/// by changing this: parallax overwrites its translation, which must therefore be in world
25/// space, so the entity's parent must sit at the world origin.
26#[derive(Clone, PartialEq, Debug, Default, Component)]
27pub struct ParallaxOrigin(pub Vec2);
28
29/// The point, relative to the origin, whose offset from the camera the depth scales, when it
30/// is not the origin itself.
31#[derive(Clone, PartialEq, Debug, Default, Component)]
32pub struct ParallaxOffset(pub Vec2);
33
34fn init_parallax(
35    mut commands: Commands,
36    entities: Query<(Entity, &Transform), (With<Parallax>, Without<ParallaxOrigin>, Without<MainCamera>)>,
37) {
38    for (entity, transform) in &entities {
39        commands.entity(entity).insert(ParallaxOrigin(transform.translation.xy()));
40    }
41}
42
43fn apply_parallax(
44    mut entities: Query<(&mut Transform, &Parallax, &ParallaxOrigin, Option<&ParallaxOffset>), Without<MainCamera>>,
45    camera: Single<&Transform, With<MainCamera>>,
46) {
47    let camera = camera.translation.xy();
48    for (mut transform, parallax, origin, offset) in &mut entities {
49        let offset = offset.map_or(Vec2::ZERO, |offset| offset.0);
50        // Deriving the position from the origin each frame, rather than moving it with the
51        // camera, keeps floating-point error from accumulating into drift.
52        let center = camera - (camera - (origin.0 + offset)) / parallax.depth;
53        let position = center - offset;
54        transform.translation.x = position.x;
55        transform.translation.y = position.y;
56    }
57}
