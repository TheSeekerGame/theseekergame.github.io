1use crate::prelude::*;
2use crate::ui::{OpenPanels, UiInputSet};

4pub struct GameStatePlugin;
5
6impl Plugin for GameStatePlugin {
7    fn build(&self, app: &mut App) {
8        app.init_state::<GameState>().init_resource::<PauseRequests>();
9        app.add_systems(PreUpdate, update_pause_requests.in_set(UiInputSet::Pause)
10            .run_if(in_state(AppState::InGame)));
11        app.add_systems(OnExit(AppState::InGame), clear_pause_requests);
12    }
13}

15#[derive(States, Clone, Eq, PartialEq, Hash, Debug, Default)]
16pub enum GameState {
17    #[default]
18    Playing,
19    Paused,
20}
21
22/// Independent owners keep closing a panel from cancelling a manual pause.
23#[derive(Resource, Default)]
24pub(crate) struct PauseRequests {
25    manual: bool,
26    modal: bool,
27}
28
29impl PauseRequests {
30    pub(crate) fn is_paused(&self) -> bool { self.manual || self.modal }
31    pub(crate) fn resume_manual(&mut self) { self.manual = false; }
32}
33
34fn update_pause_requests(
35    input: Res<ButtonInput<KeyCode>>,
36    panels: OpenPanels,
37    mut requests: ResMut<PauseRequests>,
38    state: Res<State<GameState>>,
39    mut next: ResMut<NextState<GameState>>,
40    mut transport: ResMut<GameTransport>,
41) {
42    if input.just_pressed(KeyCode::KeyP) { requests.manual = !requests.manual; }
43    requests.modal = panels.any();
44    apply_pause(&requests, state.get(), &mut next, &mut transport);
45}
46
47fn apply_pause(requests: &PauseRequests, current: &GameState, next: &mut NextState<GameState>, transport: &mut GameTransport) {
48    let desired = if requests.is_paused() { GameState::Paused } else { GameState::Playing };
49    transport.paused = requests.is_paused();
50    if *current != desired { next.set(desired); }
51}
52
53fn clear_pause_requests(
54    mut requests: ResMut<PauseRequests>,
55    mut next: ResMut<NextState<GameState>>,
56    mut transport: ResMut<GameTransport>,
57) {
58    *requests = default();
59    transport.paused = false;
60    next.set(GameState::Playing);
61}

63#[cfg(test)]
64mod tests {
65    use super::*;
66
67    #[test]
68    fn closing_a_panel_preserves_manual_pause_and_transport() {
69        let mut requests = PauseRequests { manual: true, modal: true };
70        let mut next = NextState::default();
71        let mut transport = GameTransport::default();
72        requests.modal = false;
73        apply_pause(&requests, &GameState::Playing, &mut next, &mut transport);
74        assert!(transport.paused);
75        assert!(matches!(next, NextState::Pending(GameState::Paused)));
76        requests.manual = false;
77        apply_pause(&requests, &GameState::Paused, &mut next, &mut transport);
78        assert!(!transport.paused);
79        assert!(matches!(next, NextState::Pending(GameState::Playing)));
80    }
