33        app.add_systems(OnExit(AppState::InGame), theseeker_engine::presentation::clear_presentation);
34        app.insert_resource(theseeker_engine::assets::animation::AnimationClock(
35            theseeker_engine::assets::animation::FrameClock::Global,
36        ));
37        // LDtk registration removed; entities are spawned by PNG level pipeline
38
39        // Add the plugins for each game mechanic
40        app.add_plugins(actors::ActorPlugin);
41        app.add_plugins((
42            combat::CombatPlugin,
43            interaction::InteractionPlugin,
44            physics::PhysicsPlugin,
45            effects::EffectsPlugin,
46            player::PlayerPlugin,
