22/// The share of the way to its target a claimed camera eases each tick, per axis.
23const EASE_SHARE: Vec2 = Vec2::new(0.02, 0.05);
24/// The dead zone's half size as a share of the view; the player moves freely inside it.
25const DEAD_ZONE_SHARE: Vec2 = Vec2::new(1.0 / 8.0, 1.0 / 6.0);
26/// Tracking ends once the player is back within this share of the dead zone.
27const TRACKING_RELEASE_SHARE: f32 = 0.5;
28/// How far inside the dead zone tracking holds the player, as a share of it.
29const TRACKING_INSET_SHARE: f32 = 0.8;
30/// How much of the open space's pull tracking leans by, and at most how far, per axis.
31const GEOMETRY_GUIDANCE: f32 = 0.25;
32const MAX_GEOMETRY_BIAS: Vec2 = Vec2::new(7.5, 4.5);
33/// The spring's smoothing time per axis, in seconds.
34const SMOOTH_TIME: Vec2 = Vec2::new(0.55, 0.42);
35/// The spring rests on its target once within this distance and below this speed.
36const SPRING_SETTLE_DISTANCE: f32 = 0.01;
37const SPRING_SETTLE_SPEED: f32 = 0.01;
38/// How far below a falling player the camera looks at full strength, as a share of the view.
39const FALL_LOOK_DOWN_SHARE: f32 = 0.14;
40/// The drop, as a share of the view, and the falling speed over which fall framing ramps in.
41const FALL_DEPTH_RAMP_SHARE: f32 = 0.1;
42const FALL_SPEED_RAMP: f32 = 120.0;
43/// How quickly fall framing blends toward its strength, in seconds.
44const FALL_BLEND_SECONDS: f32 = 0.12;
45/// How far from the view's center the player may stray before dragging the camera along, as
46/// shares of the view: to either side or above, and below.
47const VISIBLE_SIDE_SHARE: f32 = 0.4;
48const VISIBLE_BOTTOM_SHARE: f32 = 0.32;
49/// The open-space search covers this multiple of the view.
50const SEARCH_VIEW_SCALE: f32 = 1.1;
51/// How strongly the search favors open space toward where the player is heading.
52const SEARCH_DIRECTION_BIAS: f32 = 0.3;

107/// What claims the camera this tick.
108enum Focus {
109    /// A scripted center, or the inside of the tent the player is in: the camera holds it.

132/// What may claim the camera: a scripted focus, a spawn snap, and the player's situation.
133#[derive(SystemParam)]
134pub(super) struct FramingSubject<'w, 's> {
135    scripted: Option<Res<'w, ScriptedCameraFocus>>,
136    snap: Option<Res<'w, SnapCameraNext>>,
137    players: Query<'w, 's, FramedPlayer, (With<Player>, Without<SpikeRespawnPending>)>,
138    capsules: Query<'w, 's, &'static Transform, With<Capsule>>,
139    enemies: Query<'w, 's, &'static Transform, With<Enemy>>,
140    tent_interior: Option<Res<'w, TentInteriorAssets>>,
141}
142
143impl FramingSubject<'_, '_> {
144    /// What claims the camera this tick, strongest claim first.
145    fn focus(&self, tick_seconds: f32) -> Focus {
146        if let Some(scripted) = &self.scripted {
147            return Focus::Hold(scripted.0);
148        }
149        let Ok(player) = self.players.single() else {
150            return Focus::Settle;
151        };
152        let position = player.transform.translation.truncate();
153        let interior = player.interior.and_then(|occupant| interior_center_world(self.tent_interior.as_deref(), occupant.kind));
154        if self.snap.is_some() {
155            return Focus::Snap { position: interior.unwrap_or(position), landed: player.grounded };
156        }
157        if let Some(center) = interior {
158            return Focus::Hold(center);
159        }
160        if let Some(capsule) = player.shelter.and_then(|occupant| self.capsules.get(occupant.shelter).ok()) {
161            return Focus::Ease(Vec2::new(capsule.translation.x, position.y));
162        }
163        if player.flicker_striking {
164            return Focus::Ease(self.strike_center(position).unwrap_or(position));
165        }
166        Focus::Player(PlayerMotion {
167            position,
168            velocity: player.velocity.0 / tick_seconds.max(f32::EPSILON),
169            grounded: player.grounded,
170        })
171    }
172
173    /// The center of the enemies within a flicker strike's reach of `position`, if any.
174    fn strike_center(&self, position: Vec2) -> Option<Vec2> {
175        let range = flicker_strike_metadata().range;
176        let (sum, count) = self.enemies.iter()
177            .map(|enemy| enemy.translation.truncate())
178            .filter(|enemy| position.distance(*enemy) <= range)
179            .fold((Vec2::ZERO, 0), |(sum, count), enemy| (sum + enemy, count + 1));
180        (count > 0).then(|| sum / count as f32)
181    }
182}

184/// The level geometry the camera frames the player by.
185#[derive(SystemParam)]
186pub(super) struct FramingGeometry<'w, 's> {
187    level: Option<Res<'w, CompiledLevel>>,
188    chunks: Option<Res<'w, LevelChunkOccupancyMask>>,
189    covers: Query<'w, 's, &'static CoverCollisionTiles>,
190    search: Local<'s, OpenSpaceSearch>,
191}
192
193impl FramingGeometry<'_, '_> {
194    /// Where the open space ahead draws the camera as the player at `player` leaves the dead
195    /// zone in `direction`: its center, kept within the dead zone and off the sides crossed.
196    fn focus(&mut self, player: Vec2, view: Vec2, direction: Vec2) -> Vec2 {
197        let Some(level) = self.level.as_deref() else {
198            return player;
199        };
200        self.search.covers.clear();
201        self.search.covers.extend(self.covers.iter().flat_map(|cover| cover.tiles.iter().copied()));
202        let dead_zone = view * DEAD_ZONE_SHARE;
203        let focus = self.search.centroid(level, self.chunks.as_deref(), player, view, direction)
204            .unwrap_or(player)
205            .clamp(player - dead_zone, player + dead_zone);
206        constrain_focus_to_crossed_sides(focus, player, direction)
207    }
208}

210/// Moves the rig toward what claims the camera this tick.
211pub(super) fn follow_player(
212    mut commands: Commands,
213    mut rig: ResMut<CameraRig>,
214    subject: FramingSubject,
215    mut geometry: FramingGeometry,
216    scale: Res<GameplayPixelScale>,
217    shake: Option<Res<CameraShake>>,
218    time: Res<GameTime>,
219) {
220    rig.previous = rig.position;
221    let tick_seconds = time.seconds_per_tick() as f32;
222    let paused = shake.is_some_and(|shake| shake.pauses_follow());
223    // Easing ends within a screen pixel of the target.
224    let settle = scale.projection_scale();
225    let view = scale.visible_game_size();
226    match subject.focus(tick_seconds) {
227        Focus::Hold(center) => rig.hold(center),
228        Focus::Snap { position, landed } => {
229            rig.hold(position);
230            rig.previous = position;
231            // Spawn markers sit a few pixels above the floor; following the settle rigidly
232            // keeps the first framed view from panning.
233            if landed {
234                commands.remove_resource::<SnapCameraNext>();
235            }
236        },
237        Focus::Ease(target) => rig.ease_toward(target, paused, settle),
238        Focus::Player(player) if geometry.level.is_none() => rig.ease_toward(player.position, paused, settle),
239        Focus::Player(player) if paused => rig.keep_player_visible(player, view),
240        Focus::Player(player) => rig.frame(player, view, tick_seconds, |direction| geometry.focus(player.position, view, direction)),
241        Focus::Settle => rig.settle(tick_seconds, paused),
242    }
243}

245impl CameraRig {
246    /// Holds the camera at `center`, suspending player framing.
247    fn hold(&mut self, center: Vec2) {
248        self.suspend_framing();
249        self.target = center;
250        self.position = center;
251    }
252
253    /// Eases the camera a fixed share of the way to `target` each tick unless `paused`,
254    /// suspending player framing, and lands on it within `settle`.
255    fn ease_toward(&mut self, target: Vec2, paused: bool, settle: f32) {
256        self.suspend_framing();
257        self.target = target;
258        if paused {
259            return;
260        }
261        if (self.position - target).length() < settle {
262            self.position = target;
263        } else {
264            self.position += (target - self.position) * EASE_SHARE;
265        }
266    }
267
268    /// Settles the camera on its last framing while there is no player to frame.
269    fn settle(&mut self, tick_seconds: f32, paused: bool) {
270        if !self.framing {
271            return;
272        }
273        self.target = self.anchor;
274        if !paused {
275            self.advance(tick_seconds);
276        }
277    }
278
279    /// Stops framing the player, so it starts afresh when the player is framed again.
280    fn suspend_framing(&mut self) {
281        self.framing = false;
282        self.axes = Default::default();
283        self.fall = FallFraming::default();
284    }
285
286    /// Frames `player` in a `view`-sized view: holds still while the player stays inside the
287    /// dead zone, then tracks them, leaning toward `open_space(direction)` when they leave it
288    /// in `direction`, looking ahead during falls, and keeping them in sight.
289    fn frame(&mut self, player: PlayerMotion, view: Vec2, tick_seconds: f32, open_space: impl FnOnce(Vec2) -> Vec2) {
290        let dead_zone = view * DEAD_ZONE_SHARE;
291        let direction = self.leaving(player.position, dead_zone);
292        let lean = if direction == Vec2::ZERO { Vec2::ZERO } else { open_space(direction) - player.position };
293        for (axis, motion) in self.axes.iter_mut().enumerate() {
294            motion.begin_tracking(direction[axis], lean[axis], MAX_GEOMETRY_BIAS[axis]);
295            motion.track(&mut self.anchor[axis], player.position[axis], dead_zone[axis]);
296        }
297        self.fall.update(player, view.y, tick_seconds);
298        let falling = !player.grounded && player.velocity.y < 0.0;
299        if falling && self.fall.strength > 0.0 {
300            let fall_focus = player.position.y - view.y * FALL_LOOK_DOWN_SHARE * self.fall.strength;
301            self.anchor.y = self.anchor.y.min(fall_focus);
302        }
303        self.target = self.anchor;
304        // A critically damped camera trails a constant-speed target by speed * smooth_time.
305        // Prediction cancels that lag during long falls.
306        if falling {
307            self.target.y += player.velocity.y * SMOOTH_TIME.y * self.fall.strength;
308        }
309        self.advance(tick_seconds);
310        self.keep_player_visible(player, view);
311    }
312
313    /// The direction on each axis in which the player at `player` leaves the dead zone, if not
314    /// already tracked that way. The first framed tick instead latches the anchor on them.
315    fn leaving(&mut self, player: Vec2, dead_zone: Vec2) -> Vec2 {
316        if !self.framing {
317            self.anchor = player;
318            self.target = player;
319            self.fall.takeoff_height = player.y;
320            self.framing = true;
321            return Vec2::ZERO;
322        }
323        let offset = player - self.anchor;
324        Vec2::from_array(std::array::from_fn(|axis| self.axes[axis].leaving(offset[axis], dead_zone[axis])))
325    }
326
327    /// Keeps the player within the visible shares of the view, dragging the framing along.
328    fn keep_player_visible(&mut self, player: PlayerMotion, view: Vec2) {
329        if view.min_element() <= 0.0 {
330            return;
331        }
332        let min = player.position - view * VISIBLE_SIDE_SHARE;
333        let max = player.position + view * Vec2::new(VISIBLE_SIDE_SHARE, VISIBLE_BOTTOM_SHARE);
334        for (axis, motion) in self.axes.iter_mut().enumerate() {
335            motion.constrain(&mut self.position[axis], &mut self.anchor[axis], &mut self.target[axis],
336                player.velocity[axis], (min[axis], max[axis]));
337        }
338    }
339
340    /// Springs the camera toward its target.
341    fn advance(&mut self, tick_seconds: f32) {
342        for (axis, motion) in self.axes.iter_mut().enumerate() {
343            motion.advance(&mut self.position[axis], self.target[axis], tick_seconds, SMOOTH_TIME[axis]);
344        }
345    }
346}

348impl AxisMotion {
349    /// The direction in which the player, `offset` from the anchor, leaves the dead zone, if
350    /// not already tracked that way. Tracking ends once they are back well inside.
351    fn leaving(&mut self, offset: f32, dead_zone: f32) -> f32 {
352        if offset * self.tracking_direction <= dead_zone * TRACKING_RELEASE_SHARE {
353            self.tracking_direction = 0.0;
354        }
355        if self.tracking_direction == 0.0 && offset.abs() > dead_zone {
356            offset.signum()
357        } else {
358            0.0
359        }
360    }
361
362    /// Tracks the player in `direction`, leaning `lean` toward the open space ahead.
363    fn begin_tracking(&mut self, direction: f32, lean: f32, max_bias: f32) {
364        if direction == 0.0 {
365            return;
366        }
367        self.tracking_direction = direction;
368        self.geometry_bias = (lean * direction * GEOMETRY_GUIDANCE).clamp(0.0, max_bias);
369    }
370
371    /// Drags the anchor after a tracked player, holding them inside the dead zone by an inset
372    /// that the lean shortens; the anchor only ever moves their way.
373    fn track(&self, anchor: &mut f32, player: f32, dead_zone: f32) {
374        let inset = (dead_zone * TRACKING_INSET_SHARE - self.geometry_bias).max(dead_zone * 0.6);
375        let candidate = player - self.tracking_direction * inset;
376        if self.tracking_direction > 0.0 {
377            *anchor = anchor.max(candidate);
378        } else if self.tracking_direction < 0.0 {
379            *anchor = anchor.min(candidate);
380        }
381    }
382
383    /// Advances a critically damped spring from `position` toward `target`.
384    fn advance(&mut self, position: &mut f32, target: f32, delta_seconds: f32, smooth_time: f32) {
385        if delta_seconds <= 0.0 {
386            return;
387        }
388        let angular_frequency = 2.0 / smooth_time.max(f32::EPSILON);
389        let scaled_time = angular_frequency * delta_seconds;
390        let decay = 1.0
391            / (1.0
392                + scaled_time
393                + 0.48 * scaled_time * scaled_time
394                + 0.235 * scaled_time * scaled_time * scaled_time);
395        let displacement = *position - target;
396        let integrated_velocity = (self.velocity + angular_frequency * displacement) * delta_seconds;
397        self.velocity = (self.velocity - angular_frequency * integrated_velocity) * decay;
398        *position = target + (displacement + integrated_velocity) * decay;
399        if (*position - target).abs() <= SPRING_SETTLE_DISTANCE && self.velocity.abs() <= SPRING_SETTLE_SPEED {
400            *position = target;
401            self.velocity = 0.0;
402        }
403    }
404
405    /// Pulls `position` back within `min..=max`, bringing the anchor and target along, and
406    /// keeps the spring from pulling it out again faster than the player moves.
407    fn constrain(&mut self, position: &mut f32, anchor: &mut f32, target: &mut f32, player_velocity: f32, (min, max): (f32, f32)) {
408        if *position < min {
409            *position = min;
410            *anchor = anchor.max(min);
411            *target = target.max(min);
412            self.velocity = self.velocity.max(player_velocity);
413        } else if *position > max {
414            *position = max;
415            *anchor = anchor.min(max);
416            *target = target.min(max);
417            self.velocity = self.velocity.min(player_velocity);
418        }
419    }
420}

422impl FallFraming {
423    /// Blends the strength toward how far and how fast the player falls from their takeoff.
424    fn update(&mut self, player: PlayerMotion, view_height: f32, dt: f32) {
425        if player.grounded {
426            self.takeoff_height = player.position.y;
427        }
428        let drop = (self.takeoff_height - player.position.y).max(0.0);
429        let depth_weight = smooth_ramp(drop / (view_height * FALL_DEPTH_RAMP_SHARE).max(1.0));
430        let speed_weight = smooth_ramp(-player.velocity.y / FALL_SPEED_RAMP);
431        let desired = if player.grounded { 0.0 } else { depth_weight * speed_weight };
432        let blend = 1.0 - (-dt.max(0.0) / FALL_BLEND_SECONDS).exp();
433        self.strength += (desired - self.strength) * blend;
434    }
435}
436
437/// Smoothstep over `0..=1`.
438fn smooth_ramp(value: f32) -> f32 {
439    let t = value.clamp(0.0, 1.0);
440    t * t * (3.0 - 2.0 * t)
441}

443/// Keeps the focus from lying behind the player on any axis they left the dead zone along.
444fn constrain_focus_to_crossed_sides(focus: Vec2, player: Vec2, direction: Vec2) -> Vec2 {
445    let ahead = Vec2::select(direction.cmpgt(Vec2::ZERO), focus.max(player), focus);
446    Vec2::select(direction.cmplt(Vec2::ZERO), focus.min(player), ahead)
447}

449/// Reused buffers for the open-space search.
450#[derive(Default)]
451struct OpenSpaceSearch {
452    /// Tiles a cover fills, which count as solid.
453    covers: HashSet<IVec2>,
454    visited: Vec<bool>,
455    queue: VecDeque<IVec2>,
456}
457
458impl OpenSpaceSearch {
459    /// The center of the open space the player at `player` can reach within the search window
460    /// around them, each tile weighted by its clearance, its distance from the window's edges,
461    /// and how far it lies toward `direction`.
462    fn centroid(&mut self, level: &CompiledLevel, chunks: Option<&LevelChunkOccupancyMask>, player: Vec2, view: Vec2,
463        direction: Vec2,
464    ) -> Option<Vec2> {
465        let Self { covers, visited, queue } = self;
466        let window = FramingWindow::new(level, chunks, covers, player, view * (SEARCH_VIEW_SCALE * 0.5))?;
467        let seed = window.nearest_open(window.tile_at(player))?;
468        visited.clear();
469        visited.resize(window.len(), false);
470        visited[window.index(seed)?] = true;
471        queue.clear();
472        queue.push_back(seed);
473        let (mut weighted, mut total) = (Vec2::ZERO, 0.0);
474        while let Some(tile) = queue.pop_front() {
475            let position = window.center_of(tile);
476            let weight = window.weight(tile, direction);
477            weighted += position * weight;
478            total += weight;
479            // Each tile in the window is looked at once, open or not.
480            let neighbors = [tile + IVec2::X, tile - IVec2::X, tile + IVec2::Y, tile - IVec2::Y];
481            queue.extend(neighbors.into_iter().filter(|&neighbor| window.index(neighbor)
482                .is_some_and(|index| !std::mem::replace(&mut visited[index], true)) && window.is_open(neighbor)));
483        }
484        (total > f32::EPSILON).then(|| weighted / total)
485    }
486}

488/// The window of level tiles the open-space search covers around the player. Tile rows count
489/// down from the level's top, as in its image.
490struct FramingWindow<'a> {
491    level: &'a CompiledLevel,
492    /// The chunks the level fills, when the mask matches its size.
493    chunks: Option<&'a LevelChunkOccupancyMask>,
494    covers: &'a HashSet<IVec2>,
495    tile_size: f32,
496    center: Vec2,
497    half_size: Vec2,
498    /// The window's first and last tiles.
499    min: IVec2,
500    max: IVec2,
501}
502
503impl<'a> FramingWindow<'a> {
504    /// The level's tiles within `half_size` of `center`, or none if the level or window is empty.
505    fn new(level: &'a CompiledLevel, chunks: Option<&'a LevelChunkOccupancyMask>, covers: &'a HashSet<IVec2>,
506        center: Vec2, half_size: Vec2,
507    ) -> Option<Self> {
508        let tile_size = level.tile_size();
509        if level.width == 0 || level.height == 0 || tile_size <= 0.0 || half_size.min_element() <= 0.0 {
510            return None;
511        }
512        let chunks = chunks.filter(|mask| {
513            mask.matches_chunks(level.width.div_ceil(CHUNK_IMAGE_SIZE), level.height.div_ceil(CHUNK_IMAGE_SIZE))
514        });
515        let last = IVec2::new(level.width as i32 - 1, level.height as i32 - 1);
516        let whole = Self { level, chunks, covers, tile_size, center, half_size, min: IVec2::ZERO, max: last };
517        let top_left = whole.tile_at(center + Vec2::new(-half_size.x, half_size.y));
518        let bottom_right = whole.tile_at(center + Vec2::new(half_size.x, -half_size.y));
519        let window = Self {
520            min: top_left.min(bottom_right).clamp(IVec2::ZERO, last),
521            max: top_left.max(bottom_right).clamp(IVec2::ZERO, last),
522            ..whole
523        };
524        window.min.cmple(window.max).all().then_some(window)
525    }
526
527    fn width(&self) -> usize {
528        (self.max.x - self.min.x + 1) as usize
529    }
530
531    fn len(&self) -> usize {
532        self.width() * (self.max.y - self.min.y + 1) as usize
533    }
534
535    /// Where `tile` lies in the window's row-major tiles, if inside it.
536    fn index(&self, tile: IVec2) -> Option<usize> {
537        (tile.cmpge(self.min).all() && tile.cmple(self.max).all()).then(|| {
538            let local = (tile - self.min).as_uvec2();
539            local.y as usize * self.width() + local.x as usize
540        })
541    }
542
543    /// The tile holding `world`, which may lie outside the level.
544    fn tile_at(&self, world: Vec2) -> IVec2 {
545        let local = world - self.level.bounds.min;
546        IVec2::new(
547            (local.x / self.tile_size).floor() as i32,
548            self.level.height as i32 - 1 - (local.y / self.tile_size).floor() as i32,
549        )
550    }
551
552    /// The world position of `tile`'s center.
553    fn center_of(&self, tile: IVec2) -> Vec2 {
554        Vec2::new(
555            self.level.bounds.min.x + (tile.x as f32 + 0.5) * self.tile_size,
556            self.level.bounds.min.y + (self.level.height as f32 - tile.y as f32 - 0.5) * self.tile_size,
557        )
558    }
559
560    /// Whether `tile` lies in the level, in a chunk it fills, with no terrain or cover.
561    fn is_open(&self, tile: IVec2) -> bool {
562        let size = IVec2::new(self.level.width as i32, self.level.height as i32);
563        if tile.cmplt(IVec2::ZERO).any() || tile.cmpge(size).any() || self.covers.contains(&tile) {
564            return false;
565        }
566        let tile = tile.as_uvec2();
567        self.chunks.is_none_or(|mask| mask.is_occupied(tile.x / CHUNK_IMAGE_SIZE, tile.y / CHUNK_IMAGE_SIZE))
568            && !self.level.terrain.get(tile.y as usize * self.level.width as usize + tile.x as usize).copied().unwrap_or(true)
569    }
570
571    /// The open tile in the window nearest `tile`, at most four tiles away; the first in
572    /// reading order among equals.
573    fn nearest_open(&self, tile: IVec2) -> Option<IVec2> {
574        (0..=4).find_map(|radius| {
575            (tile.y - radius..=tile.y + radius)
576                .flat_map(|y| (tile.x - radius..=tile.x + radius).map(move |x| IVec2::new(x, y)))
577                .filter(|&candidate| self.index(candidate).is_some() && self.is_open(candidate))
578                .min_by_key(|candidate| candidate.distance_squared(tile))
579        })
580    }
581
582    /// How strongly an open tile draws the camera: less near the window's edges and in tight
583    /// spaces, more toward `direction`.
584    fn weight(&self, tile: IVec2, direction: Vec2) -> f32 {
585        let normalized_offset = (self.center_of(tile) - self.center) / self.half_size;
586        let edge_x = (1.0 - normalized_offset.x.abs().powi(4)).clamp(0.0, 1.0);
587        let edge_y = (1.0 - normalized_offset.y.abs().powi(4)).clamp(0.0, 1.0);
588        let edge_weight = 0.15 + 0.85 * edge_x * edge_y;
589        let open_neighbors = (-1..=1)
590            .flat_map(|y| (-1..=1).map(move |x| IVec2::new(x, y)))
591            .filter(|offset| self.is_open(tile + *offset))
592            .count();
593        let clearance_weight = 0.4 + 0.6 * (open_neighbors as f32 / 9.0);
594        let direction_weight = 1.0 + SEARCH_DIRECTION_BIAS * direction.dot(normalized_offset).clamp(-1.0, 1.0);
595        edge_weight * clearance_weight * direction_weight
596    }
597}
