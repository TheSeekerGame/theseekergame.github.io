6pub struct AppStatesPlugin;
7
8impl Plugin for AppStatesPlugin {
9    fn build(&self, app: &mut App) {
10        // our states
11        app.init_state::<AppState>();
12        for state in enum_iterator::all::<AppState>() {
13            app.add_systems(
14                OnExit(state),
15                // FIXME: vendor helper from iyes_bevy_extras?
16                despawn_all_recursive::<With<StateDespawnMarker>>,
17            );
18        }
19        app.add_systems(OnEnter(AppState::Restart), restart);
20
21        // Start the first expedition once all required collections and audio are ready.
22        app.add_systems(
23            Update,
24            start_first_expedition.run_if(in_state(AppState::AssetsLoading)),
25        );
26    }
27}

29/// State type: Which "screen" is the app in?
30#[derive(Debug, PartialEq, Eq, Clone, Copy, Hash, Default, States)]
31#[derive(Reflect)]
32#[derive(enum_iterator::Sequence)]
33pub enum AppState {
34    /// Initial loading screen at startup
35    #[default]
36    AssetsLoading,

38    MainMenu,
39    /// Gameplay
40    InGame,

44/// Marker for entities that should be despawned on `AppState` transition.
45///
46/// Use this on entities spawned when entering specific states, that need to

51pub fn restart(mut next_state: ResMut<NextState<AppState>>) {

56struct StartupAssets<'w> {
57    preloaded: Res<'w, PreloadedAssets>,
58    server: Res<'w, AssetServer>,

60    pixel_art: Res<'w, crate::ui::pixel_canvas::ArtLibrary>,
61}

63fn start_first_expedition(
64    mut next_state: ResMut<NextState<AppState>>,
65    time: Res<Time<Real>>,
66    audio: Option<Res<theseeker_engine::audio::AudioEngine>>,
67    assets: StartupAssets,
68    mut min_wait_timer: Local<Option<Timer>>,
69    mut commands: Commands,
70    mut starter: crate::ui::mainmenu::WorldStarter,
71) {
72    let timer = min_wait_timer.get_or_insert_with(|| {
73        Timer::from_seconds(
74            0.5,
75            TimerMode::Once,
76        )
77    });
78    timer.tick(time.delta());
79
80    if !timer.is_finished() || audio.is_some_and(|audio| !audio.preparation_ready(&assets.preloaded, &assets.server)) {
81        return;
82    }
83
84    if assets.game_ui.is_none() || !assets.pixel_art.is_ready(&assets.server) {
85        return;
86    }
87
88    info!("All startup asset collections loaded, starting the first expedition");
89    starter.generated(&mut commands, &mut next_state, AppState::AssetsLoading);
90}

92fn despawn_all_recursive<F: QueryFilter>(
93    mut commands: Commands,
94    q: Query<Entity, F>,
95) {
96    for entity in &q {
97        // Both a parent and its descendants may carry the state marker.
98        commands.entity(entity).try_despawn();
99    }
100}

103mod tests {

106    #[test]
107    fn state_cleanup_handles_marked_parents_and_children_in_one_flush() {
108        let mut app = App::new();
109        app.set_error_handler(bevy::ecs::error::panic);
110        app.add_systems(Update, despawn_all_recursive::<With<StateDespawnMarker>>);
111        let parent = app.world_mut().spawn(StateDespawnMarker).id();
112        let child = app.world_mut().spawn((StateDespawnMarker, ChildOf(parent))).id();
113        app.update();
114        assert!(app.world().get_entity(parent).is_err());
115        assert!(app.world().get_entity(child).is_err());
116    }
