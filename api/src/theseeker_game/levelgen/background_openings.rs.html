1//! Where the near background opens onto the planes behind it.
2//!
3//! The near wall follows the platforms, as the artist's does: it hugs terrain surfaces with a
4//! skin a few tiles thick, leaves others bare, swells into occasional masses beside the terrain,
5//! and bridges narrow gaps between platforms. Everything farther from terrain stays open.
6//! Openings are never slivers, the near wall never floats or thins to a strand in the air, and
7//! small openings are closed.
8use super::*;
9use distance::DistanceField;
10
11/// Range of the near wall's skin over terrain, in tiles; below one the terrain stays bare.
12const SKIN: (f32, f32) = (-2.5, 5.0);
13const SKIN_JITTER: f32 = 1.5;
14/// Where the mass noise passes `MASS_THRESHOLD`, the skin thickens by up to `MASS_REACH` tiles.
15const MASS_THRESHOLD: f32 = 0.62;
16const MASS_REACH: f32 = 6.0;
17/// Gaps up to about twice this radius, in tiles, are bridged.
18const BRIDGE_RADIUS: f32 = 3.0;
19pub(super) const MIN_OPENING: usize = 40;
20
21/// Whether each tile is covered by the near background: all terrain and the cave beside it.
22pub(super) fn near_wall(terrain: &[bool], size: UVec2, seed: u64) -> Vec<bool> {
23    let _timer = diagnostics::StageTimer::new("near_openings");
24    let (width, height) = (size.x as usize, size.y as usize);
25    let mut field = DistanceField::default();
26    field.compute(width, height, |x, y| terrain[y * width + x]);
27    let terrain_distance: Vec<f32> = field.squared.iter().map(|squared| squared.sqrt()).collect();
28    let bridged = closing(&field, width, height, &mut DistanceField::default());
29    let noise = |salt: u64, scale: Vec2, x: usize, y: usize| value_noise(Vec2::new(x as f32, y as f32) / scale, seed ^ salt);
30    let candidate: Vec<bool> = (0..width * height).map(|index| {
31        let (x, y) = (index % width, index / width);
32        let mass = ((noise(0x53, Vec2::new(28.0, 18.0), x, y) - MASS_THRESHOLD) / (1.0 - MASS_THRESHOLD)).clamp(0.0, 1.0);
33        let skin = SKIN.0 + (SKIN.1 - SKIN.0) * noise(0x51, Vec2::new(13.0, 9.0), x, y)
34            + SKIN_JITTER * (noise(0x52, Vec2::new(2.0, 3.0), x, y) - 0.5) + MASS_REACH * mass;
35        !terrain[index] && terrain_distance[index] > skin && !bridged[index]
36    }).collect();
37    let wall: Vec<bool> = candidate.iter().map(|&open| !open).collect();
38    tidy(&wall, terrain, size)
39}

41/// Terrain closed with a disk of `BRIDGE_RADIUS`: gaps narrower than the disk fill in.
42fn closing(terrain: &DistanceField, width: usize, height: usize, scratch: &mut DistanceField) -> Vec<bool> {
43    let reach = BRIDGE_RADIUS * BRIDGE_RADIUS + BRIDGE_RADIUS;
44    scratch.compute(width, height, |x, y| terrain.squared[y * width + x] > reach);
45    scratch.squared.iter().map(|&squared| squared > reach).collect()
46}

48/// Keeps a near-wall mask within the artist's rules: no one-tile openings or strands, no near
49/// wall detached from terrain, and no opening smaller than `MIN_OPENING` tiles.
50pub(super) fn tidy(wall: &[bool], terrain: &[bool], size: UVec2) -> Vec<bool> {
51    let opening: Vec<bool> = wall.iter().map(|&covered| !covered).collect();
52    let opening = open_square(&opening, size);
53    let exposed: Vec<bool> = opening.iter().map(|&open| !open).collect();
54    let mut wall: Vec<bool> = open_square(&exposed, size).iter().zip(terrain).map(|(&covered, &solid)| covered || solid).collect();
55    let touching = square(terrain, size, true);
56    let floating: Vec<bool> = wall.iter().zip(terrain).map(|(&covered, &solid)| covered && !solid).collect();
57    let detached = components(&floating, size).into_iter().filter(|component| !component.iter().any(|&index| touching[index]));
58    for index in detached.flatten() { wall[index] = false; }
59    let opening: Vec<bool> = wall.iter().map(|&covered| !covered).collect();
60    let small = components(&opening, size).into_iter().filter(|component| component.len() < MIN_OPENING);
61    for index in small.flatten() { wall[index] = true; }
62    wall
63}

65/// Morphological opening with a 3x3 square; tiles beyond the edge count as unset.
66fn open_square(mask: &[bool], size: UVec2) -> Vec<bool> {
67    square(&square(mask, size, false), size, true)
68}

70/// A 3x3 dilation (`grow`) or erosion of `mask`.
71fn square(mask: &[bool], size: UVec2, grow: bool) -> Vec<bool> {
72    let (width, height) = (size.x as i32, size.y as i32);
73    let at = |x: i32, y: i32| x >= 0 && y >= 0 && x < width && y < height && mask[(y * width + x) as usize];
74    (0..width * height).map(|index| {
75        let (x, y) = (index % width, index / width);
76        let mut neighbours = (-1..=1).flat_map(|dy| (-1..=1).map(move |dx| (x + dx, y + dy)));
77        if grow { neighbours.any(|(nx, ny)| at(nx, ny)) } else { neighbours.all(|(nx, ny)| at(nx, ny)) }
78    }).collect()
79}

81/// Four-connected components of the set tiles.
82pub(super) fn components(mask: &[bool], size: UVec2) -> Vec<Vec<usize>> {
83    let mut seen = vec![false; mask.len()];
84    let mut found = Vec::new();
85    for start in 0..mask.len() {
86        if mask[start] && !seen[start] { found.push(flood(mask, &mut seen, start, size)); }
87    }
88    found
89}
90
91fn flood(mask: &[bool], seen: &mut [bool], start: usize, size: UVec2) -> Vec<usize> {
92    let width = size.x as usize;
93    let mut component = Vec::new();
94    let mut stack = vec![start];
95    while let Some(index) = stack.pop() {
96        if seen[index] { continue; }
97        seen[index] = true;
98        component.push(index);
99        let (x, y) = (index % width, index / width);
100        let neighbours = [(x > 0).then(|| index - 1), (x + 1 < width).then(|| index + 1),
101            (y > 0).then(|| index - width), (y + 1 < size.y as usize).then(|| index + width)];
102        stack.extend(neighbours.into_iter().flatten().filter(|&cell| mask[cell] && !seen[cell]));
103    }
104    component
105}

107/// Smooth value noise in [0, 1) with one lattice point per unit of `position`.
108fn value_noise(position: Vec2, seed: u64) -> f32 {
109    let cell = position.floor();
110    let t = position - cell;
111    let t = t * t * (Vec2::splat(3.0) - 2.0 * t);
112    let lattice = |dx: i32, dy: i32| {
113        let hash = tile_hash(cell.x as i32 + dx, cell.y as i32 + dy, 0, seed);
114        (hash >> 40) as f32 / (1u64 << 24) as f32
115    };
116    let top = lattice(0, 0) + (lattice(1, 0) - lattice(0, 0)) * t.x;
117    let bottom = lattice(0, 1) + (lattice(1, 1) - lattice(0, 1)) * t.x;
118    top + (bottom - top) * t.y
119}

121#[cfg(test)]
122mod tests {
123    use super::*;
124
125    /// Platforms of assorted widths with gaps of assorted heights.
126    fn caves(size: UVec2) -> Vec<bool> {
127        (0..size.x * size.y).map(|index| {
128            let (x, y) = (index % size.x, index / size.x);
129            let band = y % 37;
130            band < 6 + x % 5 || (x / 23) % 4 == 0 && band < 20 || x < 2 || y < 2 || x + 2 >= size.x || y + 2 >= size.y
131        }).collect()
132    }
133
134    #[test]
135    fn near_wall_covers_terrain_and_follows_it_without_slivers_or_islands() {
136        let size = UVec2::new(160, 150);
137        let terrain = caves(size);
138        for seed in [1, 2, 3] {
139            let wall = near_wall(&terrain, size, seed);
140            assert!(terrain.iter().zip(&wall).all(|(&solid, &covered)| !solid || covered));
141            let open_cave = terrain.iter().filter(|&&solid| !solid).count() as f32;
142            let opening: Vec<bool> = wall.iter().map(|&covered| !covered).collect();
143            let share = opening.iter().filter(|&&open| open).count() as f32 / open_cave;
144            assert!((0.3..=0.85).contains(&share), "seed {seed}: {share}");
145            let mut field = DistanceField::default();
146            field.compute(size.x as usize, size.y as usize, |x, y| terrain[y * size.x as usize + x]);
147            let reach = SKIN.1 + SKIN_JITTER + MASS_REACH + 1.0;
148            assert!(field.squared.iter().zip(&wall).all(|(&squared, &covered)| !covered || squared <= reach * reach),
149                "seed {seed}: open cave away from platforms stays open");
150            assert!(components(&opening, size).iter().all(|component| component.len() >= MIN_OPENING));
151            assert_eq!(open_square(&opening, size), opening, "openings are at least 2 tiles across");
152            assert_eq!(tidy(&wall, &terrain, size), wall, "a tidy mask stays unchanged");
153            let touching = square(&terrain, size, true);
154            let air_wall: Vec<bool> = wall.iter().zip(&terrain).map(|(&covered, &solid)| covered && !solid).collect();
155            assert!(components(&air_wall, size).iter().all(|component| component.iter().any(|&index| touching[index])));
156        }
157    }
