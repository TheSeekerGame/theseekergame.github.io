1//! The spawn crater before and after the landing cinematic's impact.
2//!
3//! Terrain and collision are always those of the formed crater. A spawn template may have an
4//! intact-ground sibling with the same file name in `assets/levels/chunks/pre_impact/`; only its
5//! solid (`#000000`) pixels are read. Cave directly below the chunk where the intact ground
6//! closes an opening is filled too, far enough that no view before the impact reaches past it. That terrain is baked as alternate, initially
7//! hidden surface patches that a cinematic shows until the pod lands.
8use std::path::Path;
9
10use super::*;
11
12const PRE_IMPACT_DIR: &str = "pre_impact";
13/// Extent of the cave filled below a closed opening, in tiles.
14const PIT_FILL_DEPTH: u32 = 12;
15const PIT_FILL_HALF_WIDTH: u32 = 16;
16
17/// Intact terrain of the spawn chunk, row-major from the top like `CompiledLevel::terrain`.
18#[derive(Resource, Clone)]
19pub struct PreImpactTerrain {
20    /// Tile position of the spawn chunk's top-left corner.
21    origin: UVec2,
22    terrain: Vec<bool>,
23}
24
25/// Visible only once the crater has formed.
26#[derive(Component)]
27pub(crate) struct ImpactFormed;
28
29/// Visible only before the impact; spawned hidden.
30#[derive(Component)]
31pub(crate) struct ImpactIntact;
32
33impl PreImpactTerrain {
34    /// Reads the intact sibling of a spawn template placed at `origin`, if one exists.
35    pub(super) fn load(template: &Path, origin: UVec2) -> Result<Option<Self>, String> {
36        let (Some(dir), Some(name)) = (template.parent(), template.file_name()) else {
37            return Ok(None);
38        };
39        let path = dir.join(PRE_IMPACT_DIR).join(name);
40        if !path.exists() {
41            return Ok(None);
42        }
43        let image = mapgen::load_png(&path)?;
44        let size = mapgen::CHUNK_IMAGE_SIZE;
45        if image.width() != size || image.height() != size {
46            return Err(format!("{} must be {size}x{size} pixels", path.display()));
47        }
48        let pixels = image.data.as_deref().unwrap_or_default();
49        let terrain = pixels.as_chunks::<4>().0.iter()
50            .map(|pixel| pixel[3] >= 128 && pixel[..3] == [0, 0, 0])
51            .collect();
52        Ok(Some(Self { origin, terrain }))
53    }
54
55    /// The level before the impact, and the chunk regions whose terrain differs.
56    pub(super) fn intact_level(&self, level: &CompiledLevel) -> (CompiledLevel, Vec<URect>) {
57        let (intact, changed) = self.restore(level);
58        let size = mapgen::CHUNK_IMAGE_SIZE;
59        let mut regions: Vec<URect> = changed.iter().map(|tile| {
60            let min = *tile / size * size;
61            URect::from_corners(min, (min + UVec2::splat(size)).min(UVec2::new(level.width, level.height)))
62        }).collect();
63        regions.sort_by_key(|region| (region.min.y, region.min.x));
64        regions.dedup();
65        (intact, regions)
66    }
67
68    /// The tiles whose terrain the impact changes.
69    pub(super) fn changed_tiles(&self, level: &CompiledLevel) -> Vec<UVec2> {
70        self.restore(level).1
71    }
72
73    fn restore(&self, level: &CompiledLevel) -> (CompiledLevel, Vec<UVec2>) {
74        let mut intact = level.clone();
75        let size = mapgen::CHUNK_IMAGE_SIZE;
76        let mut changed: Vec<UVec2> = self.terrain.iter().enumerate()
77            .map(|(index, &solid)| (self.origin + UVec2::new(index as u32 % size, index as u32 / size), solid))
78            .filter(|&(tile, solid)| set_terrain(&mut intact, tile, solid))
79            .map(|(tile, _)| tile)
80            .collect();
81        let bottom = self.origin.y + size;
82        for column in (0..size).map(|x| self.origin.x + x) {
83            changed.extend(fill_pit_below(&mut intact, UVec2::new(column, bottom)));
84        }
85        (intact, changed)
86    }
87}
88
89fn terrain_at(level: &CompiledLevel, tile: UVec2) -> Option<bool> {
90    (tile.x < level.width && tile.y < level.height).then(|| level.terrain[(tile.y * level.width + tile.x) as usize])
91}
92
93/// Returns whether the tile changed.
94fn set_terrain(level: &mut CompiledLevel, tile: UVec2, solid: bool) -> bool {
95    if terrain_at(level, tile).is_none_or(|current| current == solid) {
96        return false;
97    }
98    level.terrain[(tile.y * level.width + tile.x) as usize] = solid;
99    true
100}

102/// Fills the cave reachable from `start` within the pit bounds when the intact spawn chunk
103/// closes it from above.
104fn fill_pit_below(level: &mut CompiledLevel, start: UVec2) -> Vec<UVec2> {
105    let closed_above = start.y > 0 && terrain_at(level, start - UVec2::Y) == Some(true);
106    if !closed_above || terrain_at(level, start) != Some(false) {
107        return Vec::new();
108    }
109    let min = UVec2::new(start.x.saturating_sub(PIT_FILL_HALF_WIDTH), start.y);
110    let max = UVec2::new(start.x + PIT_FILL_HALF_WIDTH, start.y + PIT_FILL_DEPTH - 1);
111    let mut filled = Vec::new();
112    let mut pending = vec![start];
113    while let Some(tile) = pending.pop() {
114        if !set_terrain(level, tile, true) {
115            continue;
116        }
117        filled.push(tile);
118        let neighbors = [tile.saturating_sub(UVec2::X), tile + UVec2::X, tile + UVec2::Y];
119        pending.extend(neighbors.into_iter().filter(|next| {
120            next.cmpge(min).all() && next.cmple(max).all() && terrain_at(level, *next) == Some(false)
121        }));
122    }
123    filled
124}

126/// Bakes the intact ground with the formed level's mosaics and publishes it hidden.
127pub(super) fn publish_intact_patches(
128    commands: &mut Commands, parent: Entity, (intact, regions): (&CompiledLevel, &[URect]), level: &LevelSurfaceInputs<'_>,
129    mesh: &Handle<Mesh>, images: &mut Assets<Image>, materials: &mut Assets<ColorMaterial>,
130) {
131    // Snow caps and edge detail follow the intact ground's own surface.
132    let mut placements = Vec::new();
133    for (library, allowed) in [(&level.decorations.ice, level.ice_allowed), (&level.decorations.tundra, level.tundra_allowed)] {
134        placements.extend(generate_decoration_placements(intact, library, images, Some(allowed)));
135    }
136    let decorations = surface::decoration_chunks(&placements, images, GROUND_TILE_ART_SIZE_PX);
137    let Some(surface) = terrain::prepare_main_surface(intact, level.mosaics, level.ldtk, images,
138        GROUND_TILE_ART_SIZE_PX, Some(level.biomes)) else {
139        return;
140    };
141    let patches: Vec<_> = regions.iter().map(|&region| {
142        let chunk = (region.min / mapgen::CHUNK_IMAGE_SIZE).as_ivec2();
143        let decorations = decorations.get(&chunk).map(Vec::as_slice).unwrap_or_default();
144        surface::bake_ground(&surface, intact, region, decorations, images)
145    }).collect();
146    drop(surface);
147    for patch in patches {
148        let entity = terrain::publish_ground_patch(commands, parent, level.compiled, level.tile_size,
149            patch, None, None, mesh, images, materials, &mut Vec::new());
150        let Some(entity) = entity else { continue };
151        commands.entity(entity).insert((ImpactIntact, Visibility::Hidden));
152    }
153}

167#[cfg(test)]
168mod tests {
169    use super::*;
170
171    /// Rows 0-1 solid; a one-tile opening on row 2 into an open room from row 3 down.
172    fn pit_level() -> CompiledLevel {
173        let (width, height) = (40, 20);
174        let terrain = (0..width * height).map(|index| {
175            let (x, y) = (index % width, index / width);
176            y < 2 || (y == 2 && x != 20)
177        }).collect();
178        CompiledLevel { width, height, terrain, ..default() }
179    }
180
181    #[test]
182    fn closed_opening_fills_the_cave_below_within_the_pit_bounds() {
183        let mut level = pit_level();
184        let filled = fill_pit_below(&mut level, UVec2::new(20, 2));
185        assert!(filled.contains(&UVec2::new(20, 2)));
186        assert_eq!(terrain_at(&level, UVec2::new(20 - PIT_FILL_HALF_WIDTH, 2 + PIT_FILL_DEPTH - 1)), Some(true));
187        assert_eq!(terrain_at(&level, UVec2::new(20 + PIT_FILL_HALF_WIDTH + 1, 5)), Some(false));
188        assert_eq!(terrain_at(&level, UVec2::new(20, 2 + PIT_FILL_DEPTH)), Some(false));
189    }
190
191    #[test]
192    fn opening_still_open_above_is_left_alone() {
193        let mut level = pit_level();
194        level.terrain[(level.width + 20) as usize] = false;
195        assert!(fill_pit_below(&mut level, UVec2::new(20, 2)).is_empty());
196    }
