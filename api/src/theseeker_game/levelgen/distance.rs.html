1//! Exact squared Euclidean distances with reusable separable-transform storage.
2
3const FAR: f32 = 1.0e12;
4
5#[derive(Default)]
6pub(crate) struct DistanceField {
7    pub(crate) squared: Vec<f32>,
8    input: Vec<f32>,
9    output: Vec<f32>,
10    sites: Vec<usize>,
11    cuts: Vec<f32>,
12}
13
14impl DistanceField {
15    pub(crate) fn compute(&mut self, width: usize, height: usize, feature: impl Fn(usize, usize) -> bool) {
16        let n = width.max(height);
17        self.squared.resize(width * height, FAR);
18        self.input.resize(n, FAR);
19        self.output.resize(n, FAR);
20        self.sites.resize(n, 0);
21        self.cuts.resize(n + 1, 0.0);
22        if width == 0 || height == 0 { return; }
23        for y in 0..height {
24            fill_features(&mut self.input[..width], y, &feature);
25            transform(&self.input[..width], &mut self.output, &mut self.sites, &mut self.cuts);
26            self.squared[y * width..(y + 1) * width].copy_from_slice(&self.output[..width]);
27        }
28        for x in 0..width {
29            copy_column(&mut self.input, &self.squared, x, width, height);
30            transform(&self.input[..height], &mut self.output, &mut self.sites, &mut self.cuts);
31            write_column(&mut self.squared, &self.output, x, width, height);
32        }
33    }
34}

36fn intersection(f: &[f32], p: usize, q: usize) -> f32 {
37    ((f[q] - f[p]) + (q as f32 * q as f32 - p as f32 * p as f32))
38        / (2.0 * (q as f32 - p as f32))
39}
40
41fn transform(f: &[f32], out: &mut [f32], sites: &mut [usize], cuts: &mut [f32]) {
42    let mut k = 0;
43    sites[0] = 0;
44    cuts[0] = f32::NEG_INFINITY;
45    cuts[1] = f32::INFINITY;
46    for q in 1..f.len() {
47        let mut cut = intersection(f, sites[k], q);
48        while k > 0 && cut <= cuts[k] {
49            k -= 1;
50            cut = intersection(f, sites[k], q);
51        }
52        k += 1;
53        sites[k] = q;
54        cuts[k] = cut;
55        cuts[k + 1] = f32::INFINITY;
56    }
57    k = 0;
58    for (q, value) in out.iter_mut().take(f.len()).enumerate() {
59        while cuts[k + 1] < q as f32 { k += 1; }
60        let delta = q as f32 - sites[k] as f32;
61        *value = delta * delta + f[sites[k]];
62    }
63}

66#[cfg(test)]
67fn assert_distances(features: &[(usize, usize)], squared: &[f32]) {
68    for (index, &actual) in squared.iter().enumerate() {
69        let x = index % 9;
70        let y = index / 9;
71        let expected = features.iter().map(|&(fx, fy)| {
72            let dx = x as f32 - fx as f32;
73            let dy = y as f32 - fy as f32;
74            dx * dx + dy * dy
75        }).fold(FAR, f32::min);
76        assert_eq!(actual, expected);
77    }
78}
79
80fn copy_column(out: &mut [f32], grid: &[f32], x: usize, width: usize, height: usize) {
81    for y in 0..height { out[y] = grid[y * width + x]; }
82}
83
84fn write_column(grid: &mut [f32], input: &[f32], x: usize, width: usize, height: usize) {
85    for y in 0..height { grid[y * width + x] = input[y]; }
86}
87
88fn fill_features(row: &mut [f32], y: usize, feature: &impl Fn(usize, usize) -> bool) {
89    for (x, value) in row.iter_mut().enumerate() {
90        *value = if feature(x, y) { 0.0 } else { FAR };
91    }
92}
