1//! Depth and geyser biomes, decoration/entity/ore placement, and entity publication.
2//!
3//! Rules scan open tiles in row-major order. Around its center, a rule image's green pixels require
4//! terrain, red pixels forbid it, and white pixels require a snow decoration. Every roll hashes the
5//! tile, the rule's order, and the kind's name, so a seed reproduces its placements.

11const TUNDRA_RADIUS_TILES: f32 = 60.0;

57/// Geysers grow local patches over the depth biomes before each biome's libraries place
58/// decorations, entities (ice entities may require snow), and ores. Exterior tops never consume
59/// placement budgets, and nothing stays on optional spikes.
60pub(super) fn place_level(
61    level: &CompiledLevel,
62    libraries: &GenerationLibraries,
63    images: &Assets<Image>,
64    base_biomes: Option<&LevelBiomeMask>,
65    chunk_colors: Option<&LevelChunkColorMask>,
66    occupancy: Option<&LevelChunkOccupancyMask>,
67    rooms: &[secret_rooms::SecretRoomPlacement],
68) -> LevelPlacements {
69    let mut taken = vec![false; (level.width * level.height) as usize];
70    reserve_ore_spawn_tiles(level, &mut taken);
71    // A closed room's entrance would show an entity growing inside, so rooms hold only their ore.
72    for tile in rooms.iter().flat_map(secret_rooms::SecretRoomPlacement::tiles) {
73        taken[(tile.y as u32 * level.width + tile.x as u32) as usize] = true;
74    }
75    let placeable = placement_area(level, occupancy);
76    let geysers = EntityPass { library: &libraries.entities.ice, kinds: |kind| kind == GEYSER_KIND, allowed: Some(&placeable), snow: None };
77    let mut entities = generate_entity_placements(level, images, chunk_colors, &geysers, &mut taken);
78    let biomes = build_tundra_mask_from_geysers(level, &entities, base_biomes);
79    let ice_allowed: Vec<bool> = biomes.tundra_inner.iter().zip(&placeable).map(|(&inner, &allowed)| !inner && allowed).collect();
80    let tundra_allowed: Vec<bool> = biomes.tundra_inner.iter().zip(&placeable).map(|(&inner, &allowed)| inner && allowed).collect();
81    let mut decorations = generate_decoration_placements(level, &libraries.decorations.ice, images, Some(&ice_allowed));
82    let snow = build_decoration_mask_for_kind(&decorations, "snow", level.width, level.height);
83    decorations.extend(generate_decoration_placements(level, &libraries.decorations.tundra, images, Some(&tundra_allowed)));
84    let passes = [
85        EntityPass { library: &libraries.entities.ice, kinds: |kind| kind != GEYSER_KIND, allowed: Some(&ice_allowed), snow: Some(&snow) },
86        EntityPass { library: &libraries.entities.tundra, kinds: |_| true, allowed: Some(&tundra_allowed), snow: None },
87    ];
88    for pass in &passes {
89        entities.extend(generate_entity_placements(level, images, chunk_colors, pass, &mut taken));
90    }
91    entities.extend(generate_ore_placements(level, &libraries.ores.ice, &OreCategory::ICE, Some(&ice_allowed)));
92    entities.extend(generate_ore_placements(level, &libraries.ores.tundra, &OreCategory::TUNDRA, Some(&tundra_allowed)));
93    decorations.retain(|placement| !tile_is_optional_spike(level, placement.tile_x, placement.tile_y));
94    entities.retain(|placement| !tile_is_optional_spike(level, placement.tile_x, placement.tile_y));
95    layer_entities(&mut entities, images);
96    LevelPlacements { biomes, ice_allowed, decorations, entities }
97}

99/// Gives every entity its own depth: sprites sharing a depth have no stable draw order, so
100/// overlapping ones would trade places from frame to frame. Taller sprites stand behind shorter
101/// ones, then each entity's incoming offset (an ore cluster's slot order) and placement order
102/// break ties. All entities stay in one band just behind [`ENTITY_Z`].
103fn layer_entities(entities: &mut [EntityPlacement], images: &Assets<Image>) {
104    let heights: Vec<u32> = entities.iter().map(|placement| images.get(&placement.handle).map_or(0, Image::height)).collect();
105    let mut back_to_front: Vec<usize> = (0..entities.len()).collect();
106    back_to_front.sort_by(|&a, &b| {
107        heights[b].cmp(&heights[a])
108            .then(entities[a].z_offset.total_cmp(&entities[b].z_offset))
109            .then(a.cmp(&b))
110    });
111    let frontmost = back_to_front.len().saturating_sub(1) as f32;
112    for (depth, index) in back_to_front.into_iter().enumerate() {
113        entities[index].z_offset = (depth as f32 - frontmost) * ENTITY_DEPTH_STEP;
114    }
115}

117/// Exterior sky above an isolated chunk's first rock surface is not a content habitat. This mask
118/// runs before rule quotas and spacing, leaving those budgets available for interior ledges.
119pub(crate) fn placement_area(level: &CompiledLevel, occupancy: Option<&LevelChunkOccupancyMask>) -> Vec<bool> {
120    let mut allowed = vec![true; (level.width * level.height) as usize];
121    let Some(occupancy) = occupancy else { return allowed };
122    let spawn = player_spawn(level).map(|spawn| spawn.chunk);
123    let chunks = UVec2::new(level.width.div_ceil(mapgen::CHUNK_IMAGE_SIZE), level.height.div_ceil(mapgen::CHUNK_IMAGE_SIZE));
124    for (cx, cy) in (0..chunks.y).flat_map(|y| (0..chunks.x).map(move |x| (x, y))) {
125        let occupied = occupancy.is_occupied(cx, cy);
126        let exposed = cy == 0 || !occupancy.is_occupied(cx, cy - 1);
127        let origin = UVec2::new(cx, cy) * mapgen::CHUNK_IMAGE_SIZE;
128        let end = (origin + UVec2::splat(mapgen::CHUNK_IMAGE_SIZE)).min(UVec2::new(level.width, level.height));
129        if occupied && (!exposed || spawn == Some(IVec2::new(cx as i32, cy as i32))) { continue; }
130        mask_exterior_columns(level, &mut allowed, origin, end, occupied);
131    }
132    allowed
133}
134
135fn mask_exterior_columns(level: &CompiledLevel, allowed: &mut [bool], origin: UVec2, end: UVec2, occupied: bool) {
136    for x in origin.x..end.x {
137        let bottom = (origin.y..end.y).find(|&y| occupied && level.terrain[(y * level.width + x) as usize]).unwrap_or(end.y);
138        for y in origin.y..bottom { allowed[(y * level.width + x) as usize] = false; }
139    }
140}

142/// Placements per chunk and rule, which a rule's `max` tag caps.
143type RuleCounts = HashMap<(IVec2, u32), u32>;

145/// A rule image's pixels, as offsets from its center.
146#[derive(Default)]
147struct RuleMask {
148    require: Vec<IVec2>,
149    forbid: Vec<IVec2>,
150    require_decor: Vec<IVec2>,
151}
152
153impl RuleMask {
154    fn decode(image: &Image) -> Self {
155        let mut mask = Self::default();
156        let (width, height) = (image.width() as i32, image.height() as i32);
157        let center = IVec2::new(width / 2, height / 2);
158        let pixels = image.data.as_deref().unwrap_or_default().as_chunks::<4>().0.iter().take((width * height) as usize);
159        for (index, pixel) in pixels.enumerate().filter(|(_, pixel)| pixel[3] >= 128) {
160            let offset = IVec2::new(index as i32 % width, index as i32 / width) - center;
161            let offsets = match [pixel[0], pixel[1], pixel[2]] {
162                compiler::COLOR_RULE_REQUIRE => &mut mask.require,
163                compiler::COLOR_RULE_FORBID => &mut mask.forbid,
164                [0xFF, 0xFF, 0xFF] => &mut mask.require_decor,
165                _ => continue,
166            };
167            offsets.push(offset);
168        }
169        mask
170    }
171
172    /// Whether the mask matches with its center on `anchor`, mirrored along the `flip` axes.
173    fn matches(&self, scan: &RuleScan<'_>, anchor: IVec2, flip: BVec2) -> bool {
174        let sign = IVec2::select(flip, IVec2::NEG_ONE, IVec2::ONE);
175        let set = |cells: &[bool], offset: &IVec2| {
176            tile_index(scan.level, anchor + *offset * sign).is_some_and(|index| cells.get(index) == Some(&true))
177        };
178        self.require.iter().all(|offset| set(&scan.level.terrain, offset))
179            && !self.forbid.iter().any(|offset| set(&scan.level.terrain, offset))
180            && self.require_decor.iter().all(|offset| scan.decor.is_some_and(|decor| set(decor, offset)))
181    }
182}

184/// A usable rule of a kind, keyed by its position among them.
185struct PreparedRule<'a> {

187    mask: RuleMask,

191/// Where one kind's rules are matched.
192struct RuleScan<'a> {
193    level: &'a CompiledLevel,

195    decor: Option<&'a [bool]>,

200/// A rule that matched at a tile and won its probability roll.
201struct RuleHit {
202    index: usize,
203    tile: IVec2,
204    chunk: IVec2,
205    flip: BVec2,
206    hash: u64,
207}
208
209impl RuleHit {
210    /// One of the hit's independent rolls.
211    fn roll(&self, salt: u64) -> u64 {
212        splitmix64(self.hash ^ salt)
213    }
214}

216/// The kind's rules whose images require or forbid terrain, in order.
217fn prepare_rules<'a>(kind: &'a PlacementKind, images: &Assets<Image>) -> Vec<PreparedRule<'a>> {
218    let masks = kind.rules.iter().filter_map(|image| Some((image, RuleMask::decode(images.get(&image.handle)?))));
219    masks.filter(|(_, mask)| !mask.require.is_empty() || !mask.forbid.is_empty())
220        .enumerate()
221        .map(|(key, (image, mask))| PreparedRule { image, mask, key: key as u32 })
222        .collect()
223}
224
225impl PreparedRule<'_> {
226    /// Tries the rule at a tile, as drawn and then mirrored as its ruleflip tags allow. Its randflip
227    /// tags then mirror the artwork at random.
228    fn hit(&self, scan: &RuleScan<'_>, index: usize, tile: IVec2) -> Option<RuleHit> {
229        let rule = self.image;
230        let flip = [BVec2::FALSE, BVec2::new(true, false), BVec2::new(false, true), BVec2::TRUE]
231            .into_iter()
232            .filter(|flip| (rule.ruleflip_x || !flip.x) && (rule.ruleflip_y || !flip.y))
233            .find(|&flip| self.mask.matches(scan, tile, flip))?;
234        let hash = tile_hash(tile.x, tile.y, rule.order, scan.kind_hash);
235        if !within_percent(hash, rule.prob_percent) {
236            return None;
237        }
238        let artwork = splitmix64(hash ^ 0xB5C0_FBCF);
239        let random = BVec2::new(rule.randflip_x && artwork & 2 != 0, rule.randflip_y && artwork & 4 != 0);
240        Some(RuleHit { index, tile, chunk: tile / scan.chunk_size, flip: flip | random, hash })
241    }
242
243    /// Whether the rule's `max` tag allows another placement in the chunk.
244    fn below_max(&self, counts: &RuleCounts, chunk: IVec2) -> bool {
245        self.image.max_per_chunk.is_none_or(|max| counts.get(&(chunk, self.key)).copied().unwrap_or(0) < max)
246    }
247}

249/// The index of an in-bounds tile.
250fn tile_index(level: &CompiledLevel, tile: IVec2) -> Option<usize> {
251    let size = IVec2::new(level.width as i32, level.height as i32);
252    (tile.cmpge(IVec2::ZERO).all() && tile.cmplt(size).all()).then(|| (tile.y as u32 * level.width + tile.x as u32) as usize)
253}

255/// Allowed tiles free of terrain and spikes, in row-major order.
256fn open_tiles<'a>(level: &'a CompiledLevel, allowed: Option<&'a [bool]>) -> impl Iterator<Item = (usize, IVec2)> + 'a {
257    let width = level.width as usize;
258    (0..width * level.height as usize)
259        .filter(move |&index| {
260            allowed.is_none_or(|allowed| allowed.get(index) == Some(&true))
261                && level.spike_tiles.get(index) != Some(&true)
262                && level.terrain.get(index) != Some(&true)
263        })
264        .map(move |index| (index, IVec2::new((index % width) as i32, (index / width) as i32)))
265}

271fn placement_kind_hash(kind: &str) -> u64 {
272    kind.bytes().fold(0x9E37_79B9_7F4A_7C15, |hash, byte| {
273        let hash = (hash ^ u64::from(byte)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
274        hash ^ (hash >> 27)
275    })
276}

282/// Picks a sprite with probability proportional to its weight.
283fn pick_weighted<'a>(sprites: impl Iterator<Item = &'a SpriteImage> + Clone, roll: u64) -> Option<&'a SpriteImage> {
284    let total: u32 = sprites.clone().map(|sprite| sprite.weight).sum();
285    let mut left = roll.checked_rem(u64::from(total))? as u32;
286    for sprite in sprites {
287        if left < sprite.weight {
288            return Some(sprite);
289        }
290        left -= sprite.weight;
291    }
292    None
293}

295/// Logs how many placements each kind received.
296fn log_placements<'a>(what: &str, kinds: impl Iterator<Item = &'a str>) {
297    let mut counts = BTreeMap::<&str, usize>::new();
298    for kind in kinds {
299        *counts.entry(kind).or_default() += 1;
300    }
301    let summary: Vec<String> = counts.iter().map(|(kind, count)| format!("{kind}={count}")).collect();
302    info!("levelgen: {what} placements -> {}", if summary.is_empty() { "none".to_string() } else { summary.join(", ") });
303}

305/// Decorations placed so far, and the placements stacked on each tile.
306#[derive(Default)]
307struct Decorations {
308    placements: Vec<DecorationPlacement>,
309    stacks: HashMap<usize, Vec<usize>>,
310}
311
312impl Decorations {
313    /// Whether another decoration may go on the tile. On a taken tile a third of rolls skip it, a
314    /// third stack on top, and a third replace the stack.
315    fn make_room(&mut self, index: usize, roll: u64, counts: &mut RuleCounts) -> bool {
316        let Some(stack) = self.stacks.get_mut(&index).filter(|stack| !stack.is_empty()) else {
317            return true;
318        };
319        match roll % 3 {
320            0 => false,
321            1 => true,
322            _ => {
323                stack.drain(..).for_each(|replaced| self.placements[replaced].deactivate(counts));
324                true
325            },
326        }
327    }
328}
329
330impl DecorationPlacement {
331    /// Hides a replaced decoration. Its rule's `max` count is returned under the counts of the kind
332    /// being placed, even when the replaced decoration is of another kind.
333    fn deactivate(&mut self, counts: &mut RuleCounts) {
334        if !self.active {
335            return;
336        }
337        self.active = false;
338        let chunk = IVec2::new(self.tile_x as i32, self.tile_y as i32) / CHUNK_TILES;
339        if let Some(count) = counts.get_mut(&(chunk, self.rule_key)) {
340            *count = count.saturating_sub(1);
341        }
342    }
343}

345/// Places each kind in name order, trying every rule at every open tile.
346pub(super) fn generate_decoration_placements(
347    level: &CompiledLevel,
348    library: &PlacementLibrary,
349    images: &Assets<Image>,
350    allowed: Option<&[bool]>,
351) -> Vec<DecorationPlacement> {
352    let _timer = diagnostics::StageTimer::new("generate_decoration_placements");
353    let mut decorations = Decorations::default();
354    for (name, kind) in &library.kinds {
355        place_decorations(&mut decorations, level, allowed, (name, kind), &prepare_rules(kind, images));
356    }
357    log_placements("decoration", decorations.placements.iter().filter(|p| p.active).map(|p| p.kind.as_str()));
358    decorations.placements
359}
360
361fn place_decorations(
362    decorations: &mut Decorations,
363    level: &CompiledLevel,
364    allowed: Option<&[bool]>,
365    (name, kind): (&str, &PlacementKind),
366    rules: &[PreparedRule<'_>],
367) {
368    let scan = &RuleScan { level, decor: None, kind_hash: placement_kind_hash(name), chunk_size: CHUNK_TILES };
369    let hits = open_tiles(level, allowed)
370        .flat_map(|(index, tile)| rules.iter().filter_map(move |rule| Some((rule, rule.hit(scan, index, tile)?))));
371    let mut counts = RuleCounts::new();
372    for (rule, hit) in hits {
373        if !rule.below_max(&counts, hit.chunk) {
374            continue;
375        }
376        let sprites = kind.sprites.iter().filter(|sprite| has_tags(sprite, &rule.image.tags));
377        let Some(sprite) = pick_weighted(sprites, hit.roll(0xA511_F90A)) else { continue };
378        if !decorations.make_room(hit.index, hit.roll(0xC13F_A9A9), &mut counts) {
379            continue;
380        }
381        decorations.stacks.entry(hit.index).or_default().push(decorations.placements.len());
382        decorations.placements.push(DecorationPlacement {
383            handle: sprite.handle.clone(),
384            tile_x: hit.tile.x as u32,
385            tile_y: hit.tile.y as u32,
386            flip_x: hit.flip.x,
387            flip_y: hit.flip.y,
388            kind: name.to_string(),
389            active: true,
390            offset_px: sprite.offset_px,
391            rule_key: rule.key,
392        });
393        *counts.entry((hit.chunk, rule.key)).or_default() += 1;
394    }
395}

418/// Places each kind in name order. Each open tile outside the player's chunk and six tiles clear of
419/// spikes offers its first rule hit, and `colorrule` rules only hit in chunks of their map color.
420pub(super) fn generate_entity_placements(
421    level: &CompiledLevel,
422    images: &Assets<Image>,
423    chunk_colors: Option<&LevelChunkColorMask>,
424    pass: &EntityPass<'_>,
425    taken: &mut [bool],
426) -> Vec<EntityPlacement> {
427    let _timer = diagnostics::StageTimer::new("generate_entity_placements");
428    let chunk_size = chunk_colors.map(|colors| colors.chunk_size as i32).filter(|&size| size > 0).unwrap_or(CHUNK_TILES);
429    let spawn_chunk = player_spawn(level).map(|spawn| spawn.chunk);
430    let depth_row = spawn_chunk_row(level, chunk_size);
431    let forbidden = spike_clearance(level);
432    let mut placements = Vec::new();
433    for (name, kind) in pass.library.kinds.iter().filter(|(name, _)| (pass.kinds)(name)) {
434        let rules = prepare_rules(kind, images);
435        let scan = RuleScan { level, decor: pass.snow, kind_hash: placement_kind_hash(name), chunk_size };
436        let candidates = open_tiles(level, pass.allowed)
437            .filter(|&(index, tile)| !forbidden[index] && !taken[index] && spawn_chunk != Some(tile / chunk_size))
438            .filter_map(|(index, tile)| {
439                let color = chunk_colors.and_then(|colors| colors.color_for_tile(tile.x as u32, tile.y as u32));
440                rules.iter()
441                    .filter(|rule| rule.image.map_color.is_none_or(|map| color == Some(map)))
442                    .find_map(|rule| Some((rule, rule.hit(&scan, index, tile)?)))
443            })
444            .collect();
445        place_entities(&mut placements, taken, (name, kind), depth_row, candidates);
446    }
447    log_placements("entity", placements.iter().map(|p| p.kind.as_str()));
448    placements
449}

451/// Chunks take turns top to bottom, trying their candidates in hash order. A chunk shows each sprite
452/// at most once, and a rule's `mindist` keeps its entities apart.
453fn place_entities(
454    placements: &mut Vec<EntityPlacement>,
455    taken: &mut [bool],
456    (name, kind): (&str, &PlacementKind),
457    depth_row: Option<i32>,
458    mut candidates: Vec<(&PreparedRule<'_>, RuleHit)>,
459) {
460    candidates.sort_by_key(|(_, hit)| (hit.chunk.y, hit.chunk.x, hit.roll(0xA511_F90A)));
461    let mut shown: HashMap<IVec2, HashSet<&Handle<Image>>> = HashMap::new();
462    let mut counts = RuleCounts::new();
463    let mut spaced = spatial::PlacementIndex::default();
464    for (rule, hit) in candidates {
465        let shown = shown.entry(hit.chunk).or_default();
466        if shown.len() >= kind.sprites.len() || taken[hit.index] || !rule.below_max(&counts, hit.chunk) {
467            continue;
468        }
469        let fits = |sprite: &&SpriteImage| {
470            !shown.contains(&sprite.handle)
471                && has_tags(sprite, &rule.image.tags)
472                && depth_row.is_none_or(|row| minimum_depth(sprite.harvest).is_none_or(|depth| hit.chunk.y - row >= depth))
473        };
474        let Some(sprite) = pick_weighted(kind.sprites.iter().filter(fits), hit.roll(0xC13F_A9A9)) else { continue };
475        let spacing = rule.image.min_distance_tiles as i32;
476        if spacing > 0 && !spaced.clear_square(hit.tile, spacing) {
477            continue;
478        }
479        shown.insert(&sprite.handle);
480        taken[hit.index] = true;
481        *counts.entry((hit.chunk, rule.key)).or_default() += 1;
482        spaced.insert(hit.tile);
483        placements.push(EntityPlacement {
484            handle: sprite.handle.clone(),
485            tile_x: hit.tile.x as u32,
486            tile_y: hit.tile.y as u32,
487            flip_x: hit.flip.x,
488            flip_y: hit.flip.y,
489            kind: name.to_string(),
490            chunk: hit.chunk,
491            offset_px: sprite.offset_px,
492            z_offset: 0.0,
493            harvest: sprite.harvest,
494            ore: None,
495        });
496    }

499/// Entities keep six tiles clear of spikes.
500fn spike_clearance(level: &CompiledLevel) -> Vec<bool> {
501    let (width, height) = (level.width as i32, level.height as i32);
502    let mut forbidden = vec![false; (level.width * level.height) as usize];
503    for spike in (0..forbidden.len()).filter(|&index| level.spike_tiles.get(index) == Some(&true)) {
504        let (x, y) = (spike as i32 % width, spike as i32 / width);
505        let columns = (x - 6).max(0) as usize..=(x + 6).min(width - 1) as usize;
506        for row in (y - 6).max(0)..=(y + 6).min(height - 1) {
507            let start = row as usize * width as usize;
508            forbidden[start + columns.start()..=start + columns.end()].fill(true);
509        }
510    }
511    forbidden
512}

514/// The chunk row of the player spawn, from which berry depth counts.
515fn spawn_chunk_row(level: &CompiledLevel, chunk_size: i32) -> Option<i32> {
516    let spawn = player_spawn(level)?;
517    let tile_size = level.tile_size();
518    if level.height == 0 || tile_size <= 0.0 {
519        return None;
520    }
521    let bottom = level.height as f32 - 1.0;
522    Some((bottom - (spawn.position.y / tile_size).floor()).clamp(0.0, bottom) as i32 / chunk_size)
523}

525/// Chunk rows below the spawn that berries need, authored for 192-tile chunks.
526fn minimum_depth(harvest: Option<harvestable_materials::HarvestPlantDefinition>) -> Option<i32> {
527    let reference_chunks = match harvest?.material {
528        CraftMaterial::BlueBerry => 2,
529        CraftMaterial::MossBerry => 4,
530        _ => return None,
531    };
532    Some((reference_chunks * 192_u32).div_ceil(mapgen::CHUNK_IMAGE_SIZE) as i32)
533}

535/// Keeps entities out of reach of each ore marker.
536pub(super) fn reserve_ore_spawn_tiles(level: &CompiledLevel, taken: &mut [bool]) {
537    let tile_size = level.tile_size();
538    if tile_size <= 0.0 {
539        return;
540    }
541    let radius = (ORE_ENTITY_EXCLUSION_RADIUS_PX / tile_size).ceil() as i32;
542    let reach: Vec<IVec2> = (-radius..=radius)
543        .flat_map(|y| (-radius..=radius).map(move |x| IVec2::new(x, y)))
544        .filter(|offset| {
545            (offset.as_vec2() * tile_size).length_squared() <= ORE_ENTITY_EXCLUSION_RADIUS_PX * ORE_ENTITY_EXCLUSION_RADIUS_PX
546        })
547        .collect();
548    for &marker in level.ore_spawn_tiles.iter().filter(|&&marker| tile_index(level, marker).is_some()) {
549        for index in reach.iter().filter_map(|&offset| tile_index(level, marker + offset)) {
550            taken[index] = true;
551        }
552    }
553}

555/// A cluster of two to four ores on each marker outside the player's chunk, of a random category
556/// that has sprites.
557pub(super) fn generate_ore_placements(
558    level: &CompiledLevel,
559    library: &OreLibrary,
560    categories: &[OreCategory],
561    allowed: Option<&[bool]>,
562) -> Vec<EntityPlacement> {
563    let categories: Vec<OreCategory> =
564        categories.iter().copied().filter(|&category| !library.sprites_for(category).is_empty()).collect();
565    let spawn_chunk = player_spawn(level).map(|spawn| spawn.chunk);
566    let mut seen = HashSet::new();
567    let mut placements = Vec::new();
568    for &tile in &level.ore_spawn_tiles {
569        let Some(index) = tile_index(level, tile) else { continue };
570        let chunk = tile / CHUNK_TILES;
571        if !seen.insert(tile)
572            || spawn_chunk == Some(chunk)
573            || allowed.is_some_and(|allowed| allowed.get(index) != Some(&true))
574            || categories.is_empty()
575        {
576            continue;
577        }
578        let seed = splitmix64(((tile.x as i64 as u64) << 32) ^ ((tile.y as i64 as u64) << 16) ^ 0x4DFB_FF0A_EC0D_E123);
579        let category = categories[(seed % categories.len() as u64) as usize];
580        let count = 2 + (splitmix64(seed ^ 0x517A_61A5) % 3) as usize;
581        let cluster = choose_ore_sprites(library.sprites_for(category), count, seed).into_iter().zip(ore_cluster_offsets(seed));
582        placements.extend(cluster.enumerate().map(|(slot, (ore, offset))| EntityPlacement {
583            handle: ore.handle.clone(),
584            tile_x: tile.x as u32,
585            tile_y: tile.y as u32,
586            flip_x: splitmix64(seed ^ 0xF11F_50A5 ^ (slot as u64).wrapping_mul(17)) & 1 == 1,
587            flip_y: false,
588            kind: category.dir_name().to_string(),
589            chunk,
590            offset_px: IVec2::new(offset, ORE_VERTICAL_OFFSET_PX),
591            z_offset: slot as f32 * 0.000_000_001,
592            harvest: None,
593            ore: Some(ore.definition),
594        }));
595    }
596    log_placements("ore", placements.iter().map(|p| p.kind.as_str()));
597    placements
598}

600/// Horizontal offsets of a cluster's ores: the center, either side, then one further out.
601fn ore_cluster_offsets(seed: u64) -> [i32; 4] {
602    let distance = |salt: u64| 6 + (splitmix64(seed ^ salt) % 13) as i32;
603    let (left, right, extra) = (-distance(0xB8C1_5A11), distance(0xC0B4_17E5), distance(0xD6E8_0A93));
604    [0, left, right, if splitmix64(seed ^ 0xA2F1_3C77) & 1 == 0 { left - extra } else { right + extra }]
605}
606
607/// One ore per slot, each of a different variant in a seeded order while variants last.
608fn choose_ore_sprites(sprites: &[OreImage], count: usize, seed: u64) -> Vec<&OreImage> {
609    let mut variants: Vec<u32> = sprites.iter().map(|sprite| sprite.definition.variant).collect();
610    variants.sort_unstable();
611    variants.dedup();
612    variants.sort_by_key(|&variant| splitmix64(seed ^ u64::from(variant).wrapping_mul(0x9E37_79B1)));
613    (0..count)
614        .map(|slot| {
615            let salt = slot as u64;
616            let Some(&variant) = variants.get(slot) else {
617                return &sprites[(splitmix64(seed ^ 0x8F3D_AA91 ^ salt.wrapping_mul(131)) % sprites.len() as u64) as usize];
618            };
619            let matching: Vec<&OreImage> = sprites.iter().filter(|sprite| sprite.definition.variant == variant).collect();
620            matching[(splitmix64(seed ^ u64::from(variant) ^ salt.wrapping_mul(97)) % matching.len() as u64) as usize]
621        })
622        .collect()
623}

625/// Grows tundra around each geyser over the base biomes.
626pub(super) fn build_tundra_mask_from_geysers(
627    level: &CompiledLevel,
628    geysers: &[EntityPlacement],
629    base: Option<&LevelBiomeMask>,
630) -> LevelBiomeMask {
631    let (size, len) = (UVec2::new(level.width, level.height), (level.width * level.height) as usize);
632    let cells = match base {
633        Some(mask) if mask.matches_dims(size.x, size.y) => mask.cells.clone(),
634        _ => vec![BiomeKind::Ice; len],
635    };
636    let tundra_inner = cells.iter().map(|biome| biome.is_tundra()).collect();
637    let mut mask = LevelBiomeMask { width: size.x, height: size.y, cells, tundra_inner };
638    let reach = TUNDRA_RADIUS_TILES.ceil() as u32;
639    let inner = TUNDRA_RADIUS_TILES * TUNDRA_INNER_FRACTION;
640    let stamps = geysers.iter().flat_map(|geyser| {
641        let center = UVec2::new(geyser.tile_x, geyser.tile_y);
642        let region = surface::expand(URect { min: center, max: center + UVec2::ONE }, reach, size);
643        let tiles = (region.min.y..region.max.y).flat_map(move |y| (region.min.x..region.max.x).map(move |x| UVec2::new(x, y)));
644        tiles.map(move |tile| (tile, (tile.as_vec2() - center.as_vec2()).length_squared()))
645    });
646    for (tile, distance2) in stamps.filter(|&(_, distance2)| distance2 <= TUNDRA_RADIUS_TILES * TUNDRA_RADIUS_TILES) {
647        let index = (tile.y * size.x + tile.x) as usize;
648        mask.cells[index] = BiomeKind::Tundra;
649        mask.tundra_inner[index] |= distance2 <= inner * inner;
650    }
651    mask
652}

661/// Spawns entities whose sprites loaded, foliage through its instanced renderer when it can.
662pub(super) fn spawn_entities(
663    commands: &mut Commands,
664    parent: Entity,
665    level: &CompiledLevel,
666    placements: &[EntityPlacement],
667    assets: &mut EntitySpawnAssets<'_>,
668) -> usize {
669    let tile_size = level.tile_size();
670    let (foliage, sprites): (Vec<_>, Vec<_>) = placements
671        .iter()
672        .filter(|placement| assets.images.contains(&placement.handle))
673        .partition(|placement| foliage::is_foliage_kind(&placement.kind));
674    for placement in &sprites {
675        spawn_entity(commands, parent, level, tile_size, placement);
676    }
677    let foliage_spawned = foliage::spawn_foliage(
678        commands,
679        parent,
680        level,
681        &foliage,
682        assets.images,
683        assets.meshes,
684        assets.foliage_materials,
685        assets.foliage_settings,
686    )
687    .unwrap_or_else(|error| {
688        warn!("{error}; spawning foliage as regular sprites");
689        for placement in &foliage {
690            spawn_entity(commands, parent, level, tile_size, placement);
691        }
692        foliage.len()
693    });
694    sprites.len() + foliage_spawned
695}
696
697fn spawn_entity(commands: &mut Commands, parent: Entity, level: &CompiledLevel, tile_size: f32, placement: &EntityPlacement) {
698    let bottom = (level.height as f32 - placement.tile_y as f32 - 1.0) * tile_size;
699    let anchor = Vec3::new((placement.tile_x as f32 + 0.5) * tile_size, bottom, ENTITY_Z);
700    let mut entity = commands.spawn((
701        Sprite { image: placement.handle.clone(), flip_x: placement.flip_x, flip_y: placement.flip_y, ..default() },
702        Anchor::BOTTOM_CENTER,
703        Transform::from_translation(anchor + placement.offset_px.as_vec2().extend(placement.z_offset)),
704        StateDespawnMarker,
705        Name::new(format!("Entity:{}@{},{}", placement.kind, placement.tile_x, placement.tile_y)),
706        ChildOf(parent),
707    ));
708    if let Some(definition) = placement.harvest {
709        entity.insert(harvestable_materials::HarvestablePlant::from_definition(definition));
710    }
711    if let Some(definition) = placement.ore {
712        entity.insert(harvestable_materials::HarvestableOre::from_definition(definition));
713    }
714}
