1//! Secret room covers: the ground as it was before carving, over a room and the border its
2//! carving reshades, until the room's breakable wall falls.
3//!
4//! Rooms keep two borders apart (see `carve_secret_rooms`), so each cover shows exactly what
5//! the ground would show there, whichever rooms are open. What grows inside a room stays
6//! hidden until it opens: art in its entrance would otherwise show past the cover.
7use super::*;
8use secret_rooms::{RoomTile, SecretRoomPlacement};
9
10/// Between the ground (0) and the player (~0.000015).
11const COVER_Z: f32 = 0.000_01;
12/// Below the cover, which leaves the wall's tiles open.
13const WALL_Z: f32 = 0.000_000_011;
14/// Just above the ground the room's decorations grow on.
15const CONTENTS_Z: f32 = 0.000_000_001;
16
17/// A room's art, cut from the level baked before carving.
18pub(super) struct RoomCover {
19    /// The tiles the cover spans: the room and its border.
20    region: URect,
21    /// The room and its border before carving, with the level's decorations outside rooms.
22    cover: Image,
23    /// The decorations anchored in the room, shown once it opens.
24    contents: Option<Image>,
25    /// The breakable wall, and the tiles it spans.
26    wall: Option<(Image, URect)>,
27}
28
29/// Bakes each room's cover, taking the decorations anchored in rooms out of `decorations`
30/// so the ground grows them only once their room opens.
31pub(super) fn bake(
32    original: &CompiledLevel, geometry: &TerrainGeometry<'_>, rules: &ldtk_rules::LdtkRuleData,
33    biomes: Option<&LevelBiomeMask>, decorations: &mut Vec<DecorationPlacement>, images: &Assets<Image>,
34) -> Vec<RoomCover> {
35    let rooms = &geometry.rooms;
36    if rooms.is_empty() {
37        return Vec::new();
38    }
39    let _timer = diagnostics::StageTimer::new("secret_cover_preparation");
40    let width = original.width;
41    let mut owner = vec![None; (width * original.height) as usize];
42    let tiles = rooms.iter().enumerate().flat_map(|(index, room)| room.tiles().map(move |tile| (tile, index)));
43    for (tile, room) in tiles {
44        owner[(tile.y as u32 * width + tile.x as u32) as usize] = Some(room);
45    }
46    let contents: Vec<Vec<_>> = (0..rooms.len()).map(|room| decorations
47        .extract_if(.., |decoration| owner[(decoration.tile_y * width + decoration.tile_x) as usize] == Some(room))
48        .collect()).collect();
49    let Some(surface) = prepare_main_surface(original, &geometry.mosaics, rules, images, GROUND_TILE_ART_SIZE_PX, biomes)
50    else {
51        return Vec::new();
52    };
53    let size = UVec2::new(width, original.height);
54    rooms.iter().zip(contents).map(|(room, contents)| {
55        let region = surface::expand(room.bounds(), secret_rooms::cover_border_tiles(), size);
56        let art = surface::bake_ground(&surface, original, region, decorations, images).color;
57        RoomCover::cut(room, region, art, &contents, images)
58    }).collect()
59}

61impl RoomCover {
62    /// Cuts the cover and the wall out of `art`, the level before carving baked over
63    /// `region`, and draws the room's own decorations for when it opens.
64    fn cut(room: &SecretRoomPlacement, region: URect, mut cover: Image,
65        contents: &[DecorationPlacement], images: &Assets<Image>,
66    ) -> Self {
67        let tile_px = GROUND_TILE_ART_SIZE_PX;
68        let size = region.size();
69        let local = |tile: IVec2| tile.as_uvec2() - region.min;
70        let mut footprint = vec![false; (size.x * size.y) as usize];
71        for tile in room.tiles().map(local) {
72            footprint[(tile.y * size.x + tile.x) as usize] = true;
73        }
74        let border = secret_rooms::dilate(&footprint, size, secret_rooms::cover_border_tiles());
75        let wall_tiles: HashSet<_> = room.tiles_of(RoomTile::BreakableWall).map(local).collect();
76        let wall = secret_rooms::tile_rect(room.tiles_of(RoomTile::BreakableWall)).map(|tiles| {
77            let min = tiles.min - region.min;
78            let mut image = surface::crop(&cover, URect::from_corners(min * tile_px, (tiles.max - region.min) * tile_px));
79            keep_tiles(&mut image, min, |tile| wall_tiles.contains(&tile));
80            (image, tiles)
81        });
82        keep_tiles(&mut cover, UVec2::ZERO, |tile| border[(tile.y * size.x + tile.x) as usize] && !wall_tiles.contains(&tile));
83        let contents = (!contents.is_empty()).then(|| {
84            let mut image = surface::image(size * tile_px);
85            surface::decorate(&mut image, region.min * tile_px, contents, images, tile_px);
86            image
87        });
88        Self { region, cover, contents, wall }
89    }
90}
91
92/// Clears the pixels of `image`, whose top-left tile is `origin`, in the tiles `keep` rejects.
93fn keep_tiles(image: &mut Image, origin: UVec2, keep: impl Fn(UVec2) -> bool) {
94    let width = image.width();
95    let pixels = image.data.as_mut().expect("cover pixels").as_chunks_mut::<4>().0;
96    for (index, pixel) in pixels.iter_mut().enumerate() {
97        let tile = origin + UVec2::new(index as u32 % width, index as u32 / width) / GROUND_TILE_ART_SIZE_PX;
98        if !keep(tile) {
99            *pixel = [0; 4];
100        }
101    }
102}

104/// Spawns each room's cover, contents and breakable wall, with the collision tiles that
105/// keep the room closed until the wall breaks.
106pub(super) fn spawn(commands: &mut Commands, parent: Entity, rooms: &[SecretRoomPlacement], covers: Vec<RoomCover>,
107    images: &mut Assets<Image>, level_height: u32,
108) {
109    let tile_px = GROUND_TILE_ART_SIZE_PX as f32;
110    let place = |tiles: URect, z: f32| {
111        let center = tiles.as_rect().center() * tile_px;
112        Transform::from_xyz(center.x, level_height as f32 * tile_px - center.y, z)
113    };
114    for (room, RoomCover { region, cover, contents, wall }) in rooms.iter().zip(covers) {
115        info!("Spawned cover for secret room '{}' over tiles {region:?}", room.name);
116        let cover = commands.spawn((Sprite::from_image(images.add(cover)), place(region, COVER_Z),
117            pixel_perfect_layers(), StateDespawnMarker, breakable_wall::SecretRoomCover,
118            CoverCollisionTiles { tiles: room.tiles().collect() },
119            Name::new(format!("SecretRoomCover_{}", room.name)), ChildOf(parent))).id();
120        let contents = contents.map(|image| commands.spawn((Sprite::from_image(images.add(image)),
121            place(region, CONTENTS_Z), Visibility::Hidden, pixel_perfect_layers(), StateDespawnMarker,
122            Name::new(format!("SecretRoomContents_{}", room.name)), ChildOf(parent))).id());
123        let Some((image, tiles)) = wall else { continue; };
124        let size = tiles.size().as_vec2() * tile_px;
125        commands.spawn((Sprite::from_image(images.add(image)), place(tiles, WALL_Z),
126            Collider::cuboid(size.x * 0.5, size.y * 0.5), StaticCollider,
127            // Attacks that hit enemies hit the wall; the cover's collision tiles block movement.
128            CollisionGroups::new(ENEMY, Group::all()), pixel_perfect_layers(), StateDespawnMarker,
129            breakable_wall::BreakableWall { hits_remaining: breakable_wall::BREAKABLE_WALL_HITS, cover, contents },
130            breakable_wall::SecretRoomCover, Name::new(format!("BreakableWallCover_{}", room.name)), ChildOf(parent)));
131    }
132}
