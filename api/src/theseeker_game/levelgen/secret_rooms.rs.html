1//! Secret rooms: rule images carved into the terrain, hidden behind covers until their
2//! breakable wall falls.
3//!
4//! A rule PNG is centered on the tile it is placed at. Its black, white, `#5b666f` and
5//! `#4dfbff` pixels are the room, all carved out: white is the breakable wall, `#5b666f`
6//! also gets manual near background and `#4dfbff` marks ore. Green pixels require terrain;
7//! red pixels are the entrance, which must be open cave without spikes.
8

11/// Directory containing secret room rule PNGs.
12pub(super) const SECRET_ROOMS_DIR: &str = "assets/levels/secret_rooms";

14/// What a room tile is before the room opens and it becomes air.
15#[derive(Clone, Copy, PartialEq, Eq, Debug)]
16pub(super) enum RoomTile {
17    Interior,
18    BreakableWall,
19    NearBackground,
20    OreMarker,
21}

23#[derive(Clone, Debug)]
24pub(super) struct SecretRoomRule {
25    pub name: String,
26    /// The room's tiles by offset from the rule's center, in image scan order.
27    pub room: Vec<(IVec2, RoomTile)>,
28    pub require_terrain: Vec<IVec2>,
29    /// The entrance.
30    pub forbid_terrain: Vec<IVec2>,
31    /// Whether the room may also be placed mirrored.
32    pub ruleflip_x: bool,
33}

35/// The rules, loaded once and sorted by name so placement is deterministic.
36#[derive(Resource, Default)]
37pub(super) struct SecretRoomLibrary {
38    rules: Vec<SecretRoomRule>,
39    loaded: bool,
40}

42pub(super) fn load_secret_rooms(mut library: ResMut<SecretRoomLibrary>) {
43    if std::mem::replace(&mut library.loaded, true) {
44        return;
45    }
46    let Ok(entries) = fs::read_dir(SECRET_ROOMS_DIR)
47        .inspect_err(|err| warn!("Failed to read secret rooms directory {SECRET_ROOMS_DIR}: {err}"))
48    else {
49        return;
50    };
51    let pngs = entries.flatten().map(|entry| entry.path())
52        .filter(|path| path.extension().is_some_and(|ext| ext.eq_ignore_ascii_case("png")));
53    let mut rules: Vec<_> = pngs.filter_map(|path| load_rule(&path)
54        .inspect_err(|err| warn!("Failed to load secret room rule {}: {err}", path.display())).ok()).collect();
55    rules.sort_by(|a, b| a.name.cmp(&b.name));
56    if !rules.is_empty() {
57        info!("Loaded {} secret room rules from {SECRET_ROOMS_DIR}", rules.len());
58    }
59    library.rules = rules;
60}

62pub(super) fn load_rule(path: &Path) -> Result<SecretRoomRule, String> {
63    let name = path.file_stem().and_then(|stem| stem.to_str()).unwrap_or_default();
64    let lower = name.to_ascii_lowercase();
65    parse_rule(&mapgen::load_png(path)?, name, lower.contains("ruleflip-x") || lower.contains("ruleflip_x"))
66}

68fn parse_rule(image: &Image, name: &str, ruleflip_x: bool) -> Result<SecretRoomRule, String> {
69    let data = image.data.as_deref().ok_or("image has no pixel data")?;
70    let width = image.width() as i32;
71    let center = UVec2::new(image.width() / 2, image.height() / 2).as_ivec2();
72    let mut rule = SecretRoomRule { name: name.to_owned(), room: Vec::new(), require_terrain: Vec::new(),
73        forbid_terrain: Vec::new(), ruleflip_x };
74    let pixels = data.as_chunks::<4>().0.iter().take((image.width() * image.height()) as usize);
75    for (index, pixel) in pixels.enumerate().filter(|(_, pixel)| pixel[3] >= 128) {
76        let offset = IVec2::new(index as i32 % width, index as i32 / width) - center;
77        match [pixel[0], pixel[1], pixel[2]] {
78            [0, 0, 0] => rule.room.push((offset, RoomTile::Interior)),
79            [0xFF, 0xFF, 0xFF] => rule.room.push((offset, RoomTile::BreakableWall)),
80            compiler::COLOR_NEAR_BG_MANUAL => rule.room.push((offset, RoomTile::NearBackground)),
81            compiler::COLOR_ORE_MARKER => rule.room.push((offset, RoomTile::OreMarker)),
82            compiler::COLOR_RULE_REQUIRE => rule.require_terrain.push(offset),
83            compiler::COLOR_RULE_FORBID => rule.forbid_terrain.push(offset),
84            _ => {},
85        }
86    }
87    if rule.room.is_empty() {
88        return Err("no room pixels: black, white, #5b666f or #4dfbff".into());
89    }
90    Ok(rule)
91}

93/// A carved room.
94#[derive(Clone, Debug)]
95pub(super) struct SecretRoomPlacement {
96    pub name: String,
97    /// The room's level tiles, which lie inside the level: rules match only on terrain.
98    pub room: Vec<(IVec2, RoomTile)>,
99}

101impl SecretRoomPlacement {
102    pub fn tiles(&self) -> impl Iterator<Item = IVec2> + Clone + '_ {
103        self.room.iter().map(|&(tile, _)| tile)
104    }
105
106    /// The room's tiles of one kind, in rule scan order.
107    pub fn tiles_of(&self, kind: RoomTile) -> impl Iterator<Item = IVec2> + '_ {
108        self.room.iter().filter(move |&&(_, tile_kind)| tile_kind == kind).map(|&(tile, _)| tile)
109    }
110
111    pub fn bounds(&self) -> URect {
112        tile_rect(self.tiles()).expect("rules have room tiles")
113    }
114}

116/// The smallest rectangle holding `tiles`.
117pub(super) fn tile_rect(tiles: impl Iterator<Item = IVec2>) -> Option<URect> {
118    tiles.map(|tile| IRect::from_corners(tile, tile + 1)).reduce(|a, b| a.union(b)).map(|rect| rect.as_urect())
119}

121/// How far, in tiles, carving a room reshades the ground around it (the reach of edge
122/// shading, rounded up) plus a tile of margin. A room's cover hides this border too.
123pub(super) fn cover_border_tiles() -> u32 {
124    let reach_px = GROUND_DARKEN_START_PX as f32 + GROUND_DARKEN_RANGE_PX;
125    (reach_px / GROUND_TILE_ART_SIZE_PX as f32).ceil() as u32 + 1
126}

128pub(super) struct CarvingResult {
129    pub terrain: Vec<bool>,
130    pub placements: Vec<SecretRoomPlacement>,
131}
132
133/// Places each rule at most once and carves its room out of the terrain.
134///
135/// Rooms stay two cover borders away from each other, from optional spikes and from
136/// `crater`, the tiles the intro impact changes. So a room's carving never reaches another
137/// room's cover, and no cover ever hides art that changes during play.
138pub(super) fn carve_secret_rooms(level: &CompiledLevel, library: &SecretRoomLibrary, crater: &[UVec2]) -> CarvingResult {
139    let _timer = diagnostics::StageTimer::new("secret_room_carving");
140    let size = UVec2::new(level.width, level.height);
141    let index = |tile: UVec2| (tile.y * size.x + tile.x) as usize;
142    let gap = 2 * cover_border_tiles();
143    let mut changing = level.optional_spike_tiles.clone();
144    changing.resize(level.terrain.len(), false);
145    for &tile in crater {
146        changing[index(tile)] = true;
147    }
148    let mut reserved = dilate(&changing, size, gap);
149    let mut terrain = level.terrain.clone();
150    let mut placements = Vec::new();
151    for rule in &library.rules {
152        let Some((anchor, flip)) = best_placement(level, &reserved, rule) else { continue; };
153        let room: Vec<_> = rule.room.iter().map(|&(offset, kind)| (anchor + mirrored(offset, flip), kind)).collect();
154        let mut footprint = vec![false; terrain.len()];
155        for tile in room.iter().map(|&(tile, _)| index(tile.as_uvec2())) {
156            terrain[tile] = false;
157            footprint[tile] = true;
158        }
159        reserved.iter_mut().zip(dilate(&footprint, size, gap)).for_each(|(reserved, near)| *reserved |= near);
160        info!("Secret room '{}' carved at {anchor}, flipped={flip}", rule.name);
161        placements.push(SecretRoomPlacement { name: rule.name.clone(), room });
162    }
163    CarvingResult { terrain, placements }
164}

166/// The fitting anchor and orientation with the lowest hash, the first one on ties.
167fn best_placement(level: &CompiledLevel, reserved: &[bool], rule: &SecretRoomRule) -> Option<(IVec2, bool)> {
168    let flips: &[bool] = if rule.ruleflip_x { &[false, true] } else { &[false] };
169    let anchors = (0..level.height as i32).flat_map(|y| (0..level.width as i32).map(move |x| IVec2::new(x, y)));
170    let mut best: Option<(u64, IVec2, bool)> = None;
171    for (anchor, flip) in anchors.flat_map(|anchor| flips.iter().map(move |&flip| (anchor, flip))) {
172        let hash = anchor_hash(anchor, flip);
173        if best.is_none_or(|(lowest, ..)| hash < lowest) && fits(level, reserved, rule, anchor, flip) {
174            best = Some((hash, anchor, flip));
175        }
176    }
177    best.map(|(_, anchor, flip)| (anchor, flip))
178}

180/// Rooms go where the anchor hash is lowest; a mirrored room ranks just after the unmirrored one.
181fn anchor_hash(anchor: IVec2, flip: bool) -> u64 {
182    let position = ((anchor.x as i64 as u64) << 32) ^ ((anchor.y as i64 as u64) << 16);
183    super::hash::mix64(position ^ 0xCAFE_BABE_DEAD_BEEF).wrapping_add(u64::from(flip))
184}

186/// Whether the room fits: an open entrance in the authored cave, the required tiles solid,
187/// and the room itself unreserved terrain.
188fn fits(level: &CompiledLevel, reserved: &[bool], rule: &SecretRoomRule, anchor: IVec2, flip: bool) -> bool {
189    let at = |offset: IVec2| {
190        let tile = anchor + mirrored(offset, flip);
191        let inside = tile.cmpge(IVec2::ZERO).all() && tile.cmplt(UVec2::new(level.width, level.height).as_ivec2()).all();
192        inside.then(|| (tile.y as u32 * level.width + tile.x as u32) as usize)
193    };
194    let open = |offset: &IVec2| at(*offset).is_some_and(|index| !level.terrain[index] && level.spike_tiles.get(index) != Some(&true));
195    let solid = |offset: &IVec2| at(*offset).is_some_and(|index| level.terrain[index]);
196    let free = |(offset, _): &(IVec2, RoomTile)| at(*offset).is_some_and(|index| level.terrain[index] && !reserved[index]);
197    rule.forbid_terrain.iter().all(open) && rule.require_terrain.iter().all(solid) && rule.room.iter().all(free)
198}
199
200fn mirrored(offset: IVec2, flip: bool) -> IVec2 {
201    if flip { IVec2::new(-offset.x, offset.y) } else { offset }
202}

204/// Marks every tile within `radius` tiles of a marked one along both axes.
205pub(super) fn dilate(mask: &[bool], size: UVec2, radius: u32) -> Vec<bool> {
206    let (width, height, radius) = (size.x as usize, size.y as usize, radius as usize);
207    let rows = dilate_lines(mask, (width, 1), (height, width), radius);
208    dilate_lines(&rows, (height, width), (width, 1), radius)
209}
210
211/// Dilates each line of `mask` along itself: `len` cells `step` apart, in `lines` lines
212/// whose starts lie `stride` apart.
213fn dilate_lines(mask: &[bool], (len, step): (usize, usize), (lines, stride): (usize, usize), radius: usize) -> Vec<bool> {
214    let mut out = vec![false; mask.len()];
215    let mut marked = vec![0; len + 1];
216    for line in 0..lines {
217        let cell = |i: usize| line * stride + i * step;
218        for i in 0..len {
219            marked[i + 1] = marked[i] + usize::from(mask[cell(i)]);
220        }
221        for i in 0..len {
222            out[cell(i)] = marked[(i + radius + 1).min(len)] > marked[i.saturating_sub(radius)];
223        }
224    }
225    out
226}
