1//! Exemplar-driven near-background detailing: image analogies on an opening mask.
2//!
3//! An exemplar pairs a context (near wall, openings, and solid terrain) with the artist's
4//! IntGrid values for the same cells. Synthesis reads a target context and copies each value
5//! from the exemplar cell whose surroundings match best, preferring to continue the copies of
6//! already-written neighbours so the artist's strokes and clusters stay whole. Cells deep inside
7//! the wall take its base value and cells deep inside an opening stay empty.
8//! `tools/backgrounds/synthesis.py` details the middle and far planes the same way.
9use super::*;
10use std::collections::HashMap;
11
12/// Context windows span 9x9 cells; the inner 5x5 counts double.
13const RADIUS: i32 = 4;
14const SIDE: i32 = 2 * RADIUS + 1;
15/// Cells written before a cell in row order, within two cells of it.
16const CAUSAL: [(i32, i32); 11] = [(-2, -2), (-2, -1), (-2, 0), (-2, 1), (-2, 2), (-1, -2), (-1, -1), (-1, 0), (-1, 1), (0, -2), (0, -1)];
17/// Adjacent neighbours count double.
18const CAUSAL_WEIGHT: [u32; 11] = [3, 3, 3, 3, 3, 3, 6, 6, 6, 3, 6];
19const SAMPLES: u64 = 10;
20/// How much worse a continued copy may match than the best fresh sample and still win.
21const COHERENCE: f32 = 2.0;
22/// Openings may lose cells this far in from their edge.
23const OPENING_REACH: i32 = 2;
24/// Detailing reaches at most this far into the wall.
25const MOST_REACH: i32 = 8;
26
27/// Exemplar colors (`tools/backgrounds/exemplars.py`): IntGrid values 0–4, black for terrain.
28const VALUE_COLORS: [[u8; 3]; 5] = [[255, 255, 255], [220, 90, 90], [90, 150, 220], [150, 150, 150], [120, 190, 120]];
29const SOLID_COLOR: [u8; 3] = [0, 0, 0];
30
31/// Bit masks over a 9x9 window (row-major, bit `y * 9 + x`).
32struct WindowMasks {
33    inner: u128,
34    outer: u128,
35    small: u128,
36}
37
38fn window_masks() -> WindowMasks {
39    let mut masks = WindowMasks { inner: 0, outer: 0, small: 0 };
40    for bit in 0..SIDE * SIDE {
41        let ring = (bit % SIDE - RADIUS).abs().max((bit / SIDE - RADIUS).abs());
42        let flag = 1u128 << bit;
43        if ring <= 2 { masks.inner |= flag } else { masks.outer |= flag }
44        if ring <= 1 { masks.small |= flag }
45    }
46    masks
47}

49/// What synthesis matches: walls, openings, and solid terrain around every cell.
50pub(super) struct Context {
51    size: UVec2,
52    wall: Vec<bool>,
53    solid: Vec<bool>,
54    /// Chessboard distance to the other side, positive in walls and negative in openings.
55    distance: Vec<i32>,
56    wall_windows: Vec<u128>,
57    open_windows: Vec<u128>,
58}
59
60impl Context {
61    pub(super) fn new(wall: Vec<bool>, solid: Vec<bool>, size: UVec2) -> Self {
62        let inside = chessboard_distance(&wall, size);
63        let opening: Vec<bool> = wall.iter().map(|&covered| !covered).collect();
64        let outside = chessboard_distance(&opening, size);
65        let distance = wall.iter().enumerate()
66            .map(|(index, &covered)| if covered { inside[index] } else { -outside[index] }).collect();
67        let not_solid: Vec<bool> = solid.iter().map(|&set| !set).collect();
68        Self { wall_windows: windows(&wall, size), open_windows: windows(&not_solid, size), size, wall, solid, distance }
69    }
70
71    fn clipped(&self, index: usize, reach: i32) -> i32 {
72        self.distance[index].clamp(-OPENING_REACH - 1, reach + 1)
73    }
74
75    fn keys(&self, index: usize, reach: i32, masks: &WindowMasks) -> [(u128, u128, i32); 3] {
76        let (wall, open, distance) = (self.wall_windows[index], self.open_windows[index], self.clipped(index, reach));
77        [(wall & masks.inner, open & masks.inner, distance), (wall & masks.small, open & masks.small, distance), (0, 0, distance)]
78    }
79}

81/// Each cell's 9x9 neighbourhood of `mask` as bits; cells beyond the edge count as set.
82fn windows(mask: &[bool], size: UVec2) -> Vec<u128> {
83    let (width, height) = (size.x as i32, size.y as i32);
84    let at = |x: i32, y: i32| x < 0 || y < 0 || x >= width || y >= height || mask[(y * width + x) as usize];
85    let rows: Vec<u16> = (0..width * height).map(|index| {
86        let (x, y) = (index % width, index / width);
87        (0..SIDE).fold(0, |bits, dx| bits | u16::from(at(x + dx - RADIUS, y)) << dx)
88    }).collect();
89    let row_bits = |x: i32, row: i32| if (0..height).contains(&row) { rows[(row * width + x) as usize] } else { (1 << SIDE) - 1 };
90    (0..width * height).map(|index| {
91        let (x, y) = (index % width, index / width);
92        (0..SIDE).fold(0, |bits, dy| bits | u128::from(row_bits(x, y + dy - RADIUS)) << (dy * SIDE))
93    }).collect()
94}

96/// Chessboard distance from each set cell to the nearest unset cell; zero on unset cells.
97fn chessboard_distance(mask: &[bool], size: UVec2) -> Vec<i32> {
98    let width = size.x as usize;
99    let far = (size.x + size.y) as i32;
100    let mut distance: Vec<i32> = mask.iter().map(|&set| if set { far } else { 0 }).collect();
101    let neighbour = |index: usize, (dx, dy): (i32, i32)| {
102        let (x, y) = ((index % width) as i32 + dx, (index / width) as i32 + dy);
103        (x >= 0 && y >= 0 && x < size.x as i32 && y < size.y as i32).then(|| y as usize * width + x as usize)
104    };
105    let passes: [(Vec<usize>, [(i32, i32); 4]); 2] = [
106        ((0..mask.len()).collect(), [(-1, 0), (0, -1), (-1, -1), (1, -1)]),
107        ((0..mask.len()).rev().collect(), [(1, 0), (0, 1), (1, 1), (-1, 1)]),
108    ];
109    for (order, offsets) in passes {
110        for index in order {
111            let nearest = offsets.iter().filter_map(|&offset| neighbour(index, offset)).map(|cell| distance[cell] + 1).min();
112            distance[index] = distance[index].min(nearest.unwrap_or(far));
113        }
114    }
115    distance
116}

118/// The artist's near background for one biome, with the context its values answer.
119pub(super) struct Exemplar {
120    context: Context,
121    values: Vec<i32>,
122    base: i32,
123    reach: i32,
124    /// Exemplar cells by exact, coarse, and distance-only context key.
125    pools: [HashMap<(u128, u128, i32), Vec<usize>>; 3],
126    valid: Vec<bool>,
127    masks: WindowMasks,
128}
129
130impl Exemplar {
131    /// Reads an exemplar image and derives its context the way near openings are tidied.
132    pub(super) fn from_image(image: &Image) -> Option<Self> {
133        let size = image.size();
134        let pixels = image.data.as_ref()?.as_chunks::<4>().0.iter().take((size.x * size.y) as usize);
135        let (values, solid): (Vec<i32>, Vec<bool>) = pixels.map(decode_pixel).unzip();
136        let known: Vec<bool> = values.iter().zip(&solid).map(|(&value, &set)| value >= 0 || set).collect();
137        let painted: Vec<bool> = values.iter().zip(&solid).map(|(&value, &set)| value != 0 || set).collect();
138        let wall = background_openings::tidy(&painted, &solid, size);
139        Some(Self::new(Context::new(wall, solid, size), values, &known))
140    }
141
142    fn new(context: Context, mut values: Vec<i32>, known: &[bool]) -> Self {
143        let own: Vec<bool> = known.iter().zip(&context.solid).map(|(&set, &solid)| set && !solid).collect();
144        let base = most_common((0..values.len())
145            .filter(|&index| own[index] && context.wall[index] && context.distance[index] > 6)
146            .map(|index| values[index]));
147        let mut detail: Vec<i32> = (0..values.len())
148            .filter(|&index| own[index] && context.wall[index] && values[index] != base)
149            .map(|index| context.distance[index]).collect();
150        detail.sort_unstable();
151        let reach = detail.get(detail.len() * 99 / 100).copied().unwrap_or(1).clamp(1, MOST_REACH);
152        // Terrain hides the near plane; the artist's cells under it count as plain wall.
153        for (value, _) in values.iter_mut().zip(&context.solid).filter(|(_, &solid)| solid) { *value = base; }
154        let known_distance = chessboard_distance(known, context.size);
155        let (width, height) = (context.size.x as i32, context.size.y as i32);
156        let valid: Vec<bool> = (0..values.len()).map(|index| {
157            let (x, y) = (index as i32 % width, index as i32 / width);
158            let inside = x >= RADIUS && y >= RADIUS && x < width - RADIUS && y < height - RADIUS;
159            inside && own[index] && known_distance[index] > RADIUS && (-OPENING_REACH..=reach).contains(&context.distance[index])
160        }).collect();
161        let masks = window_masks();
162        let mut pools: [HashMap<_, Vec<usize>>; 3] = Default::default();
163        let keyed = (0..values.len()).filter(|&index| valid[index])
164            .flat_map(|index| context.keys(index, reach, &masks).into_iter().enumerate().map(move |(level, key)| (level, key, index)));
165        for (level, key, index) in keyed {
166            pools[level].entry(key).or_default().push(index);
167        }
168        Self { context, values, base, reach, pools, valid, masks }
169    }
170
171    fn width(&self) -> i32 {
172        self.context.size.x as i32
173    }
174
175    /// The exemplar cell `offset` away from `cell`; valid cells keep their causal window inside.
176    fn offset(&self, cell: usize, (dy, dx): (i32, i32)) -> usize {
177        (cell as i32 + dy * self.width() + dx) as usize
178    }
179}

181fn decode_pixel(&[red, green, blue, alpha]: &[u8; 4]) -> (i32, bool) {
182    let color = [red, green, blue];
183    if alpha < 128 { return (-1, false); }
184    let value = VALUE_COLORS.iter().position(|&known| known == color).map_or(-1, |value| value as i32);
185    (value, color == SOLID_COLOR)
186}
187
188fn most_common(values: impl Iterator<Item = i32>) -> i32 {
189    let mut counts = [0usize; VALUE_COLORS.len()];
190    for value in values.filter_map(|value| usize::try_from(value).ok()).filter(|&value| value < VALUE_COLORS.len()) {
191        counts[value] += 1;
192    }
193    (0..counts.len()).max_by_key(|&value| counts[value]).unwrap_or(0) as i32
194}

196/// IntGrid values for the target's cells where `detail` holds: 0 in openings, the exemplar's
197/// values near their edges, and the exemplar's base value deeper in the wall.
198pub(super) fn synthesize(target: &Context, exemplar: &Exemplar, detail: impl Fn(usize) -> bool, seed: u64) -> Vec<i32> {
199    let _timer = diagnostics::StageTimer::new("near_detail_synthesis");
200    let mut synthesis = Synthesis::new(target, exemplar, seed);
201    let reach = exemplar.reach;
202    let band = |index: usize| !target.solid[index] && (-OPENING_REACH..=reach).contains(&target.distance[index]) && detail(index);
203    let pending: Vec<usize> = (0..target.wall.len()).filter(|&index| band(index)).collect();
204    for &index in &pending { synthesis.out[index] = -1; }
205    for index in pending { synthesis.write(index); }
206    synthesis.out
207}

209/// Row-order synthesis state: written values and the exemplar cell each was copied from.
210struct Synthesis<'a> {
211    target: &'a Context,
212    exemplar: &'a Exemplar,
213    seed: u64,
214    out: Vec<i32>,
215    source: Vec<Option<usize>>,
216    candidates: Vec<(usize, bool)>,
217}
218
219impl<'a> Synthesis<'a> {
220    fn new(target: &'a Context, exemplar: &'a Exemplar, seed: u64) -> Self {
221        let out = target.wall.iter().map(|&covered| if covered { exemplar.base } else { 0 }).collect();
222        let source = vec![None; target.wall.len()];
223        Self { target, exemplar, seed, out, source, candidates: Vec::with_capacity(CAUSAL.len() + SAMPLES as usize) }
224    }
225
226    fn neighbour(&self, index: usize, (dy, dx): (i32, i32)) -> Option<usize> {
227        let width = self.target.size.x as i32;
228        let (x, y) = (index as i32 % width + dx, index as i32 / width + dy);
229        (x >= 0 && x < width && y >= 0).then(|| (y * width + x) as usize)
230    }
231
232    fn write(&mut self, index: usize) {
233        self.gather(index);
234        let scored: Vec<_> = self.candidates.iter().enumerate()
235            .map(|(slot, &(candidate, coherent))| (self.score(index, candidate, slot as u64), candidate, coherent)).collect();
236        let best = |coherent: bool| scored.iter().filter(|entry| entry.2 == coherent)
237            .min_by(|a, b| a.0.total_cmp(&b.0)).map_or((f32::INFINITY, None), |entry| (entry.0, Some(entry.1)));
238        let (best_coherent, best_sample) = (best(true), best(false));
239        let chosen = if best_coherent.0 <= best_sample.0 + COHERENCE { best_coherent.1 } else { best_sample.1 };
240        let fallback = if self.target.wall[index] { self.exemplar.base } else { 0 };
241        self.out[index] = chosen.map_or(fallback, |cell| self.exemplar.values[cell]);
242        self.source[index] = chosen;
243    }
244
245    /// Continuations of the neighbours' copies, then fresh samples with the closest context key.
246    fn gather(&mut self, index: usize) {
247        let ex = self.exemplar;
248        self.candidates.clear();
249        for offset in CAUSAL {
250            let copied = self.neighbour(index, offset).and_then(|cell| self.source[cell]);
251            let continued = copied.map(|cell| ex.offset(cell, (-offset.0, -offset.1))).filter(|&cell| ex.valid[cell]);
252            self.candidates.extend(continued.map(|cell| (cell, true)));
253        }
254        let keys = self.target.keys(index, ex.reach, &ex.masks);
255        let Some(pool) = ex.pools.iter().zip(keys).find_map(|(pool, key)| pool.get(&key)) else { return };
256        let samples = (0..SAMPLES).map(|sample| {
257            let hash = splitmix64(self.seed ^ (index as u64).wrapping_mul(0x9E37_79B9) ^ sample << 56);
258            (pool[(hash % pool.len() as u64) as usize], false)
259        });
260        self.candidates.extend(samples);
261    }
262
263    /// Context bits, edge distance, and already-written neighbours that differ, plus a small
264    /// random share that breaks ties.
265    fn score(&self, index: usize, candidate: usize, slot: u64) -> f32 {
266        let (ex, target, masks) = (self.exemplar, self.target, &self.exemplar.masks);
267        let differs = |a: u128, b: u128| 2 * ((a ^ b) & masks.inner).count_ones() + ((a ^ b) & masks.outer).count_ones();
268        let mut total = differs(ex.context.wall_windows[candidate], target.wall_windows[index])
269            + differs(ex.context.open_windows[candidate], target.open_windows[index])
270            + ex.context.clipped(candidate, ex.reach).abs_diff(target.clipped(index, ex.reach));
271        total += CAUSAL.into_iter().zip(CAUSAL_WEIGHT).filter(|&(offset, _)| {
272            let written = self.neighbour(index, offset).map_or(ex.base, |cell| self.out[cell]);
273            written >= 0 && ex.values[ex.offset(candidate, offset)] != written
274        }).map(|(_, weight)| weight).sum::<u32>();
275        let noise = splitmix64(self.seed ^ slot ^ (index as u64) << 20) >> 41;
276        total as f32 + noise as f32 / (1u64 << 23) as f32 * 0.5
277    }
278}

287    /// The artist's exemplar, its terrain (cells outside the painted area count as terrain), and
288    /// its decoded values.
289    fn artist(path: &str) -> (Exemplar, Vec<bool>, Vec<i32>, UVec2) {
290        let image = mapgen::load_png(Path::new(path)).unwrap();
291        let (values, solid): (Vec<i32>, Vec<bool>) = image.data.as_ref().unwrap().as_chunks::<4>().0.iter().map(decode_pixel).unzip();
292        let terrain = values.iter().zip(&solid).map(|(&value, &set)| set || value < 0).collect();
293        (Exemplar::from_image(&image).unwrap(), terrain, values, image.size())
294    }

296    #[test]
297    fn near_detailing_copies_the_artist_values_around_openings_only() {
298        for path in [ICE, TUNDRA] {
299            let (exemplar, terrain, values, size) = artist(path);
300            let wall = background_openings::near_wall(&terrain, size, 7);
301            let target = Context::new(wall.clone(), terrain.clone(), size);
302            let grid = synthesize(&target, &exemplar, |_| true, 11);
303            assert_eq!(grid, synthesize(&target, &exemplar, |_| true, 11), "{path}: a seed reproduces its detailing");
304            assert!(grid.iter().all(|value| values.contains(value)), "{path}: only the artist's values");
305            let deep_opening = |index: usize| target.distance[index] < -OPENING_REACH;
306            let deep_wall = |index: usize| target.distance[index] > exemplar.reach || terrain[index];
307            assert!((0..grid.len()).filter(|&index| deep_opening(index)).all(|index| grid[index] == 0), "{path}");
308            assert!((0..grid.len()).filter(|&index| deep_wall(index)).all(|index| grid[index] == exemplar.base), "{path}");
309            let detail = grid.iter().zip(&terrain).filter(|&(&value, &solid)| !solid && value != 0 && value != exemplar.base).count();
310            assert!(detail > 0, "{path}: openings receive detailing");
311        }
312    }

336    fn shade_terrain(image: &mut Image, terrain: &[bool], size: UVec2) {
337        let width = image.width() as usize;
338        let bytes = image.data.as_mut().unwrap();
339        for (index, _) in terrain.iter().enumerate().filter(|&(_, &solid)| solid) {
340            let (x, y) = (index % size.x as usize * 6, index / size.x as usize * 6);
341            let pixels = (0..36).map(|pixel| ((y + pixel / 6) * width + x + pixel % 6) * 4);
342            pixels.for_each(|offset| bytes[offset..offset + 4].copy_from_slice(&[18, 20, 28, 255]));
343        }
