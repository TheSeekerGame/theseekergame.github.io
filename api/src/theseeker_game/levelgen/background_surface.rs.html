19/// The player spawn chunk, left empty so the surface backdrop shows through it.
20pub(super) struct SpawnChunkCutout {
21    chunk_origin: IVec2,
22    chunk_size: u32,
23}
24
25impl SpawnChunkCutout {
26    pub(super) fn new(level: &CompiledLevel, tile_px: u32) -> Option<Self> {
27        let spawn = level.spawns.iter().find(|spawn| spawn.kind == SpawnKind::Player)?;
28        let chunk_size = mapgen::CHUNK_IMAGE_SIZE * tile_px;
29        Some(Self { chunk_origin: spawn.chunk * chunk_size as i32, chunk_size })
30    }
31
32    fn apply(&self, image: &mut Image, region_origin: UVec2) {
33        clear_image_rect(image, self.chunk_origin - region_origin.as_ivec2(), UVec2::splat(self.chunk_size));
34    }
35}

37pub(super) fn bake(surface: &surface::TileSurface<'_>, blur: f32, cutout: Option<SpawnChunkCutout>, world_size: Vec2) -> BackgroundBake {
38    let regions = surface::chunk_regions(surface.size);
39    let mut patches = surface::bake_jobs(&regions, |region| bake_patch(surface, region, blur, cutout.as_ref()));
40    patches.sort_by_key(|patch| (patch.region.min.y, patch.region.min.x));
41    BackgroundBake { patches, size: surface.size, tile_px: surface.tile_px, world_size }
42}

44fn bake_patch(surface: &surface::TileSurface<'_>, region: URect, blur: f32, cutout: Option<&SpawnChunkCutout>) -> BackgroundPatch {
45    // The output owns one tile region. One extra texel supplies filtering across its edge.
46    let support = crate::graphics::background_blur::support(blur) + 3;
47    let padded = surface::expand(region, support.div_ceil(surface.tile_px), surface.size);
48    let mut color = surface.bake(padded);
49    let origin = padded.min * surface.tile_px;
50    if let Some(cutout) = cutout { cutout.apply(&mut color, origin); }
51    let pixels = URect { min: region.min * surface.tile_px, max: region.max * surface.tile_px };
52    let local = URect { min: pixels.min - origin, max: pixels.max - origin };
53    let coverage = surface::alpha(&surface::crop(&color, local));
54    finalize_background_group_image(&mut color, blur);
55    let texture_region = surface::expand(pixels, 1, surface.size * surface.tile_px);
56    let mut color = surface::crop(&color, URect { min: texture_region.min - origin, max: texture_region.max - origin });
57    color.sampler = if blur > 0.0001 { ImageSampler::linear() } else { ImageSampler::nearest() };
58    BackgroundPatch { region, texture_region, color, coverage }
59}

61/// A background plane's parallax depth, the inverse of the share of camera motion it follows.
62pub(super) fn parallax_depth(group: usize) -> f32 {
63    1.0 / [0.95, 0.85, 0.75][group.min(2)]
64}

66#[expect(clippy::too_many_arguments, reason = "Publication owns Bevy asset stores and the shared coverage atlas.")]
67pub(super) fn publish(commands: &mut Commands, root: Entity, group: usize, bake: BackgroundBake, center: Vec2,
68    masks: Option<&mut DarknessBakeMaskAtlas>, images: &mut Assets<Image>, meshes: &mut Assets<Mesh>, materials: &mut Assets<ColorMaterial>,
69) {
70    let depth = parallax_depth(group);
71    let mut coverage = Vec::with_capacity(bake.patches.len());
72    for patch in bake.patches {
73        let pixels = URect { min: patch.region.min * bake.tile_px, max: patch.region.max * bake.tile_px };
74        coverage.push((pixels.min, pixels.width(), pixels.height(), patch.coverage));
75        if !surface::visible(&patch.color) { continue; }
76        let mut mesh = Mesh::from(Rectangle::default());
77        let uv_min = (pixels.min - patch.texture_region.min).as_vec2() / patch.texture_region.size().as_vec2();
78        let uv_scale = pixels.size().as_vec2() / patch.texture_region.size().as_vec2();
79        inset_uvs(&mut mesh, uv_min, uv_scale);
80        let size = patch.region.size().as_vec2() / bake.size.as_vec2() * bake.world_size;
81        let local_center = (patch.region.min.as_vec2() + patch.region.size().as_vec2() * 0.5) / bake.size.as_vec2();
82        let position = center + (local_center - Vec2::splat(0.5)) * bake.world_size * Vec2::new(1.0, -1.0);
83        let texture = images.add(patch.color);
84        commands.spawn((Mesh2d(meshes.add(mesh)), bevy::sprite_render::MeshMaterial2d(materials.add(ColorMaterial::from(texture))),
85            Transform::from_xyz(position.x, position.y, -(group as f32 + 1.0)).with_scale(size.extend(1.0)),
86            RenderLayers::layer(0), crate::parallax::Parallax { depth }, crate::parallax::ParallaxOffset(center - position), ChildOf(root), StateDespawnMarker,
87            Name::new(format!("Background{}Chunk", group + 1))));
88    }
89    if let Some(masks) = masks {
90        let layer = match group { 0 => &mut masks.near, 1 => &mut masks.middle, _ => &mut masks.far };
91        layer.set_chunks(bake.size * bake.tile_px, coverage, center, bake.world_size, depth);
92    }
93}

95fn inset_uvs(mesh: &mut Mesh, min: Vec2, scale: Vec2) {
96    let Some(VertexAttributeValues::Float32x2(uvs)) = mesh.attribute_mut(Mesh::ATTRIBUTE_UV_0) else { return; };
97    for uv in uvs { *uv = (min + Vec2::from_array(*uv) * scale).to_array(); }
