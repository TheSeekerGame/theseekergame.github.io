4const FLIPS: [BVec2; 4] = [BVec2::FALSE, BVec2::new(true, false), BVec2::new(false, true), BVec2::TRUE];
5
6pub(super) fn run_mosaic_generative_layers(
7    compiled: &CompiledLevel, mosaics: &[MosaicRules; 3], images: &Assets<Image>,
8) -> [Vec<bool>; 3] {
9    let _timer = diagnostics::StageTimer::new("run_mosaic_generative_layers");
10    let mut outputs: [Vec<bool>; 3] = std::array::from_fn(|_| Vec::new());
11    for index in 0..3 {
12        let source = if index == 0 { &compiled.terrain } else { &outputs[index - 1] };
13        let layer = apply_mosaic_layer_rules(source, &compiled.terrain, compiled, &mosaics[index], images);
14        // The highest matching layer owns each cell.
15        for lower in &mut outputs[..index] {
16            lower.iter_mut().zip(&layer).filter(|(_, claimed)| **claimed).for_each(|(cell, _)| *cell = false);
17        }
18        outputs[index] = layer;
19    }
20    outputs
21}

23/// Runs a layer's rules in order over `source`. Each rule claims matching source cells into
24/// the layer, so later rules see only what earlier ones left.
25pub(super) fn apply_mosaic_layer_rules(
26    source: &[bool],
27    primary: &[bool],
28    compiled: &CompiledLevel,
29    rules: &MosaicRules,
30    images: &Assets<Image>,
31) -> Vec<bool> {
32    let decoded = rules.decoded.get_or_init(|| {
33        let mut decoded: Vec<_> = rules.rules.iter().filter_map(|entry| decode_rule(entry, images)).collect();
34        decoded.sort_by_key(|rule| rule.order);
35        decoded
36    });
37    let mut layer = vec![false; source.len()];
38    let mut unclaimed = source.to_vec();
39    for rule in decoded {
40        apply_rule(rule, primary, compiled, &mut layer, &mut unclaimed);
41    }
42    layer
43}

45fn apply_rule(rule: &MosaicDecodedRule, primary: &[bool], compiled: &CompiledLevel, layer: &mut [bool], unclaimed: &mut [bool]) {
46    let (source, layer_before) = (unclaimed.to_vec(), layer.to_vec());
47    let view = RuleView { width: compiled.width, height: compiled.height, source: &source, primary, layer: &layer_before };
48    // A center requirement holds under any flip, so it rejects cells up front.
49    let needs_center = rule.source.require.first() == Some(&(0, 0));
50    let cells = (0..compiled.height as i32).flat_map(|y| (0..compiled.width as i32).map(move |x| IVec2::new(x, y)));
51    for cell in cells {
52        if needs_center && !source[view.index(cell).expect("cells lie in the grid")] {
53            continue;
54        }
55        let Some(claims) = view.claims(rule, cell, unclaimed) else { continue; };
56        for index in claims {
57            layer[index] = true;
58            unclaimed[index] = false;
59        }
60    }
61}

63/// The grids a rule reads, as they were when the rule started.
64struct RuleView<'a> {
65    width: u32,
66    height: u32,
67    source: &'a [bool],
68    primary: &'a [bool],
69    layer: &'a [bool],
70}
71
72impl RuleView<'_> {
73    fn index(&self, cell: IVec2) -> Option<usize> {
74        let inside = cell.x >= 0 && cell.y >= 0 && cell.x < self.width as i32 && cell.y < self.height as i32;
75        inside.then(|| (cell.y as u32 * self.width + cell.x as u32) as usize)
76    }
77
78    /// The grid index of `offset` from `cell`, mirrored per `flip`.
79    fn at(&self, cell: IVec2, (dx, dy): (i32, i32), flip: BVec2) -> Option<usize> {
80        self.index(cell + IVec2::new(if flip.x { -dx } else { dx }, if flip.y { -dy } else { dy }))
81    }
82
83    fn set(&self, grid: &[bool], cell: IVec2, offset: (i32, i32), flip: BVec2) -> bool {
84        self.at(cell, offset, flip).is_some_and(|index| grid.get(index) == Some(&true))
85    }
86
87    fn matches(&self, rule: &MosaicDecodedRule, cell: IVec2, flip: BVec2) -> bool {
88        [(self.source, &rule.source), (self.primary, &rule.primary), (self.layer, &rule.layer)]
89            .into_iter()
90            .all(|(grid, conditions)| {
91                conditions.require.iter().all(|&offset| self.set(grid, cell, offset, flip))
92                    && !conditions.forbid.iter().any(|&offset| self.set(grid, cell, offset, flip))
93            })
94    }
95
96    /// The cells `rule` claims when centered on `cell`, if it fires there: the first allowed
97    /// orientation that matches, its required source still unclaimed, then a probability roll
98    /// that may also mirror the claimed cells. Every claimed cell must still be unclaimed.
99    fn claims(&self, rule: &MosaicDecodedRule, cell: IVec2, unclaimed: &[bool]) -> Option<Vec<usize>> {
100        let flip = FLIPS
101            .into_iter()
102            .filter(|flip| (!flip.x || rule.ruleflip_x) && (!flip.y || rule.ruleflip_y))
103            .find(|&flip| self.matches(rule, cell, flip))?;
104        if !rule.source.require.iter().all(|&offset| self.set(unclaimed, cell, offset, flip)) {
105            return None;
106        }
107        let hash = tile_hash(cell.x, cell.y, rule.order, 0);
108        if !within_percent(hash, rule.prob_percent) {
109            return None;
110        }
111        let claim_flip = BVec2::new(
112            flip.x || rule.randflip_x && (hash >> 1) & 1 == 1,
113            flip.y || rule.randflip_y && (hash >> 2) & 1 == 1,
114        );
115        rule.claims
116            .iter()
117            .map(|&offset| self.at(cell, offset, claim_flip).filter(|&index| unclaimed[index]))
118            .collect()
119    }
120}

122/// Decodes a rule mask: each opaque pixel's color adds its offset from the center to one
123/// of the rule's conditions. Rules that claim nothing are dropped.
124fn decode_rule(entry: &RuleImage, images: &Assets<Image>) -> Option<MosaicDecodedRule> {
125    let image = images.get(&entry.handle)?;
126    let mut rule = MosaicDecodedRule {
127        order: entry.order,
128        source: MaskConditions::default(),
129        primary: MaskConditions::default(),
130        layer: MaskConditions::default(),
131        claims: Vec::new(),
132        ruleflip_x: entry.ruleflip_x,
133        ruleflip_y: entry.ruleflip_y,
134        randflip_x: entry.randflip_x,
135        randflip_y: entry.randflip_y,
136        prob_percent: entry.prob_percent,
137    };
138    let width = image.width() as i32;
139    let center = ((image.width() / 2) as i32, (image.height() / 2) as i32);
140    let pixels = image.data.as_deref().unwrap_or_default().as_chunks::<4>().0;
141    let opaque = pixels.iter().take((image.width() * image.height()) as usize).enumerate()
142        .filter(|(_, pixel)| pixel[3] >= 128);
143    for (index, pixel) in opaque {
144        let offset = (index as i32 % width - center.0, index as i32 / width - center.1);
145        rule.add_mask_pixel([pixel[0], pixel[1], pixel[2]], offset);
146    }
147    // Nearest requirements first, so most rejections are cheap.
148    rule.source.require.sort_by_key(|&(x, y)| x.abs() + y.abs());
149    (!rule.claims.is_empty()).then_some(rule)
150}
151
152impl MosaicDecodedRule {
153    fn add_mask_pixel(&mut self, rgb: [u8; 3], offset: (i32, i32)) {
154        match rgb {
155            // Black cells must be source, and are what the rule claims.
156            [0, 0, 0] => {
157                self.source.require.push(offset);
158                self.claims.push(offset);
159            },
160            compiler::COLOR_RULE_REQUIRE => self.source.require.push(offset),
161            compiler::COLOR_RULE_FORBID => self.source.forbid.push(offset),
162            compiler::MOSAIC_RULE_REQUIRE_PRIMARY_COLOR => self.primary.require.push(offset),
163            compiler::MOSAIC_RULE_FORBID_PRIMARY_COLOR => self.primary.forbid.push(offset),
164            compiler::MOSAIC_RULE_REQUIRE_SELF_COLOR => self.layer.require.push(offset),
165            compiler::MOSAIC_RULE_FORBID_SELF_COLOR => self.layer.forbid.push(offset),
166            _ => {},
167        }
168    }
