42#[derive(Resource, Default)]
43pub struct ChunkLibrary {
44    templates: Vec<ChunkTemplate>,
45    loaded: bool,
46}
47
48impl ChunkLibrary {
49    pub fn ensure_loaded(&mut self) -> Result<(), String> {
50        if self.loaded {
51            return Ok(());
52        }
53        let dir = Path::new(CHUNKS_DIR);
54        let read_dir = fs::read_dir(dir).map_err(|err| {
55            format!(
56                "failed to read {}: {err}",
57                dir.display()
58            )
59        })?;
60        let mut templates: Vec<ChunkTemplate> = read_dir
61            .flatten()
62            .filter_map(chunk_template_from_entry)
63            .collect();
64        if templates.is_empty() {
65            return Err("no chunk templates found".into());
66        }
67        let spawn_count = templates.iter().filter(|t| t.is_spawn).count();
68        if spawn_count != 1 {
69            return Err(format!(
70                "expected exactly one spawn chunk, found {spawn_count}"
71            ));
72        }
73        templates.sort_by(|a, b| a.name.cmp(&b.name));
74        debug!(
75            "Loaded {} chunk templates",
76            templates.len()
77        );
78        self.templates = templates;
79        self.loaded = true;
80        Ok(())
81    }
82
83    pub fn templates(&self) -> &[ChunkTemplate] {
84        &self.templates
85    }
86
87    pub fn spawn_index(&self) -> Option<usize> {
88        self.templates.iter().position(|t| t.is_spawn)
89    }
90}

164#[derive(Clone, Copy, Default, Debug)]
165struct MapBounds {
166    min: IVec2,
167    max: IVec2,
168}
169
170impl MapBounds {
171    fn single(pos: IVec2) -> Self {
172        Self { min: pos, max: pos }
173    }
174
175    fn include(&mut self, pos: IVec2) {
176        self.min = IVec2::new(self.min.x.min(pos.x), self.min.y.min(pos.y));
177        self.max = IVec2::new(self.max.x.max(pos.x), self.max.y.max(pos.y));
178    }
179
180    fn width_chunks(&self) -> u32 {
181        (self.max.x - self.min.x + 1).max(0) as u32
182    }
183
184    fn height_chunks(&self) -> u32 {
185        (self.max.y - self.min.y + 1).max(0) as u32
186    }

197pub fn generate_map(chunk_lib: &mut ChunkLibrary, map_lib: &mut MapLibrary,
198    request: &super::pipeline::GenerationRequest,
199) -> Result<GeneratedMap, String> {
200    let _timer = super::diagnostics::StageTimer::new("assemble_map");
201    chunk_lib.ensure_loaded()?;
202    map_lib.ensure_loaded()?;
203    let spawn_index = chunk_lib.spawn_index().ok_or("spawn chunk missing")?;
204    let mut rng = StdRng::seed_from_u64(request.seed);
205    let entry = match &request.map {
206        Some(name) => map_lib.maps.iter().find(|map| &map.name == name).ok_or("requested map not found")?,
207        None => &map_lib.maps[rng.random_range(0..map_lib.maps.len())],
208    };
209    let map = load_map_description(entry)?;
210    let placed = assign_compatible_templates(&map.slots, chunk_lib.templates(), spawn_index, &mut rng)?;
211    let chunks_w = map.bounds.width_chunks();
212    let chunks_h = map.bounds.height_chunks();
213    let width = chunks_w.checked_mul(CHUNK_IMAGE_SIZE).ok_or("map width overflow")?;
214    let height = chunks_h.checked_mul(CHUNK_IMAGE_SIZE).ok_or("map height overflow")?;
215    let mut decoded = HashMap::new();
216    for &SlotTemplate { template: index, .. } in &placed {
217        if let std::collections::hash_map::Entry::Vacant(entry) = decoded.entry(index) {
218            let template = &chunk_lib.templates[index];
219            entry.insert(load_chunk(template)?);
220        }
221    }
222    let pixels = map.slots.iter().zip(&placed).flat_map(|(slot, &SlotTemplate { template: index, flip, source_origin })| {
223        let origin = (slot.grid_pos - map.bounds.min).as_uvec2() * CHUNK_IMAGE_SIZE;
224        let template = &chunk_lib.templates[index];
225        let data = decoded[&index].data.as_deref().unwrap_or_default();
226        (0..CHUNK_IMAGE_SIZE * CHUNK_IMAGE_SIZE).filter_map(move |i| {
227            let local = UVec2::new(i % CHUNK_IMAGE_SIZE, i / CHUNK_IMAGE_SIZE);
228            let source = source_origin + local;
229            let rgba = chunk_source_pixel(data, source.x, source.y, template.footprint * CHUNK_IMAGE_SIZE, flip)?;
230            (!should_clear_template_pixel(template, rgba)).then_some((origin + local, rgba))
231        })
232    });
233    let compiled = super::compiler::compile_pixels(width, height, &request.config, pixels);
234    let pre_impact = map.slots.iter().zip(&placed).find(|(_, placement)| placement.template == spawn_index)
235        .map(|(slot, _)| {
236            let origin = (slot.grid_pos - map.bounds.min).as_uvec2() * CHUNK_IMAGE_SIZE;
237            super::impact_crater::PreImpactTerrain::load(&chunk_lib.templates[spawn_index].path, origin)
238        })
239        .transpose()?
240        .flatten();
241    let mut chunk_colors = vec![[0; 3]; (chunks_w * chunks_h) as usize];
242    let mut chunk_occupied = vec![false; chunk_colors.len()];
243    for slot in &map.slots {
244        let pos = (slot.grid_pos - map.bounds.min).as_uvec2();
245        let index = (pos.y * chunks_w + pos.x) as usize;
246        chunk_colors[index] = slot.map_color;
247        chunk_occupied[index] = true;
248    }
249    let selected_templates = map.slots.iter().zip(&placed).filter(|(_, placement)| placement.source_origin == UVec2::ZERO).map(|(slot, placement)| {
250        (chunk_lib.templates[placement.template].name.clone(), slot.grid_pos - map.bounds.min, placement.flip)
251    }).collect();
252    let spawn_row = map.slots.iter().find(|slot| slot.kind == SlotKind::Spawn).map_or(0, |slot| (slot.grid_pos.y - map.bounds.min.y) as u32);
253    let biomes = depth_biomes(width, height, spawn_row);
254    Ok(GeneratedMap { compiled, seed: request.seed, selected_templates, map_name: entry.name.clone(),
255        synthetic_path: format!("{}{}:{:016x}", GENERATED_LEVEL_PREFIX, entry.name, request.seed),
256        width, height, chunks_w, chunks_h, biomes,
257        chunk_colors, chunk_occupied, pre_impact })
258}

260/// Coarse deterministic patches blend the depth bands without scattering isolated biome tiles.
261fn depth_biomes(width: u32, height: u32, spawn_row: u32) -> Vec<BiomeKind> {
262    let start = (spawn_row + TUNDRA_TRANSITION_START_CHUNKS) * CHUNK_IMAGE_SIZE;
263    let end = (spawn_row + TUNDRA_TRANSITION_END_CHUNKS) * CHUNK_IMAGE_SIZE;
264    (0..width * height).map(|index| {
265        let (x, y) = (index % width, index / width);
266        let roll = super::hash::tile_hash((x / 8) as i32, (y / 8) as i32, 0, 0xB10A_E123) % u64::from(end - start);
267        if y >= end || (y >= start && roll < u64::from(y - start)) { BiomeKind::Tundra } else { BiomeKind::Ice }
268    }).collect()
269}

271/// One template per slot, in slot order.
272fn assign_compatible_templates(slots: &[MapSlot], templates: &[ChunkTemplate], spawn: usize, rng: &mut StdRng)
273    -> Result<Vec<SlotTemplate>, String> {
274    let mut pools: [Vec<(usize, bool)>; 16] = std::array::from_fn(|mask| {
275        compatible_orientations(templates, mask as u8, 1)
276    });
277    for pool in &mut pools { pool.shuffle(rng); }
278    let mut used = vec![false; templates.len()];
279    let mut cursors = [0usize; 16];
280    let mut placed = assign_large_templates(slots, templates, rng, &mut used);
281    for (slot, placement) in slots.iter().zip(&mut placed) {
282        if placement.is_some() {
283            continue;
284        }
285        let (template, flip) = choose_template(slot, templates, spawn, &pools, &mut used, &mut cursors)?;
286        *placement = Some(SlotTemplate { template, flip, source_origin: UVec2::ZERO });
287    }
288    placed.into_iter().map(|placement| placement.ok_or_else(|| "unassigned map slot".into())).collect()
289}
290
291fn compatible_orientations(templates: &[ChunkTemplate], mask: u8, footprint: u32) -> Vec<(usize, bool)> {
292    templates.iter().enumerate().filter(|(_, template)| !template.is_spawn && template.footprint == footprint).flat_map(|(index, template)| {
293        [false, true].into_iter().filter_map(move |flip| {
294            let connectors = [template.connectors, swap_left_right(template.connectors)][usize::from(flip)];
295            (connectors == mask).then_some((index, flip))
296        })
297    }).collect()
298}

300/// Reserve complete rooms before single-cell assignment. A large room only exposes the middle
301/// of each edge; any external neighbor at a corner makes that block ineligible.
302fn assign_large_templates(slots: &[MapSlot], templates: &[ChunkTemplate], rng: &mut StdRng, used: &mut [bool])
303    -> Vec<Option<SlotTemplate>> {
304    let lookup: HashMap<_, _> = slots.iter().enumerate().map(|(index, slot)| (slot.grid_pos, index)).collect();
305    let mut placed = vec![None; slots.len()];
306    let mut pools: [Vec<(usize, bool)>; 16] = std::array::from_fn(|mask| compatible_orientations(templates, mask as u8, 3));
307    for pool in &mut pools { pool.shuffle(rng); }
308    for slot in slots {
309        let Some((members, mask)) = large_room_slots(slot.grid_pos, slots, &lookup) else { continue };
310        if members.iter().any(|&index| placed[index].is_some()) { continue; }
311        let Some((template, flip)) = pools[mask as usize].iter().copied().find(|&(index, _)| !used[index]) else { continue };
312        used[template] = true;
313        for (local, index) in members.into_iter().enumerate() {
314            let source_origin = UVec2::new(local as u32 % 3, local as u32 / 3) * CHUNK_IMAGE_SIZE;
315            placed[index] = Some(SlotTemplate { template, flip, source_origin });
316        }
317    }
318    placed
319}

321fn large_room_slots(origin: IVec2, slots: &[MapSlot], lookup: &HashMap<IVec2, usize>) -> Option<([usize; 9], u8)> {
322    let mut members = [0; 9];
323    for (local, member) in members.iter_mut().enumerate() {
324        *member = *lookup.get(&(origin + IVec2::new(local as i32 % 3, local as i32 / 3)))?;
325        if slots[*member].kind == SlotKind::Spawn { return None; }
326    }
327    let mut mask = 0;
328    for (delta, direction) in [(IVec2::new(0, -1), DIR_UP), (IVec2::new(3, 0), DIR_RIGHT),
329        (IVec2::new(0, 3), DIR_DOWN), (IVec2::new(-1, 0), DIR_LEFT)] {
330        mask |= room_boundary_mask(origin + delta, direction, lookup)?;
331    }
332    (mask != 0).then_some((members, mask))
333}
334
335fn room_boundary_mask(origin: IVec2, direction: u8, lookup: &HashMap<IVec2, usize>) -> Option<u8> {
336    let mut mask = 0;
337    for edge in 0..3 {
338        let occupied = lookup.contains_key(&(origin + room_edge_offset(edge, direction)));
339        if occupied && edge != 1 { return None; }
340        if occupied { mask = direction; }
341    }
342    Some(mask)
343}
344
345fn room_edge_offset(edge: i32, direction: u8) -> IVec2 {
346    if direction == DIR_UP || direction == DIR_DOWN { IVec2::new(edge, 0) } else { IVec2::new(0, edge) }
347}

349fn choose_template(slot: &MapSlot, templates: &[ChunkTemplate], spawn: usize, pools: &[Vec<(usize, bool)>; 16],
350    used: &mut [bool], cursors: &mut [usize; 16],
351) -> Result<(usize, bool), String> {
352    if slot.kind == SlotKind::Spawn {
353        return (templates[spawn].connectors == slot.adjacency_mask).then_some((spawn, false))
354            .ok_or_else(|| "spawn connectors do not match map".into());
355    }
356    let mask = slot.adjacency_mask as usize;
357    let pool = &pools[mask];
358    let choice = pool.iter().copied().find(|&(index, _)| !used[index])
359        .or_else(|| pool.get(cursors[mask] % pool.len().max(1)).copied())
360        .ok_or_else(|| format!("no compatible template for connector mask {mask:04b}"))?;
361    used[choice.0] = true;
362    cursors[mask] += 1;
363    Ok(choice)
364}

366fn chunk_template_from_entry(entry: fs::DirEntry) -> Option<ChunkTemplate> {
367    let path = entry.path();
368    let stem = png_file_stem(&path)?;
369    let name = stem.to_string();
370    let is_spawn = stem.starts_with("player_spawn");
371    let connectors = parse_connector_suffix(stem);
372    let footprint = if stem.starts_with("large_chunk_3x3_") { 3 } else { 1 };
373    Some(ChunkTemplate {
374        name,
375        path,
376        connectors,
377        is_spawn,
378        footprint,
379    })
380}

402fn parse_connector_suffix(stem: &str) -> u8 {
403    let mut mask = 0u8;
404    for ch in stem.chars().rev() {
405        match ch {
406            'U' => mask |= DIR_UP,
407            'D' => mask |= DIR_DOWN,
408            'L' => mask |= DIR_LEFT,
409            'R' => mask |= DIR_RIGHT,
410            _ => break,
411        }
412    }
413    mask
414}

416fn load_map_description(entry: &MapEntry) -> Result<MapDescription, String> {
417    let image = load_png(&entry.path)?;
418    
419    let data = image.data.as_ref().ok_or("map image missing data")?;
420    let (mut slots, spawn_count, bounds) = collect_map_slots(data, image.width());
421    if spawn_count != 1 {
422        return Err(format!(
423            "map {} must contain exactly one white spawn pixel (found {})",
424            entry.name, spawn_count
425        ));
426    }
427    let Some(bounds) = bounds else {
428        return Err(format!("map {} contains no chunks", entry.name));
429    };
430
431    assign_adjacency_masks(&mut slots);
432    Ok(MapDescription { slots, bounds })
433}

435fn collect_map_slots(data: &[u8], width: u32) -> (Vec<MapSlot>, usize, Option<MapBounds>) {
436    let mut slots = Vec::new();
437    let mut spawn_count = 0usize;
438    let mut bounds = None;
439
440    for (pixel_idx, pixel) in data
441        .as_chunks::<4>().0.iter()
442        .enumerate()
443    {
444        let Some(slot) = slot_from_map_pixel(pixel_idx, pixel, width) else {
445            continue;
446        };
447        if slot.kind == SlotKind::Spawn {
448            spawn_count += 1;
449        }
450        include_slot_bounds(&mut bounds, slot.grid_pos);
451        slots.push(slot);
452    }
453
454    (slots, spawn_count, bounds)
455}

457fn slot_from_map_pixel(pixel_idx: usize, pixel: &[u8], width: u32) -> Option<MapSlot> {
458    if pixel[3] < 128 {
459        return None;
460    }
461
462    let color = [pixel[0], pixel[1], pixel[2]];
463    map_slot_from_color(map_pixel_position(pixel_idx, width), color)
464}
465
466fn map_pixel_position(pixel_idx: usize, width: u32) -> IVec2 {
467    let x = pixel_idx % width as usize;
468    let y = pixel_idx / width as usize;
469    IVec2::new(x as i32, y as i32)
470}

472fn map_slot_from_color(pos: IVec2, color: [u8; 3]) -> Option<MapSlot> {
473    let kind = match color {
474        MAP_CHUNK_COLOR | MAP_GEYSER_COLOR => SlotKind::Regular,
475        MAP_SPAWN_COLOR => SlotKind::Spawn,
476        _ => return None,
477    };
478
479    Some(MapSlot {
480        grid_pos: pos,
481        adjacency_mask: 0,
482        kind,
483        map_color: color,
484    })
485}
486
487fn include_slot_bounds(bounds: &mut Option<MapBounds>, pos: IVec2) {
488    let Some(bounds) = bounds else {
489        *bounds = Some(MapBounds::single(pos));
490        return;
491    };
492
493    bounds.include(pos);
494}

496fn assign_adjacency_masks(slots: &mut [MapSlot]) {
497    let lookup: HashMap<IVec2, ()> =
498        slots.iter().map(|slot| (slot.grid_pos, ())).collect();
499    for slot in slots {
500        slot.adjacency_mask = adjacency_mask(slot.grid_pos, &lookup);
501    }
502}
503
504fn adjacency_mask(pos: IVec2, lookup: &HashMap<IVec2, ()>) -> u8 {
505    let neighbor_offsets = [
506        (IVec2::new(0, -1), DIR_UP),
507        (IVec2::new(1, 0), DIR_RIGHT),
508        (IVec2::new(0, 1), DIR_DOWN),
509        (IVec2::new(-1, 0), DIR_LEFT),
510    ];
511    let mut mask = 0u8;
512    for (delta, bit) in neighbor_offsets {
513        if lookup.contains_key(&(pos + delta)) {
514            mask |= bit;
515        }
516    }
517    mask
518}

520fn chunk_source_pixel(
521    src: &[u8],
522    x: u32,
523    y: u32,
524    size: u32,
525    flip: bool,
526) -> Option<[u8; 4]> {
527    let src_x = if flip { size - 1 - x } else { x };
528    let src_idx = ((y * size + src_x) * 4) as usize;
529    if src_idx + 3 >= src.len() {
530        return None;
531    }
532
533    Some([
534        src[src_idx],
535        src[src_idx + 1],
536        src[src_idx + 2],
537        src[src_idx + 3],
538    ])
539}

541fn should_clear_template_pixel(template: &ChunkTemplate, pixel: [u8; 4]) -> bool {
542    !template.is_spawn && pixel_matches_rgb(pixel, CHUNK_SPAWN_COLOR)
543}
544
545fn pixel_matches_rgb(pixel: [u8; 4], color: [u8; 3]) -> bool {
546    pixel[0] == color[0] && pixel[1] == color[1] && pixel[2] == color[2]
547}
548
549fn swap_left_right(mask: u8) -> u8 {
550    let mut swapped = mask & (DIR_UP | DIR_DOWN);
551    if mask & DIR_LEFT != 0 {
552        swapped |= DIR_RIGHT;
553    }
554    if mask & DIR_RIGHT != 0 {
555        swapped |= DIR_LEFT;
556    }
557    swapped
558}

560pub(super) fn load_png(path: &Path) -> Result<Image, String> {
561    let bytes = fs::read(path).map_err(|err| {
562        format!(
563            "failed to read {}: {err}",
564            path.display()
565        )
566    })?;
567    let mut image = Image::from_buffer(
568        &bytes,
569        ImageType::Extension("png"),
570        CompressedImageFormats::default(),
571        true,
572        ImageSampler::Default,
573        RenderAssetUsages::default(),
574    )
575    .map_err(|err| {
576        format!(
577            "failed to decode {}: {err}",
578            path.display()
579        )
580    })?;
581    image.sampler = ImageSampler::nearest();
582    Ok(image)
583}

586fn load_chunk(template: &ChunkTemplate) -> Result<Image, String> {
587    let image = load_png(&template.path)?;
588    let size = template.footprint * CHUNK_IMAGE_SIZE;
589    if image.width() != size || image.height() != size {
590        return Err(format!("chunk {} must be {size}x{size} pixels", template.name));
591    }
592    Ok(image)
593}
