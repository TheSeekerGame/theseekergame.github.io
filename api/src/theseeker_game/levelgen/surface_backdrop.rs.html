1//! Sky and distant landscape seen above the player spawn chunk.
2//!
3//! `tools/surface_art/generate.py` generates one image per layer, each exactly one chunk wide.
4//! A layer's parallax is measured from the player spawn, so the resting capsule view shows
5//! the artwork as authored. Below that view the layers stay fixed to the world; they never
6//! sink into the caves under the surface.
7use super::*;
8use bevy::image::{ImageLoaderSettings, ImageSampler};
9use crate::camera::MainCamera;
10use crate::graphics::post_processing::foreground_fog::OpenSkyRegion;
11
12/// Spawn-chunk rows between which foreground fog fades out toward the open sky:
13/// fully present below the ground line, gone just above the crater rim.
14const FOG_CLEAR_ROWS: (f32, f32) = (560.0, 490.0);
15
16struct LayerSpec {
17    image: &'static str,
18    /// Shows only once the impact has formed the crater.
19    formed_crater: bool,
20    /// Spawn-chunk pixel row of the image's top edge (y downward from the chunk top).
21    top_row: f32,
22    /// Distance factor: 1 moves with the world; larger values move less.
23    depth: f32,
24    z: f32,
25}
26
27/// Back to front. Top rows must match `LAYER_TOPS` in the generator. The cave backgrounds
28/// parallax around the level center, which shifts their content under the spawn chunk; the
29/// opaque sky covers the middle and far ones (z -2 and -3), and the crater's far wall covers
30/// the near one (z -1) where it shows through the rail shaft.
31const LAYERS: [LayerSpec; 5] = [
32    LayerSpec { image: "levels/surface/Sky.png", formed_crater: false, top_row: -80.0, depth: 4.0, z: -1.9 },
33    LayerSpec { image: "levels/surface/Clouds.png", formed_crater: false, top_row: -440.0, depth: 2.0, z: -1.8 },
34    LayerSpec { image: "levels/surface/FarRange.png", formed_crater: false, top_row: 400.0, depth: 1.6, z: -1.7 },
35    LayerSpec { image: "levels/surface/MidHills.png", formed_crater: false, top_row: 440.0, depth: 1.25, z: -1.6 },
36    LayerSpec { image: "levels/surface/CraterRim.png", formed_crater: true, top_row: 470.0, depth: 1.05, z: -0.5 },
37];
38
39#[derive(Component)]
40pub(crate) struct SurfaceLayer {
41    depth: f32,
42    /// Top-left corner when the camera rests at `pivot`.
43    rest_position: Vec2,
44    pivot: Vec2,
45}
46
47/// Handles for every backdrop image, so a caller can wait until the backdrop is drawable.
48#[derive(Resource, Default)]
49pub(crate) struct SurfaceBackdropImages(pub Vec<Handle<Image>>);

51pub(super) fn spawn_surface_backdrop(
52    mut commands: Commands,
53    compiled: Option<Res<CompiledLevel>>,
54    root: Option<Res<LevelRootEntity>>,
55    asset_server: Res<AssetServer>,
56    mut spawned_for_root: Local<Option<Entity>>,
57) {
58    let (Some(compiled), Some(root)) = (compiled, root) else {
59        return;
60    };
61    if *spawned_for_root == Some(root.0) {
62        return;
63    }
64    let Some(spawn) = compiled.spawns.iter().find(|spawn| spawn.kind == SpawnKind::Player) else {
65        return;
66    };
67    *spawned_for_root = Some(root.0);
68
69    let chunk_px = compiled.tile_size() * mapgen::CHUNK_IMAGE_SIZE as f32;
70    let chunk_top_left = Vec2::new(
71        spawn.chunk.x as f32 * chunk_px,
72        compiled.bounds.max.y - spawn.chunk.y as f32 * chunk_px,
73    );
74    let mut handles = Vec::with_capacity(LAYERS.len());
75    for layer in &LAYERS {
76        // Smoothly shaded like the cave backgrounds, so filtered rather than pixel-sampled.
77        let image = asset_server.load_builder().with_settings(|settings: &mut ImageLoaderSettings| {
78            settings.sampler = ImageSampler::linear();
79        }).load(layer.image);
80        handles.push(image.clone());
81        let rest_position = chunk_top_left - Vec2::new(0.0, layer.top_row);
82        let mut entity = commands.spawn((
83            Name::new("SurfaceBackdropLayer"),
84            Sprite { image, ..default() },
85            Anchor::TOP_LEFT,
86            Transform::from_translation(rest_position.extend(layer.z)),
87            RenderLayers::layer(0),
88            SurfaceLayer { depth: layer.depth, rest_position, pivot: spawn.position },
89            ChildOf(root.0),
90            StateDespawnMarker,
91        ));
92        if layer.formed_crater {
93            entity.insert(super::impact_crater::ImpactFormed);
94        }
95    }
96    commands.insert_resource(SurfaceBackdropImages(handles));
97    commands.insert_resource(OpenSkyRegion {
98        x_range: Vec2::new(chunk_top_left.x, chunk_top_left.x + chunk_px),
99        fade_y: Vec2::new(chunk_top_left.y - FOG_CLEAR_ROWS.0, chunk_top_left.y - FOG_CLEAR_ROWS.1),
100    });
101}

103pub(super) fn place_surface_layers(
104    camera: Query<&Transform, (With<MainCamera>, Without<SurfaceLayer>)>,
105    mut layers: Query<(&SurfaceLayer, &mut Transform)>,
106) {
107    let Ok(camera) = camera.single() else {
108        return;
109    };
110    for (layer, mut transform) in &mut layers {
111        let offset = camera.translation.xy() - layer.pivot;
112        let offset = Vec2::new(offset.x, offset.y.max(0.0));
113        let position = layer.rest_position + offset * (1.0 - 1.0 / layer.depth);
114        transform.translation.x = position.x;
115        transform.translation.y = position.y;
116    }
117}
