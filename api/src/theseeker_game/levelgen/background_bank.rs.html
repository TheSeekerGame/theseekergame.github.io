1//! Load-only adapter for the paired backgrounds produced by tools/backgrounds/generate.py.
2use super::*;
3use serde::Deserialize;
4use bevy::tasks::{ComputeTaskPool, TaskPool};

6const BANK_DIR: &str = "assets/levels/background_bank";
7const FORMAT_VERSION: u32 = 2;
8const PADDING: u32 = 2;
9
10#[derive(Deserialize)]
11struct Manifest {
12    version: u32,
13    map_hash: u64,
14    size: [u32; 2],
15    tile_px: u32,
16    patches: Vec<Patch>,
17}

19#[derive(Deserialize)]
20struct Patch {
21    min: [u32; 2],
22    max: [u32; 2],
23    texture_min: [u32; 2],
24    texture_max: [u32; 2],

27pub(super) fn load_pair(config: &LevelPngConfig, level: &CompiledLevel, seed: u64) -> Result<[background_surface::BackgroundBake; 2], String> {
28    let _timer = diagnostics::StageTimer::new("shipped_background_decode");
29    let (name, seed) = selection(&config.path, seed)?;
30    let variant = std::env::var("SEEKER_BACKGROUND_VARIANT").ok()
31        .and_then(|value| value.parse::<usize>().ok()).unwrap_or(seed as usize % 3);
32    if variant >= 3 { return Err("Background variant must be 0, 1, or 2".into()); }
33    let size = UVec2::new(level.width, level.height) + UVec2::splat(PADDING * 2);
34    let source = if config.path.starts_with(mapgen::GENERATED_LEVEL_PREFIX) {
35        Path::new("assets/levels/maps").join(format!("{name}.png"))
36    } else { Path::new("assets").join(&config.path) };
37    let map_hash = fs::read(&source).map_err(|err| format!("{}: {err}", source.display()))?
38        .into_iter().fold(0xcbf2_9ce4_8422_2325u64, |hash, byte| (hash ^ u64::from(byte)).wrapping_mul(0x100_0000_01b3));
39    let load = |group| load_layer(&Path::new(BANK_DIR).join(&name).join(format!("{variant}-{group}")), size, config.tile_size, map_hash);
40    let pair = [load(1)?, load(2)?];
41    info!("levelgen: selected shipped background pair {name}/{variant}");
42    Ok(pair)
43}

45fn selection(path: &str, seed: u64) -> Result<(String, u64), String> {
46    if let Some(rest) = path.strip_prefix(mapgen::GENERATED_LEVEL_PREFIX) {
47        let (name, seed) = rest.rsplit_once(':').ok_or("Invalid generated-map identity")?;
48        return Ok((name.into(), u64::from_str_radix(seed, 16).map_err(|err| err.to_string())?));
49    }
50    let name = Path::new(path).file_stem().and_then(|stem| stem.to_str()).ok_or("Invalid level filename")?;
51    Ok((format!("custom/{name}"), seed))
52}

54fn load_layer(root: &Path, size: UVec2, tile_size: f32, map_hash: u64) -> Result<background_surface::BackgroundBake, String> {
55    let path = root.join("manifest.json");
56    let data = fs::read(&path).map_err(|err| format!("{}: {err}. Build this map's assets with tools/backgrounds/generate.py", path.display()))?;
57    let manifest: Manifest = serde_json::from_slice(&data).map_err(|err| format!("{}: {err}", path.display()))?;
58    if manifest.version != FORMAT_VERSION || manifest.map_hash != map_hash || UVec2::from_array(manifest.size) != size || manifest.tile_px != GROUND_TILE_ART_SIZE_PX {
59        return Err(format!("{}: background dimensions or format changed; regenerate its assets", path.display()));
60    }
61    let patches = load_patches(root, &manifest)?;
62    Ok(background_surface::BackgroundBake { patches, size, tile_px: manifest.tile_px, world_size: size.as_vec2() * tile_size })
63}
64
65fn load_patches(root: &Path, manifest: &Manifest) -> Result<Vec<background_surface::BackgroundPatch>, String> {
66    if manifest.patches.is_empty() { return Ok(Vec::new()); }
67    let pool = ComputeTaskPool::get_or_init(TaskPool::new);
68    let batch_size = manifest.patches.len().div_ceil(pool.thread_num().clamp(1, 4));
69    let batches = pool.scope(|scope| {
70        for (batch_index, batch) in manifest.patches.chunks(batch_size).enumerate() {
71            scope.spawn(load_patch_batch(root, manifest, batch, batch_index * batch_size));
72        }
73    });
74    let mut ordered: Vec<_> = batches.into_iter().flatten().collect();
75    ordered.sort_by_key(|(index, _)| *index);
76    ordered.into_iter().map(|(_, patch)| patch).collect()
77}

79async fn load_patch_batch(root: &Path, manifest: &Manifest, batch: &[Patch], start: usize)
80    -> Vec<(usize, Result<background_surface::BackgroundPatch, String>)> {
81    batch.iter().enumerate().map(|(offset, patch)| {
82        let index = start + offset;
83        (index, read_patch(root, index, patch, manifest))
84    }).collect()
85}
86
87fn read_patch(root: &Path, index: usize, patch: &Patch, manifest: &Manifest) -> Result<background_surface::BackgroundPatch, String> {
88    let region = URect { min: UVec2::from_array(patch.min), max: UVec2::from_array(patch.max) };
89    let texture_region = URect { min: UVec2::from_array(patch.texture_min), max: UVec2::from_array(patch.texture_max) };
90    validate_region(region, texture_region, manifest)?;
91    let mut color = mapgen::load_png(&root.join(format!("{index}.png")))?;
92    color.sampler = ImageSampler::linear();
93    let coverage = mapgen::load_png(&root.join(format!("{index}-coverage.png")))?;
94    if UVec2::new(color.width(), color.height()) != texture_region.size()
95        || UVec2::new(coverage.width(), coverage.height()) != region.size() * manifest.tile_px {
96        return Err(format!("{}: texture {index} has incorrect dimensions", root.display()));
97    }
98    let coverage = coverage.data.as_ref().ok_or("Missing background coverage pixels")?
99        .as_chunks::<4>().0.iter().map(|px| u8::from(px[0] != 0)).collect();
100    Ok(background_surface::BackgroundPatch { region, texture_region, color, coverage })
101}
102
103fn validate_region(region: URect, texture: URect, manifest: &Manifest) -> Result<(), String> {
104    let size = UVec2::from_array(manifest.size);
105    if !region.min.cmplt(region.max).all() || !region.max.cmple(size).all()
106        || !region.min.x.is_multiple_of(mapgen::CHUNK_IMAGE_SIZE) || !region.min.y.is_multiple_of(mapgen::CHUNK_IMAGE_SIZE)
107        || !region.size().cmple(UVec2::splat(mapgen::CHUNK_IMAGE_SIZE)).all() {
108        return Err("Invalid background chunk bounds".into());
109    }
110    let pixels = URect { min: region.min * manifest.tile_px, max: region.max * manifest.tile_px };
111    if texture != surface::expand(pixels, 1, size * manifest.tile_px) {
112        return Err("Background texture must include its one-pixel filtering gutter".into());
113    }
114    Ok(())
