5/// Separates the tundra's detailing rolls from the ice's.
6const TUNDRA_DETAIL_SALT: u64 = 0x7D5A_11E3;
7
8pub(super) fn build_background_biome_allowed_masks(
9    biome_mask: Option<&LevelBiomeMask>,
10    len: usize,
11) -> (Vec<bool>, Vec<bool>) {
12    let mut tundra_allowed = vec![false; len];
13    let mut ice_allowed = vec![true; len];
14    let Some(biome_mask) = biome_mask else {
15        return (tundra_allowed, ice_allowed);
16    };
17
18    ice_allowed.fill(false);
19    for (idx, cell) in biome_mask.cells.iter().enumerate() {
20        mark_background_biome_allowed_cell(
21            &mut tundra_allowed,
22            &mut ice_allowed,
23            idx,
24            cell.is_tundra(),
25        );
26    }
27    (tundra_allowed, ice_allowed)
28}
29
30pub(super) fn mark_background_biome_allowed_cell(
31    tundra_allowed: &mut [bool],
32    ice_allowed: &mut [bool],
33    idx: usize,
34    is_tundra: bool,
35) {
36    if is_tundra {
37        tundra_allowed[idx] = true;
38        return;
39    }
40    ice_allowed[idx] = true;
41}

43pub(super) fn combine_background_biome_near_grids(
44    len: usize,
45    tundra_allowed: &[bool],
46    ice_near_grid: &[i32],
47    tundra_near_grid: Option<&[i32]>,
48) -> Vec<i32> {
49    let mut near_grid = vec![0i32; len];
50    for (idx, cell) in near_grid.iter_mut().enumerate() {
51        *cell = background_biome_near_grid_value(
52            idx,
53            tundra_allowed,
54            ice_near_grid,
55            tundra_near_grid,
56        );
57    }
58    near_grid
59}

61pub(super) fn background_biome_near_grid_value(
62    idx: usize,
63    tundra_allowed: &[bool],
64    ice_near_grid: &[i32],
65    tundra_near_grid: Option<&[i32]>,
66) -> i32 {
67    if !tundra_allowed.get(idx).copied().unwrap_or(false) {
68        return ice_near_grid.get(idx).copied().unwrap_or(0);
69    }
70    tundra_near_grid
71        .and_then(|grid| grid.get(idx).copied())
72        .unwrap_or(0)
73}

85pub(super) fn build_near_background_layer(
86    near_grid: &[i32],
87    inputs: NearBackgroundLayerInputs,
88) -> Option<background_surface::BackgroundBake> {
89    if !near_grid.iter().any(|&v| v != 0) {
90        return None;
91    }
92
93    let (near_grid, biome_mask_warped) = warp_bg_int_grid_for_parallax(
94        inputs.compiled.width,
95        inputs.compiled.height,
96        near_grid,
97        inputs.biome_mask,
98        inputs.near_depth,
99    );
100    let biome_mask_for_bake = biome_mask_warped.as_ref().or(inputs.biome_mask);
101    let render_mask: Vec<bool> = near_grid.iter().map(|&v| v != 0).collect();
102    let surface = prepare_background_surface(
103        inputs.compiled,
104        &near_grid,
105        &render_mask,
106        biome_mask_for_bake,
107        inputs.ldtk_data,
108        inputs.images,
109    )?;
110
111    let cutout = background_surface::SpawnChunkCutout::new(inputs.compiled, surface.tile_px);
112    Some(background_surface::bake(&surface, inputs.blur_settings.blur_for_group(0),
113        cutout, Vec2::new(inputs.compiled.bounds.width(), inputs.compiled.bounds.height())))
114}

116pub(super) fn finalize_background_group_image(group_image: &mut Image, blur_px: f32) {
117    if blur_px > 0.0001 {
118        apply_background_blur(group_image, blur_px);
119        group_image.sampler = ImageSampler::linear();
120        return;
121    }
122    group_image.sampler = ImageSampler::nearest();
123}

125#[expect(clippy::too_many_arguments, reason = "Bevy system dependencies are expressed as explicit system parameters.")]
126pub(super) fn build_background_layers(
127    mut commands: Commands,
128    compiled: Option<Res<CompiledLevel>>,
129    root: Option<Res<LevelRootEntity>>,
130    biomes: Option<Res<LevelBiomeMask>>,
131    libraries: Res<GenerationLibraries>,
132    cfg: Res<LevelPngConfig>,
133    request: Option<Res<pipeline::GenerationRequest>>,
134    ldtk_data: Res<ldtk_rules::LdtkRuleData>,
135    mut images: ResMut<Assets<Image>>,
136    mut meshes: ResMut<Assets<Mesh>>,
137    mut color_materials: ResMut<Assets<ColorMaterial>>,
138    mut built_flag: ResMut<pipeline::GenerationBuild>,
139    blur_settings: Res<BackgroundBlurSettings>,
140    mut darkness_masks: Option<ResMut<DarknessBakeMaskAtlas>>,
141
142) {
143    if built_flag.phase != pipeline::BuildPhase::Ground {
144        return;
145    }
146    let Some(compiled) = compiled else { return; };
147    let root_e = match root {
148        Some(r) => r.0,
149        None => return,
150    };
151    let biome_mask = biomes.as_ref().and_then(|b| {
152        if b.matches_dims(compiled.width, compiled.height) {
153            Some(b.as_ref())
154        } else {
155            warn!(
156                "Biome mask dimensions {}x{} do not match compiled level {}x{}",
157                b.width, b.height, compiled.width, compiled.height
158            );
159            None
160        }
161    });
162    if !ldtk_ready(&ldtk_data, &images) || !libraries.loaded(&images) {
163        return;
164    }
165
166    let len = (compiled.width * compiled.height) as usize;
167    let manual_near_mask: Option<&[bool]> = if compiled
168        .manual_near_bg_mask
169        .len()
170        == len
171    {
172        Some(compiled.manual_near_bg_mask.as_slice())
173    } else if compiled.manual_near_bg_mask.is_empty() {
174        None
175    } else {
176        warn!(
177                "Manual near background mask has length {}, expected {} ({}x{}); ignoring manual markers",
178                compiled.manual_near_bg_mask.len(),
179                len,
180                compiled.width,
181                compiled.height
182            );
183        None
184    };
185    let (tundra_allowed, ice_allowed) =
186        build_background_biome_allowed_masks(biome_mask, len);
187    let tundra_present = tundra_allowed.iter().any(|&b| b);
188
189    let seed = request.as_ref().map_or(0, |r| r.seed);
190    let banks = match background_bank::load_pair(&cfg, &compiled, seed) {
191        Ok(banks) => banks,
192        Err(error) => { built_flag.fail(error); return; }
193    };
194
195    if let Some(mask_atlas) = darkness_masks.as_mut() {
196        mask_atlas.clear_background_layers();
197    }
198
199    let level_world_w = compiled.bounds.width();
200    let level_world_h = compiled.bounds.height();
201    let level_center = Vec2::new(level_world_w * 0.5, level_world_h * 0.5);
202
203    let near_depth = background_surface::parallax_depth(0);
204
205    let shared_images = images.as_ref();
206    let exemplar = |handle: &Handle<Image>| shared_images.get(handle).and_then(Exemplar::from_image);
207    let (Some(ice), Some(tundra)) = (exemplar(&libraries.backgrounds.ice), exemplar(&libraries.backgrounds.tundra)) else {
208        built_flag.fail("Near-background exemplars are missing or unreadable".into());
209        return;
210    };
211    // One near wall for every biome; each biome details it after its own exemplar.
212    let size = UVec2::new(compiled.width, compiled.height);
213    let mut wall = background_openings::near_wall(&compiled.terrain, size, seed);
214    for (covered, &marked) in wall.iter_mut().zip(manual_near_mask.unwrap_or_default()) {
215        *covered |= marked;
216    }
217    let target = Context::new(wall, compiled.terrain.clone(), size);
218    let ice_near_grid = background_synthesis::synthesize(&target, &ice, |index| ice_allowed[index], seed);
219    let tundra_near_grid = tundra_present.then(|| {
220        background_synthesis::synthesize(&target, &tundra, |index| tundra_allowed[index], seed ^ TUNDRA_DETAIL_SALT)
221    });
222    let near_grid = combine_background_biome_near_grids(len, &tundra_allowed, &ice_near_grid, tundra_near_grid.as_deref());
223    let near_layer = build_near_background_layer(&near_grid, NearBackgroundLayerInputs {
224        compiled: &compiled,
225        biome_mask,
226        ldtk_data: &ldtk_data,
227        images: shared_images,
228        blur_settings: &blur_settings,
229        near_depth,
230    });
231    let [mid_layer, far_layer] = banks;
232
233    for (group, layer) in [near_layer, Some(mid_layer), Some(far_layer)].into_iter().enumerate() {
234        let Some(layer) = layer else { continue; };
235        background_surface::publish(&mut commands, root_e, group, layer, level_center,
236            darkness_masks.as_deref_mut(), &mut images, &mut meshes, &mut color_materials);
237    }
238
239    built_flag.phase = pipeline::BuildPhase::Surfaces;
240}

242pub(super) fn warp_bg_int_grid_for_parallax(
243    width: u32,
244    height: u32,
245    grid: &[i32],
246    biome_mask: Option<&LevelBiomeMask>,
247    parallax_depth: f32,
248) -> (Vec<i32>, Option<LevelBiomeMask>) {
249    let len = (width * height) as usize;
250    if width == 0 || height == 0 || grid.len() != len {
251        return (grid.to_vec(), None);
252    }
253    if !parallax_depth.is_finite() || parallax_depth <= 0.0 {
254        return (grid.to_vec(), None);
255    }
256
257    let f = 1.0 / parallax_depth;
258    if !f.is_finite() || f <= 0.0 || (f - 1.0).abs() < 0.0001 {
259        return (grid.to_vec(), None);
260    }
261
262    let origin_x = width as f32 * 0.5;
263    let origin_y = height as f32 * 0.5;
264    let w_i = width as i32;
265    let h_i = height as i32;
266
267    let src_cells = biome_mask.and_then(|b| {
268        if b.matches_dims(width, height) {
269            Some(b.cells.as_slice())
270        } else {
271            None
272        }
273    });
274    let mut warped_cells = src_cells.map(|_| vec![BiomeKind::default(); len]);
275    let src_inner = biome_mask.and_then(|b| {
276        if b.matches_dims(width, height) && b.tundra_inner.len() == len {
277            Some(b.tundra_inner.as_slice())
278        } else {
279            None
280        }
281    });
282    let mut warped_inner = src_inner.map(|_| vec![false; len]);
283
284    let mut warped_grid = vec![0i32; len];
285    let sources = BgParallaxWarpSources {
286        width,
287        origin_x,
288        origin_y,
289        scale_factor: f,
290        max_x: w_i - 1,
291        max_y: h_i - 1,
292        grid,
293        src_cells,
294        src_inner,
295    };
296    for y_out in 0..height {
297        warp_bg_int_grid_row(
298            y_out,
299            &sources,
300            &mut warped_grid,
301            &mut warped_cells,
302            &mut warped_inner,
303        );
304    }
305
306    let warped_biome_mask = warped_cells.map(|cells| LevelBiomeMask {
307        width,
308        height,
309        cells,
310        tundra_inner: warped_inner.unwrap_or_default(),
311    });
312    (warped_grid, warped_biome_mask)
313}

315pub(super) struct BgParallaxWarpSources<'a> {
316    pub width: u32,
317    pub origin_x: f32,
318    pub origin_y: f32,
319    pub scale_factor: f32,
320    pub max_x: i32,
321    pub max_y: i32,
322    pub grid: &'a [i32],
323    pub src_cells: Option<&'a [BiomeKind]>,
324    pub src_inner: Option<&'a [bool]>,
325}

327pub(super) fn warp_bg_int_grid_row(
328    y_out: u32,
329    sources: &BgParallaxWarpSources,
330    warped_grid: &mut [i32],
331    warped_cells: &mut Option<Vec<BiomeKind>>,
332    warped_inner: &mut Option<Vec<bool>>,
333) {
334    for x_out in 0..sources.width {
335        warp_bg_int_grid_cell(
336            x_out,
337            y_out,
338            sources,
339            warped_grid,
340            warped_cells,
341            warped_inner,
342        );
343    }
344}
345
346pub(super) fn warp_bg_int_grid_cell(
347    x_out: u32,
348    y_out: u32,
349    sources: &BgParallaxWarpSources,
350    warped_grid: &mut [i32],
351    warped_cells: &mut Option<Vec<BiomeKind>>,
352    warped_inner: &mut Option<Vec<bool>>,
353) {
354    let src_idx = parallax_warp_source_index(x_out, y_out, sources);
355    let dst_idx = (y_out * sources.width + x_out) as usize;
356    warped_grid[dst_idx] = sources.grid[src_idx];
357    copy_warped_biome_cell(sources.src_cells, warped_cells, src_idx, dst_idx);
358    copy_warped_inner_cell(sources.src_inner, warped_inner, src_idx, dst_idx);
359}

361pub(super) fn parallax_warp_source_index(
362    x_out: u32,
363    y_out: u32,
364    sources: &BgParallaxWarpSources,
365) -> usize {
366    let pos_out_x = x_out as f32 + 0.5;
367    let pos_out_y = y_out as f32 + 0.5;
368    let pos_src_x =
369        sources.origin_x + (pos_out_x - sources.origin_x) / sources.scale_factor;
370    let pos_src_y =
371        sources.origin_y + (pos_out_y - sources.origin_y) / sources.scale_factor;
372
373    let src_x = ((pos_src_x - 0.5).round() as i32).clamp(0, sources.max_x) as u32;
374    let src_y = ((pos_src_y - 0.5).round() as i32).clamp(0, sources.max_y) as u32;
375    (src_y * sources.width + src_x) as usize
376}

378pub(super) fn copy_warped_biome_cell(
379    src_cells: Option<&[BiomeKind]>,
380    warped_cells: &mut Option<Vec<BiomeKind>>,
381    src_idx: usize,
382    dst_idx: usize,
383) {
384    let (Some(src_cells), Some(dst_cells)) = (src_cells, warped_cells.as_mut()) else {
385        return;
386    };
387    dst_cells[dst_idx] = src_cells[src_idx];
388}

390pub(super) fn copy_warped_inner_cell(
391    src_inner: Option<&[bool]>,
392    warped_inner: &mut Option<Vec<bool>>,
393    src_idx: usize,
394    dst_idx: usize,
395) {
396    let (Some(src_inner), Some(dst_inner)) = (src_inner, warped_inner.as_mut()) else {
397        return;
398    };
399    dst_inner[dst_idx] = src_inner[src_idx];
400}

402pub(super) fn prepare_background_surface<'a>(
403    compiled: &CompiledLevel, int_grid: &[i32], render_mask: &[bool], biome_mask: Option<&LevelBiomeMask>,
404    ldtk_data: &'a ldtk_rules::LdtkRuleData, images: &'a Assets<Image>,
405) -> Option<surface::TileSurface<'a>> {
406    let _timer = diagnostics::StageTimer::new("bake_background_group_ldtk_from_intgrid");
407    let size = UVec2::new(compiled.width, compiled.height);
408    let assets = biome_layers(ldtk_data, images, None, biome_mask, ["NearBackground", "TundraNearBackground"])?;
409    let tile_px = assets[0].tile_size;
410    let rules: Vec<_> = assets.iter().enumerate().map(|(index, asset)| {
411        let seed = 0x1234_5678_9ABC_DEF0u64.wrapping_add(0x1111 + index as u64 * 0x10);
412        (ldtk_rules::RuleProgram::background(int_grid, size, asset.layer, seed),
413            ldtk_rules::discover_bg_fallback_tiles(asset.layer))
414    }).collect();
415    let cells = int_grid.iter().enumerate().map(|(index, &value)| {
416        if !render_mask[index] { return None; }
417        let source = usize::from(cell_uses_tundra(biome_mask, assets.len(), index));
418        let (program, fallback) = &rules[source];
419        let tile = program.at(index).or_else(|| {
420            (value != 0).then(|| fallback.get(value)).flatten()
421                .map(|tile_id| ldtk_rules::TilePlacement { tile_id, flip_x: false, flip_y: false })
422        });
423        tile.map(|tile| surface::TilePaint { tile, source })
424    }).collect();
425    let sources = assets.into_iter().map(|asset| (asset.image, asset.tiles_per_row)).collect();
426    Some(surface::TileSurface { size, tile_px, sources, cells })
427}
