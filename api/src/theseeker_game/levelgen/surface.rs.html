21pub(super) fn image(size: UVec2) -> Image {
22    let mut image = Image::new_fill(
23        Extent3d { width: size.x, height: size.y, depth_or_array_layers: 1 },
24        TextureDimension::D2, &[0, 0, 0, 0], TextureFormat::Rgba8UnormSrgb,
25        RenderAssetUsages::RENDER_WORLD,
26    );
27    image.sampler = ImageSampler::nearest();
28    image
29}

31pub(super) fn chunk_regions(size: UVec2) -> Vec<URect> {
32    let step = mapgen::CHUNK_IMAGE_SIZE;
33    let cols = size.x.div_ceil(step);
34    let rows = size.y.div_ceil(step);
35    (0..cols * rows).map(|index| {
36        let min = UVec2::new(index % cols, index / cols) * step;
37        URect { min, max: (min + UVec2::splat(step)).min(size) }
38    }).collect()
39}

41pub(super) fn expand(region: URect, margin: u32, size: UVec2) -> URect {
42    URect {
43        min: UVec2::new(region.min.x.saturating_sub(margin), region.min.y.saturating_sub(margin)),
44        max: region.max.saturating_add(UVec2::splat(margin)).min(size),
45    }
46}

48pub(super) fn bake_jobs<T: Send + 'static>(regions: &[URect], bake: impl Fn(URect) -> T + Sync) -> Vec<T> {
49    if regions.is_empty() { return Vec::new(); }
50    let pool = ComputeTaskPool::get_or_init(TaskPool::new);
51    let workers = pool.thread_num().clamp(1, 4);
52    let batch_size = regions.len().div_ceil(workers);
53    pool.scope(|scope| {
54        for batch in regions.chunks(batch_size) {
55            let bake = &bake;
56            scope.spawn(bake_batch(batch, bake));
57        }
58    }).into_iter().flatten().collect()
59}

61impl TileSurface<'_> {
62    pub(super) fn bake(&self, region: URect) -> Image {
63        let mut output = image(region.size() * self.tile_px);
64        let width = output.width();
65        let bytes = output.data.as_mut().expect("surface pixels");
66        for y in region.min.y..region.max.y {
67            blit_row(self, bytes, width, region, y);
68        }
69        output
70    }
71
72}

74fn blit_row(surface: &TileSurface<'_>, bytes: &mut [u8], width: u32, region: URect, y: u32) {
75    for x in region.min.x..region.max.x {
76        let Some(paint) = &surface.cells[(y * surface.size.x + x) as usize] else { continue; };
77        let (source, tiles_per_row) = surface.sources[paint.source];
78        let tile = paint.tile.tile_id;
79        let from = UVec2::new(tile % tiles_per_row, tile / tiles_per_row) * surface.tile_px;
80        let to = (UVec2::new(x, y) - region.min) * surface.tile_px;
81        let flip = BVec2::new(paint.tile.flip_x, paint.tile.flip_y);
82        paint_tile(bytes, width, to, source, from, surface.tile_px, flip);
83    }
84}

86/// Paints the `tile_px` tile at `from` in `source` onto `dest` at `to`, mirrored per
87/// `flip`, over what is already there.
88fn paint_tile(dest: &mut [u8], dest_width: u32, to: UVec2, source: &Image, from: UVec2, tile_px: u32, flip: BVec2) {
89    let Some(src) = source.data.as_deref() else { return; };
90    let src_width = source.width();
91    for (tx, ty) in (0..tile_px).flat_map(|ty| (0..tile_px).map(move |tx| (tx, ty))) {
92        let sx = from.x + if flip.x { tile_px - 1 - tx } else { tx };
93        let sy = from.y + if flip.y { tile_px - 1 - ty } else { ty };
94        let src_idx = ((sy * src_width + sx) * 4) as usize;
95        let dst_idx = ((to.y + ty) * dest_width + (to.x + tx)) as usize * 4;
96        if src_idx + 3 < src.len() && dst_idx + 3 < dest.len() {
97            blend_over(&mut dest[dst_idx..dst_idx + 4], &src[src_idx..src_idx + 4]);
98        }
99    }
100}

102/// Source-over alpha blending of one RGBA pixel.
103fn blend_over(dst: &mut [u8], src: &[u8]) {
104    let alpha = src[3];
105    if alpha == 0 {
106        return;
107    }
108    if alpha == 255 {
109        dst.copy_from_slice(&[src[0], src[1], src[2], 255]);
110        return;
111    }
112    let sa = alpha as f32 / 255.0;
113    let da = dst[3] as f32 / 255.0;
114    let out_a = sa + da * (1.0 - sa);
115    if out_a <= 0.0 {
116        return;
117    }
118    for channel in 0..3 {
119        dst[channel] = ((src[channel] as f32 * sa + dst[channel] as f32 * da * (1.0 - sa)) / out_a) as u8;
120    }
121    dst[3] = (out_a * 255.0) as u8;
122}

124pub(super) fn crop(source: &Image, rect: URect) -> Image {
125    let mut output = image(rect.size());
126    let Some(src) = &source.data else { return output; };
127    let dst = output.data.as_mut().expect("crop pixels");
128    let src_stride = source.width() as usize * 4;
129    let dst_stride = rect.width() as usize * 4;
130    for (y, row) in dst.chunks_exact_mut(dst_stride).enumerate() {
131        let start = (rect.min.y as usize + y) * src_stride + rect.min.x as usize * 4;
132        row.copy_from_slice(&src[start..start + dst_stride]);
133    }
134    output
135}

152pub(super) fn bake_ground(
153    surface: &TileSurface<'_>, level: &CompiledLevel, region: URect,
154    decorations: &[DecorationPlacement], images: &Assets<Image>,
155) -> GroundPatch {
156    let _timer = diagnostics::StageTimer::new("ground_region_bake");
157    let padded = expand(region, ground_shading::SUPPORT_PX.div_ceil(surface.tile_px), surface.size);
158    let mut color = surface.bake(padded);
159    if !visible(&color) && decorations.is_empty() {
160        return GroundPatch { region, color: image(region.size() * surface.tile_px), overlay: None };
161    }
162    let origin = padded.min * surface.tile_px;
163    let width = color.width();
164    let height = color.height();
165    let bytes = color.data.as_mut().expect("ground pixels");
166    let spike_mask = spike_pixels(bytes, width, origin, level, surface.tile_px);
167    ground_shading::shade(bytes, width as usize * 4, origin, level, surface.tile_px);
168    let overlay = build_spike_light_overlay_image(bytes, width, height, &spike_mask);
169    decorate(&mut color, origin, decorations, images, surface.tile_px);
170    let crop_rect = URect { min: (region.min - padded.min) * surface.tile_px,
171        max: (region.max - padded.min) * surface.tile_px };
172    GroundPatch { region, color: crop(&color, crop_rect),
173        overlay: overlay.map(|image| crop(&image, crop_rect)) }
174}

176fn spike_pixels(bytes: &[u8], width: u32, origin: UVec2, level: &CompiledLevel, tile_px: u32) -> Vec<u8> {
177    bytes.as_chunks::<4>().0.iter().enumerate().map(|(index, pixel)| {
178        let tile = (origin + UVec2::new(index as u32 % width, index as u32 / width)) / tile_px;
179        let spike = level.spike_tiles.get((tile.y * level.width + tile.x) as usize).copied().unwrap_or(false);
180        u8::from(spike && pixel[3] != 0 && pixel[..3] == SPIKE_LIGHT_OVERLAY_COLOR)
181    }).collect()
182}

184/// Paints active decorations over an image whose top-left corner sits at `origin` in level pixels.
185pub(super) fn decorate(image: &mut Image, origin: UVec2, placements: &[DecorationPlacement], images: &Assets<Image>, tile_px: u32) {
186    let size = IVec2::new(image.width() as i32, image.height() as i32);
187    let origin = origin.as_ivec2();
188    let overlaps = |art: &DecorationArt<'_>| (art.origin - origin).cmplt(size).all() && (art.origin + art.size).cmpgt(origin).all();
189    let arts = placements.iter().filter(|placement| placement.active).filter_map(|placement| DecorationArt::new(placement, images, tile_px));
190    let pixels = arts.filter(overlaps).flat_map(DecorationArt::pixels).map(|(at, pixel)| (at - origin, pixel));
191    let data = image.data.as_mut().expect("decoration pixels");
192    for (at, pixel) in pixels.filter(|(at, _)| at.cmpge(IVec2::ZERO).all() && at.cmplt(size).all()) {
193        let index = (at.y * size.x + at.x) as usize * 4;
194        data[index..index + 4].copy_from_slice(pixel);
195    }
196}

198/// Level pixels, counted bottom-up, that snow decorations cover, so blood stays behind the snow.
199pub(super) fn snow_occlusion(placements: &[DecorationPlacement], size: UVec2, tile_px: u32, images: &Assets<Image>) -> BloodForegroundOcclusion {
200    let size = size.as_ivec2();
201    let snow = placements.iter().filter(|placement| placement.active && placement.kind == "snow");
202    let pixels = snow.filter_map(|placement| DecorationArt::new(placement, images, tile_px)).flat_map(DecorationArt::pixels);
203    let inside = pixels.map(|(at, _)| at).filter(|at| at.cmpge(IVec2::ZERO).all() && at.cmplt(size).all());
204    BloodForegroundOcclusion::from_pixels(inside.map(|at| IVec2::new(at.x, size.y - 1 - at.y)))
205}

207/// A decoration's sprite, hung 10px below its tile's bottom center and shifted by its offset (positive y up).
208#[derive(Clone, Copy)]
209struct DecorationArt<'a> {
210    bytes: &'a [u8],
211    size: IVec2,
212    /// The top-left corner in level pixels.
213    origin: IVec2,
214    flip: BVec2,
215}
216
217impl<'a> DecorationArt<'a> {
218    fn new(placement: &DecorationPlacement, images: &'a Assets<Image>, tile_px: u32) -> Option<Self> {
219        let image = images.get(&placement.handle)?;
220        let size = IVec2::new(image.width() as i32, image.height() as i32);
221        if size.min_element() == 0 {
222            return None;
223        }
224        let tile = tile_px as i32;
225        let anchor = IVec2::new(placement.tile_x as i32, placement.tile_y as i32) * tile + IVec2::new(tile / 2, tile + 10);
226        let origin = anchor - IVec2::new(size.x / 2, size.y) + placement.offset_px * IVec2::new(1, -1);
227        Some(Self { bytes: image.data.as_deref()?, size, origin, flip: BVec2::new(placement.flip_x, placement.flip_y) })
228    }
229
230    /// Each opaque pixel's position in level pixels, with its color.
231    fn pixels(self) -> impl Iterator<Item = (IVec2, &'a [u8])> {
232        let positions = (0..self.size.y).flat_map(move |y| (0..self.size.x).map(move |x| IVec2::new(x, y)));
233        positions.filter_map(move |at| {
234            let source = IVec2::select(self.flip, self.size - 1 - at, at);
235            let index = (source.y * self.size.x + source.x) as usize * 4;
236            self.bytes.get(index..index + 4).filter(|pixel| pixel[3] != 0).map(|pixel| (self.origin + at, pixel))
237        })
238    }
239}

241async fn bake_batch<T>(regions: &[URect], bake: &(impl Fn(URect) -> T + Sync)) -> Vec<T> {

245/// Decoration ownership is indexed once, including artwork that overhangs a neighboring chunk.
246pub(super) fn decoration_chunks(placements: &[DecorationPlacement], images: &Assets<Image>, tile_px: u32) -> HashMap<IVec2, Vec<DecorationPlacement>> {
247    let edge = IVec2::splat((mapgen::CHUNK_IMAGE_SIZE * tile_px) as i32);
248    let mut chunks: HashMap<IVec2, Vec<DecorationPlacement>> = HashMap::new();
249    for placement in placements.iter().filter(|placement| placement.active) {
250        let Some(art) = DecorationArt::new(placement, images, tile_px) else { continue };
251        let (min, max) = (art.origin.div_euclid(edge), (art.origin + art.size - 1).div_euclid(edge));
252        for chunk in (min.y..=max.y).flat_map(|y| (min.x..=max.x).map(move |x| IVec2::new(x, y))) {
253            chunks.entry(chunk).or_default().push(placement.clone());
254        }
255    }
256    chunks
