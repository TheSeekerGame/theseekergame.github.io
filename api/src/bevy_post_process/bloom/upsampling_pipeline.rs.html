43pub fn init_bloom_upscaling_pipeline(
44    mut commands: Commands,
45    fullscreen_shader: Res<FullscreenShader>,
46    asset_server: Res<AssetServer>,
47) {
48    let bind_group_layout = BindGroupLayoutDescriptor::new(
49        "bloom_upsampling_bind_group_layout",
50        &BindGroupLayoutEntries::sequential(
51            ShaderStages::FRAGMENT,
52            (
53                // Input texture
54                texture_2d(TextureSampleType::Float { filterable: true }),
55                // Sampler
56                sampler(SamplerBindingType::Filtering),
57                // BloomUniforms
58                uniform_buffer::<BloomUniforms>(true),
59            ),
60        ),
61    );
62
63    commands.insert_resource(BloomUpsamplingPipeline {
64        bind_group_layout,
65        fullscreen_shader: fullscreen_shader.clone(),
66        fragment_shader: load_embedded_asset!(asset_server.as_ref(), "bloom.wgsl"),
67    });
68}

70impl SpecializedRenderPipeline for BloomUpsamplingPipeline {
71    type Key = BloomUpsamplingPipelineKeys;
72
73    fn specialize(&self, key: Self::Key) -> RenderPipelineDescriptor {
74        let color_blend = match key.composite_mode {
75            BloomCompositeMode::EnergyConserving => {
76                // At the time of developing this we decided to blend our
77                // blur pyramid levels using native WGPU render pass blend
78                // constants. They are set in the bloom node's run function.
79                // This seemed like a good approach at the time which allowed
80                // us to perform complex calculations for blend levels on the CPU,
81                // however, we missed the fact that this prevented us from using
82                // textures to customize bloom appearance on individual parts
83                // of the screen and create effects such as lens dirt or
84                // screen blur behind certain UI elements.
85                //
86                // TODO: Use alpha instead of blend constants and move
87                // compute_blend_factor to the shader. The shader
88                // will likely need to know current mip number or
89                // mip "angle" (original texture is 0deg, max mip is 90deg)
90                // so make sure you give it that as a uniform.
91                // That does have to be provided per each pass unlike other
92                // uniforms that are set once.
93                BlendComponent {
94                    src_factor: BlendFactor::Constant,
95                    dst_factor: BlendFactor::OneMinusConstant,
96                    operation: BlendOperation::Add,
97                }
98            }
99            BloomCompositeMode::Additive => BlendComponent {
100                src_factor: BlendFactor::Constant,
101                dst_factor: BlendFactor::One,
102                operation: BlendOperation::Add,
103            },
104        };
105
106        RenderPipelineDescriptor {
107            label: Some("bloom_upsampling_pipeline".into()),
108            layout: vec![self.bind_group_layout.clone()],
109            vertex: self.fullscreen_shader.to_vertex_state(),
110            fragment: Some(FragmentState {
111                shader: self.fragment_shader.clone(),
112                entry_point: Some("upsample".into()),
113                targets: vec![Some(ColorTargetState {
114                    format: key.target_format,
115                    blend: Some(BlendState {
116                        color: color_blend,
117                        alpha: BlendComponent {
118                            src_factor: BlendFactor::Zero,
119                            dst_factor: BlendFactor::One,
120                            operation: BlendOperation::Add,
121                        },
122                    }),
123                    write_mask: ColorWrites::ALL,
124                })],
125                ..default()
126            }),
127            ..default()
128        }
129    }
130}

132pub fn prepare_upsampling_pipeline(
133    mut commands: Commands,
134    pipeline_cache: Res<PipelineCache>,
135    mut pipelines: ResMut<SpecializedRenderPipelines<BloomUpsamplingPipeline>>,
136    pipeline: Res<BloomUpsamplingPipeline>,
137    views: Query<(&ExtractedView, Entity, &Bloom)>,
138) {
139    for (view, entity, bloom) in &views {
140        let pipeline_id = pipelines.specialize(
141            &pipeline_cache,
142            &pipeline,
143            BloomUpsamplingPipelineKeys {
144                composite_mode: bloom.composite_mode,
145                target_format: BLOOM_TEXTURE_FORMAT,
146            },
147        );
148
149        let pipeline_final_id = pipelines.specialize(
150            &pipeline_cache,
151            &pipeline,
152            BloomUpsamplingPipelineKeys {
153                composite_mode: bloom.composite_mode,
154                target_format: view.target_format,
155            },
156        );
157
158        commands.entity(entity).insert(UpsamplingPipelineIds {
159            id_main: pipeline_id,
160            id_final: pipeline_final_id,
161        });
162    }
