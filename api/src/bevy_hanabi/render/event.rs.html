42/// GPU buffer storing the spawn events emitted by parent effects for their
43/// children.
44///
45/// The event buffer contains for each effect the number of particles to spawn
46/// this frame. That number is incremented by another effect when it
47/// emits a spawn event, and reset to zero on next frame after the indirect init
48/// pass spawned new particles, and before the new update pass of the
49/// source effect optionally emits more spawn events.
50///
51/// GPU spawn events are never accumulated over frames; if a source emits too
52/// many events and the target effect cannot spawn that many particles, for
53/// example because it reached its capacity, then the extra events are
54/// discarded. This is consistent with the CPU behavior of
55/// [`EffectSpawner::spawn_count`].
56///
57/// Note that the number of allocated events in the buffer slice associated with
58/// a child effect instance is not recorded here; instead it's stored in
59/// [`GpuChildInfo::event_count`]. This buffer only stores the events
60/// themselves.
61pub struct EventBuffer {
62    /// GPU buffer storing the spawn events.
63    buffer: Buffer,
64    /// Buffer capacity, in words (4 bytes).
65    capacity: u32,
66    /// Allocated (used) buffer size, in words (4 bytes).
67    size: u32,
68    /// Slices into the GPU buffer where event sub-allocations for each effect
69    /// are located. Slices are stored ordered by location in the buffer, for
70    /// convenience of allocation.
71    slices: Vec<EventSlice>,
72}
73
74impl EventBuffer {
75    /// Create a new event buffer to store the spawn events of the specified
76    /// child effect.
77    pub fn new(buffer: Buffer, capacity: u32) -> Self {
78        Self {
79            buffer,
80            capacity,
81            size: 0,
82            slices: vec![],
83        }
84    }
85
86    /// Get a reference to the underlying GPU buffer.
87    pub fn buffer(&self) -> &Buffer {
88        &self.buffer
89    }
90
91    /// Allocate a new slice for a child effect.
92    pub fn allocate(&mut self, size: u32) -> Option<EventSlice> {
93        if self.size + size > self.capacity {
94            return None;
95        }
96
97        if self.slices.is_empty() {
98            let slice = EventSlice { slice: 0..size };
99            self.slices.push(slice.clone());
100            self.size += size;
101            return Some(slice);
102        }
103
104        let mut start = 0;
105        for (idx, es) in self.slices.iter().enumerate() {
106            let avail_size = es.slice.start - start;
107            if size > avail_size {
108                start = es.slice.end;
109                continue;
110            }
111
112            let slice = EventSlice {
113                slice: start..start + size,
114            };
115            self.slices.insert(idx, slice.clone());
116            self.size += size;
117            return Some(slice);
118        }
119
120        if start + size <= self.capacity {
121            let slice = EventSlice {
122                slice: start..start + size,
123            };
124            self.slices.push(slice.clone());
125            self.size += size;
126            Some(slice)
127        } else {
128            None
129        }
130    }
131
132    /// Free the slice of a consumer effect once that effect is deallocated.
133    pub fn free(&mut self, slice: &EventSlice) -> SlabState {
134        // Note: could use binary search, but likely not enough elements to be worth it
135        if let Some(idx) = self.slices.iter().position(|es| es == slice) {
136            self.slices.remove(idx);
137        }
138        if self.slices.is_empty() {
139            SlabState::Free
140        } else {
141            SlabState::Used
142        }
143    }
144}

244/// Allocate storage for GPU events for all child effects.
245///
246/// This system manages allocating storage for GPU events of child effects, and
247/// spawning the [`CachedEffectEvents`] storing that allocation.
248pub(crate) fn allocate_events(
249    mut commands: Commands,
250    mut event_cache: ResMut<EventCache>,
251    mut q_child_effects: Query<(Entity, Option<&mut CachedEffectEvents>), With<ChildEffectOf>>,
252    q_old_child_effects: Query<Entity, (With<CachedEffectEvents>, Without<ChildEffectOf>)>,
253) {
254    #[cfg(feature = "trace")]
255    let _span = bevy::log::info_span!("allocate_events").entered();
256    trace!("allocate_events");
257
258    event_cache.clear_previous_frame_resizes();
259
260    // Allocate storage and add the component to childs missing it.
261    for (entity, maybe_cached_events) in &mut q_child_effects {
262        if let Some(_cached_events) = maybe_cached_events {
263            // Nothing really to do for now because we hardcode a number of
264            // events, so the allocation won't ever change...
265        } else {
266            const FIXME_HARD_CODED_EVENT_COUNT: u32 = 256;
267            let cached_effect_events = event_cache.allocate(FIXME_HARD_CODED_EVENT_COUNT);
268            commands.entity(entity).insert(cached_effect_events);
269        }
270    }
271
272    // Remove the component from effects which are not a child anymore. This should
273    // be pretty rare; in general the effect is just despawned.
274    for entity in &q_old_child_effects {
275        commands.entity(entity).remove::<CachedEffectEvents>();
276    }
277}

280/// which indicates that the effect doesn't use GPU events anymore.
281pub(crate) fn on_remove_cached_effect_events(
282    trigger: On<Remove, CachedEffectEvents>,
283    query: Query<(Entity, &CachedEffectEvents)>,
284    mut event_cache: ResMut<EventCache>,
285) {
286    #[cfg(feature = "trace")]
287    let _span = bevy::log::info_span!("on_remove_cached_effect_events").entered();
288    trace!("on_remove_cached_effect_events");
289
290    if let Ok((entity, cached_effect_event)) = query.get(trigger.event().entity) {
291        // TODO - handle SlabState return value to invalidate property bind groups!!
292        if let Err(err) = event_cache.free(cached_effect_event) {
293            error!("Error while freeing cached events for effect {entity:?}: {err:?}");
294        }
295    };

311/// Cache for effect events.
312#[derive(Resource)]
313pub struct EventCache {
314    /// Render device to allocate GPU resources as needed.
315    device: RenderDevice,
316    /// Single shared GPU buffer storing all the [`GpuChildInfo`] structs
317    /// for all the parent effects.
318    child_infos_buffer: HybridAlignedBufferVec,
319    /// Collection of event buffers managed by this cache. Some buffers might
320    /// be `None` if the entry is not used. Since the buffers are referenced
321    /// by index, we cannot move them once they're allocated.
322    buffers: Vec<Option<EventBuffer>>,
323    /// Single shared GPU buffer storing all the [`GpuDispatchIndirectArgs`]
324    /// structs for all the indirect init passes. Any effect allocating storage
325    /// for GPU events also get an entry into this buffer, to allow consuming
326    /// the events from an init pass indirectly dispatched (GPU-driven).
327    // FIXME - merge with the update pass one, we don't need 2 buffers storing the same type; on
328    // the other hand if we sync the allocations with GpuChildInfo we can guarantee a perfect
329    // batching for the init fill dispatch pass (single dispatch for all instances at once).
330    init_indirect_dispatch_buffer: GpuBuffer<GpuDispatchIndirectArgs>,
331    /// Bind group layout for the indirect dispatch pass, which clears the GPU
332    /// event counts ([`GpuChildInfo::event_count`]).
333    indirect_child_info_buffer_bind_group_layout: BindGroupLayout,
334    /// Bind group for the indirect dispatch pass, which clears the GPU event
335    /// counts ([`GpuChildInfo::event_count`]).
336    indirect_child_info_buffer_bind_group: Option<BindGroup>,
337}

394    /// Allocate a memory block to store the given number of GPU events.
395    ///
396    /// The allocation always succeeds, allocating a new GPU event buffer if
397    /// none of the existing ones can store the requested number of events.
398    ///
399    /// # Returns
400    ///
401    /// The [`CachedEffectEvents`] component representing the allocation.
402    ///
403    /// # Panics
404    ///
405    /// Panics if the number of events `num_events` is zero.
406    pub fn allocate(&mut self, num_events: u32) -> CachedEffectEvents {
407        assert!(num_events > 0);
408
409        // Allocate an entry into the indirect dispatch buffer
410        // The value pushed is a dummy; see allocate_frame_buffers().
411        let init_indirect_dispatch_index = self.init_indirect_dispatch_buffer.allocate();
412
413        // Try to find an allocated GPU buffer with enough capacity
414        let mut empty_index = None;
415        for (buffer_index, buffer) in self.buffers.iter_mut().enumerate() {
416            let Some(buffer) = buffer.as_mut() else {
417                // Remember the first empty slot in case we need to allocate a new GPU buffer
418                if empty_index.is_none() {
419                    empty_index = Some(buffer_index);
420                }
421                continue;
422            };
423
424            // Try to allocate a slice into the buffer
425            if let Some(event_slice) = buffer.allocate(num_events) {
426                trace!("Allocate new slice in event buffer #{buffer_index} for {num_events} events: range={event_slice:?}");
427                return CachedEffectEvents {
428                    buffer_index: buffer_index as u32,
429                    init_indirect_dispatch_index,
430                    range: event_slice.slice,
431                };
432            }
433        }
434
435        // Cannot find any suitable GPU event buffer; allocate a new one
436
437        // Compute the slot where to store the new buffer
438        let buffer_index = empty_index.unwrap_or(self.buffers.len());
439
440        // Create the GPU bfufer
441        let label = format!("hanabi:buffer:event_buffer{buffer_index}");
442        let align = self.device.limits().min_storage_buffer_offset_alignment;
443        let capacity = num_events.max(16 * 1024); // min capacity
444        let byte_size = (capacity as u64 * 4).next_multiple_of(align as u64);
445        let capacity = (byte_size / 4) as u32;
446        // In debug, fill the buffer with some debug marker
447        #[cfg(debug_assertions)]
448        let buffer = {
449            let mut contents: Vec<u32> = Vec::with_capacity(capacity as usize);
450            contents.resize(capacity as usize, 0xDEADBEEF);
451            self.device.create_buffer_with_data(&BufferInitDescriptor {
452                label: Some(&label[..]),
453                usage: BufferUsages::COPY_DST | BufferUsages::STORAGE,
454                contents: bytemuck::cast_slice(contents.as_slice()),
455            })
456        };
457        // In release, don't initialize the buffer for performance
458        #[cfg(not(debug_assertions))]
459        let buffer = self.device.create_buffer(&BufferDescriptor {
460            label: Some(&label[..]),
461            size: byte_size,
462            usage: BufferUsages::COPY_DST | BufferUsages::STORAGE,
463            mapped_at_creation: false,
464        });
465        trace!("Created new event buffer #{buffer_index} '{label}' with {byte_size} bytes ({capacity} events; align={align}B)");
466        let mut buffer = EventBuffer::new(buffer, capacity);
467
468        // Allocate a slice from the new event buffer
469        let event_slice = buffer.allocate(num_events).expect("Failed to allocate event slice inside new buffer specifically created for this allocation.");
470        trace!("Allocate new slice in event buffer #{buffer_index} for {num_events} events: range={event_slice:?}");
471
472        // Store the event buffer at the selected slot
473        if buffer_index >= self.buffers.len() {
474            self.buffers.push(Some(buffer));
475        } else {
476            debug_assert!(self.buffers[buffer_index].is_none());
477            self.buffers[buffer_index] = Some(buffer);
478        }
479
480        CachedEffectEvents {
481            buffer_index: buffer_index as u32,
482            init_indirect_dispatch_index,
483            range: event_slice.slice,
484        }
485    }

487    /// Deallocated and remove an event block allocation from the cache.
488    pub fn free(
489        &mut self,
490        cached_effect_events: &CachedEffectEvents,
491    ) -> Result<SlabState, CachedEventsError> {
492        trace!(
493            "Removing cached event {:?} from cache.",
494            cached_effect_events
495        );
496
497        self.init_indirect_dispatch_buffer
498            .free(cached_effect_events.init_indirect_dispatch_index);
499
500        let entry = self
501            .buffers
502            .get_mut(cached_effect_events.buffer_index as usize)
503            .ok_or(CachedEventsError::InvalidBufferIndex(
504                cached_effect_events.buffer_index,
505            ))?;
506        let buffer = entry.as_mut().ok_or(CachedEventsError::BufferDeallocated(
507            cached_effect_events.buffer_index,
508        ))?;
509        if buffer.free(&EventSlice {
510            slice: cached_effect_events.range.clone(),
511        }) == SlabState::Free
512        {
513            let buffer = entry.take().unwrap();
514            buffer.buffer.destroy();
515            Ok(SlabState::Free)
516        } else {
517            Ok(SlabState::Used)
518        }
519    }

521    /// Allocate a new block of [`GpuChildInfo`] structures for a list of

523    pub fn allocate_child_infos(
524        &mut self,
525        parent_entity: Entity,
526        children: Vec<(Entity, BufferBindingSource)>,
527        child_infos: &[GpuChildInfo],

541            children,

546    /// Re-allocate a block of [`GpuChildInfo`] structures for a modified list
547    /// of children.
548    pub fn reallocate_child_infos(
549        &mut self,
550        parent_entity: Entity,
551        children: Vec<(Entity, BufferBindingSource)>,
552        child_infos: &[GpuChildInfo],
553        cached_parent_info: &mut CachedParentInfo,

564        assert_eq!(

568        trace!(
569            "Parent {:?}: Allocated new ChildInfo[] entry at byte range {:?}",
570            parent_entity,
571            byte_range
572        );
573
574        cached_parent_info.children = children;
575        cached_parent_info.byte_range = byte_range;
576    }
