119pub(crate) struct SortedEffectBatches {
120    /// Effect batches in the order they were inserted by [`push()`], indexed by
121    /// the returned [`EffectBatchIndex`].
122    ///
123    /// [`push()`]: Self::push
124    batches: Vec<EffectBatch>,
125    /// Index of the dispatch queue used for indirect fill dispatch and
126    /// submitted to [`GpuBufferOperations`].
127    pub(super) dispatch_queue_index: Option<u32>,
128}
129
130impl SortedEffectBatches {
131    pub fn clear(&mut self) {
132        self.batches.clear();
133        self.dispatch_queue_index = None;
134    }
135
136    /// Insert a new batch into the collection.
137    ///
138    /// # Returns
139    ///
140    /// This returns the index of the new batch if the inserted one couldn't be
141    /// merged with a previous batch. Otherwise the input batch was merged with
142    /// an existing one, and therefore share its index; in that case `None` is
143    /// returned.
144    pub fn push(&mut self, effect_batch: EffectBatch) -> EffectBatchIndex {
145        let index = self.batches.len() as u32;
146        self.batches.push(effect_batch);
147        EffectBatchIndex(index)
148    }
149
150    #[allow(dead_code)]
151    pub fn len(&self) -> usize {
152        self.batches.len()
153    }
154
155    pub fn is_empty(&self) -> bool {
156        self.batches.is_empty()
157    }
158
159    /// Get an iterator over the sorted sequence of effect batches.
160    #[inline]
161    pub fn iter(&self) -> &[EffectBatch] {
162        &self.batches
163    }
164
165    pub fn get(&self, index: EffectBatchIndex) -> Option<&EffectBatch> {
166        if index.0 < self.batches.len() as u32 {
167            Some(&self.batches[index.0 as usize])
168        } else {
169            None
170        }
171    }
172}

195    level: u32,

204/// Sorts effects into the proper order for batching.
205///
206/// This places parents before children and also tries to place effects in the
207/// same slab together.
208pub(crate) struct EffectSorter {
209    /// Information that we keep about each effect.
210    pub effects: Vec<EffectToBeSorted>,
211    /// A mapping from a child to its parent, if it has one.
212    pub child_to_parent: EntityHashMap<Entity>,
213}
214
215impl EffectSorter {
216    /// Creates a new [`EffectSorter`].
217    pub fn new() -> EffectSorter {
218        EffectSorter {
219            effects: vec![],
220            child_to_parent: default(),
221        }
222    }
223
224    /// Insert an effect to be sorted.
225    pub fn insert(
226        &mut self,
227        entity: Entity,
228        slab_id: SlabId,
229        base_instance: u32,
230        parent: Option<Entity>,
231    ) {
232        self.effects.push(EffectToBeSorted {
233            entity,
234            slab_id,
235            base_instance,
236        });
237        if let Some(parent) = parent {
238            self.child_to_parent.insert(entity, parent);
239        }
240    }
241
242    /// Sorts all the effects into the optimal order for batching.
243    pub fn sort(&mut self) {
244        // trace!("Sorting {} effects...", self.effects.len());
245        // for effect in &self.effects {
246        //     trace!(
247        //         "+ {}: slab={:?} base_instance={:?}",
248        //         effect.entity,
249        //         effect.slab_id,
250        //         effect.base_instance
251        //     );
252        // }
253        // trace!("child->parent:");
254        // for (k, v) in &self.child_to_parent {
255        //     trace!("+ c[{k}] -> p[{v}]");
256        // }
257
258        // First, create a map of entity to index.
259        let mut entity_to_index = EntityHashMap::default();
260        for (index, effect) in self.effects.iter().enumerate() {
261            entity_to_index.insert(effect.entity, index);
262        }
263
264        // Next, create a map of children to their parents.
265        let mut children_to_parent: Vec<_> = (0..self.effects.len()).map(|_| vec![]).collect();
266        for (kid, parent) in self.child_to_parent.iter() {
267            let (parent_index, kid_index) = (entity_to_index[parent], entity_to_index[kid]);
268            children_to_parent[kid_index].push(parent_index);
269        }
270
271        // Now topologically sort the graph. Create an ordering that places
272        // children before parents.
273        // https://en.wikipedia.org/wiki/Topological_sorting#Depth-first_search
274        let mut ordering = vec![0; self.effects.len()];
275        let mut visiting = FixedBitSet::with_capacity(self.effects.len());
276        let mut visited = FixedBitSet::with_capacity(self.effects.len());
277        while let Some(effect_index) = visited.zeroes().next() {
278            visit(
279                &mut ordering,
280                &mut visiting,
281                &mut visited,
282                &children_to_parent,
283                effect_index,
284            );
285        }
286
287        // Compute levels.
288        let mut levels = vec![0; self.effects.len()];
289        for effect_index in ordering.into_iter().rev() {
290            let level = levels[effect_index];
291            for &parent in &children_to_parent[effect_index] {
292                levels[parent] = levels[parent].max(level + 1);
293            }
294        }
295
296        // Now sort the result.
297        self.effects.sort_unstable_by_key(|effect| EffectSortKey {
298            level: levels[entity_to_index[&effect.entity]],
299            slab_id: effect.slab_id,
300            base_instance: effect.base_instance,
301        });
302
303        // Helper function for topologically sorting the effect dependency graph
304        fn visit(
305            ordering: &mut Vec<usize>,
306            visiting: &mut FixedBitSet,
307            visited: &mut FixedBitSet,
308            children_to_parent: &[Vec<usize>],
309            effect_index: usize,
310        ) {
311            if visited.contains(effect_index) {
312                return;
313            }
314            debug_assert!(
315                !visiting.contains(effect_index),
316                "Parent-child effect relation contains a cycle"
317            );
318
319            visiting.insert(effect_index);
320
321            for &parent in &children_to_parent[effect_index] {
322                visit(ordering, visiting, visited, children_to_parent, parent);
323            }
324
325            visited.insert(effect_index);
326            ordering.push(effect_index);
327        }
328    }
329}

343    /// batch.

345    /// Position of the emitter so we can compute distance to camera.

352impl EffectBatch {
353    /// Create a new batch from a single input.
354    pub fn from_input(
355        main_entity: Entity,
356        extracted_effect: &ExtractedEffect,
357        extracted_spawner: &ExtractedSpawner,
358        cached_mesh: &ExtractedEffectMesh,
359        cached_effect_events: Option<&CachedEffectEvents>,
360        cached_child_info: Option<&CachedChildInfo>,
361        spawner_index: u32,
362        input: &mut BatchInput,
363        draw_indirect_buffer_row_index: BufferTableId,
364        metadata_table_id: BufferTableId,
365        property_key: Option<PropertyBindGroupKey>,
366    ) -> EffectBatch {
367        assert_eq!(
368            input.event_buffer_index.is_some(),
369            input.init_indirect_dispatch_index.is_some()
370        );
371
372        let spawn_info = if let Some(event_buffer_index) = input.event_buffer_index {
373            BatchSpawnInfo::GpuSpawner {
374                init_indirect_dispatch_index: input.init_indirect_dispatch_index.unwrap(),
375                event_buffer_index,
376            }
377        } else {
378            BatchSpawnInfo::CpuSpawner {
379                total_spawn_count: extracted_spawner.spawn_count,
380            }
381        };
382
383        EffectBatch {
384            batch_info_id: u32::MAX, // allocated later once the batch is completed
385            handle: extracted_effect.handle.clone(),
386            slab_id: input.effect_slice.slab_id,
387            slice: input.effect_slice.slice.clone(),
388            spawn_info,
389            init_and_update_pipeline_ids: input.init_and_update_pipeline_ids,
390            render_shader: extracted_effect.effect_shaders.render.clone(),
391            parent_slab_id: input.parent_slab_id,
392            parent_min_binding_size: cached_child_info
393                .map(|cci| cci.parent_particle_layout.min_binding_size32()),
394            parent_binding_source: cached_child_info
395                .map(|cci| cci.parent_buffer_binding_source.clone()),
396            child_event_buffers: input.child_effects.clone(),
397            property_key,
398            spawner_base: spawner_index,
399            particle_layout: input.effect_slice.particle_layout.clone(),
400            draw_indirect_buffer_row_index,
401            metadata_table_id,
402            layout_flags: extracted_effect.layout_flags,
403            mesh: cached_mesh.mesh,
404            texture_layout: extracted_effect.texture_layout.clone(),
405            textures: extracted_effect.textures.clone(),
406            alpha_mode: extracted_effect.alpha_mode,
407            entities: vec![main_entity.index_u32()],
408            cached_effect_events: cached_effect_events.cloned(),
409            sort_fill_indirect_dispatch_index: None, // set later as needed
410        }
411    }
412}

463    #[test]
464    fn toposort_batches() {
465        let mut sorter = EffectSorter::new();
466
467        // Some "parent" effect
468        let e1 = Entity::from_raw_u32(1).unwrap();
469        insert_entry(&mut sorter, e1, SlabId::new(42), 0, None);
470        assert_eq!(sorter.effects.len(), 1);
471        assert_eq!(sorter.effects[0].entity, e1);
472        assert!(sorter.child_to_parent.is_empty());
473
474        // Some "child" effect in a different buffer
475        let e2 = Entity::from_raw_u32(2).unwrap();
476        insert_entry(&mut sorter, e2, SlabId::new(5), 30, Some(e1));
477        assert_eq!(sorter.effects.len(), 2);
478        assert_eq!(sorter.effects[0].entity, e1);
479        assert_eq!(sorter.effects[1].entity, e2);
480        assert_eq!(sorter.child_to_parent.len(), 1);
481        assert_eq!(sorter.child_to_parent[&e2], e1);
482
483        sorter.sort();
484        assert_eq!(sorter.effects.len(), 2);
485        assert_eq!(sorter.effects[0].entity, e2); // child first
486        assert_eq!(sorter.effects[1].entity, e1); // parent after
487        assert_eq!(sorter.child_to_parent.len(), 1); // unchanged
488        assert_eq!(sorter.child_to_parent[&e2], e1); // unchanged
489
490        // Some "child" effect in the same buffer as its parent
491        let e3 = Entity::from_raw_u32(3).unwrap();
492        insert_entry(&mut sorter, e3, SlabId::new(42), 20, Some(e1));
493        assert_eq!(sorter.effects.len(), 3);
494        assert_eq!(sorter.effects[0].entity, e2); // from previous sort
495        assert_eq!(sorter.effects[1].entity, e1); // from previous sort
496        assert_eq!(sorter.effects[2].entity, e3); // simply appended
497        assert_eq!(sorter.child_to_parent.len(), 2);
498        assert_eq!(sorter.child_to_parent[&e2], e1);
499        assert_eq!(sorter.child_to_parent[&e3], e1);
500
501        sorter.sort();
502        assert_eq!(sorter.effects.len(), 3);
503        assert_eq!(sorter.effects[0].entity, e2); // child first
504        assert_eq!(sorter.effects[1].entity, e3); // other child next (in same buffer as parent)
505        assert_eq!(sorter.effects[2].entity, e1); // finally, parent
506        assert_eq!(sorter.child_to_parent.len(), 2);
507        assert_eq!(sorter.child_to_parent[&e2], e1);
508        assert_eq!(sorter.child_to_parent[&e3], e1);
509    }
