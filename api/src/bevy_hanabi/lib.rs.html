18//! 🎆 Hanabi -- a GPU particle system plugin for the Bevy game engine.
19//!
20//! The 🎆 Hanabi particle system is a GPU-based particle system integrated with
21//! the built-in Bevy renderer. It allows creating complex visual effects with
22//! millions of particles simulated in real time, by leveraging the power of
23//! compute shaders and offloading most of the work of particle simulating to
24//! the GPU.
25//!
26//! The library supports the `wasm` target (WebAssembly) via the WebGPU renderer
27//! only. Compute shaders are not available via the legacy WebGL2 renderer.
28//! See Bevy's own [WebGL2 and WebGPU](https://github.com/bevyengine/bevy/tree/latest/examples#webgl2-and-webgpu)
29//! section of the examples README for more information on how to run Wasm
30//! builds with WebGPU.
31//!
32//! # 2D vs. 3D
33//!
34//! 🎆 Hanabi integrates both with the 2D and the 3D core pipelines of Bevy. The
35//! 2D pipeline integration is controlled by the `2d` cargo feature, while the
36//! 3D pipeline integration is controlled by the `3d` cargo feature. Both
37//! features are enabled by default for convenience. As an optimization, users
38//! can disable default features and re-enable only one of the two modes. At
39//! least one of the `2d` or `3d` features must be enabled.
40//!
41//! ```toml
42//! # Example: enable only 3D integration
43//! bevy_hanabi = { version = "0.19", default-features = false, features = ["3d"] }
44//! ```
45//!
46//! # Example
47//!
48//! Add the 🎆 Hanabi plugin to your app:
49//!
50//! ```no_run
51//! # use bevy::prelude::*;
52//! use bevy_hanabi::prelude::*;
53//!
54//! App::default()
55//!     .add_plugins(DefaultPlugins)
56//!     .add_plugins(HanabiPlugin)
57//!     .run();
58//! ```
59//!
60//! Create an [`EffectAsset`] describing a visual effect:
61//!
62//! ```
63//! # use bevy::prelude::*;
64//! # use bevy_hanabi::prelude::*;
65//! fn setup(mut effects: ResMut<Assets<EffectAsset>>) {
66//!     // Define a color gradient from red to transparent black
67//!     let mut gradient = bevy_hanabi::Gradient::new();
68//!     gradient.add_key(0.0, Vec4::new(1., 0., 0., 1.));
69//!     gradient.add_key(1.0, Vec4::ZERO);
70//!
71//!     // Create a new expression module
72//!     let mut module = Module::default();
73//!
74//!     // On spawn, randomly initialize the position of the particle
75//!     // to be over the surface of a sphere of radius 2 units.
76//!     let init_pos = SetPositionSphereModifier {
77//!         center: module.lit(Vec3::ZERO),
78//!         radius: module.lit(2.),
79//!         dimension: ShapeDimension::Surface,
80//!     };
81//!
82//!     // Also initialize a radial initial velocity to 6 units/sec
83//!     // away from the (same) sphere center.
84//!     let init_vel = SetVelocitySphereModifier {
85//!         center: module.lit(Vec3::ZERO),
86//!         speed: module.lit(6.),
87//!     };
88//!
89//!     // Initialize the total lifetime of the particle, that is
90//!     // the time for which it's simulated and rendered. This modifier
91//!     // is almost always required, otherwise the particles will stay
92//!     // alive forever, and new particles can't be spawned instead.
93//!     let lifetime = module.lit(10.); // literal value "10.0"
94//!     let init_lifetime = SetAttributeModifier::new(Attribute::LIFETIME, lifetime);
95//!
96//!     // Every frame, add a gravity-like acceleration downward
97//!     let accel = module.lit(Vec3::new(0., -3., 0.));
98//!     let update_accel = AccelModifier::new(accel);
99//!
100//!     // Create the effect asset
101//!     let effect = EffectAsset::new(
102//!         // Maximum number of particles alive at a time
103//!         1024,
104//!         // Spawn at a rate of 5 particles per second
105//!         SpawnerSettings::rate(5.0.into()),
106//!         // Move the expression module into the asset
107//!         module,
108//!     )
109//!     .with_name("MyEffect")
110//!     .init(init_pos)
111//!     .init(init_vel)
112//!     .init(init_lifetime)
113//!     .update(update_accel)
114//!     // Render the particles with a color gradient over their
115//!     // lifetime. This maps the gradient key 0 to the particle spawn
116//!     // time, and the gradient key 1 to the particle death (10s).
117//!     .render(ColorOverLifetimeModifier {
118//!         gradient,
119//!         blend: ColorBlendMode::Overwrite,
120//!         mask: ColorBlendMask::RGBA,
121//!     });
122//!
123//!     // Insert into the asset system
124//!     let effect_asset = effects.add(effect);
125//! }
126//! ```
127//!
128//! Then add an instance of that effect to an entity by spawning a
129//! [`ParticleEffect`] component referencing the asset.
130//!
131//! ```
132//! # use bevy::prelude::*;
133//! # use bevy_hanabi::prelude::*;
134//! # fn spawn_effect(mut commands: Commands) {
135//! #   let effect_asset = Handle::<EffectAsset>::default();
136//! commands.spawn((
137//!     ParticleEffect::new(effect_asset),
138//!     Transform::from_translation(Vec3::Y),
139//! ));
140//! # }
141//! ```
142//!
143//! # Workflow
144//!
145//! Authoring and using a particle effect follows this workflow:
146//!
147//! 1. Create an [`EffectAsset`] representing the definition of the particle
148//!    effect. This asset is a proper Bevy [`Asset`], expected to be authored in
149//!    advance, serialized, and shipped with your application. Creating an
150//!    [`EffectAsset`] at runtime while the application is running is also
151//!    supported, though. In any case however, the asset doesn't do anything by
152//!    itself.
153//! 2. At runtime, when the application is running, create an actual particle
154//!    effect instance by spawning a [`ParticleEffect`] component. The component
155//!    references the [`EffectAsset`] via its `handle` field. Multiple instances
156//!    can reference the same asset at the same time, and some changes to the
157//!    asset are reflected to its instances, although not all changes are
158//!    supported. In general, avoid changing an [`EffectAsset`] while it's in
159//!    use by one or more [`ParticleEffect`].
160//! 3. If using properties, spawn an [`EffectProperties`] component on the same
161//!    entity. Then update properties through that component at any time while
162//!    the effect is active. This allows some moderate CPU-side control over the
163//!    simulation and rendering of the effect, without having to destroy the
164//!    effect and re-create a new one.
165//! 4. If using textures, spawn an [`EffectMaterial`] component to define which
166//!    texture is bound to which slot in the effect. An [`EffectAsset`] only
167//!    defines "slots" of textures, not the actual assets bound to those slots.
168//!    This way, you can reuse the same effect asset multiple times with
169//!    different textures, like you'd do with a regular rendering mesh.
170//! 5. For advanced VFX composed of multiple hierarchical effects, where two or
171//!    more effects are connected to each other in a parent-child relationship,
172//!    spawn an [`EffectParent`] on any child effect instance to specify its
173//!    parent instance. See also the [`EmitSpawnEventModifier`].
174//!
175//! The [`EffectAsset`] is intended to be the serialized effect format, which
176//! authors can save to disk and ship with their application. At this time
177//! however serialization and deserialization is still a work in progress. In
178//! particular, serialization and deserialization of all
179//! [modifiers](crate::modifier) is currently not supported on `wasm` target.

233/// Extension trait to convert an object to WGSL code.
234///
235/// This is mainly used for floating-point constants. This is required because
236/// WGSL doesn't support a floating point constant without a decimal separator
237/// (_e.g._ `0.` instead of `0`), which would be what a regular string
238/// formatting function like [`format!()`] would produce, but which is
239/// interpreted as an integral type by WGSL instead.
240///
241/// # Example
242///
243/// ```
244/// # use bevy_hanabi::ToWgslString;
245/// let x = 2.0_f32;
246/// assert_eq!("let x = 2.;", format!("let x = {};", x.to_wgsl_string()));
247/// ```
248///
249/// # Remark
250///
251/// This trait is soft-deprecated. It serves the same purpose as
252/// [`EvalContext::eval()`], however it lacks any context for the evaluation of
253/// an expression producing WGSL code, so its use is limited. It's still useful
254/// for constant (literal) expressions, which do not require any context to
255/// evaluate.
256///
257/// [`format!()`]: std::format
258/// [`EvalContext::eval()`]: crate::graph::expr::EvalContext::eval
259pub trait ToWgslString {
260    /// Convert an object to a string representing its WGSL code.
261    fn to_wgsl_string(&self) -> String;
262}

484/// Simulation space for the particles of an effect.
485#[derive(Debug, Default, Clone, Copy, PartialEq, Eq, Reflect, Serialize, Deserialize)]
486#[non_exhaustive]
487pub enum SimulationSpace {
488    /// Particles are simulated in global space.
489    ///
490    /// The global space is the Bevy world space. Particles simulated in global
491    /// space are "detached" from the emitter when they spawn, and not
492    /// influenced anymore by the emitter's [`GlobalTransform`] after spawning.
493    /// The particle's [`Attribute::POSITION`] is the world space position
494    /// of the particle.
495    ///
496    /// This is the default.
497    #[default]
498    Global,
499
500    /// Particles are simulated in local effect space.
501    ///
502    /// The local space is the space associated with the [`GlobalTransform`] of
503    /// the [`ParticleEffect`] component being simulated. Particles
504    /// simulated in local effect space are "attached" to the effect, and
505    /// will be affected by its [`GlobalTransform`]. The particle's
506    /// [`Attribute::POSITION`] is the position of the particle relative to
507    /// the effect's [`GlobalTransform`].
508    Local,
509}
510
511impl SimulationSpace {
512    /// Evaluate the simulation space expression.
513    ///
514    /// - In the init and udpate contexts, this expression transforms the
515    ///   particle's position from simulation space to storage space.
516    /// - In the render context, this expression transforms the particle's
517    ///   position from simulation space to view space.
518    pub fn eval(&self, context: &dyn EvalContext) -> Result<String, ExprError> {
519        match context.modifier_context() {
520            ModifierContext::Init | ModifierContext::Update => match *self {
521                SimulationSpace::Global => {
522                    if !context.particle_layout().contains(Attribute::POSITION) {
523                        return Err(ExprError::GraphEvalError(format!("Global-space simulation requires that the particles have a {} attribute.", Attribute::POSITION.name())));
524                    }
525                    Ok(format!(
526                        "particle.{} += transform[3].xyz;", // TODO: get_view_position()
527                        Attribute::POSITION.name()
528                    ))
529                }
530                SimulationSpace::Local => Ok("".to_string()),
531            },
532            ModifierContext::Render => Ok(match *self {
533                // TODO: cast vec3 -> vec4 auomatically
534                SimulationSpace::Global => "vec4<f32>(local_position, 1.0)",
535                // TODO: transform_world_to_view(...)
536                SimulationSpace::Local => "transform * vec4<f32>(local_position, 1.0)",
537            }
538            .to_string()),
539            _ => Err(ExprError::GraphEvalError(
540                "Invalid modifier context value.".to_string(),
541            )),
542        }
543    }
544}

584/// Particle-based visual effect instance.
585///
586/// The particle effect component represents a single instance of a visual
587/// effect. The visual effect itself is described by a handle to an
588/// [`EffectAsset`].
589///
590/// This instance is associated to an [`Entity`], inheriting
591/// its [`GlobalTransform`] as the origin frame for its particle spawning.
592///
593/// # Content
594///
595/// The values in this component, with the exception of the handle to the
596/// underlying asset itself ([`ParticleEffect::handle`]), are optional
597/// per-instance overrides taking precedence over the fallback value set in the
598/// [`EffectAsset`].
599///
600/// Note that the effect graph itself including all its modifiers cannot be
601/// overridden per instance. For minor variations use different assets. If you
602/// need too many variants try to use a property instead.
603///
604/// # Component dependencies
605///
606/// ## Mandatory components
607///
608/// The [`ParticleEffect`] component requires, in the sense of ECS, some other
609/// components for the particle effect to work. Those mandatory components are
610/// automatically added by Bevy if not otherwise provided during spawning.
611///
612/// - A [`CompiledParticleEffect`] component, which contains some precomputed
613///   data and allocated resources, in particular GPU resources.
614/// - A [`Visibility`] component (and the components it requires in turn) to
615///   make use of the Bevy visibility system and optimize rendering of the
616///   effects. This influences simulation when using
617///   [`SimulationCondition::WhenVisible`].
618/// - A [`Transform`] component to define the position of the particle emitter.
619///
620/// ## Optional components
621///
622/// - The [`EffectMaterial`] defines the "material" of the particle effect,
623///   which contains for example the textures used by the particle system. This
624///   component is optional, because not all effects make use of a material.
625/// - The [`EffectParent`] defines the parent effect of this effect. This is
626///   used for hierarchical effect construction, and to use GPU spawn events to
627///   make the parent effect trigger spawning particles in this effect.
628/// - The [`EffectProperties`] defines the runtime values of the properties of
629///   the effect. If the effect doesn't use properties, this component is not
630///   used.
631///
632/// # Change detection
633///
634/// The [`CompiledParticleEffect`] component located on the same [`Entity`] as
635/// the current [`ParticleEffect`] component is automatically updated when a
636/// change occurs to this component. This separation is designed to leverage the
637/// change detection system of Bevy, in order to ensure that any manual change
638/// by the user is performed on this component and any automated change is
639/// performed on the [`CompiledParticleEffect`] component. This allows
640/// efficiently detecting when a user change requires an update to the compiled
641/// particle effect, while ensuring conversely that internal mutations do not
642/// invalidate the compiled effect and do not trigger a costly shader rebuild
643/// for example.
644#[derive(Debug, Default, Clone, Component, Reflect)]
645#[reflect(Component)]
646#[require(
647    CompiledParticleEffect,
648    Transform,
649    Visibility,
650    VisibilityClass,
651    SyncToRenderWorld
652)]
653#[component(on_add = bevy::camera::visibility::add_visibility_class::<EffectVisibilityClass>)]
654pub struct ParticleEffect {
655    /// Handle of the effect to instantiate.
656    pub handle: Handle<EffectAsset>,
657    /// Optional per-instance PRNG seed.
658    ///
659    /// Set this value to `Some(seed)` to override the default value set in
660    /// [`EffectAsset::prng_seed`] for this effect instance. By default this is
661    /// `None`, and the instance uses the same PRNG seed as its [`EffectAsset`].
662    pub prng_seed: Option<u32>,
663}
664
665impl ParticleEffect {
666    /// Create a new particle effect instance from an existing asset.
667    pub fn new(handle: Handle<EffectAsset>) -> Self {
668        Self {
669            handle,
670            prng_seed: None,
671        }
672    }
673}

683/// Material for an effect instance.
684///
685/// A material component contains the render resources (textures) for a single
686/// effect instance. Those textures are automatically bound during rendering to
687/// the slots defined with [`Module::add_texture_slot()`]. Using this, multiple
688/// effect instances sharing a same source [`EffectAsset`] can be instantiated
689/// and rendered with different sets of textures, without changing the asset.
690///
691/// The [`EffectMaterial`] component needs to be spawned on the same entity as
692/// the [`ParticleEffect`] component representing the effect instance.
693#[derive(Debug, Default, Clone, Component)]
694pub struct EffectMaterial {
695    /// List of texture images to use to render the effect instance.
696    ///
697    /// The images are ordered by [slot index] into the corresponding
698    /// [`TextureLayout`].
699    ///
700    /// [slot index]: crate::TextureLayout::get_slot_by_name
701    pub images: Vec<Handle<Image>>,
702}
703
704/// Texture slot of a [`Module`].
705///
706/// A texture slot defines a named bind point where a texture can be attached
707/// and sampled by an effect during rendering. A slot also has an implicit
708/// unique index corresponding to its position in the [`TextureLayout::layout`]
709/// array of the effect.
710#[derive(Debug, Clone, PartialEq, Eq, Hash, Reflect, Serialize, Deserialize)]
711pub struct TextureSlot {
712    /// Unique slot name.
713    pub name: String,
714}
715
716/// Texture layout.
717///
718/// Defines the list of texture slots for an effect.
719#[derive(Debug, Default, Clone, PartialEq, Eq, Hash, Reflect, Serialize, Deserialize)]
720pub struct TextureLayout {
721    /// The list of slots.
722    ///
723    /// The slots index corresponds to the index inside this array. Each image
724    /// in [`EffectMaterial::images`] maps by index to a unique slot in this
725    /// array.
726    pub layout: Vec<TextureSlot>,
727}
728
729impl TextureLayout {
730    /// Find a texture slot by name, and return its index.
731    ///
732    /// The index uniquely identify the slot (like its name), and indexes into
733    /// the [`EffectMaterial::images`] array to determine which texture is bound
734    /// to it.
735    pub fn get_slot_by_name(&self, name: &str) -> Option<usize> {
736        self.layout.iter().position(|slot| slot.name == name)
737    }
738}

747/// Effect shaders.

763/// Source code (WGSL) of an effect.
764///
765/// The source code is generated from an [`EffectAsset`] by applying all
766/// modifiers. The resulting source code is _configured_ (the Hanabi variables
767/// `{{VARIABLE}}` are replaced with the relevant WGSL code) but is not
768/// _specialized_ (the conditional directives like `#if` are still present).
769///
770/// This is mainly used internally by Hanabi as an intermediate step toward
771/// generating the final [`EffectShaders`], and is exposed mainly for debugging
772/// and inspection (editor). In general, you don't need to use this type
773/// directly.
774#[derive(Debug, Default, Clone, PartialEq, Eq, PartialOrd, Ord, Hash)]
775pub struct EffectShaderSources {
776    /// WGSL source code for the init compute shader.
777    pub init_shader_source: String,
778    /// WGSL source code for the update compute shader.
779    pub update_shader_source: String,
780    /// WGSL source code for the render graphics shader (both vertex and
781    /// fragment).
782    pub render_shader_source: String,
783    /// Effect flags the source codes were generated with.
784    pub layout_flags: LayoutFlags,
785}
786
787/// Error resulting from the generating of the WGSL shader code of an
788/// [`EffectAsset`].
789#[derive(Debug, Error)]
790pub enum ShaderGenerateError {
791    /// Error related to an [`Expr`].
792    #[error("Expression error: {0}")]
793    Expr(ExprError),
794
795    /// Shader validation error.
796    #[error("Validation error: {0}")]
797    Validate(String),
798}
799
800impl EffectShaderSources {
801    /// Generate the effect shader WGSL source code.
802    ///
803    /// This takes a base asset effect and generate the WGSL code for the
804    /// various shaders (init/update/render).
805    pub fn generate(
806        asset: &EffectAsset,
807        // Note: ideally the below fields are folded into the asset, but currently the parent
808        // relationship and GPU event one are not encoded in assets.
809        parent_layout: Option<&ParticleLayout>,
810        num_event_bindings: u32,
811    ) -> Result<EffectShaderSources, ShaderGenerateError> {
812        trace!(
813            "Generating shader sources for asset '{}' with {} event bindings",
814            asset.name,
815            num_event_bindings
816        );
817
818        let particle_layout = asset.particle_layout();
819
820        // The particle layout cannot be empty currently because we always emit some
821        // Particle{} struct and it needs at least one field. There's probably no use
822        // case for an empty layout anyway.
823        if particle_layout.size() == 0 {
824            return Err(ShaderGenerateError::Validate(format!(
825                "Asset {} has invalid empty particle layout.",
826                asset.name
827            )));
828        }
829        if let Some(parent_layout) = parent_layout {
830            if parent_layout.size() == 0 {
831                return Err(ShaderGenerateError::Validate(format!(
832                    "Effect using asset {} has invalid empty parent particle layout.",
833                    asset.name
834                )));
835            }
836        }
837
838        // Currently the POSITION attribute is mandatory, as it's always used by the
839        // render shader.
840        if !particle_layout.contains(Attribute::POSITION) {
841            return Err(ShaderGenerateError::Validate(format!(
842                "The particle layout of asset '{}' is missing the '{}' attribute. Add a modifier using that attribute, for example the SetAttributeModifier.",
843                asset.name, Attribute::POSITION.name().to_ascii_uppercase()
844            )));
845        }
846
847        // Currently ribbon rendering requires AGE, so warn if it's missing because
848        // everything will break with some weird error aboud bind groups or whatnot.
849        if particle_layout.contains(Attribute::RIBBON_ID)
850            && !particle_layout.contains(Attribute::AGE)
851        {
852            return Err(ShaderGenerateError::Validate(format!(
853                "The particle layout of asset '{}' uses ribbons (has the '{}' attribute), but is missing the '{}' attribute, which is mandatory for ribbons. Add a modifier using that attribute, for example the SetAttributeModifier.",
854                asset.name, Attribute::RIBBON_ID.name().to_ascii_uppercase(), Attribute::AGE.name().to_ascii_uppercase()
855            )));
856        }
857
858        // Generate the WGSL code declaring all the attributes inside the Particle
859        // struct.
860        let attributes_code = particle_layout.generate_code();
861        let parent_attributes_code = parent_layout
862            .map(|layout| layout.generate_code())
863            .unwrap_or_default();
864
865        // For the renderer, assign all its inputs to the values of the attributes
866        // present, or a default value.
867        let mut inputs_code = String::new();
868        // All required attributes, except the size/color which are variadic
869        let required_attributes =
870            HashSet::from_iter([Attribute::AXIS_X, Attribute::AXIS_Y, Attribute::AXIS_Z]);
871        let mut present_attributes = HashSet::new();
872        let mut has_size = false;
873        let mut has_color = false;
874        for attr_layout in particle_layout.attributes() {
875            let attr = attr_layout.attribute;
876            if attr == Attribute::SIZE {
877                if !has_size {
878                    inputs_code += &format!(
879                        "var size = vec3<f32>(particle.{0}, particle.{0}, particle.{0});\n",
880                        Attribute::SIZE.name()
881                    );
882                    has_size = true;
883                    present_attributes.insert(attr);
884                } else {
885                    warn!("Attribute SIZE conflicts with another size attribute; ignored.");
886                }
887            } else if attr == Attribute::SIZE2 {
888                if !has_size {
889                    inputs_code += &format!(
890                        "var size = vec3<f32>(particle.{0}, 1.0);\n",
891                        Attribute::SIZE2.name()
892                    );
893                    has_size = true;
894                    present_attributes.insert(attr);
895                } else {
896                    warn!("Attribute SIZE2 conflicts with another size attribute; ignored.");
897                }
898            } else if attr == Attribute::SIZE3 {
899                if !has_size {
900                    inputs_code += &format!("var size = particle.{0};\n", Attribute::SIZE3.name());
901                    has_size = true;
902                    present_attributes.insert(attr);
903                } else {
904                    warn!("Attribute SIZE3 conflicts with another size attribute; ignored.");
905                }
906            } else if attr == Attribute::HDR_COLOR {
907                if !has_color {
908                    inputs_code +=
909                        &format!("var color = particle.{};\n", Attribute::HDR_COLOR.name());
910                    has_color = true;
911                    present_attributes.insert(attr);
912                } else {
913                    warn!("Attribute HDR_COLOR conflicts with another color attribute; ignored.");
914                }
915            } else if attr == Attribute::COLOR {
916                if !has_color {
917                    inputs_code += &format!(
918                        "var color = unpack4x8unorm(particle.{0});\n",
919                        Attribute::COLOR.name()
920                    );
921                    has_color = true;
922                    present_attributes.insert(attr);
923                } else {
924                    warn!("Attribute COLOR conflicts with another color attribute; ignored.");
925                }
926            } else {
927                inputs_code += &format!("var {0} = particle.{0};\n", attr.name());
928                present_attributes.insert(attr);
929            }
930        }
931        // For all attributes required by the render shader, if they're not explicitly
932        // stored in the particle layout, define a variable with their default value.
933        if !has_size {
934            inputs_code += &format!(
935                "var size = {0};\n",
936                Attribute::SIZE3.default_value().to_wgsl_string() // TODO - or SIZE?
937            );
938        }
939        if !has_color {
940            inputs_code += &format!(
941                "var color = {};\n",
942                Attribute::HDR_COLOR.default_value().to_wgsl_string() // TODO - or COLOR?
943            );
944        }
945        for &attr in required_attributes.difference(&present_attributes) {
946            inputs_code += &format!(
947                "var {} = {};\n",
948                attr.name(),
949                attr.default_value().to_wgsl_string()
950            );
951        }
952
953        // Generate the shader code defining the per-effect properties, if any
954        let property_layout = asset.property_layout();
955        let properties_code = property_layout
956            .generate_property_struct_code()
957            .unwrap_or_default();
958        let properties_binding_code = if property_layout.is_empty() {
959            "// (no properties)".to_string()
960        } else {
961            "@group(2) @binding(3) var<storage, read> properties : array<Properties>;".to_string()
962        };
963
964        // Event buffer bindings for the update pass, if the effect emits GPU events to
965        // one or more other effects.
966        let mut emit_event_buffer_bindings_code = String::with_capacity(256);
967        emit_event_buffer_bindings_code.push_str(
968            "@group(3) @binding(1) var<storage, read_write> child_info_buffer : ChildInfoBuffer;\n",
969        );
970        let mut emit_event_buffer_append_funcs_code = String::with_capacity(1024);
971        let base_binding_index = 2;
972        for i in 0..num_event_bindings {
973            let binding_index = base_binding_index + i;
974            emit_event_buffer_bindings_code.push_str(&format!(
975                "@group(3) @binding({binding_index}) var<storage, read_write> event_buffer_{i} : EventBuffer;\n"));
976            emit_event_buffer_append_funcs_code.push_str(&format!(
977                r##"/// Append one or more spawn events to the event buffer.
978fn append_spawn_events_{0}(base_child_index: u32, particle_index: u32, count: u32) {{
979    // Optimize this case.
980    if (count == 0u) {{
981        return;
982    }}
983
984    let capacity = arrayLength(&event_buffer_{0}.spawn_events);
985    let base = min(u32(atomicAdd(&child_info_buffer.rows[base_child_index + {0}].event_count, i32(count))), capacity);
986    let capped_count = min(count, capacity - base);
987    for (var i = 0u; i < capped_count; i += 1u) {{
988        event_buffer_{0}.spawn_events[base + i].particle_index = particle_index;
989    }}
990}}
991"##,
992                i,
993            ));
994        }
995        emit_event_buffer_bindings_code.pop();
996        if emit_event_buffer_bindings_code.is_empty() {
997            emit_event_buffer_bindings_code = "// (not emitting GPU events)".into();
998        }
999        emit_event_buffer_append_funcs_code.pop();
1000        if emit_event_buffer_append_funcs_code.is_empty() {
1001            emit_event_buffer_append_funcs_code = "// (not emitting GPU events)".into();
1002        }
1003
1004        // Start from the base module containing the expressions actually serialized in
1005        // the asset. We will add the ones created on-the-fly by applying the
1006        // modifiers to the contexts.
1007        let mut module = asset.module().clone();
1008
1009        let mut layout_flags = LayoutFlags::NONE;
1010        if asset.simulation_space == SimulationSpace::Local {
1011            layout_flags |= LayoutFlags::LOCAL_SPACE_SIMULATION;
1012        }
1013        match &asset.alpha_mode {
1014            AlphaMode::Mask(_) => layout_flags.insert(LayoutFlags::USE_ALPHA_MASK),
1015            AlphaMode::Opaque => layout_flags.insert(LayoutFlags::OPAQUE),
1016            _ => layout_flags.remove(LayoutFlags::USE_ALPHA_MASK | LayoutFlags::OPAQUE),
1017        }
1018        if particle_layout.contains(Attribute::RIBBON_ID) {
1019            layout_flags |= LayoutFlags::RIBBONS;
1020        }
1021        if parent_layout.is_some() {
1022            layout_flags |= LayoutFlags::READ_PARENT_PARTICLE;
1023        }
1024
1025        // Generate the shader code for the initializing shader
1026        let (init_code, init_extra, init_sim_space_transform_code, consume_gpu_spawn_events) = {
1027            // Apply all the init modifiers
1028            let mut init_context =
1029                ShaderWriter::new(ModifierContext::Init, &property_layout, &particle_layout);
1030            for m in asset.init_modifiers() {
1031                if let Err(err) = m.apply(&mut module, &mut init_context) {
1032                    error!(
1033                        "Failed to compile effect '{}', error in init context: {}",
1034                        asset.name, err
1035                    );
1036                    return Err(ShaderGenerateError::Expr(err));
1037                }
1038            }
1039
1040            let sim_space_transform_code =
1041                asset.simulation_space.eval(&init_context).map_err(|err| {
1042                    error!("Failed to compile effect's simulation space: {}", err);
1043                    ShaderGenerateError::Expr(err)
1044                })?;
1045
1046            // Effect uses GPU spawn events if it produces them, or if it has a parent and
1047            // consumes them from that parent.
1048            let consume_gpu_spawn_events =
1049                init_context.emits_gpu_spawn_events().unwrap_or(false) || parent_layout.is_some();
1050
1051            (
1052                init_context.main_code,
1053                init_context.extra_code,
1054                sim_space_transform_code,
1055                consume_gpu_spawn_events,
1056            )
1057        };
1058
1059        let init_shader_source = PARTICLES_INIT_SHADER_TEMPLATE
1060            .replace("{{ATTRIBUTES}}", &attributes_code)
1061            .replace("{{PARENT_ATTRIBUTES}}", &parent_attributes_code)
1062            .replace("{{INIT_CODE}}", &init_code)
1063            .replace("{{INIT_EXTRA}}", &init_extra)
1064            .replace("{{PROPERTIES}}", &properties_code)
1065            .replace("{{PROPERTIES_BINDING}}", &properties_binding_code)
1066            .replace(
1067                "{{SIMULATION_SPACE_TRANSFORM_PARTICLE}}",
1068                &init_sim_space_transform_code,
1069            );
1070        trace!(
1071            "Configured init shader for '{}':\n{}",
1072            asset.name,
1073            init_shader_source
1074        );
1075
1076        // Generate the shader code for the update shader
1077        let (mut update_code, update_extra, emit_gpu_spawn_events) = {
1078            let mut update_context =
1079                ShaderWriter::new(ModifierContext::Update, &property_layout, &particle_layout);
1080            for m in asset.update_modifiers() {
1081                if let Err(err) = m.apply(&mut module, &mut update_context) {
1082                    error!(
1083                        "Failed to compile effect '{}', error in update context: {}",
1084                        asset.name, err
1085                    );
1086                    return Err(ShaderGenerateError::Expr(err));
1087                }
1088            }
1089
1090            let emit_gpu_spawn_events = update_context.emits_gpu_spawn_events().unwrap_or(false);
1091
1092            (
1093                update_context.main_code,
1094                update_context.extra_code,
1095                emit_gpu_spawn_events,
1096            )
1097        };
1098
1099        if consume_gpu_spawn_events {
1100            layout_flags |= LayoutFlags::CONSUME_GPU_SPAWN_EVENTS;
1101        }
1102        if emit_gpu_spawn_events {
1103            layout_flags |= LayoutFlags::EMIT_GPU_SPAWN_EVENTS;
1104        }
1105
1106        // Insert Euler motion integration if needed.
1107        let has_position = present_attributes.contains(&Attribute::POSITION);
1108        let has_velocity = present_attributes.contains(&Attribute::VELOCITY);
1109        if asset.motion_integration != MotionIntegration::None {
1110            if has_position && has_velocity {
1111                // Note the prepended "\n" to prevent appending to a comment line.
1112                let code = format!(
1113                    "\nparticle.{0} += particle.{1} * sim_params.delta_time;\n",
1114                    Attribute::POSITION.name(),
1115                    Attribute::VELOCITY.name()
1116                );
1117                if asset.motion_integration == MotionIntegration::PreUpdate {
1118                    update_code.insert_str(0, &code);
1119                } else {
1120                    update_code += &code;
1121                }
1122            } else {
1123                warn!(
1124                        "Asset '{}' specifies motion integration but is missing {}. Particles won't move unless the POSITION attribute is explicitly assigned. Set MotionIntegration::None to remove this warning.",
1125                        asset.name,
1126                        if has_position {
1127                            "Attribute::VELOCITY"
1128                        } else {
1129                            "Attribute::POSITION"
1130                        }
1131                    )
1132            }
1133        }
1134
1135        // Generate the shader code for the render shader
1136        let (
1137            vertex_code,
1138            fragment_code,
1139            render_extra,
1140            alpha_cutoff_code,
1141            flipbook_scale_code,
1142            flipbook_row_count_code,
1143            material_bindings_code,
1144        ) = {
1145            let texture_layout = module.texture_layout();
1146            let mut render_context =
1147                RenderContext::new(&property_layout, &particle_layout, &texture_layout);
1148            for m in asset.render_modifiers() {
1149                m.apply_render(&mut module, &mut render_context)
1150                    .map_err(ShaderGenerateError::Expr)?;
1151            }
1152
1153            if render_context.needs_uv {
1154                layout_flags |= LayoutFlags::NEEDS_UV;
1155            }
1156            if render_context.needs_normal {
1157                layout_flags |= LayoutFlags::NEEDS_NORMAL;
1158            }
1159            if render_context.needs_particle_fragment {
1160                layout_flags |= LayoutFlags::NEEDS_PARTICLE_FRAGMENT;
1161            }
1162
1163            let alpha_cutoff_code = if let AlphaMode::Mask(cutoff) = &asset.alpha_mode {
1164                render_context.eval(&module, *cutoff).unwrap_or_else(|err| {
1165                    error!(
1166                        "Failed to evaluate the expression for AlphaMode::Mask, error: {}",
1167                        err
1168                    );
1169
1170                    // In Debug, show everything to help diagnosing
1171                    #[cfg(debug_assertions)]
1172                    return 1_f32.to_wgsl_string();
1173
1174                    // In Release, hide everything with an error
1175                    #[cfg(not(debug_assertions))]
1176                    return 0_f32.to_wgsl_string();
1177                })
1178            } else {
1179                String::new()
1180            };
1181
1182            let (flipbook_scale_code, flipbook_row_count_code) = if let Some(grid_size) =
1183                render_context.sprite_grid_size
1184            {
1185                layout_flags |= LayoutFlags::FLIPBOOK;
1186                // Note: row_count needs to be i32, not u32, because of sprite_index
1187                let flipbook_row_count_code = (grid_size.x as i32).to_wgsl_string();
1188                let flipbook_scale_code =
1189                    Vec2::new(1.0 / grid_size.x as f32, 1.0 / grid_size.y as f32).to_wgsl_string();
1190                (flipbook_scale_code, flipbook_row_count_code)
1191            } else {
1192                (String::new(), String::new())
1193            };
1194
1195            trace!(
1196                "Generating material bindings code for layout: {:?}",
1197                texture_layout
1198            );
1199            let mut material_bindings_code = String::new();
1200            let mut bind_index = 0;
1201            for (slot, _) in texture_layout.layout.iter().enumerate() {
1202                let tex_index = bind_index;
1203                let sampler_index = bind_index + 1;
1204                material_bindings_code.push_str(&format!(
1205                    "@group(3) @binding({tex_index}) var material_texture_{slot}: texture_2d<f32>;
1206@group(3) @binding({sampler_index}) var material_sampler_{slot}: sampler;
1207"
1208                ));
1209                bind_index += 2;
1210            }
1211
1212            (
1213                render_context.vertex_code,
1214                render_context.fragment_code,
1215                render_context.render_extra,
1216                alpha_cutoff_code,
1217                flipbook_scale_code,
1218                flipbook_row_count_code,
1219                material_bindings_code,
1220            )
1221        };
1222
1223        // Configure aging code
1224        let has_age = present_attributes.contains(&Attribute::AGE);
1225        let has_lifetime = present_attributes.contains(&Attribute::LIFETIME);
1226        let mut age_code = String::new();
1227        if has_age {
1228            if has_lifetime {
1229                age_code += &format!(
1230                    "\n    let was_alive = particle.{0} < particle.{1};",
1231                    Attribute::AGE.name(),
1232                    Attribute::LIFETIME.name()
1233                );
1234            }
1235
1236            age_code += &format!(
1237                "\n    particle.{0} = particle.{0} + sim_params.delta_time;",
1238                Attribute::AGE.name()
1239            );
1240
1241            if has_lifetime {
1242                age_code += &format!(
1243                    "\n    var is_alive = particle.{0} < particle.{1};",
1244                    Attribute::AGE.name(),
1245                    Attribute::LIFETIME.name()
1246                );
1247            }
1248        } else {
1249            // Since we're using a dead index buffer, all particles that make it to the
1250            // update compute shader are guaranteed to be alive (we never
1251            // simulate dead particles).
1252            age_code = "\n    let was_alive = true;\n    var is_alive = true;".to_string();
1253        }
1254
1255        // Configure reaping code
1256        let reap_code = if has_age && has_lifetime {
1257            format!(
1258                "is_alive = is_alive && (particle.{0} < particle.{1});",
1259                Attribute::AGE.name(),
1260                Attribute::LIFETIME.name()
1261            )
1262        } else {
1263            "".to_string()
1264        };
1265
1266        // Assign attributes individually instead of using struct
1267        // assignment. Otherwise we might race on `PREV` and `NEXT`
1268        // attributes, which might be updated behind our back when adjacent
1269        // particles die.
1270        let mut writeback_code = "".to_owned();
1271        for attribute in present_attributes
1272            .iter()
1273            .filter(|attribute| **attribute != Attribute::PREV && **attribute != Attribute::NEXT)
1274        {
1275            writeln!(
1276                &mut writeback_code,
1277                "    particle_buffer.particles[base_particle + particle_index].{0} = particle.{0};",
1278                attribute.name()
1279            )
1280            .unwrap();
1281        }
1282
1283        // Configure the update shader template, and make sure a corresponding shader
1284        // asset exists
1285        let update_shader_source = PARTICLES_UPDATE_SHADER_TEMPLATE
1286            .replace("{{ATTRIBUTES}}", &attributes_code)
1287            .replace("{{PARENT_ATTRIBUTES}}", &parent_attributes_code)
1288            .replace("{{AGE_CODE}}", &age_code)

1419    /// Update the compiled effect from its asset and instance.
1420    pub(crate) fn update(
1421        &mut self,
1422        rebuild: bool,
1423        instance: &ParticleEffect,
1424        material: Option<&EffectMaterial>,
1425        mesh: Option<&EffectMesh>,
1426        asset: &EffectAsset,
1427        parent_entity: Option<Entity>,
1428        child_entities: Vec<Entity>,
1429        parent_layout: Option<ParticleLayout>,
1430        asset_server: &AssetServer,
1431        shaders: &mut ResMut<Assets<Shader>>,
1432        shader_cache: &mut ResMut<ShaderCache>,
1433    ) {
1434        trace!(
1435            "Updating (rebuild:{}) compiled particle effect '{}' ({:?})",
1436            rebuild,
1437            asset.name,
1438            instance.handle,
1439        );
1440
1441        // #289 - Panic in fn extract_effects
1442        // We now keep a strong handle. Since CompiledParticleEffect is kept in sync
1443        // with the source ParticleEffect, this shouldn't produce any strong cyclic
1444        // dependency.
1445        debug_assert!(instance.handle.is_strong());
1446
1447        // Note: if something marked the ParticleEffect as changed (via Mut for example)
1448        // but didn't actually change anything, or at least didn't change the asset,
1449        // then we may end up here with the same asset handle. Don't try to be
1450        // too smart, and rebuild everything anyway, it's easier than trying to
1451        // diff what may or may not have changed.
1452        self.asset = instance.handle.clone();
1453        self.simulation_condition = asset.simulation_condition;
1454        self.prng_seed = instance.prng_seed.unwrap_or(asset.prng_seed);
1455
1456        // Check if the instance changed. If so, rebuild some data from this compiled
1457        // effect based on the new data of the effect instance.
1458        if rebuild {
1459            // Clear the compiled effect if the effect instance changed. We could try to get
1460            // smarter here, only invalidate what changed, but for now just wipe everything
1461            // and rebuild from scratch all three shaders together.
1462            self.effect_shader = None;
1463        }
1464
1465        // If the shaders are already compiled, there's nothing more to do
1466        if self.effect_shader.is_some() {
1467            return;
1468        }
1469
1470        self.parent = parent_entity;
1471        self.children = child_entities;
1472
1473        let num_event_bindings = self.children.len() as u32;
1474        let shader_source = match EffectShaderSources::generate(
1475            asset,
1476            parent_layout.as_ref(),
1477            num_event_bindings,
1478        ) {
1479            Ok(shader_source) => shader_source,
1480            Err(err) => {
1481                error!(
1482                    "Failed to generate shaders for effect asset '{}': {}",
1483                    asset.name, err
1484                );
1485                return;
1486            }
1487        };
1488
1489        self.layout_flags = shader_source.layout_flags;
1490        self.alpha_mode = asset.alpha_mode;
1491        self.parent_particle_layout = parent_layout;
1492        trace!(
1493            "Compiled effect sources: layout_flags={:?} alpha_mode={:?}",
1494            self.layout_flags,
1495            self.alpha_mode
1496        );
1497
1498        // TODO - Replace with Option<EffectShader { handle: Handle<Shader>, hash:
1499        // u64 }> where the hash takes into account the code and extra code
1500        // for each pass (and any other varying item). We don't need to keep
1501        // around the entire shader code, only a hash of it for compare (or, maybe safer
1502        // to avoid hash collisions, an index into a shader cache). The only
1503        // use is to be able to compare 2 instances and see if they can be
1504        // batched together.
1505        self.effect_shader = Some(EffectShaders {
1506            init: shader_cache.get_or_insert(
1507                &asset.name,
1508                "init",
1509                &shader_source.init_shader_source,
1510                shaders,
1511            ),
1512            update: shader_cache.get_or_insert(
1513                &asset.name,
1514                "update",
1515                &shader_source.update_shader_source,
1516                shaders,
1517            ),
1518            render: shader_cache.get_or_insert(
1519                &asset.name,
1520                "render",
1521                &shader_source.render_shader_source,
1522                shaders,
1523            ),
1524        });
1525
1526        trace!(
1527            "CompiledParticleEffect::update(): shaders={:?} texture_count={} layout_flags={:?}",
1528            self.effect_shader.as_ref().unwrap(),
1529            material.map(|mat| mat.images.len()).unwrap_or(0),
1530            self.layout_flags,
1531        );
1532
1533        if let Some(mesh) = mesh {
1534            self.mesh = Some(mesh.0.clone());
1535        } else {
1536            self.mesh = asset
1537                .mesh
1538                .as_ref()
1539                .map(|path: &bevy::asset::AssetPath<'_>| asset_server.load::<Mesh>(path));
1540        }
1541
1542        self.textures = material.map(|mat| &mat.images).cloned().unwrap_or_default();
1543    }
1544
1545    /// Get the effect shader if configured, or `None` otherwise.
1546    ///
1547    /// The returned assets are the shaders actually compiled and used. You
1548    /// should never mutate those shader assets directly; changing them without
1549    /// updating the rest of the effect will cause discrepancies in the render
1550    /// pipeline and most likely panics and crashes. This getter is provided
1551    /// mainly for debugging and inspection (editor).
1552    pub fn get_configured_shaders(&self) -> Option<&EffectShaders> {
1553        self.effect_shader.as_ref()
1554    }

1567impl ShaderCode for Gradient<Vec2> {
1568    fn to_shader_code(&self, input: &str) -> String {
1569        if self.keys().is_empty() {
1570            return String::new();
1571        }
1572        let mut s: String = self
1573            .keys()
1574            .iter()
1575            .enumerate()
1576            .map(|(index, key)| {
1577                format!(
1578                    "let t{0} = {1};\nlet v{0} = {2};",
1579                    index,
1580                    key.ratio().to_wgsl_string(),
1581                    key.value.to_wgsl_string()
1582                )
1583            })
1584            .fold("// Gradient\n".into(), |s, key| s + &key + "\n");
1585        if self.keys().len() == 1 {
1586            s + "return v0;\n"
1587        } else {
1588            s += &format!("if ({input} <= t0) {{ return v0; }}\n");
1589            let mut s = self
1590                .keys()
1591                .iter()
1592                .skip(1)
1593                .enumerate()
1594                .map(|(index, _key)| {
1595                    format!(
1596                        "else if ({input} <= t{1}) {{ return mix(v{0}, v{1}, ({input} - t{0}) / (t{1} - t{0})); }}\n",
1597                        index,
1598                        index + 1
1599                    )
1600                })
1601                .fold(s, |s, key| s + &key);
1602            let _ = writeln!(s, "else {{ return v{}; }}", self.keys().len() - 1);
1603            s
1604        }
1605    }
1606}

1690/// Compile all the [`ParticleEffect`] components into a
1691/// [`CompiledParticleEffect`].
1692///
1693/// This system runs in the [`EffectSystems::CompileEffects`] set of the
1694/// [`PostUpdate`] schedule. It gathers all new instances of [`ParticleEffect`],
1695/// as well as instances which changed, and (re)compile them into an optimized
1696/// form saved into the [`CompiledParticleEffect`] component.
1697///
1698/// Hidden instances are compiled like visible ones, both to allow users to
1699/// compile "in the background" by spawning a hidden effect, and also to prevent
1700/// having mixed state where only some effects are compiled, and effects
1701/// becoming visible later need to be special casing. If you want to avoid
1702/// compiling an effect, don't spawn it.
1703fn compile_effects(
1704    asset_server: Res<AssetServer>,
1705    effects: Res<Assets<EffectAsset>>,
1706    mut shaders: ResMut<Assets<Shader>>,
1707    mut shader_cache: ResMut<ShaderCache>,
1708    mut q_effects: Query<(
1709        Entity,
1710        Ref<ParticleEffect>,
1711        Option<Ref<EffectMaterial>>,
1712        Option<Ref<EffectMesh>>,
1713        Option<Ref<EffectParent>>,
1714        &mut CompiledParticleEffect,
1715    )>,
1716) {
1717    trace!("compile_effects: {} effect(s)", q_effects.iter().len());
1718
1719    // Loop over all existing effects and collect the valid ones. We do a separate
1720    // pass because we can't borrow mutably while doing a double lookup on the
1721    // query. This map is used to lookup valid parents, and filter out effects with
1722    // a declared parent but unresolved parent asset.
1723    let particle_layouts_and_parents: HashMap<Entity, (ParticleLayout, Option<Entity>)> = q_effects
1724        .iter()
1725        .filter_map(|(entity, effect, _, _, parent, _)| {
1726            effects
1727                .get(&effect.handle)
1728                .map(|asset| (entity, (asset.particle_layout(), parent.map(|p| p.entity))))
1729        })
1730        .collect();
1731
1732    // Count children
1733    let mut children: HashMap<Entity, Vec<Entity>> =
1734        HashMap::with_capacity_and_hasher(particle_layouts_and_parents.len(), Default::default());
1735    for (child, (_, parent)) in particle_layouts_and_parents.iter() {
1736        if let Some(parent) = parent.as_ref() {
1737            children.entry(*parent).or_default().push(*child);
1738        }
1739    }
1740
1741    // Loop over all existing effects to update them, including invisible ones
1742    for (
1743        asset,
1744        entity,
1745        effect,
1746        material,
1747        mesh,
1748        parent_entity,
1749        parent_layout,
1750        mut compiled_effect,
1751    ) in q_effects.iter_mut().filter_map(
1752        |(entity, effect, material, mesh, parent, compiled_effect)| {
1753            // Check if asset is available, otherwise silently ignore as we can't check for
1754            // changes, and conceptually it makes no sense to render a particle effect whose
1755            // asset was unloaded.
1756            let asset = effects.get(&effect.handle)?;
1757
1758            // Same for the parent asset, if any.
1759            let (parent_entity, parent_layout) = if let Some(parent) = &parent {
1760                let Some((parent_layout, _)) = particle_layouts_and_parents.get(&parent.entity)
1761                else {
1762                    // There's a parent declared, but not found. Skip the current asset.
1763                    return None;
1764                };
1765                // Declared parent with found parent asset, child asset is valid.
1766                (Some(parent.entity), Some(parent_layout.clone()))
1767            } else {
1768                // No declared parent, asset is valid.
1769                (None, None)
1770            };
1771
1772            Some((
1773                asset,
1774                entity,
1775                effect,
1776                material,
1777                mesh,
1778                parent_entity,
1779                parent_layout,
1780                compiled_effect,
1781            ))
1782        },
1783    ) {
1784        let child_entities = children
1785            .get_mut(&entity)
1786            .map(std::mem::take)
1787            .unwrap_or_default();
1788
1789        // If the ParticleEffect didn't change, and the compiled one is for the correct
1790        // asset, then there's nothing to do.
1791        let material_changed = material.as_ref().is_some_and(|r| r.is_changed());
1792        let need_rebuild =
1793            effect.is_changed() || material_changed || compiled_effect.children != child_entities;
1794        if need_rebuild || (compiled_effect.asset != effect.handle) {
1795            if need_rebuild {
1796                debug!("Invalidating the compiled cache for effect on entity {:?} due to changes in the ParticleEffect component. If you see this message too much, then performance might be affected. Find why the change detection of the ParticleEffect is triggered.", entity);
1797            }
1798
1799            compiled_effect.update(
1800                need_rebuild,
1801                &effect,
1802                material.map(|r| r.into_inner()),
1803                mesh.map(|r| r.into_inner()),
1804                asset,
1805                parent_entity,
1806                child_entities,
1807                parent_layout,
1808                &asset_server,
1809                &mut shaders,
1810                &mut shader_cache,
1811            );
1812        } else {
1813            // Update the PRNG seed. Unfortunately at the minute the "seed" (which
1814            // really is the internal PRNG state rather) is not cached on GPU, and
1815            // is re-uploaded each frame, so if it's not changed every frame then
1816            // there's no randomness anymore, because the uses of the previous frame
1817            // are "forgotten".
1818            let mut rng = rand::rngs::StdRng::seed_from_u64(compiled_effect.prng_seed as u64);
1819            compiled_effect.prng_seed = rng.random();
1820        }
1821    }
1822
1823    // Clear removed effects, to allow them to be released by the asset server
1824    for (_, effect, _, _, parent, mut compiled_effect) in q_effects.iter_mut() {
1825        // If the effect has no asset, clear its compilation
1826        if effects.get(&effect.handle).is_none() {
1827            compiled_effect.clear();
1828        }
1829
1830        // If the effect has a parent, and that parent has no asset, also clear the
1831        // child's compilation.
1832        if let Some(parent) = parent {
1833            if particle_layouts_and_parents.get(&parent.entity).is_none() {
1834                compiled_effect.clear();
1835            }
1836        }
1837    }
1838}
