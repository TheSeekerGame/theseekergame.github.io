59/// Labels for the Hanabi systems.
60#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, SystemSet)]
61pub enum EffectSystems {
62    /// Tick all effect instances to generate particle spawn counts.
63    ///
64    /// This system runs during the [`PostUpdate`] schedule. Any system which
65    /// modifies an effect spawner should run before this set to ensure the
66    /// spawner takes into account the newly set values during its ticking.
67    TickSpawners,
68
69    /// Compile the effect instances, updating the [`CompiledParticleEffect`]
70    /// components.
71    ///
72    /// This system runs during the [`PostUpdate`] schedule. This is largely an
73    /// internal task which can be ignored by most users.
74    ///
75    /// [`CompiledParticleEffect`]: crate::CompiledParticleEffect
76    CompileEffects,
77
78    /// Update the properties of the effect instance based on the declared
79    /// properties in the [`EffectAsset`], updating the associated
80    /// [`EffectProperties`] component.
81    ///
82    /// This system runs during the [`PostUpdate`] schedule, after the assets
83    /// have been updated. Any system which modifies an [`EffectAsset`]'s
84    /// declared properties should run before this set in order for changes to
85    /// be taken into account in the same frame.
86    UpdatePropertiesFromAsset,
87
88    /// Prepare effect assets for the extracted effects.
89    ///
90    /// Part of Bevy's own [`RenderSystems::PrepareAssets`].
91    PrepareEffectAssets,
92
93    /// Queue the GPU commands for the extracted effects.
94    ///
95    /// Part of Bevy's own [`RenderSystems::Queue`].
96    QueueEffects,
97
98    /// Prepare GPU data for the queued effects.
99    ///
100    /// Part of Bevy's own [`RenderSystems::PrepareResources`].
101    PrepareEffectGpuResources,
102
103    /// Prepare the GPU bind groups once all buffers have been (re-)allocated
104    /// and won't change this frame.
105    ///
106    /// Part of Bevy's own [`RenderSystems::PrepareBindGroups`].
107    PrepareBindGroups,
108}

117impl HanabiPlugin {
118    /// Create the `vfx_common.wgsl` shader with proper alignment.
119    ///
120    /// This creates a new [`Shader`] from the `vfx_common.wgsl` template file,
121    /// by applying the given alignment for storage buffers. This produces a
122    /// shader ready for the specific GPU device associated with that
123    /// alignment.
124    pub(crate) fn make_common_shader(min_storage_buffer_offset_alignment: u32) -> Shader {
125        let spawner_padding_code =
126            GpuSpawnerParams::padding_code(min_storage_buffer_offset_alignment);
127        let batch_info_padding_code =
128            GpuBatchInfo::padding_code(min_storage_buffer_offset_alignment);
129        let effect_metadata_padding_code =
130            GpuEffectMetadata::padding_code(min_storage_buffer_offset_alignment);
131        let render_effect_indirect_size =
132            GpuEffectMetadata::aligned_size(min_storage_buffer_offset_alignment);
133        let effect_metadata_stride_code =
134            (render_effect_indirect_size.get() as u32).to_wgsl_string();
135        let common_code = include_str!("render/vfx_common.wgsl")
136            .replace("{{SPAWNER_PADDING}}", &spawner_padding_code)
137            .replace("{{BATCH_INFO_PADDING}}", &batch_info_padding_code)
138            .replace("{{EFFECT_METADATA_PADDING}}", &effect_metadata_padding_code)
139            .replace("{{EFFECT_METADATA_STRIDE}}", &effect_metadata_stride_code);
140        Shader::from_wgsl(
141            common_code,
142            std::path::Path::new(file!())
143                .parent()
144                .unwrap()
145                .join(format!(
146                    "render/vfx_common_{}.wgsl",
147                    min_storage_buffer_offset_alignment
148                ))
149                .to_string_lossy(),
150        )
151    }
152
153    /// Create the `vfx_indirect.wgsl` shader with proper alignment.
154    ///
155    /// This creates a new [`Shader`] from the `vfx_indirect.wgsl` template
156    /// file, by applying the given alignment for storage buffers. This
157    /// produces a shader ready for the specific GPU device associated with
158    /// that alignment.
159    pub(crate) fn make_indirect_shader(
160        min_storage_buffer_offset_alignment: u32,
161        has_events: bool,
162    ) -> Shader {
163        let render_effect_indirect_size =
164            GpuEffectMetadata::aligned_size(min_storage_buffer_offset_alignment);
165        let render_effect_indirect_stride_code =
166            (render_effect_indirect_size.get() as u32).to_wgsl_string();
167        let indirect_code = include_str!("render/vfx_indirect.wgsl").replace(
168            "{{EFFECT_METADATA_STRIDE}}",
169            &render_effect_indirect_stride_code,
170        );
171        Shader::from_wgsl(
172            indirect_code,
173            std::path::Path::new(file!())
174                .parent()
175                .unwrap()
176                .join(format!(
177                    "render/vfx_indirect_{}_{}.wgsl",
178                    min_storage_buffer_offset_alignment,
179                    if has_events { "events" } else { "noevent" },
180                ))
181                .to_string_lossy(),
182        )
183    }
184
185    /// Create the `vfx_prefix_sum.wgsl` shader.
186    ///
187    /// This creates a new [`Shader`] from the `vfx_prefix_sum.wgsl` template
188    /// file.
189    pub(crate) fn make_prefix_sum_shader() -> Shader {
190        let prefix_sum_code = include_str!("render/vfx_prefix_sum.wgsl");
191        Shader::from_wgsl(
192            prefix_sum_code,
193            std::path::Path::new(file!())
194                .parent()
195                .unwrap()
196                .join("render/vfx_prefix_sum.wgsl")
197                .to_string_lossy(),
198        )
199    }
200}

202impl Plugin for HanabiPlugin {
203    fn build(&self, app: &mut App) {
204        // Register modifiers
205        {
206            let type_registry = app.world().resource::<AppTypeRegistry>();
207            register_modifiers(type_registry);
208        }
209
210        // Register asset
211        app.init_asset::<EffectAsset>()
212            .insert_resource(Random(spawn::new_rng()))
213            .add_plugins(HanabiRenderPlugin)
214            .add_plugins(ExtractComponentPlugin::<EffectVisibilityClass>::default())
215            .init_resource::<DefaultMesh>()
216            .init_resource::<ShaderCache>()
217            .init_resource::<DebugSettings>()
218            .init_resource::<Time<EffectSimulation>>()
219            .configure_sets(
220                PostUpdate,
221                (
222                    EffectSystems::TickSpawners
223                        // This checks the visibility to skip work, so needs to run after
224                        // ComputedVisibility was updated.
225                        .after(VisibilitySystems::VisibilityPropagate),
226                    EffectSystems::CompileEffects,
227                ),
228            )
229            .configure_sets(
230                PreUpdate,
231                EffectSystems::UpdatePropertiesFromAsset.after(bevy::asset::AssetTrackingSystems),
232            )
233            .add_systems(
234                First,
235                effect_simulation_time_system
236                    .after(time_system)
237                    .in_set(TimeSystems),
238            )
239            .add_systems(
240                PostUpdate,
241                (
242                    tick_spawners.in_set(EffectSystems::TickSpawners),
243                    compile_effects.in_set(EffectSystems::CompileEffects),
244                    update_properties_from_asset.in_set(EffectSystems::UpdatePropertiesFromAsset),
245                ),
246            );
247
248        app.init_asset_loader::<EffectAssetLoader>();
249
250        // Register types with reflection
251        app.register_type::<EffectAsset>()
252            .register_type::<ParticleEffect>()
253            .register_type::<EffectProperties>()
254            .register_type::<SpawnerSettings>()
255            .register_type::<Time<EffectSimulation>>();
256    }
257
258    fn finish(&self, app: &mut App) {
259        let render_device = app
260            .sub_app(RenderApp)
261            .world()
262            .resource::<RenderDevice>()
263            .clone();
264
265        let adapter_name = app
266            .world()
267            .get_resource::<RenderAdapterInfo>()
268            .map(|ai| &ai.name[..])
269            .unwrap_or("<unknown>");
270
271        // Check device limits
272        let limits = render_device.limits();
273        if limits.max_bind_groups < 4 {
274            error!("Hanabi requires a GPU device supporting at least 4 bind groups (Limits::max_bind_groups).\n  Current adapter: {}\n  Supported bind groups: {}", adapter_name, limits.max_bind_groups);
275            return;
276        } else {
277            info!("Initializing Hanabi for GPU adapter {}", adapter_name);
278        }
279
280        // Insert the properly aligned `vfx_common.wgsl` shader into Assets<Shader>, so
281        // that the automated Bevy shader processing finds it as an import. This is used
282        // for init/update/render shaders (but not the indirect one).
283        {
284            let common_shader = HanabiPlugin::make_common_shader(
285                render_device.limits().min_storage_buffer_offset_alignment,
286            );
287            let mut assets = app.world_mut().resource_mut::<Assets<Shader>>();
288            assets
289                .insert(&HANABI_COMMON_TEMPLATE_HANDLE, common_shader)
290                .unwrap();
291        }
292
293        // Insert the two variants of the properly aligned `vfx_indirect.wgsl` shaders
294        // into Assets<Shader>.
295        let (
296            indirect_shader_noevent,
297            indirect_shader_events,
298            prefix_sum_shader,
299            sort_fill_shader,
300            sort_shader,
301            sort_copy_shader,
302        ) = {
303            let align = render_device.limits().min_storage_buffer_offset_alignment;
304            let indirect_shader_noevent = HanabiPlugin::make_indirect_shader(align, false);
305            let indirect_shader_events = HanabiPlugin::make_indirect_shader(align, true);
306            let prefix_sum_shader = HanabiPlugin::make_prefix_sum_shader();
307            let sort_fill_shader = Shader::from_wgsl(
308                include_str!("render/vfx_sort_fill.wgsl"),
309                std::path::Path::new(file!())
310                    .parent()
311                    .unwrap()
312                    .join("render/vfx_sort_fill.wgsl")
313                    .to_string_lossy(),
314            );
315            let sort_shader = Shader::from_wgsl(
316                VFX_SORT_WGSL,
317                std::path::Path::new(file!())
318                    .parent()
319                    .unwrap()
320                    .join("render/vfx_sort.wgsl")
321                    .to_string_lossy(),
322            );
323            let sort_copy_shader = Shader::from_wgsl(
324                include_str!("render/vfx_sort_copy.wgsl"),
325                std::path::Path::new(file!())
326                    .parent()
327                    .unwrap()
328                    .join("render/vfx_sort_copy.wgsl")
329                    .to_string_lossy(),
330            );
331
332            let mut assets = app.world_mut().resource_mut::<Assets<Shader>>();
333            let indirect_shader_noevent = assets.add(indirect_shader_noevent);
334            let indirect_shader_events = assets.add(indirect_shader_events);
335            let prefix_sum_shader = assets.add(prefix_sum_shader);
336            let sort_fill_shader = assets.add(sort_fill_shader);
337            let sort_shader = assets.add(sort_shader);
338            let sort_copy_shader = assets.add(sort_copy_shader);
339
340            (
341                indirect_shader_noevent,
342                indirect_shader_events,
343                prefix_sum_shader,
344                sort_fill_shader,
345                sort_shader,
346                sort_copy_shader,
347            )
348        };
349
350        let effects_meta = EffectsMeta::new(
351            render_device.clone(),
352            indirect_shader_noevent,
353            indirect_shader_events,
354            prefix_sum_shader,
355        );
356
357        let effect_cache = EffectCache::new(render_device.clone());
358        let property_cache = PropertyCache::new(&render_device);
359        let event_cache = EventCache::new(render_device);
360
361        let render_app = app.sub_app_mut(RenderApp);
362        let sort_bind_groups = SortBindGroups::new(
363            render_app.world_mut(),
364            sort_fill_shader,
365            sort_shader,
366            sort_copy_shader,
367        );
368
369        // Register the custom render pipeline
370        render_app
371            .insert_resource(effects_meta)
372            .insert_resource(effect_cache)
373            .insert_resource(property_cache)
374            .insert_resource(event_cache)
375            .init_resource::<RenderDebugSettings>()
376            .init_resource::<EffectBindGroups>()
377            .init_resource::<PropertyBindGroups>()
378            .init_resource::<InitFillDispatchQueue>()
379            .init_resource::<SortFillDispatchQueue>()
380            .insert_resource(sort_bind_groups)
381            .init_resource::<UtilsPipeline>()
382            .init_resource::<GpuBufferOperations>()
383            .init_resource::<PrefixSumPipeline>()
384            .init_resource::<DispatchIndirectPipeline>()
385            .init_resource::<SpecializedComputePipelines<DispatchIndirectPipeline>>()
386            .init_resource::<ParticlesInitPipeline>()
387            .init_resource::<SpecializedComputePipelines<ParticlesInitPipeline>>()
388            .init_resource::<ParticlesInitPipeline>()
389            .init_resource::<SpecializedComputePipelines<ParticlesInitPipeline>>()
390            .init_resource::<ParticlesUpdatePipeline>()
391            .init_resource::<SpecializedComputePipelines<ParticlesUpdatePipeline>>()
392            .init_resource::<ParticlesRenderPipeline>()
393            .init_resource::<SpecializedRenderPipelines<ParticlesRenderPipeline>>()
394            .init_resource::<EffectAssetEvents>()
395            .init_resource::<SimParams>()
396            .init_resource::<SortedEffectBatches>()
397            .configure_sets(
398                Render,
399                (
400                    EffectSystems::PrepareEffectAssets.in_set(RenderSystems::PrepareAssets),
401                    EffectSystems::QueueEffects.in_set(RenderSystems::Queue),
402                    EffectSystems::PrepareEffectGpuResources
403                        .in_set(RenderSystems::PrepareResources),
404                    EffectSystems::PrepareBindGroups.in_set(RenderSystems::PrepareBindGroups),
405                ),
406            )
407            .edit_schedule(ExtractSchedule, |schedule| {
408                schedule.add_systems((
409                    start_stop_gpu_debug_capture,
410                    report_ready_state.before(extract_effects),
411                    extract_effects,
412                    extract_sim_params,
413                    extract_effect_events,
414                ));
415            })
416            .add_systems(
417                Render,
418                (
419                    (
420                        // Do all clears from previous frame; they can run in parallel as they
421                        // clear different resources.
422                        (clear_transient_batch_inputs, clear_previous_frame_resizes),
423                        // Allocate GPU resources depending only on the extracted data; they can
424                        // run in parallel as they touch different components.
425                        (
426                            // Allocate GPU storage for the effect particles
427                            allocate_effects,
428                            // Allocate GPU storage for GPU events (for child effects)
429                            allocate_events,
430                            // Allocate GPU storage for properties
431                            allocate_properties,
432                            // Update draw indirect args if Bevy relocated a render mesh
433                            update_mesh_locations
434                                // Need Bevy to have allocated the mesh in the MeshAllocator
435                                .after(bevy::render::mesh::allocator::allocate_and_free_meshes)
436                                // Need Bevy to have prepared the RenderMesh to read it
437                                .after(prepare_assets::<bevy::render::mesh::RenderMesh>),
438                            // Allocate GPU effect metadata
439                            allocate_metadata,
440                        ),
441                        // Allocate parent and child infos. Those need all effects allocated and
442                        // all parents resolved first, as well as event buffers allocated.
443                        allocate_parent_child_infos
444                            // Need the effects allocated to fetch the parent's slab ID
445                            .after(allocate_effects)
446                            // Need the events allocated to fetch the event buffer of children
447                            .after(allocate_events),
448                        fixup_parents
449                            // Second pass fixup after allocate_parent_child_infos()
450                            .after(allocate_parent_child_infos),
451                        // Prepare pipelines; they can run in parallel as they touch different
452                        // resources.
453                        (
454                            // Resolve the init and update pipelines, queue them if needed, and
455                            // check their state to determine if the
456                            // effect can be used this frame.
457                            prepare_init_update_pipelines
458                                // Need the bind group layout for the effect itself, which depends
459                                // on the particle layout.
460                                .after(allocate_effects)
461                                // Need the bind group layout for properties, which depends on the
462                                // property layout.
463                                .after(allocate_properties)
464                                // Need the number of event buffers to bind
465                                .after(fixup_parents),
466                            // Prepare the indirect pipeline depending on whether there's any child
467                            // info.
468                            prepare_indirect_pipeline
469                                // Need to know if any GPU event using effect is active or not
470                                .after(allocate_events),
471                        ),
472                        propagate_ready_state
473                            // Need the ready state of parents, which depends on the init/update
474                            // pipeline states
475                            .after(prepare_init_update_pipelines),
476                        prepare_batch_inputs,
477                        batch_effects,
478                    )
479                        // TODO: remove this chain() once all system dependencies are setup
480                        // correctly above.
481                        .chain()
482                        .in_set(EffectSystems::PrepareEffectAssets),
483                    // Once batched, queue the effects/batches which are ready to be
484                    // updated/rendered this frame.
485                    queue_effects
486                        .in_set(EffectSystems::QueueEffects)
487                        .after(batch_effects),
488                    // Queue the dispatch ops to fill the indirect dispatch args of the init pass
489                    // of child effects.
490                    queue_init_indirect_workgroup_update
491                        .in_set(EffectSystems::QueueEffects)
492                        .after(batch_effects)
493                        .after(fixup_parents),
494                    prepare_gpu_resources
495                        .in_set(EffectSystems::PrepareEffectGpuResources)
496                        // This creates the bind group for the view
497                        .after(prepare_view_uniforms)
498                        // Upload and optionally resize the draw indirect args buffer
499                        .after(update_mesh_locations)
500                        // Bind groups depend on buffers being re-/allocated
501                        .before(prepare_bind_groups),
502                    prepare_property_buffers
503                        .in_set(EffectSystems::PrepareEffectGpuResources)
504                        .before(prepare_bind_groups),
505                    prepare_effect_metadata
506                        .in_set(EffectSystems::PrepareEffectGpuResources)
507                        // Need DispatchBufferIndices to be allocated
508                        .after(allocate_effects)
509                        // Need the draw indirect args to be allocated
510                        .after(update_mesh_locations)
511                        // Need the local/global/base child index
512                        .after(fixup_parents)
513                        // Need the indirect dispatch args index for GPU event based init pass
514                        .after(allocate_events)
515                        // Need the properties block offset in GPU slab
516                        .after(allocate_properties)
517                        // This may invalidate some bind groups when resizing the metadata buffer
518                        .before(prepare_bind_groups),
519                    // Queue the dispatch ops to fill the indirect dispatch args of the ribbon
520                    // particle sort pass. Deferred from batch_effects() so it runs after the
521                    // effect metadata buffer has been (re-)allocated to this frame's size.
522                    queue_sort_fill_dispatch_ops
523                        .in_set(EffectSystems::PrepareEffectGpuResources)
524                        // Need the metadata buffer (re-)allocated so the captured handle and
525                        // dynamic offsets are correct and in-bounds
526                        .after(prepare_effect_metadata)
527                        // Must submit into the shared GpuBufferOperations before
528                        // queue_init_fill_dispatch_ops uploads its args buffer (end_frame)
529                        .before(queue_init_fill_dispatch_ops)
530                        .before(prepare_bind_groups),
531                    queue_init_fill_dispatch_ops
532                        .in_set(EffectSystems::PrepareEffectGpuResources)
533                        .after(prepare_gpu_resources)
534                        .before(prepare_bind_groups),
535                    // Prepare the bind groups
536                    prepare_bind_groups
537                        .in_set(EffectSystems::PrepareBindGroups)
538                        .after(queue_effects)
539                        .after(prepare_assets::<GpuImage>),
540                ),
541            );
542
543        // Register observers to deallocate GPU resources
544        {
545            let world = render_app.world_mut();
546            world.add_observer(on_remove_cached_effect);
547            world.add_observer(on_remove_cached_metadata);
548            world.add_observer(on_remove_cached_draw_indirect_args);
549            world.add_observer(on_remove_cached_effect_events);
550            world.add_observer(on_remove_cached_properties);
551        }
552
553        // Register the draw function for drawing the particles. This will be called
554        // during the main 2D/3D pass, at the Transparent2d/3d phase, after the
555        // opaque objects have been rendered (or, rather, commands for those
556        // have been recorded).
557        #[cfg(feature = "2d")]
558        {
559            let draw_particles = DrawEffects::new(render_app.world_mut());
560            render_app
561                .world()
562                .get_resource::<DrawFunctions<Transparent2d>>()
563                .unwrap()
564                .write()
565                .add(draw_particles);
566        }
567        #[cfg(feature = "3d")]
568        {
569            let draw_particles = DrawEffects::new(render_app.world_mut());
570            render_app
571                .world()
572                .get_resource::<DrawFunctions<Transparent3d>>()
573                .unwrap()
574                .write()
575                .add(draw_particles);
576
577            let draw_particles = DrawEffects::new(render_app.world_mut());
578            render_app
579                .world()
580                .get_resource::<DrawFunctions<AlphaMask3d>>()
581                .unwrap()
582                .write()
583                .add(draw_particles);
584
585            let draw_particles = DrawEffects::new(render_app.world_mut());
586            render_app
587                .world()
588                .get_resource::<DrawFunctions<Opaque3d>>()
589                .unwrap()
590                .write()
591                .add(draw_particles);
592        }
593    }
