1//! Type registration for modifiers.
2//!
3//! This module contains utilities used to reflect and serialize
4//! [`EffectAsset`]. In general, you don't need to directly interact with those;
5//! instead, use [`EffectAsset::serialize`], as well as the
6//! [`EffectAssetSerializer`] and [`EffectAssetDeserializer`].
7//!
8//! _Modifiers_ are objects implementing the [`Modifier`] trait. Because a trait
9//! abstracts the concrete object type, it cannot be used as is with the Bevy
10//! reflection and serialization infrastructure. Bevy does allow reflecting
11//! traits with `#[reflect_trait]`, but due to other limitations (notably,
12//! `Box<T>` not being reflected), this is not sufficient to easily make
13//! modifiers reflected and serializable.
14//!
15//! This module contains some utilities to enable reflecting and serializing
16//! modifiers. At the core, we register for each conrete modifier type some
17//! [`ReflectModifier`] type data, which essentially contains a factory function
18//! used to construct an instance of that type during deserialization. We also
19//! define [`Modifiers`], an ordered collection (`Vec`) of [`BoxedModifier`]
20//! objects, and provide some implementation of Bevy's [`SerializeWithRegistry`]
21//! and [`DeserializeWithRegistry`]. Those trait are automatically picked up by
22//! Bevy's reflect-based serialization utilities like [`TypedReflectSerializer`]
23//! and [`TypedReflectDeserializer`]. They are in turn used by
24//! [`EffectAssetSerializer`] and [`EffectAssetDeserializer`], which are custom
25//! de/serializer implementations for [`EffectAsset`] which handle the
26//! collections of modifiers for the init, update, and render passes of each
27//! effect.
28//!
29//! [`EffectAsset`]: crate::EffectAsset
30//! [`EffectAsset::serialize`]: crate::EffectAsset::serialize
31//! [`EffectAssetSerializer`]: crate::EffectAssetSerializer
32//! [`EffectAssetDeserializer`]: crate::EffectAssetDeserializer
33//! [`SerializeWithRegistry`]: bevy::reflect::serde::SerializeWithRegistry
34//! [`DeserializeWithRegistry`]: bevy::reflect::serde::DeserializeWithRegistry
35//! [`TypedReflectSerializer`]: bevy::reflect::serde::TypedReflectSerializer
36//! [`TypedReflectDeserializer`]: bevy::reflect::serde::TypedReflectDeserializer
37

56/// Factory function to build a default modifier instance.
57///
58/// The factory can use the given [`Module`] to register some expressions in
59/// order to default-initialize itself.
60pub type ModifierFactory = fn(&mut Module) -> BoxedModifier;
61
62/// Type data attached to a modifier type's registration in
63/// [`AppTypeRegistry`]. Carrying both fields here means the editor
64/// never needs to construct an instance just to query
65/// `Modifier::context()`.
66#[derive(Clone, Copy)]
67pub struct ReflectModifier {
68    /// Factory function.
69    pub factory: ModifierFactory,
70    /// Modifier context, cached from [`Modifier::context()`].
71    pub context: ModifierContext,
72}
73
74impl ReflectModifier {}
75
76/// Register a [`ReflectModifier`] type data for the given [`Modifier`].
77pub fn register_reflect_modifier<T: Modifier>(
78    type_registry: &AppTypeRegistry,
79    factory: ModifierFactory,
80) {
81    // Create a dummy instance to cache its context. This also serves as a
82    // validation layer to check the returned object type is correct.
83    let context = {
84        let mut module = Module::default();
85        let modifier = factory(&mut module);
86        let context = modifier.context();
87        let any = modifier.into_any();
88        assert_eq!(
89            any.type_id(),
90            TypeId::of::<T>(),
91            "Factory for modifier type '{}' returned a different object of type.",
92            std::any::type_name::<T>()
93        );
94        context
95    };
96
97    let reflect_modifier = ReflectModifier { factory, context };
98
99    match type_registry.write().get_mut(TypeId::of::<T>()) {
100        Some(type_registration) => type_registration.insert(reflect_modifier),
101        None => warn!(
102            "insert_reflect_modifier: type {} not found in TypeRegistry",
103            std::any::type_name::<T>()
104        ),
105    }
106}

108// Serialize a Box<dyn Modifier> by delegating to the reflect serializer. The
109// reflect serializer emits the type tag and the inner data as a single map
110// entry.
111impl bevy::reflect::serde::SerializeWithRegistry for BoxedModifier {
112    fn serialize<S>(&self, serializer: S, registry: &TypeRegistry) -> Result<S::Ok, S::Error>
113    where
114        S: Serializer,
115    {
116        // Get an &dyn Reflect for the boxed modifier
117        let reflect: &dyn Reflect = Reflect::as_reflect(&**self);
118
119        // Serialize as a single-entry map { "full::type::Path": <typed-data> }
120        let type_path = reflect
121            .get_represented_type_info()
122            .ok_or_else(|| {
123                S::Error::custom("cannot serialize dynamic value without represented type")
124            })?
125            .type_path();
126
127        let mut map = serializer.serialize_map(Some(1))?;
128        map.serialize_entry(type_path, &TypedReflectSerializer::new(reflect, registry))?;
129        map.end()
130    }
131}

133// Deserialize a Box<dyn Modifier> using the reflect deserializer and the
134// ReflectModifier type data to construct the concrete Modifier instance.
135impl<'de> bevy::reflect::serde::DeserializeWithRegistry<'de> for BoxedModifier {
136    fn deserialize<D>(deserializer: D, registry: &TypeRegistry) -> Result<Self, D::Error>
137    where
138        D: serde::de::Deserializer<'de>,
139    {
140        // First, use the generic reflect deserializer which expects a single-entry map
141        // { "full::type::Path": <value> } and returns a Box<dyn PartialReflect> (e.g.
142        // DynamicStruct).
143        let reflect_seed = ReflectDeserializer::new(registry);
144        let boxed_partial: Box<dyn PartialReflect> = reflect_seed
145            .deserialize(deserializer)
146            .map_err(serde::de::Error::custom)?;
147
148        // Obtain the type id represented by the deserialized reflect value.
149        let type_info = boxed_partial.get_represented_type_info().ok_or_else(|| {
150            serde::de::Error::custom("reflected value has no represented type info")
151        })?;
152        let type_id = type_info.type_id();
153
154        // Lookup the ReflectModifier type data that must have been registered for this
155        // concrete modifier type (register_reflect_modifier<T>). That type-data
156        // contains the factory used to create a default instance and cached
157        // ModifierContext.
158        let reflect_modifier = registry
159            .get_type_data::<ReflectModifier>(type_id)
160            .ok_or_else(|| {
161                serde::de::Error::custom(format!(
162                    "no ReflectModifier type data for '{}'",
163                    type_info.type_path()
164                ))
165            })?;
166
167        // Build a default instance using the stored factory. The factory currently
168        // takes a &mut Module; construct a temporary Module (same as
169        // register_reflect_modifier).
170        let mut module = Module::default();
171        let mut modifier: BoxedModifier = (reflect_modifier.factory)(&mut module);
172
173        // Apply the deserialized partial reflect value into the instance.
174        let reflect_mut: &mut dyn Reflect = Reflect::as_reflect_mut(&mut *modifier);
175        reflect_mut.apply(boxed_partial.as_partial_reflect());
176
177        Ok(modifier)
178    }
179}

188/// Ordered collection of [`BoxedModifier`] objects.
189///
190/// This is a wrapper over a `Vec<BoxedModifier>`, with additional reflection
191/// and serialization features.
192#[derive(Default, Clone, Reflect)]
193#[reflect(SerializeWithRegistry, DeserializeWithRegistry, from_reflect = false)]
194pub struct Modifiers(#[reflect(ignore)] pub Vec<BoxedModifier>);
195
196impl Deref for Modifiers {
197    type Target = Vec<BoxedModifier>;
198    fn deref(&self) -> &Self::Target {
199        &self.0
200    }
201}
202
203impl DerefMut for Modifiers {
204    fn deref_mut(&mut self) -> &mut Self::Target {
205        &mut self.0
206    }
207}

209impl bevy::reflect::serde::SerializeWithRegistry for Modifiers {
210    fn serialize<S>(&self, serializer: S, registry: &TypeRegistry) -> Result<S::Ok, S::Error>
211    where
212        S: Serializer,
213    {
214        struct Elem<'a> {
215            boxed: &'a BoxedModifier,
216            registry: &'a TypeRegistry,
217        }
218
219        impl<'a> serde::Serialize for Elem<'a> {
220            fn serialize<S2>(&self, serializer: S2) -> Result<S2::Ok, S2::Error>
221            where
222                S2: serde::Serializer,
223            {
224                use bevy::reflect::serde::SerializeWithRegistry;
225                self.boxed.serialize(serializer, self.registry)
226            }
227        }
228
229        let mut seq = serializer.serialize_seq(Some(self.0.len()))?;
230        for m in &self.0 {
231            seq.serialize_element(&Elem { boxed: m, registry })?;
232        }
233        seq.end()
234    }
235}

237impl<'de> bevy::reflect::serde::DeserializeWithRegistry<'de> for Modifiers {
238    fn deserialize<D>(deserializer: D, registry: &TypeRegistry) -> Result<Self, D::Error>
239    where
240        D: serde::de::Deserializer<'de>,
241    {
242        // Deserialize the Modifiers collection, which is a sequence of single-item
243        // {typename:value} maps.
244
245        struct ModifiersVisitor<'a> {
246            registry: &'a TypeRegistry,
247        }
248
249        impl<'a, 'de> Visitor<'de> for ModifiersVisitor<'a> {
250            type Value = Modifiers;
251
252            fn expecting(&self, formatter: &mut Formatter) -> std::fmt::Result {
253                write!(formatter, "a list of modifiers")
254            }
255
256            fn visit_seq<A>(self, mut seq: A) -> Result<Self::Value, A::Error>
257            where
258                A: SeqAccess<'de>,
259            {
260                // Each element is a single-entry map { "full::type::Path": <value> }.
261                // Implement a DeserializeSeed that parses that map into a BoxedModifier.
262                struct ElemSeed<'a> {
263                    registry: &'a TypeRegistry,
264                }
265
266                impl<'de2, 'a> serde::de::DeserializeSeed<'de2> for ElemSeed<'a> {
267                    type Value = BoxedModifier;
268
269                    fn deserialize<D2>(self, deserializer: D2) -> Result<Self::Value, D2::Error>
270                    where
271                        D2: serde::de::Deserializer<'de2>,
272                    {
273                        use bevy::reflect::serde::DeserializeWithRegistry;
274                        BoxedModifier::deserialize(deserializer, self.registry)
275                    }
276                }
277
278                let mut vec = Vec::new();
279                while let Some(modifier) = seq.next_element_seed(ElemSeed {
280                    registry: self.registry,
281                })? {
282                    vec.push(modifier);
283                }
284
285                Ok(Modifiers(vec))
286            }
287        }
288
289        let modifiers: Self = deserializer.deserialize_seq(ModifiersVisitor { registry })?;
290        Ok(modifiers)
291    }
292}

306    /// Serialize and deserialize a [`Modifiers`] container.
307    #[test]
308    fn serde_modifiers() {
309        use bevy::reflect::{
310            serde::{TypedReflectDeserializer, TypedReflectSerializer},
311            Reflect,
312        };
313
314        use crate::{Attribute, SetAttributeModifier, SetPositionSphereModifier, ShapeDimension};
315
316        let mut module = Module::default();
317
318        // Create a Modifiers container with 2 different modifiers
319        let mut modifiers = vec![];
320        let m1 = SetAttributeModifier::new(Attribute::SIZE, module.lit(2.));
321        let bm1: BoxedModifier = Box::new(m1);
322        modifiers.push(bm1);
323        let m2 = SetPositionSphereModifier {
324            center: module.lit(Vec3::ZERO),
325            radius: module.lit(1.),
326            dimension: ShapeDimension::Surface,
327        };
328        let bm2: BoxedModifier = Box::new(m2);
329        modifiers.push(bm2);
330        let modifiers = Modifiers(modifiers);
331
332        // Prepare a type registry with all known modifiers registered.
333        let type_registry = AppTypeRegistry::new_with_derived_types();
334        register_modifiers(&type_registry);
335        let registry = type_registry.read();
336
337        // Serialize via reflection, so that the SerializeWithRegistry impl of Modifiers
338        // get automatically picked up and used.
339        let s = ron::ser::to_string_pretty(
340            &TypedReflectSerializer::new(modifiers.as_reflect(), &registry),
341            PrettyConfig::default(),
342        )
343        .unwrap();
344        println!("{s}");
345
346        // The serialized string must contain the type names of all modifiers, because
347        // they use dynamic dispatching.
348        assert!(s.contains(std::any::type_name::<SetAttributeModifier>()));
349        assert!(s.contains(std::any::type_name::<SetPositionSphereModifier>()));
350        // The Modifiers wrapper itself does not necessarily appear in the serialized
351        // form here because we're serializing the inner representation
352        // directly.
353        assert!(!s.contains(std::any::type_name::<Modifiers>()));
354
355        // Deserialize via reflection again. This also picks up the
356        // DeserializeWithRegistry impl of Modifiers automatically, which recovers the
357        // concrete object via the ReflectModifier type data factory.
358        let mut de = ron::de::Deserializer::from_str(&s).unwrap();
359        let type_registration = registry.get(std::any::TypeId::of::<Modifiers>()).unwrap();
360        let deserializer = TypedReflectDeserializer::new(type_registration, &registry);
361        let mods = deserializer.deserialize(&mut de).unwrap();
362
363        // Recover and check the deserialized Modifiers
364        assert!(mods.represents::<Modifiers>());
365        let mods = mods.try_downcast::<Modifiers>().unwrap();
366        assert_eq!(mods.0.len(), modifiers.0.len());
367        for (mi, smi) in mods.0.iter().zip(modifiers.0.iter()) {
368            use std::any::Any;
369
370            assert_eq!(
371                mi.get_represented_type_info().type_id(),
372                smi.get_represented_type_info().type_id()
373            );
374        }
375
376        // Recover and check the individual modifiers
377        let serde_m1 = mods.0[0]
378            .as_reflect()
379            .downcast_ref::<SetAttributeModifier>()
380            .unwrap();
381        let serde_m2 = mods.0[1]
382            .as_reflect()
383            .downcast_ref::<SetPositionSphereModifier>()
384            .unwrap();
385        assert_eq!(m1, *serde_m1);
386        assert_eq!(m2, *serde_m2);
387    }

389    fn cmp_modifiers<'a>(
390        a: impl Iterator<Item = &'a dyn Modifier>,
391        b: impl Iterator<Item = &'a dyn Modifier>,
392    ) {
393        let a = a.collect::<Vec<_>>();
394        let b = b.collect::<Vec<_>>();
395        assert_eq!(a.len(), b.len());
396        for (a, b) in a.iter().zip(b.iter()) {
397            assert_eq!(a.context(), b.context());
398        }

413    /// Serialize and deserialize a full [`EffectAsset`].
414    #[test]
415    fn serde_asset() {
416        let mut module = Module::default();
417        let accel_mod = AccelModifier::new(module.lit(Vec3::X));
418        let asset =
419            EffectAsset::new(24, SpawnerSettings::once(3.0.into()), module).update(accel_mod);
420
421        let type_registry = AppTypeRegistry::new_with_derived_types();
422        register_modifiers(&type_registry);
423
424        // ser
425        let registry = type_registry.read();
426        let serializer = EffectAssetSerializer::new(&asset, &registry);
427        let json = ron::ser::to_string_pretty(&serializer, PrettyConfig::default()).unwrap();
428        println!("{json}");
429
430        // de
431        let mut deserializer = ron::de::Deserializer::from_str(&json).unwrap();
432        let deserialize = EffectAssetDeserializer::new(&registry);
433        let serde_asset = deserialize.deserialize(&mut deserializer).unwrap();
434
435        // Validate what can be
436        assert_eq!(asset.name, serde_asset.name);
437        assert_eq!(asset.capacity(), serde_asset.capacity());
438        assert_eq!(asset.spawner, serde_asset.spawner);
439        assert_eq!(asset.z_layer_2d, serde_asset.z_layer_2d);
440        assert_eq!(asset.simulation_space, serde_asset.simulation_space);
441        assert_eq!(asset.simulation_condition, serde_asset.simulation_condition);
442        assert_eq!(asset.prng_seed, serde_asset.prng_seed);
443        assert_eq!(asset.motion_integration, serde_asset.motion_integration);
444        assert_eq!(asset.alpha_mode, serde_asset.alpha_mode);
445        cmp_modifiers(asset.init_modifiers(), serde_asset.init_modifiers());
446        cmp_modifiers(asset.update_modifiers(), serde_asset.update_modifiers());
