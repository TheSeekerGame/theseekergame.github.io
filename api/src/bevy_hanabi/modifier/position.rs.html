15/// A modifier to set the position of particles on or inside a circle/disc,
16/// randomly.
17///
18/// # Attributes
19///
20/// This modifier requires the following particle attributes:
21/// - [`Attribute::POSITION`]
22#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Reflect, Serialize, Deserialize)]
23pub struct SetPositionCircleModifier {
24    /// The circle center, relative to the emitter position.
25    ///
26    /// Expression type: `Vec3`
27    pub center: ExprHandle,
28    /// The circle axis, which is the normalized normal of the circle's plane.
29    /// Set this to `Vec3::Z` for a 2D game.
30    ///
31    /// Expression type: `Vec3`
32    pub axis: ExprHandle,
33    /// The circle radius.
34    ///
35    /// Expression type: `f32`
36    pub radius: ExprHandle,
37    /// The shape dimension to set the position to.
38    ///
39    /// Note the particular interpretation of the dimension for this shape,
40    /// which unlike other shapes is a 2D one to begin with:
41    /// - [`ShapeDimension::Volume`] randomly position the particle anywhere on
42    ///   the "volume" of the shape, which here is understood to be the 2D disc
43    ///   surface including its origin (`dist <= r`).
44    /// - [`ShapeDimension::Surface`] randomly position the particle anywhere on
45    ///   the "surface" of the shape, which here is understood to be the
46    ///   perimeter circle, the set of points at a distance from the center
47    ///   exactly equal to the radius (`dist == r`).
48    pub dimension: ShapeDimension,
49}
50
51impl SetPositionCircleModifier {
52    fn eval(
53        &self,
54        module: &mut Module,
55        context: &mut dyn EvalContext,
56    ) -> Result<String, ExprError> {
57        let func_id = calc_func_id(self);
58        let func_name = format!("set_position_circle_{0:016X}", func_id);
59
60        context.make_fn(
61            &func_name,
62            "particle: ptr<function, Particle>",
63            module,
64            &mut |m: &mut Module, ctx: &mut dyn EvalContext| -> Result<String, ExprError> {
65                let center = ctx.eval(m, self.center)?;
66                let axis = ctx.eval(m, self.axis)?;
67
68                let radius = match self.dimension {
69                    ShapeDimension::Surface => {
70                        // Constant radius
71                        format!("let r = {};", ctx.eval(m, self.radius)?)
72                    }
73                    ShapeDimension::Volume => {
74                        // Radius uniformly distributed in [0:1], then square-rooted
75                        // to account for the increased perimeter covered by increased radii.
76                        format!("let r = sqrt(frand()) * ({});", ctx.eval(m, self.radius)?)
77                    }
78                };
79
80                Ok(format!(
81                    r##"    // Circle center
82    let c = {};
83    // Circle basis
84    let n = {};
85    let sign = step(0.0, n.z) * 2.0 - 1.0;
86    let a = -1.0 / (sign + n.z);
87    let b = n.x * n.y * a;
88    let tangent = vec3<f32>(1.0 + sign * n.x * n.x * a, sign * b, -sign * n.x);
89    let bitangent = vec3<f32>(b, sign + n.y * n.y * a, -n.y);
90    // Circle radius
91    {}
92    // Spawn random point on/in circle
93    let theta = frand() * tau;
94    let dir = tangent * cos(theta) + bitangent * sin(theta);
95    (*particle).{} = c + r * dir;
96"##,
97                    center,
98                    axis,
99                    radius,
100                    Attribute::POSITION.name(),
101                ))
102            },
103        )?;
104
105        let code = format!("{}(&particle);\n", func_name);
106
107        Ok(code)
108    }
109}

131/// A modifier to set the position of particles on or inside a sphere, randomly.
132///
133/// # Attributes
134///
135/// This modifier requires the following particle attributes:
136/// - [`Attribute::POSITION`]
137#[derive(Debug, Clone, Copy, PartialEq, Hash, Reflect, Serialize, Deserialize)]
138pub struct SetPositionSphereModifier {
139    /// The sphere center, relative to the emitter position.
140    ///
141    /// Expression type: `Vec3`
142    pub center: ExprHandle,
143    /// The sphere radius.
144    ///
145    /// Expression type: `f32`
146    pub radius: ExprHandle,
147    /// The shape dimension to set the position to.
148    pub dimension: ShapeDimension,
149}
150
151impl SetPositionSphereModifier {
152    fn eval(
153        &self,
154        module: &mut Module,
155        context: &mut dyn EvalContext,
156    ) -> Result<String, ExprError> {
157        let func_id = calc_func_id(self);
158        let func_name = format!("set_position_sphere_{0:016X}", func_id);
159
160        context.make_fn(
161            &func_name,
162            "particle: ptr<function, Particle>",
163            module,
164            &mut |m: &mut Module, ctx: &mut dyn EvalContext| -> Result<String, ExprError> {
165                let center = ctx.eval(m, self.center)?;
166
167                let radius = match self.dimension {
168                    ShapeDimension::Surface => {
169                        // Constant radius
170                        format!("let r = {};", ctx.eval(m, self.radius)?)
171                    }
172                    ShapeDimension::Volume => {
173                        // Radius uniformly distributed in [0:1], then scaled by ^(1/3) in 3D
174                        // to account for the increased surface covered by increased radii.
175                        // https://stackoverflow.com/questions/54544971/how-to-generate-uniform-random-points-inside-d-dimension-ball-sphere
176                        format!(
177                            "let r = pow(frand(), 1./3.) * ({});",
178                            ctx.eval(m, self.radius)?
179                        )
180                    }
181                };
182
183                Ok(format!(
184                    r##"    // Sphere center
185    let c = {};
186
187    // Sphere radius
188    {}
189
190    // Spawn randomly along the sphere surface using Archimedes's theorem
191    let theta = frand() * tau;
192    let z = frand() * 2. - 1.;
193    let phi = acos(z);
194    let sinphi = sin(phi);
195    let x = sinphi * cos(theta);
196    let y = sinphi * sin(theta);
197    let dir = vec3<f32>(x, y, z);
198    (*particle).{} = c + r * dir;
199"##,
200                    center,
201                    radius,
202                    Attribute::POSITION.name(),
203                ))
204            },
205        )?;
206
207        let code = format!("{}(&particle);\n", func_name);
208
209        Ok(code)
210    }
211}

233/// A modifier to set the position of particles on a truncated 3D cone.
234///
235/// The 3D cone is oriented along the Y axis, with its origin at the center of
236/// the base circle of the cone. The center of the top circle truncating the
237/// cone is located at a positive Y.
238///
239/// Particles are moved somewhere inside the volume or on the surface of a
240/// truncated 3D cone defined by its base radius, its top radius, and the height
241/// of the cone section.
242///
243/// # Attributes
244///
245/// This modifier requires the following particle attributes:
246/// - [`Attribute::POSITION`]
247#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Reflect, Serialize, Deserialize)]
248pub struct SetPositionCone3dModifier {
249    /// The cone height along its axis, between the base and top radii.
250    ///
251    /// Expression type: `f32`
252    pub height: ExprHandle,
253    /// The cone radius at its base, perpendicularly to its axis.
254    ///
255    /// Expression type: `f32`
256    pub base_radius: ExprHandle,
257    /// The cone radius at its truncated top, perpendicularly to its axis.
258    /// This can be set to zero to get a non-truncated cone.
259    ///
260    /// Expression type: `f32`
261    pub top_radius: ExprHandle,
262    /// The shape dimension to set the position to.
263    pub dimension: ShapeDimension,
264}
265
266impl SetPositionCone3dModifier {
267    fn eval(
268        &self,
269        module: &mut Module,
270        context: &mut dyn EvalContext,
271    ) -> Result<String, ExprError> {
272        let func_id = calc_func_id(self);
273        let func_name = format!("set_position_cone3d_{0:016X}", func_id);
274
275        context.make_fn(
276            &func_name,
277            "transform: mat4x4<f32>, particle: ptr<function, Particle>",
278            module,
279            &mut |m: &mut Module, ctx: &mut dyn EvalContext| -> Result<String, ExprError> {
280                let height = ctx.eval(m, self.height)?;
281                let top_radius = ctx.eval(m, self.top_radius)?;
282                let base_radius = ctx.eval(m, self.base_radius)?;
283
284                Ok(format!(
285                    r##"    // Truncated cone height
286    let h0 = {0};
287    // Random height ratio
288    let alpha_h = pow(frand(), 1.0 / 3.0);
289    // Random delta height from top
290    let h = h0 * alpha_h;
291    // Top radius
292    let rt = {1};
293    // Bottom radius
294    let rb = {2};
295    // Radius at height h
296    let r0 = rb + (rt - rb) * alpha_h;
297    // Random delta radius
298    let alpha_r = sqrt(frand());
299    // Random radius at height h
300    let r = r0 * alpha_r;
301    // Random base angle
302    let theta = frand() * tau;
303    let cost = cos(theta);
304    let sint = sin(theta);
305    // Random position relative to truncated cone origin (not apex)
306    let x = r * cost;
307    let y = h;
308    let z = r * sint;
309    let p = vec3<f32>(x, y, z);
310    let p2 = transform * vec4<f32>(p, 0.0);
311    (*particle).{3} = p2.xyz;
312"##,
313                    height,
314                    top_radius,
315                    base_radius,
316                    Attribute::POSITION.name(),
317                ))
318            },
319        )?;
320
321        let code = format!("{}(transform, &particle);\n", func_name);
322
323        Ok(code)
324    }
325}
