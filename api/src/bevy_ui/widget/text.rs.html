29/// UI text system flags.
30///
31/// Used internally by [`measure_text_system`] and [`text_system`] to schedule text for processing.
32#[derive(Component, Debug, Clone, Reflect)]
33#[reflect(Component, Default, Debug, Clone)]
34pub struct TextNodeFlags {
35    /// If set then a new measure function for the text node will be created.
36    needs_measure_fn: bool,
37    /// If set then the text will be recomputed.
38    needs_recompute: bool,
39}
40
41impl Default for TextNodeFlags {
42    fn default() -> Self {
43        Self {
44            needs_measure_fn: true,
45            needs_recompute: true,
46        }
47    }
48}

50/// The top-level UI text component.
51///
52/// Adding [`Text`] to an entity will pull in required components for setting up a UI text node.
53///
54/// The string in this component is the first 'text span' in a hierarchy of text spans that are collected into
55/// a [`ComputedTextBlock`]. See [`TextSpan`](bevy_text::TextSpan) for the component used by children of entities with [`Text`].
56///
57/// Note that [`Transform`](bevy_transform::components::Transform) on this entity is managed automatically by the UI layout system.
58///
59///
60/// ```
61/// # use bevy_asset::Handle;
62/// # use bevy_color::Color;
63/// # use bevy_color::palettes::basic::BLUE;
64/// # use bevy_ecs::world::World;
65/// # use bevy_text::{Font, FontSize, Justify, TextLayout, TextFont, TextColor, TextSpan};
66/// # use bevy_ui::prelude::Text;
67/// #
68/// # let font_handle: Handle<Font> = Default::default();
69/// # let mut world = World::default();
70/// #
71/// // Basic usage.
72/// world.spawn(Text::new("hello world!"));
73///
74/// // With non-default style.
75/// world.spawn((
76///     Text::new("hello world!"),
77///     TextFont {
78///         font: font_handle.clone().into(),
79///         font_size: FontSize::Px(60.0),
80///         ..Default::default()
81///     },
82///     TextColor(BLUE.into()),
83/// ));
84///
85/// // With text justification.
86/// world.spawn((
87///     Text::new("hello world\nand bevy!"),
88///     TextLayout::justify(Justify::Center)
89/// ));
90///
91/// // With spans
92/// world.spawn(Text::new("hello ")).with_children(|parent| {
93///     parent.spawn(TextSpan::new("world"));
94///     parent.spawn((TextSpan::new("!"), TextColor(BLUE.into())));
95/// });
96/// ```
97#[derive(Component, Debug, Default, Clone, Deref, DerefMut, Reflect, PartialEq)]
98#[reflect(Component, Default, Debug, PartialEq, Clone)]
99#[require(
100    Node,
101    TextLayout,
102    TextFont,
103    TextColor,
104    LineHeight,
105    LetterSpacing,
106    TextNodeFlags,
107    ContentSize,
108    // Hinting is enabled by default as UI text is normally pixel.
109    FontHinting::Enabled
110)]
111pub struct Text(pub String);

169/// Text measurement for UI layout. See [`NodeMeasure`].
170pub struct TextMeasure {
171    pub info: TextMeasureInfo,
172}
173
174impl TextMeasure {
175    /// Checks if the Parley text layout is needed for measuring the text.
176    #[inline]
177    pub const fn needs_buffer(height: Option<f32>, available_width: AvailableSpace) -> bool {
178        height.is_none() && matches!(available_width, AvailableSpace::Definite(_))
179    }
180}
181
182impl Measure for TextMeasure {
183    fn measure(&mut self, measure_args: MeasureArgs) -> Vec2 {
184        let mut width = measure_args.resolve_width();
185        let height = measure_args.resolve_height();
186
187        let MeasureArgs {
188            available_width,
189            buffer,
190            font_system,
191            style,
192            ..
193        } = measure_args;
194
195        // The text is wrapped inside the content box, so subtract horizontal padding and border.
196        if style.box_sizing == taffy::style::BoxSizing::BorderBox {
197            let context = taffy::Size {
198                width: width.effective,
199                height: height.effective,
200            };
201            let calc = |_, _| 0.;
202            let padding = style.padding.resolve_or_zero(context, calc);
203            let border = style.border.resolve_or_zero(context, calc);
204            let total_x_inset = padding.left + padding.right + border.left + border.right;
205            width.min = width.min.map(|min| (min - total_x_inset).max(0.));
206            width.max = width.max.map(|max| (max - total_x_inset).max(0.));
207            width.effective = width
208                .effective
209                .map(|effective| (effective - total_x_inset).max(0.));
210        }
211
212        let x = width
213            .effective
214            .unwrap_or_else(|| match available_width {
215                AvailableSpace::Definite(x) => {
216                    // It is possible for the "min content width" to be larger than
217                    // the "max content width" when soft-wrapping right-aligned text
218                    // and possibly other situations.
219
220                    x.max(self.info.min.x).min(self.info.max.x)
221                }
222                AvailableSpace::MinContent => self.info.min.x,
223                AvailableSpace::MaxContent => self.info.max.x,
224            })
225            .maybe_clamp(width.min, width.max);
226
227        let size = height.effective.map_or_else(
228            || match available_width {
229                AvailableSpace::Definite(_) => {
230                    if let Some(buffer) = buffer {
231                        self.info
232                            .compute_size(TextBounds::new_horizontal(x), buffer, font_system)
233                    } else {
234                        error!("text measure failed, buffer is missing");
235                        Vec2::default()
236                    }
237                }
238                AvailableSpace::MinContent => Vec2::new(x, self.info.min.y),
239                AvailableSpace::MaxContent => Vec2::new(x, self.info.max.y),
240            },
241            |y| Vec2::new(x, y),
242        );
243
244        Vec2::new(
245            size.x.maybe_clamp(width.min, width.max),
246            size.y.maybe_clamp(height.min, height.max),
247        )
248        .ceil()
249    }
250}

252/// Generates a new [`Measure`] for a text node on changes to its [`Text`] component.
253///
254/// A `Measure` is used by the UI's layout algorithm to determine the appropriate amount of space
255/// to provide for the text given the fonts, the text itself and the constraints of the layout.
256///
257/// * Measures are regenerated on changes to either [`ComputedTextBlock`] or [`ComputedUiRenderTargetInfo`].
258/// * Changes that only modify the colors of a `Text` do not require a new `Measure`. This system
259///   is only able to detect that a `Text` component has changed and will regenerate the `Measure` on
260///   color changes. This can be expensive, particularly for large blocks of text, and the [`bypass_change_detection`](bevy_ecs::change_detection::DetectChangesMut::bypass_change_detection)
261///   method should be called when only changing the `Text`'s colors.
262pub fn measure_text_system(
263    fonts: Res<Assets<Font>>,
264    mut text_query: Query<
265        (
266            Entity,
267            Ref<Text>,
268            Ref<TextLayout>,
269            &mut ContentSize,
270            &mut TextNodeFlags,
271            &mut ComputedTextBlock,
272            Ref<ComputedUiRenderTargetInfo>,
273            &ComputedNode,
274        ),
275        With<Node>,
276    >,
277    mut text_reader: TextUiReader,
278    mut text_pipeline: ResMut<TextPipeline>,
279    mut font_system: ResMut<FontCx>,
280    mut layout_cx: ResMut<LayoutCx>,
281    rem_size: Res<RemSize>,
282) {
283    for (
284        entity,
285        text,
286        block,
287        mut content_size,
288        mut text_flags,
289        mut computed,
290        computed_target,
291        computed_node,
292    ) in &mut text_query
293    {
294        // Note: the ComputedTextBlock::needs_rerender bool is cleared in create_text_measure().
295        // 1e-5 epsilon to ignore tiny scale factor float errors
296        if !(1e-5
297            < (computed_target.scale_factor() - computed_node.inverse_scale_factor.recip()).abs()
298            || computed.needs_rerender(computed_target.is_changed(), rem_size.is_changed())
299            || text.is_changed()
300            || text_flags.needs_measure_fn
301            || content_size.is_added())
302        {
303            continue;
304        }
305
306        match text_pipeline.create_text_measure(
307            entity,
308            fonts.as_ref(),
309            text_reader.iter(entity),
310            computed_target.scale_factor,
311            &block,
312            computed.as_mut(),
313            &mut font_system,
314            &mut layout_cx,
315            computed_target.logical_size(),
316            rem_size.0,
317        ) {
318            Ok(measure) => {
319                if block.linebreak == LineBreak::NoWrap {
320                    content_size.set(NodeMeasure::Fixed(FixedMeasure { size: measure.max }));
321                } else {
322                    content_size.set(NodeMeasure::Text(TextMeasure { info: measure }));
323                }
324
325                // Text measure func created successfully, so set `TextNodeFlags` to schedule a recompute
326                text_flags.needs_measure_fn = false;
327                text_flags.needs_recompute = true;
328            }
329            Err(
330                TextError::NoSuchFont
331                | TextError::NoSuchFontFamily(_)
332                | TextError::DegenerateScaleFactor,
333            ) => {
334                // Try again next frame
335                text_flags.needs_measure_fn = true;
336            }
337            Err(
338                e @ (TextError::FailedToAddGlyph(_)
339                | TextError::FailedToGetGlyphImage(_)
340                | TextError::MissingAtlasLayout
341                | TextError::MissingAtlasTexture
342                | TextError::InconsistentAtlasState),
343            ) => {
344                panic!("Fatal error when processing text: {e}.");
345            }
346        };
347    }
348}

350/// Updates the layout and size information for a UI text node on changes to the size value of its [`Node`] component,
351/// or when the `needs_recompute` field of [`TextNodeFlags`] is set to true.
352/// This information is computed by the [`TextPipeline`] and then stored in [`TextLayoutInfo`].
353///
354/// ## World Resources
355///
356/// [`ResMut<Assets<Image>>`](Assets<Image>) -- This system only adds new [`Image`] assets.
357/// It does not modify or observe existing ones. The exception is when adding new glyphs to a [`bevy_text::FontAtlas`].
358pub fn text_system(
359    mut textures: ResMut<Assets<Image>>,
360    mut font_atlas_set: ResMut<FontAtlasSet>,
361    mut text_pipeline: ResMut<TextPipeline>,
362    mut text_query: Query<(
363        Ref<ComputedNode>,
364        &TextLayout,
365        &mut TextLayoutInfo,
366        &mut TextNodeFlags,
367        &mut ComputedTextBlock,
368        Ref<FontHinting>,
369    )>,
370    mut scale_cx: ResMut<ScaleCx>,
371) {
372    for (node, block, mut text_layout_info, mut text_flags, mut computed, hinting) in
373        &mut text_query
374    {
375        if node.is_changed() || text_flags.needs_recompute || hinting.is_changed() {
376            // Skip the text node if it is waiting for a new measure func
377            if text_flags.needs_measure_fn {
378                continue;
379            }
380
381            let physical_node_size = if block.linebreak == LineBreak::NoWrap {
382                // With `NoWrap` set, no constraints are placed on the width of the text.
383                TextBounds::UNBOUNDED
384            } else {
385                let content_box_size = node.content_box().size();
386                TextBounds::new(content_box_size.x, content_box_size.y)
387            };
388
389            match text_pipeline.update_text_layout_info(
390                &mut text_layout_info,
391                &mut font_atlas_set,
392                &mut textures,
393                &mut computed,
394                &mut scale_cx,
395                physical_node_size,
396                block.justify,
397                *hinting,
398            ) {
399                Err(
400                    TextError::NoSuchFont
401                    | TextError::NoSuchFontFamily(_)
402                    | TextError::DegenerateScaleFactor,
403                ) => {
404                    // There was an error processing the text layout, try again next frame
405                    text_flags.needs_recompute = true;
406                }
407                Err(e @ TextError::FailedToGetGlyphImage(_)) => {
408                    warn_once!("{e}.");
409                    text_flags.needs_recompute = false;
410                    text_layout_info.clear();
411                }
412                Err(
413                    e @ (TextError::FailedToAddGlyph(_)
414                    | TextError::MissingAtlasLayout
415                    | TextError::MissingAtlasTexture
416                    | TextError::InconsistentAtlasState),
417                ) => {
418                    panic!("Fatal error when processing text: {e}.");
419                }
420                Ok(()) => {
421                    text_layout_info.scale_factor = node.inverse_scale_factor().recip();
422                    text_layout_info.size *= node.inverse_scale_factor();
423                    text_flags.needs_recompute = false;
424                }
425            }
426        }
427    }
