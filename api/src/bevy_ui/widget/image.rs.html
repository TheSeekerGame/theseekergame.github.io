14/// A UI Node that renders an image.
15#[derive(Component, Clone, Debug, Reflect, FromTemplate)]
16#[reflect(Component, Default, Debug, Clone)]
17#[require(Node, ImageNodeSize)]
18pub struct ImageNode {
19    /// The tint color used to draw the image.
20    ///
21    /// This is multiplied by the color of each pixel in the image.
22    /// The field value defaults to solid white, which will pass the image through unmodified.
23    pub color: Color,
24    /// Handle to the texture.
25    ///
26    /// This defaults to a [`TRANSPARENT_IMAGE_HANDLE`], which points to a fully transparent 1x1 texture.
27    pub image: Handle<Image>,
28    /// The (optional) texture atlas used to render the image.
29    #[template(built_in)]
30    pub texture_atlas: Option<TextureAtlas>,
31    /// Whether the image should be flipped along its x-axis.
32    pub flip_x: bool,
33    /// Whether the image should be flipped along its y-axis.
34    pub flip_y: bool,
35    /// An optional rectangle representing the region of the image to render, instead of rendering
36    /// the full image. This is an easy one-off alternative to using a [`TextureAtlas`].
37    ///
38    /// When used with a [`TextureAtlas`], the rect
39    /// is offset by the atlas's minimal (top-left) corner position.
40    pub rect: Option<Rect>,
41    /// Controls how the image is altered to fit within the layout and how the layout algorithm determines the space to allocate for the image.
42    pub image_mode: NodeImageMode,
43    /// Which region of the UI node the image should be drawn within.
44    pub visual_box: VisualBox,
45}

47impl Default for ImageNode {
48    /// A transparent 1x1 image with a solid white tint.
49    ///
50    /// # Warning
51    ///
52    /// This will be invisible by default.
53    /// To set this to a visible image, you need to set the `texture` field to a valid image handle,
54    /// or use [`Handle<Image>`]'s default 1x1 solid white texture (as is done in [`ImageNode::solid_color`]).
55    fn default() -> Self {
56        ImageNode {
57            // This should be white because the tint is multiplied with the image,
58            // so if you set an actual image with default tint you'd want its original colors
59            color: Color::WHITE,
60            texture_atlas: None,
61            // This texture needs to be transparent by default, to avoid covering the background color
62            image: TRANSPARENT_IMAGE_HANDLE,
63            flip_x: false,
64            flip_y: false,
65            rect: None,
66            image_mode: NodeImageMode::Auto,
67            visual_box: VisualBox::ContentBox,
68        }
69    }
70}

155/// Controls how the image is altered to fit within the layout and how the layout algorithm determines the space in the layout for the image
156#[derive(Default, Debug, Clone, PartialEq, Reflect)]
157#[reflect(Clone, Default, PartialEq)]
158pub enum NodeImageMode {
159    /// The image will be sized automatically by taking the size of the source image and applying any layout constraints.
160    #[default]
161    Auto,
162    /// The image will be resized to match the size of the node. The image's original size and aspect ratio will be ignored.
163    Stretch,
164    /// The texture will be cut in 9 slices, keeping the texture in proportions on resize
165    Sliced(TextureSlicer),
166    /// The texture will be repeated if stretched beyond `stretched_value`
167    Tiled {
168        /// Should the image repeat horizontally
169        tile_x: bool,
170        /// Should the image repeat vertically
171        tile_y: bool,
172        /// The texture will repeat when the ratio between the *drawing dimensions* of texture and the
173        /// *original texture size* are above this value.
174        stretch_value: f32,
175    },
176}
177
178impl NodeImageMode {
179    /// Returns true if this mode uses slices internally ([`NodeImageMode::Sliced`] or [`NodeImageMode::Tiled`])
180    #[inline]
181    pub const fn uses_slices(&self) -> bool {
182        matches!(
183            self,
184            NodeImageMode::Sliced(..) | NodeImageMode::Tiled { .. }
185        )
186    }
187}

213    pub size: Vec2,
214    /// The region of the UI node containing the image
215    pub visual_box: VisualBox,
216}

218impl Measure for ImageMeasure {
219    fn measure(&mut self, measure_args: MeasureArgs) -> Vec2 {
220        let mut width = measure_args.resolve_width();
221        let mut height = measure_args.resolve_height();
222
223        let calc = |_, _| 0.;
224        let padding = measure_args.style.padding.resolve_or_zero(
225            taffy::Size {
226                width: width.effective,
227                height: height.effective,
228            },
229            calc,
230        );
231        let border = measure_args.style.border.resolve_or_zero(
232            taffy::Size {
233                width: width.effective,
234                height: height.effective,
235            },
236            calc,
237        );
238        let content_inset = Vec2::new(
239            padding.left + padding.right + border.left + border.right,
240            padding.top + padding.bottom + border.top + border.bottom,
241        );
242
243        if measure_args.style.box_sizing == taffy::style::BoxSizing::BorderBox {
244            width.min = width.min.map(|min| (min - content_inset.x).max(0.));
245            width.preferred = width
246                .preferred
247                .map(|preferred| (preferred - content_inset.x).max(0.));
248            width.max = width.max.map(|max| (max - content_inset.x).max(0.));
249            width.effective = width
250                .effective
251                .map(|effective| (effective - content_inset.x).max(0.));
252
253            height.min = height.min.map(|min| (min - content_inset.y).max(0.));
254            height.preferred = height
255                .preferred
256                .map(|preferred| (preferred - content_inset.y).max(0.));
257            height.max = height.max.map(|max| (max - content_inset.y).max(0.));
258            height.effective = height
259                .effective
260                .map(|effective| (effective - content_inset.y).max(0.));
261        }
262
263        let inset = match self.visual_box {
264            VisualBox::ContentBox => Vec2::ZERO,
265            VisualBox::PaddingBox => {
266                Vec2::new(padding.left + padding.right, padding.top + padding.bottom)
267            }
268            VisualBox::BorderBox => Vec2::new(content_inset.x, content_inset.y),
269        };
270
271        let width = ResolvedAxis {
272            min: width.min.map(|min| min + inset.x),
273            preferred: width.preferred.map(|preferred| preferred + inset.x),
274            max: width.max.map(|max| max + inset.x),
275            effective: width.effective.map(|effective| effective + inset.x),
276        };
277        let height = ResolvedAxis {
278            min: height.min.map(|min| min + inset.y),
279            preferred: height.preferred.map(|preferred| preferred + inset.y),
280            max: height.max.map(|max| max + inset.y),
281            effective: height.effective.map(|effective| effective + inset.y),
282        };
283
284        // Use aspect_ratio from style, fall back to inherent aspect ratio
285        let aspect_ratio = measure_args
286            .style
287            .aspect_ratio
288            .unwrap_or_else(|| self.size.x / self.size.y);
289
290        // Apply aspect ratio
291        // If only one of width or height was determined at this point, then the other is set beyond this point using the aspect ratio.
292        let taffy_size = taffy::Size {
293            width: width.effective,
294            height: height.effective,
295        }
296        .maybe_apply_aspect_ratio(Some(aspect_ratio));
297
298        (Vec2::new(
299            taffy_size
300                .width
301                .unwrap_or(self.size.x)
302                .maybe_clamp(width.min, width.max),
303            taffy_size
304                .height
305                .unwrap_or(self.size.y)
306                .maybe_clamp(height.min, height.max),
307        ) - inset)
308            .max(Vec2::ZERO)
309    }
310}

314/// Updates content size of the node based on the image provided
315pub fn update_image_content_size_system(
316    textures: Res<Assets<Image>>,
317    atlases: Res<Assets<TextureAtlasLayout>>,
318    mut query: Query<
319        (
320            &mut ContentSize,
321            Ref<ImageNode>,
322            &mut ImageNodeSize,
323            Ref<ComputedUiRenderTargetInfo>,
324        ),
325        UpdateImageFilter,
326    >,
327) {
328    for (mut content_size, image, mut image_size, computed_target) in &mut query {
329        if !matches!(image.image_mode, NodeImageMode::Auto)
330            || image.image.id() == TRANSPARENT_IMAGE_HANDLE.id()
331        {
332            if image.is_changed() {
333                // Remove any existing measure.
334                content_size.clear();
335            }
336            continue;
337        }
338
339        if let Some(size) =
340            image
341                .rect
342                .map(|rect| rect.size().as_uvec2())
343                .or_else(|| match &image.texture_atlas {
344                    Some(atlas) => atlas.texture_rect(&atlases).map(|t| t.size()),
345                    None => textures.get(&image.image).map(Image::size),
346                })
347        {
348            // Update only if size or scale factor has changed to avoid needless layout calculations
349            if size != image_size.size
350                || computed_target.is_changed()
351                || content_size.is_added()
352                || image.is_changed()
353            {
354                image_size.size = size;
355                content_size.set(NodeMeasure::Image(ImageMeasure {
356                    // multiply the image size by the scale factor to get the physical size
357                    size: size.as_vec2() * computed_target.scale_factor(),
358                    visual_box: image.visual_box,
359                }));
360            }
361        }
362    }
