31/// Component used to render a [`RenderTarget`]  to a node.
32///
33/// # See Also
34///
35/// [`update_viewport_render_target_size`]
36#[derive(Component, Debug, Clone, Copy, Reflect)]
37#[reflect(Component, Debug)]
38#[require(Node)]
39#[cfg_attr(
40    feature = "bevy_picking",
41    require(PointerId::Custom(uuid::Uuid::new_v4()))
42)]
43pub struct ViewportNode {
44    /// The entity representing the [`Camera`] associated with this viewport.
45    ///
46    /// Note: Removing the [`ViewportNode`] component will not despawn this
47    /// entity.
48    ///
49    /// Note: Despawning the camera entity will leave a viewport node with an
50    /// invalid camera. It will automatically be set to none when
51    /// [`update_viewport_render_target_size`] runs next.
52    pub camera: Option<Entity>,
53}
54
55impl ViewportNode {
56    /// Creates a new [`ViewportNode`] with a given `camera`.
57    #[inline]
58    pub const fn new(camera: Entity) -> Self {
59        Self {
60            camera: Some(camera),
61        }
62    }
63}

66/// Handles viewport picking logic.
67///
68/// Viewport entities that are being hovered or dragged will have all pointer inputs sent to them.
69pub fn viewport_picking(
70    mut commands: Commands,
71    mut viewport_query: Query<(
72        Entity,
73        &mut ViewportNode,
74        &PointerId,
75        &mut PointerLocation,
76        &ComputedNode,
77        &UiGlobalTransform,
78    )>,
79    camera_query: Query<(&Camera, &RenderTarget)>,
80    hover_map: Res<HoverMap>,
81    pointer_state: Res<PointerState>,
82    mut pointer_inputs: MessageReader<PointerInput>,
83) {
84    use bevy_camera::NormalizedRenderTarget;
85    use bevy_math::Rect;
86    // Handle hovered entities.
87    let mut viewport_picks: EntityHashMap<PointerId> = hover_map
88        .iter()
89        .flat_map(|(hover_pointer_id, hits)| {
90            hits.iter()
91                .filter(|(entity, _)| viewport_query.contains(**entity))
92                .map(|(entity, _)| (*entity, *hover_pointer_id))
93        })
94        .collect();
95
96    // Handle dragged entities, which need to be considered for dragging in and out of viewports.
97    for ((pointer_id, _), pointer_state) in pointer_state.pointer_buttons.iter() {
98        for &target in pointer_state
99            .dragging
100            .keys()
101            .filter(|&entity| viewport_query.contains(*entity))
102        {
103            viewport_picks.insert(target, *pointer_id);
104        }
105    }
106
107    for (
108        viewport_entity,
109        mut viewport,
110        &viewport_pointer_id,
111        mut viewport_pointer_location,
112        computed_node,
113        global_transform,
114    ) in &mut viewport_query
115    {
116        let Some(pick_pointer_id) = viewport_picks.get(&viewport_entity) else {
117            // Lift the viewport pointer if it's not being used.
118            viewport_pointer_location.location = None;
119            continue;
120        };
121        let Some(camera_entity) = viewport.camera else {
122            continue;
123        };
124        let Ok((camera, render_target)) = camera_query.get(camera_entity) else {
125            viewport.camera = None;
126            continue;
127        };
128        let Some(cam_viewport_size) = camera.logical_viewport_size() else {
129            continue;
130        };
131
132        // Create a `Rect` in *physical* coordinates centered at the node's GlobalTransform
133        let node_rect =
134            Rect::from_center_size(global_transform.translation.trunc(), computed_node.size());
135        // Location::position uses *logical* coordinates
136        let top_left = node_rect.min * computed_node.inverse_scale_factor();
137        let logical_size = computed_node.size() * computed_node.inverse_scale_factor();
138
139        let Some(target) = render_target.as_image() else {
140            continue;
141        };
142
143        for input in pointer_inputs
144            .read()
145            .filter(|input| &input.pointer_id == pick_pointer_id)
146        {
147            let local_position = (input.location.position - top_left) / logical_size;
148            let position = local_position * cam_viewport_size;
149
150            let location = Location {
151                position,
152                target: NormalizedRenderTarget::Image(target.clone().into()),
153            };
154            viewport_pointer_location.location = Some(location.clone());
155
156            commands.write_message(PointerInput {
157                location,
158                pointer_id: viewport_pointer_id,
159                action: input.action,
160            });
161        }
162    }
163}

165/// Updates the size of the associated render target for viewports when the node size changes.
166pub fn update_viewport_render_target_size(
167    mut viewport_query: Query<
168        (&mut ViewportNode, &ComputedNode),
169        Or<(Changed<ComputedNode>, Changed<ViewportNode>)>,
170    >,
171    camera_query: Query<&RenderTarget>,
172    mut images: ResMut<Assets<Image>>,
173) {
174    for (mut viewport, computed_node) in &mut viewport_query {
175        let Some(camera) = viewport.camera else {
176            continue;
177        };
178        let Ok(render_target) = camera_query.get(camera) else {
179            viewport.camera = None;
180            continue;
181        };
182        let size = computed_node.size();
183
184        let Some(image_handle) = render_target.as_image() else {
185            continue;
186        };
187        let size = size.as_uvec2().max(UVec2::ONE).to_extents();
188        images.get_mut(image_handle).unwrap().resize(size);
189    }
