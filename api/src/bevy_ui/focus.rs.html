25/// Describes what type of input interaction has occurred for a UI node.
26///
27/// This is commonly queried with a `Changed<Interaction>` filter.
28///
29/// Updated in [`ui_focus_system`].
30///
31/// If a UI node has both [`Interaction`] and [`InheritedVisibility`] components,
32/// [`Interaction`] will always be [`Interaction::None`]
33/// when [`InheritedVisibility::get()`] is false.
34/// This ensures that hidden UI nodes are not interactable,
35/// and do not end up stuck in an active state if hidden at the wrong time.
36///
37/// Note that you can also control the visibility of a node using the [`Display`](crate::ui_node::Display) property,
38/// which fully collapses it during layout calculations.
39///
40/// # See also
41///
42/// - [`Button`](crate::widget::Button) which requires this component
43/// - [`RelativeCursorPosition`] to obtain the position of the cursor relative to current node
44#[derive(Component, Copy, Clone, Eq, PartialEq, Debug, Reflect)]
45#[reflect(Component, Default, PartialEq, Debug, Clone)]
46#[cfg_attr(
47    feature = "serialize",
48    derive(serde::Serialize, serde::Deserialize),
49    reflect(Serialize, Deserialize)
50)]
51pub enum Interaction {
52    /// The node has been pressed.
53    ///
54    /// Note: This does not capture click/press-release action.
55    Pressed,
56    /// The node has been hovered over
57    Hovered,
58    /// Nothing has happened
59    None,
60}
61
62impl Interaction {
63    const DEFAULT: Self = Self::None;
64}
65
66impl Default for Interaction {
67    fn default() -> Self {
68        Self::DEFAULT
69    }
70}

72/// A component storing the position of the mouse relative to the node, (0., 0.) being the center and (0.5, 0.5) being the bottom-right
73/// If the mouse is not over the node, the value will go beyond the range of (-0.5, -0.5) to (0.5, 0.5)
74///
75/// It can be used alongside [`Interaction`] to get the position of the press.
76///
77/// The component is updated when it is in the same entity with [`Node`].
78#[derive(Component, Copy, Clone, Default, PartialEq, Debug, Reflect)]
79#[reflect(Component, Default, PartialEq, Debug, Clone)]
80#[cfg_attr(
81    feature = "serialize",
82    derive(serde::Serialize, serde::Deserialize),
83    reflect(Serialize, Deserialize)
84)]
85pub struct RelativeCursorPosition {
86    /// True if the cursor position is over an unclipped area of the Node.
87    pub cursor_over: bool,
88    /// Cursor position relative to the size and position of the Node.
89    /// A None value indicates that the cursor position is unknown.
90    pub normalized: Option<Vec2>,
91}
92
93impl RelativeCursorPosition {
94    /// A helper function to check if the mouse is over the node
95    pub fn cursor_over(&self) -> bool {
96        self.cursor_over
97    }
98}

100/// Describes whether the node should block interactions with lower nodes
101#[derive(Component, Copy, Clone, Eq, PartialEq, Debug, Reflect)]

108pub enum FocusPolicy {
109    /// Blocks interaction
110    Block,
111    /// Lets interaction pass through
112    Pass,
113}

117}

145/// The system that sets Interaction for all UI elements based on the mouse cursor activity
146///
147/// Entities with a hidden [`InheritedVisibility`] are always treated as released.
148pub fn ui_focus_system(
149    mut hovered_nodes: Local<Vec<Entity>>,
150    mut state: Local<State>,
151    camera_query: Query<(Entity, &Camera, &RenderTarget)>,
152    primary_window: Query<Entity, With<PrimaryWindow>>,
153    windows: Query<&Window>,
154    mouse_button_input: Res<ButtonInput<MouseButton>>,
155    touches_input: Res<Touches>,
156    ui_stack: Res<UiStack>,
157    mut node_query: Query<NodeQuery>,
158    clipping_query: Query<(&ComputedNode, &UiGlobalTransform, &Node)>,
159    child_of_query: Query<&ChildOf, Without<OverrideClip>>,
160) {
161    let primary_window = primary_window.iter().next();
162
163    // reset entities that were both clicked and released in the last frame
164    for entity in state.entities_to_reset.drain(..) {
165        if let Ok(NodeQueryItem {
166            interaction: Some(mut interaction),
167            ..
168        }) = node_query.get_mut(entity)
169        {
170            *interaction = Interaction::None;
171        }
172    }
173
174    let mouse_released =
175        mouse_button_input.just_released(MouseButton::Left) || touches_input.any_just_released();
176    if mouse_released {
177        for node in &mut node_query {
178            if let Some(mut interaction) = node.interaction
179                && *interaction == Interaction::Pressed
180            {
181                *interaction = Interaction::None;
182            }
183        }
184    }
185
186    let mouse_clicked =
187        mouse_button_input.just_pressed(MouseButton::Left) || touches_input.any_just_pressed();
188
189    let camera_cursor_positions: EntityHashMap<Vec2> = camera_query
190        .iter()
191        .filter_map(|(entity, camera, render_target)| {
192            // Interactions are only supported for cameras rendering to a window.
193            let Some(NormalizedRenderTarget::Window(window_ref)) =
194                render_target.normalize(primary_window)
195            else {
196                return None;
197            };
198            let window = windows.get(window_ref.entity()).ok()?;
199
200            let viewport_position = camera
201                .physical_viewport_rect()
202                .map(|rect| rect.min.as_vec2())
203                .unwrap_or_default();
204            window
205                .physical_cursor_position()
206                .or_else(|| {
207                    touches_input
208                        .first_pressed_position()
209                        .map(|pos| pos * window.scale_factor())
210                })
211                .map(|cursor_position| (entity, cursor_position - viewport_position))
212        })
213        .collect();
214
215    // prepare an iterator that contains all the nodes that have the cursor in their rect,
216    // from the top node to the bottom one. this will also reset the interaction to `None`
217    // for all nodes encountered that are no longer hovered.
218
219    hovered_nodes.clear();
220    // reverse the iterator to traverse the tree from closest slice to furthest
221    for uinodes in ui_stack
222        .partition
223        .iter()
224        .rev()
225        .map(|range| &ui_stack.uinodes[range.clone()])
226    {
227        // Retrieve the first node and resolve its camera target.
228        // Only need to do this once per slice, as all the nodes in the slice share the same camera.
229        let Ok(root_node) = node_query.get_mut(uinodes[0]) else {
230            continue;
231        };
232
233        let Some(camera_entity) = root_node.target_camera.get() else {
234            continue;
235        };
236
237        let cursor_position = camera_cursor_positions.get(&camera_entity);
238
239        for entity in uinodes.iter().rev().cloned() {
240            let Ok(node) = node_query.get_mut(entity) else {
241                continue;
242            };
243
244            let Some(inherited_visibility) = node.inherited_visibility else {
245                continue;
246            };
247
248            // Nodes that are not rendered should not be interactable
249            if !inherited_visibility.get() {
250                // Reset their interaction to None to avoid strange stuck state
251                if let Some(mut interaction) = node.interaction {
252                    // We cannot simply set the interaction to None, as that will trigger change detection repeatedly
253                    interaction.set_if_neq(Interaction::None);
254                }
255                continue;
256            }
257
258            let contains_cursor = cursor_position.is_some_and(|point| {
259                node.node.contains_point(*node.transform, *point)
260                    && clip_check_recursive(*point, entity, &clipping_query, &child_of_query)
261            });
262
263            // The mouse position relative to the node
264            // (-0.5, -0.5) is the top-left corner, (0.5, 0.5) is the bottom-right corner
265            // Coordinates are relative to the entire node, not just the visible region.
266            let normalized_cursor_position = cursor_position.and_then(|cursor_position| {
267                // ensure node size is non-zero in all dimensions, otherwise relative position will be
268                // +/-inf. if the node is hidden, the visible rect min/max will also be -inf leading to
269                // false positives for mouse_over (#12395)
270                node.node.normalize_point(*node.transform, *cursor_position)
271            });
272
273            // If the current cursor position is within the bounds of the node's visible area, consider it for
274            // clicking
275            let relative_cursor_position_component = RelativeCursorPosition {
276                cursor_over: contains_cursor,
277                normalized: normalized_cursor_position,
278            };
279
280            // Save the relative cursor position to the correct component
281            if let Some(mut node_relative_cursor_position_component) = node.relative_cursor_position
282            {
283                // Avoid triggering change detection when not necessary.
284                node_relative_cursor_position_component
285                    .set_if_neq(relative_cursor_position_component);
286            }
287
288            if contains_cursor {
289                hovered_nodes.push(entity);
290            } else {
291                if let Some(mut interaction) = node.interaction
292                    && (*interaction == Interaction::Hovered
293                        || (normalized_cursor_position.is_none()))
294                {
295                    interaction.set_if_neq(Interaction::None);
296                }
297                continue;
298            }
299        }
300    }
301
302    // set Pressed or Hovered on top nodes. as soon as a node with a `Block` focus policy is detected,
303    // the iteration will stop on it because it "captures" the interaction.
304    let mut hovered_nodes = hovered_nodes.iter();
305    let mut iter = node_query.iter_many_mut(hovered_nodes.by_ref());
306    while let Some(node) = iter.fetch_next() {
307        if let Some(mut interaction) = node.interaction {
308            if mouse_clicked {
309                // only consider nodes with Interaction "pressed"
310                if *interaction != Interaction::Pressed {
311                    *interaction = Interaction::Pressed;
312                    // if the mouse was simultaneously released, reset this Interaction in the next
313                    // frame
314                    if mouse_released {
315                        state.entities_to_reset.push(node.entity);
316                    }
317                }
318            } else if *interaction == Interaction::None {
319                *interaction = Interaction::Hovered;
320            }
321        }
322
323        match node.focus_policy.unwrap_or(&FocusPolicy::Block) {
324            FocusPolicy::Block => {
325                break;
326            }
327            FocusPolicy::Pass => { /* allow the next node to be hovered/pressed */ }
328        }
329    }
330    // reset `Interaction` for the remaining lower nodes to `None`. those are the nodes that remain in
331    // `moused_over_nodes` after the previous loop is exited.
332    let mut iter = node_query.iter_many_mut(hovered_nodes);
333    while let Some(node) = iter.fetch_next() {
334        if let Some(mut interaction) = node.interaction {
335            // don't reset pressed nodes because they're handled separately
336            if *interaction != Interaction::Pressed {
337                interaction.set_if_neq(Interaction::None);
338            }
339        }
340    }
341}

343/// Walk up the tree child-to-parent checking that `point` is not clipped by any ancestor node.
344/// If `entity` has an [`OverrideClip`] component it ignores any inherited clipping and returns true.
345pub fn clip_check_recursive(
346    point: Vec2,
347    entity: Entity,
348    clipping_query: &Query<'_, '_, (&ComputedNode, &UiGlobalTransform, &Node)>,
349    child_of_query: &Query<&ChildOf, Without<OverrideClip>>,
350) -> bool {
351    let Ok(child_of) = child_of_query.get(entity) else {
352        // Reached root, point unclipped by all ancestors
353        return true;
354    };
355    if let Ok((computed_node, transform, node)) = clipping_query.get(child_of.0)
356        && !node.overflow.is_visible()
357        && transform.try_inverse().is_none_or(|affine| {
358            !computed_node
359                .resolve_clip_rect(node.overflow, node.overflow_clip_margin)
360                .contains(affine.transform_point2(point))
361        })
362    {
363        // The point is clipped (or transform not invertible) → ignore for picking
364        return false;
365    }
366
367    clip_check_recursive(point, child_of.0, clipping_query, child_of_query)
