20/// Inputs provided to [`Measure::measure`].
21pub struct MeasureArgs<'a> {
22    /// The known width from Taffy if this size is definite.
23    pub known_width: Option<f32>,
24    /// The known height from Taffy if this size is definite.
25    pub known_height: Option<f32>,
26    /// The horizontal space available for this UI node.
27    pub available_width: AvailableSpace,
28    /// The vertical space available for this UI node.
29    pub available_height: AvailableSpace,
30    /// Parley font database, needed for text measurement.
31    pub font_system: &'a mut FontCx,
32    /// Text layout buffer used to compute intrinsic text size when required.
33    pub buffer: Option<&'a mut bevy_text::ComputedTextBlock>,
34    /// Resolved style for this node in physical pixels.
35    pub style: &'a taffy::Style,
36}

39/// Resolved values for per-axis size constraints.
40pub struct ResolvedAxis {
41    /// Resolved minimum size along this axis.
42    pub min: Option<f32>,
43    /// Resolved preferred size along this axis.
44    pub preferred: Option<f32>,
45    /// Resolved maximum size along this axis.
46    pub max: Option<f32>,
47    /// Effective size along this axis after applying known size and min/max clamping.
48    pub effective: Option<f32>,
49}
50
51fn resolve_axis(
52    known_size: Option<f32>,
53    available_space: AvailableSpace,
54    min_dim: taffy::style::Dimension,
55    size_dim: taffy::style::Dimension,
56    max_dim: taffy::style::Dimension,
57) -> ResolvedAxis {
58    let calc = |_, _| 0.;
59    let available = available_space.into_option();
60    let min = min_dim.maybe_resolve(available, calc);
61    let preferred = size_dim.maybe_resolve(available, calc);
62    let max = max_dim.maybe_resolve(available, calc);
63    ResolvedAxis {
64        min,
65        preferred,
66        max,
67        effective: known_size.or(preferred.or(min).maybe_clamp(min, max)),
68    }
69}

71impl MeasureArgs<'_> {
72    /// Resolve the node's width constraints and the effective width.
73    pub fn resolve_width(&self) -> ResolvedAxis {
74        resolve_axis(
75            self.known_width,
76            self.available_width,
77            self.style.min_size.width,
78            self.style.size.width,
79            self.style.max_size.width,
80        )
81    }

83    /// Resolve the node's height constraints and the effective height.
84    pub fn resolve_height(&self) -> ResolvedAxis {
85        resolve_axis(

87            self.available_height,
88            self.style.min_size.height,
89            self.style.size.height,
90            self.style.max_size.height,
91        )
92    }

95/// A `Measure` is used to compute the size of a ui node
96/// when the size of that node is based on its content.
97pub trait Measure: Send + Sync + 'static {
98    /// Calculate the size of the node given the constraints.
99    fn measure(&mut self, measure_args: MeasureArgs<'_>) -> Vec2;
100}
101
102/// A type to serve as Taffy's node context (which allows the content size of leaf nodes to be computed)
103///
104/// It has specific variants for common built-in types to avoid making them opaque and needing to box them
105/// by wrapping them in a closure and a Custom variant that allows arbitrary measurement closures if required.
106pub enum NodeMeasure {
107    Fixed(FixedMeasure),
108    Text(TextMeasure),
109    Image(ImageMeasure),
110    Custom(Box<dyn Measure>),
111}
112
113impl Measure for NodeMeasure {
114    fn measure(&mut self, measure_args: MeasureArgs) -> Vec2 {
115        match self {
116            NodeMeasure::Fixed(fixed) => fixed.measure(measure_args),
117            NodeMeasure::Text(text) => text.measure(measure_args),
118            NodeMeasure::Image(image) => image.measure(measure_args),
119            NodeMeasure::Custom(custom) => custom.measure(measure_args),
120        }
121    }
122}
123
124/// A `FixedMeasure` is a `Measure` that ignores all constraints and
125/// always returns the same size.
126#[derive(Default, Clone)]
127pub struct FixedMeasure {
128    pub size: Vec2,
129}
130
131impl Measure for FixedMeasure {
132    fn measure(&mut self, _: MeasureArgs) -> Vec2 {
133        self.size
134    }
135}

137/// A node with a `ContentSize` component is a node where its size
138/// is based on its content.
139#[derive(Component, Reflect, Default)]
140#[reflect(Component, Default)]
141pub struct ContentSize {
142    /// The `Measure` used to compute the intrinsic size
143    #[reflect(ignore)]
144    pub(crate) measure: Option<NodeMeasure>,
145}
146
147impl ContentSize {
148    /// Set a `Measure` for the UI node entity with this component
149    pub fn set(&mut self, measure: NodeMeasure) {
150        self.measure = Some(measure);
151    }
152
153    /// Clear the current `Measure` for this UI node.
154    pub fn clear(&mut self) {
155        self.measure = None;
156    }
157
158    /// Creates a `ContentSize` with a `Measure` that always returns given `size` argument, regardless of the UI layout's constraints.
159    pub fn fixed_size(size: Vec2) -> ContentSize {
160        let mut content_size = Self::default();
161        content_size.set(NodeMeasure::Fixed(FixedMeasure { size }));
162        content_size
163    }
