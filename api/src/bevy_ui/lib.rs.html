93/// The label enum labeling the types of systems in the Bevy UI
94#[derive(Debug, Hash, PartialEq, Eq, Clone, SystemSet)]
95pub enum UiSystems {
96    /// After this label, input interactions with UI entities have been updated for this frame.
97    ///
98    /// Runs in [`PreUpdate`].
99    Focus,
100    /// All UI systems in [`PostUpdate`] will run in or after this label.
101    Prepare,
102    /// Propagate UI component values needed by layout.
103    Propagate,
104    /// Update content requirements before layout.
105    Content,
106    /// After this label, the ui layout state has been updated.
107    ///
108    /// Runs in [`PostUpdate`].
109    Layout,
110    /// UI systems ordered after [`UiSystems::Layout`].
111    ///
112    /// Runs in [`PostUpdate`].
113    PostLayout,
114    /// After this label, the [`UiStack`] resource has been updated.
115    ///
116    /// Runs in [`PostUpdate`].
117    Stack,
118}

142impl Plugin for UiPlugin {
143    fn build(&self, app: &mut App) {
144        app.init_resource::<UiSurface>()
145            .init_resource::<UiScale>()
146            .init_resource::<UiStack>()
147            .configure_sets(
148                PostUpdate,
149                (
150                    CameraUpdateSystems,
151                    UiSystems::Prepare.after(AnimationSystems),
152                    UiSystems::Propagate,
153                    UiSystems::Content,
154                    UiSystems::Layout,
155                    UiSystems::PostLayout,
156                )
157                    .chain(),
158            )
159            .configure_sets(
160                PostUpdate,
161                PropagateSet::<ComputedUiTargetCamera>::default().in_set(UiSystems::Propagate),
162            )
163            .add_plugins(HierarchyPropagatePlugin::<ComputedUiTargetCamera>::new(
164                PostUpdate,
165            ))
166            .configure_sets(
167                PostUpdate,
168                PropagateSet::<ComputedUiRenderTargetInfo>::default().in_set(UiSystems::Propagate),
169            )
170            .add_plugins(HierarchyPropagatePlugin::<ComputedUiRenderTargetInfo>::new(
171                PostUpdate,
172            ))
173            .add_systems(
174                PreUpdate,
175                ui_focus_system.in_set(UiSystems::Focus).after(InputSystems),
176            );
177
178        #[cfg(feature = "bevy_picking")]
179        app.add_plugins(picking_backend::UiPickingPlugin)
180            .add_systems(
181                First,
182                widget::viewport_picking.in_set(PickingSystems::PostInput),
183            );
184
185        ui_layout_system
186            .in_set(UiSystems::Layout)
187            .before(TransformSystems::Propagate)
188            // Text and Text2D operate on disjoint sets of entities
189            .ambiguous_with(bevy_sprite::update_text2d_layout);
190
191        app.add_systems(
192            PostUpdate,
193            (
194                propagate_ui_target_cameras
195                    .in_set(UiSystems::Prepare)
196                    .before(bevy_app::TransformGizmoRenderStep),
197                ui_layout_system
198                    .in_set(UiSystems::Layout)
199                    .ambiguous_with(bevy_sprite::update_text2d_layout),
200                ui_stack_system.in_set(UiSystems::Stack),
201                update_clipping_system.in_set(UiSystems::PostLayout),
202                // Potential conflicts: `Assets<Image>`
203                // They run independently since `widget::image_node_system` will only ever observe
204                // its own ImageNode, and `widget::text_system` & `bevy_text::update_text2d_layout`
205                // will never modify a pre-existing `Image` asset.
206                widget::update_image_content_size_system
207                    .in_set(UiSystems::Content)
208                    .in_set(AmbiguousWithText)
209                    .in_set(AmbiguousWithUpdateText2dLayout),
210                // Potential conflicts: `Assets<Image>`
211                // `widget::text_system` and `bevy_text::update_text2d_layout` run independently
212                // since this system will only ever update viewport images.
213                widget::update_viewport_render_target_size
214                    .in_set(UiSystems::PostLayout)
215                    .in_set(AmbiguousWithText)
216                    .in_set(AmbiguousWithUpdateText2dLayout),
217            ),
218        );
219
220        app.add_plugins(accessibility::AccessibilityPlugin);
221
222        app.add_observer(interaction_states::on_add_disabled)
223            .add_observer(interaction_states::on_remove_disabled)
224            .add_observer(interaction_states::on_add_checkable)
225            .add_observer(interaction_states::on_remove_checkable)
226            .add_observer(interaction_states::on_add_checked)
227            .add_observer(interaction_states::on_remove_checked)
228            .add_observer(interaction_states::on_add_selectable)
229            .add_observer(interaction_states::on_remove_selectable)
230            .add_observer(interaction_states::on_add_selected)
231            .add_observer(interaction_states::on_remove_selected);
232
233        build_text_interop(app);
234    }
235}

237fn build_text_interop(app: &mut App) {
238    app.add_systems(
239        PostUpdate,
240        (
241            widget::measure_text_system
242                .chain()
243                .after(detect_text_needs_rerender)
244                .after(bevy_text::load_font_assets_into_font_collection)
245                .in_set(UiSystems::Content)
246                // Text and Text2d are independent.
247                // Potential conflict: `Assets<Image>`
248                // Since both systems will only ever insert new [`Image`] assets,
249                // they will never observe each other's effects.
250                .ambiguous_with(bevy_sprite::update_text2d_layout)
251                // We assume Text is on disjoint UI entities to ImageNode and UiTextureAtlasImage
252                // FIXME: Add an archetype invariant for this https://github.com/bevyengine/bevy/issues/1481.
253                .ambiguous_with(widget::update_image_content_size_system)
254                .ambiguous_with(EditableTextSystems),
255            widget::text_system
256                .in_set(UiSystems::PostLayout)
257                .after(bevy_text::load_font_assets_into_font_collection)
258                .before(bevy_asset::AssetEventSystems)
259                // Text2d and bevy_ui text are entirely on separate entities
260                .ambiguous_with(bevy_sprite::update_text2d_layout)
261                .ambiguous_with(bevy_sprite::calculate_bounds_text2d),
262            (
263                widget::update_editable_text_content_size,
264                widget::update_editable_text_styles,
265            )
266                .chain()
267                .in_set(UiSystems::Content)
268                .after(bevy_text::load_font_assets_into_font_collection)
269                .before(EditableTextSystems)
270                .ambiguous_with(widget::update_image_content_size_system)
271                .ambiguous_with(widget::measure_text_system)
272                .ambiguous_with(bevy_sprite::update_text2d_layout),
273            (
274                widget::update_editable_text_layout.before(bevy_asset::AssetEventSystems),
275                widget::scroll_editable_text,
276            )
277                .chain()
278                .in_set(UiSystems::PostLayout)
279                // This is unlikely to result in real conflicts,
280                // as FocusChangeEvents only mutates internal state of InputFocus,
281                // and editable_text_system only reads from it.
282                // However, in case this changes in the future, this is a safer choice,
283                // as editable_text_system or related systems could generate focus changes
284                // which should be processed ASAP.
285                .before(bevy_input_focus::InputFocusSystems::FocusChangeEvents)
286                .ambiguous_with(widget::text_system)
287                .ambiguous_with(bevy_sprite::update_text2d_layout)
288                .ambiguous_with(bevy_sprite::calculate_bounds_text2d),
289        ),
290    );
291
292    app.configure_sets(
293        PostUpdate,
294        AmbiguousWithText.ambiguous_with(widget::text_system),
295    );
296
297    app.configure_sets(
298        PostUpdate,
299        AmbiguousWithUpdateText2dLayout.ambiguous_with(bevy_sprite::update_text2d_layout),
300    );
301
302    // We cannot set this up in bevy_text as this would create a circular dependency between bevy_ui and bevy_text
303    app.configure_sets(PostUpdate, EditableTextSystems.in_set(UiSystems::Content));
304}
