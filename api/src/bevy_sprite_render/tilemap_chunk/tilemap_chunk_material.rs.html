23/// Material used for rendering tilemap chunks.
24///
25/// This material is used internally by the tilemap system to render chunks of tiles
26/// efficiently using a single draw call per chunk.
27#[derive(Asset, TypePath, AsBindGroup, Debug, Clone)]
28pub struct TilemapChunkMaterial {
29    pub alpha_mode: AlphaMode2d,
30
31    #[texture(0, dimension = "2d_array")]
32    #[sampler(1)]
33    pub tileset: Handle<Image>,
34
35    #[texture(2, sample_type = "u_int")]
36    pub tile_data: Handle<Image>,
37}
38
39impl Material2d for TilemapChunkMaterial {
40    fn fragment_shader() -> ShaderRef {
41        ShaderRef::Path(
42            AssetPath::from_path_buf(embedded_path!("tilemap_chunk_material.wgsl"))
43                .with_source("embedded"),
44        )
45    }
46
47    fn alpha_mode(&self) -> AlphaMode2d {
48        self.alpha_mode
49    }
50}

52/// Packed per-tile data for use in the `Rgba16Uint` tile data texture in `TilemapChunkMaterial`.
53#[repr(C)]
54#[derive(Clone, Copy, Debug, Pod, Zeroable)]
55pub struct PackedTileData {
56    tileset_index: u16, // red channel
57    color: [u8; 4],     // green and blue channels
58    flags: u16,         // alpha channel
59}
60
61impl PackedTileData {
62    fn empty() -> Self {
63        Self {
64            tileset_index: u16::MAX,
65            color: [0, 0, 0, 0],
66            flags: 0,
67        }
68    }
69}
70
71impl From<TileData> for PackedTileData {
72    fn from(
73        TileData {
74            tileset_index,
75            color,
76            visible,
77            orientation,
78        }: TileData,
79    ) -> Self {
80        Self {
81            tileset_index,
82            color: color.to_srgba().to_u8_array(),
83            flags: (visible as u16) | ((orientation as u16) << 1),
84        }
85    }
86}
87
88impl From<Option<TileData>> for PackedTileData {
89    fn from(maybe_tile_data: Option<TileData>) -> Self {
90        maybe_tile_data
91            .map(Into::into)
92            .unwrap_or(PackedTileData::empty())
93    }
94}

96pub fn make_chunk_tile_data_image(size: &UVec2, data: &[PackedTileData]) -> Image {
97    Image {
98        data: Some(bytemuck::cast_slice(data).to_vec()),
99        data_order: TextureDataOrder::default(),
100        texture_descriptor: TextureDescriptor {
101            size: size.to_extents(),
102            dimension: TextureDimension::D2,
103            format: TextureFormat::Rgba16Uint,
104            label: None,
105            mip_level_count: 1,
106            sample_count: 1,
107            usage: TextureUsages::TEXTURE_BINDING | TextureUsages::COPY_DST,
108            view_formats: &[],
109        },
110        sampler: ImageSampler::nearest(),
111        texture_view_descriptor: None,
112        asset_usage: RenderAssetUsages::RENDER_WORLD | RenderAssetUsages::MAIN_WORLD,
113        copy_on_resize: false,
114    }
