74        // These bindings should be loaded as a shader library, but it depends on runtime

85                .init_resource::<RenderMaterial2dIds>()
86                .allow_ambiguous_resource::<RenderMaterial2dIds>()
87                .init_gpu_resource::<SpecializedMeshPipelines<Mesh2dPipeline>>()
88                .add_systems(

92                        init_batched_instance_buffer,

99                .add_systems(

126pub fn check_views_need_specialization(
127    mut view_key_cache: ResMut<ViewKeyCache>,
128    mut dirty_specializations: ResMut<DirtySpecializations>,
129    cameras: Query<(
130        &MainEntity,
131        &ExtractedView,
132        &ExtractedCamera,
133        &Msaa,
134        Option<&Tonemapping>,
135        Option<&DebandDither>,
136    )>,
137) {
138    for (view_entity, view, camera, msaa, tonemapping, dither) in &cameras {
139        let mut view_key = Mesh2dPipelineKey::from_msaa_samples(msaa.samples())
140            | Mesh2dPipelineKey::from_target_format(view.target_format);
141
142        if camera
143            .compositing_space
144            .is_some_and(|s| s == CompositingSpace::Srgb)
145        {
146            view_key |= Mesh2dPipelineKey::SRGB_COMPOSITING;
147        }
148        if camera
149            .compositing_space
150            .is_some_and(|s| s == CompositingSpace::Oklab)
151        {
152            view_key |= Mesh2dPipelineKey::OKLAB_COMPOSITING;
153        }
154
155        if !camera.hdr {
156            if let Some(tonemapping) = tonemapping {
157                view_key |= Mesh2dPipelineKey::TONEMAP_IN_SHADER;
158                view_key |= tonemapping_pipeline_key(*tonemapping);
159            }
160            if let Some(DebandDither::Enabled) = dither {
161                view_key |= Mesh2dPipelineKey::DEBAND_DITHER;
162            }
163        }
164
165        if !view_key_cache
166            .get_mut(view_entity)
167            .is_some_and(|current_key| *current_key == view_key)
168        {
169            view_key_cache.insert(*view_entity, view_key);
170            dirty_specializations
171                .views
172                .insert(view.retained_view_entity);
173        }
174    }
175}

183fn load_mesh2d_bindings(render_device: Res<RenderDevice>, asset_server: Res<AssetServer>) {
184    let mut mesh_bindings_shader_defs = Vec::with_capacity(1);
185
186    if let Some(per_object_buffer_batch_size) =
187        GpuArrayBuffer::<Mesh2dUniform>::batch_size(&render_device.limits())
188    {
189        mesh_bindings_shader_defs.push(ShaderDefVal::UInt(
190            "PER_OBJECT_BUFFER_BATCH_SIZE".into(),
191            per_object_buffer_batch_size,
192        ));
193    }
194
195    // Load the mesh_bindings shader module here as it depends on runtime information about
196    // whether storage buffers are supported, or the maximum uniform buffer binding size.
197    let handle: Handle<Shader> = load_embedded_asset!(
198        asset_server.as_ref(),
199        "mesh2d_bindings.wgsl",
200        move |settings| {
201            *settings = ShaderSettings {
202                shader_defs: mesh_bindings_shader_defs.clone(),
203            }
204        }
205    );
206    // Forget the handle so we don't have to store it anywhere, and we keep the embedded asset
207    // loaded. Note: This is what happens in `load_shader_library` internally.
208    core::mem::forget(handle);
209}

218pub struct Mesh2dUniform {
219    // Affine 4x3 matrix transposed to 3x4
220    pub world_from_local: [Vec4; 3],
221    // 3x3 matrix packed in mat2x4 and f32 as:
222    //   [0].xyz, [1].x,
223    //   [1].yz, [2].xy
224    //   [2].z
225    pub local_from_world_transpose_a: [Vec4; 2],
226    pub local_from_world_transpose_b: f32,
227    pub flags: u32,
228    pub tag: u32,
229}
230
231impl Mesh2dUniform {
232    fn from_components(mesh_transforms: &Mesh2dTransforms, tag: u32) -> Self {
233        let (local_from_world_transpose_a, local_from_world_transpose_b) =
234            mesh_transforms.world_from_local.inverse_transpose_3x3();
235        Self {
236            world_from_local: mesh_transforms.world_from_local.to_transpose(),
237            local_from_world_transpose_a,
238            local_from_world_transpose_b,
239            flags: mesh_transforms.flags,
240            tag,
241        }
242    }
243}

268pub fn extract_mesh2d(
269    mut render_mesh_instances: ResMut<RenderMesh2dInstances>,
270    render_material_2d_bind_group_ids: Res<RenderMaterial2dBindGroupIds>,
271    render_material_instances: Res<RenderMaterial2dIds>,
272    query: Extract<
273        Query<(
274            Entity,
275            &ViewVisibility,
276            &GlobalTransform,
277            &Mesh2d,
278            Option<&MeshTag>,
279            Has<NoAutomaticBatching>,
280        )>,
281    >,
282) {
283    render_mesh_instances.clear();
284
285    for (entity, view_visibility, transform, handle, tag, no_automatic_batching) in &query {
286        if !view_visibility.get() {
287            continue;
288        }
289        let main_entity = entity.into();
290        let material_bind_group_id = render_material_instances
291            .get(&main_entity)
292            .and_then(|material_id| render_material_2d_bind_group_ids.get(material_id))
293            .copied()
294            .unwrap_or_default();
295        render_mesh_instances.insert(
296            main_entity,
297            RenderMesh2dInstance {
298                transforms: Mesh2dTransforms {
299                    world_from_local: transform.affine().into(),
300                    flags: MeshFlags::empty().bits(),
301                },
302                mesh_asset_id: handle.0.id(),
303                material_bind_group_id,
304                automatic_batching: !no_automatic_batching,
305                tag: tag.map_or(0, |i| **i),
306            },
307        );
308    }
309}

319pub fn init_mesh_2d_pipeline(
320    mut commands: Commands,
321    render_device: Res<RenderDevice>,
322    asset_server: Res<AssetServer>,
323) {
324    let tonemapping_lut_entries = get_lut_bind_group_layout_entries();
325    let view_layout = BindGroupLayoutDescriptor::new(
326        "mesh2d_view_layout",
327        &BindGroupLayoutEntries::sequential(
328            ShaderStages::VERTEX_FRAGMENT,
329            (
330                uniform_buffer::<ViewUniform>(true),
331                uniform_buffer::<GlobalsUniform>(false),
332                tonemapping_lut_entries[0].visibility(ShaderStages::FRAGMENT),
333                tonemapping_lut_entries[1].visibility(ShaderStages::FRAGMENT),
334            ),
335        ),
336    );
337
338    let mesh_layout = BindGroupLayoutDescriptor::new(
339        "mesh2d_layout",
340        &BindGroupLayoutEntries::single(
341            ShaderStages::VERTEX_FRAGMENT,
342            GpuArrayBuffer::<Mesh2dUniform>::binding_layout(&render_device.limits()),
343        ),
344    );
345
346    commands.insert_resource(Mesh2dPipeline {
347        view_layout,
348        mesh_layout,
349        per_object_buffer_batch_size: GpuArrayBuffer::<Mesh2dUniform>::batch_size(
350            &render_device.limits(),
351        ),
352        shader: load_embedded_asset!(asset_server.as_ref(), "mesh2d.wgsl"),
353    });
354}

456bitflags::bitflags! {
457    #[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
458    #[repr(transparent)]
459    // NOTE: Apparently quadro drivers support up to 64x MSAA.
460    // MSAA uses the highest 3 bits for the MSAA log2(sample count) to support up to 128x MSAA.
461    // FIXME: make normals optional?
462    pub struct Mesh2dPipelineKey: u32 {
463        const NONE                              = 0;
464        const TONEMAP_IN_SHADER                 = 1 << 0;
465        const DEBAND_DITHER                     = 1 << 1;
466        const BLEND_ALPHA                       = 1 << 2;
467        const MAY_DISCARD                       = 1 << 3;
468        const SRGB_COMPOSITING                  = 1 << 4;
469        const OKLAB_COMPOSITING                 = 1 << 5;
470        const COLOR_TARGET_FORMAT_RESERVED_BITS = Self::COLOR_TARGET_FORMAT_MASK_BITS << Self::COLOR_TARGET_FORMAT_SHIFT_BITS;
471        const MSAA_RESERVED_BITS                = Self::MSAA_MASK_BITS << Self::MSAA_SHIFT_BITS;
472        const PRIMITIVE_TOPOLOGY_RESERVED_BITS  = Self::PRIMITIVE_TOPOLOGY_MASK_BITS << Self::PRIMITIVE_TOPOLOGY_SHIFT_BITS;
473        const TONEMAP_METHOD_RESERVED_BITS      = Self::TONEMAP_METHOD_MASK_BITS << Self::TONEMAP_METHOD_SHIFT_BITS;
474        const TONEMAP_METHOD_NONE               = 0 << Self::TONEMAP_METHOD_SHIFT_BITS;
475        const TONEMAP_METHOD_REINHARD           = 1 << Self::TONEMAP_METHOD_SHIFT_BITS;
476        const TONEMAP_METHOD_REINHARD_LUMINANCE = 2 << Self::TONEMAP_METHOD_SHIFT_BITS;
477        const TONEMAP_METHOD_ACES_FITTED        = 3 << Self::TONEMAP_METHOD_SHIFT_BITS;
478        const TONEMAP_METHOD_AGX                = 4 << Self::TONEMAP_METHOD_SHIFT_BITS;
479        const TONEMAP_METHOD_SOMEWHAT_BORING_DISPLAY_TRANSFORM = 5 << Self::TONEMAP_METHOD_SHIFT_BITS;
480        const TONEMAP_METHOD_TONY_MC_MAPFACE    = 6 << Self::TONEMAP_METHOD_SHIFT_BITS;
481        const TONEMAP_METHOD_BLENDER_FILMIC     = 7 << Self::TONEMAP_METHOD_SHIFT_BITS;
482        const TONEMAP_METHOD_PBR_NEUTRAL        = 8 << Self::TONEMAP_METHOD_SHIFT_BITS;
483        const STRIP_INDEX_FORMAT_RESERVED_BITS        = Self::INDEX_FORMAT_MASK_BITS << Self::INDEX_FORMAT_SHIFT_BITS;
484        const STRIP_INDEX_FORMAT_NONE                 = 0 << Self::INDEX_FORMAT_SHIFT_BITS;
485        const STRIP_INDEX_FORMAT_U32                  = 1 << Self::INDEX_FORMAT_SHIFT_BITS;
486        const STRIP_INDEX_FORMAT_U16                  = 2 << Self::INDEX_FORMAT_SHIFT_BITS;
487    }
488}

490impl Mesh2dPipelineKey {
491    const COLOR_TARGET_FORMAT_MASK_BITS: u32 = bevy_render::view::COLOR_TARGET_FORMAT_MASK_BITS;
492    const COLOR_TARGET_FORMAT_SHIFT_BITS: u32 = 6;
493    const MSAA_MASK_BITS: u32 = 0b111;
494    const MSAA_SHIFT_BITS: u32 = 32 - Self::MSAA_MASK_BITS.count_ones();
495    const PRIMITIVE_TOPOLOGY_MASK_BITS: u32 = 0b111;
496    const PRIMITIVE_TOPOLOGY_SHIFT_BITS: u32 = Self::MSAA_SHIFT_BITS - 3;
497    const TONEMAP_METHOD_MASK_BITS: u32 = 0b1111;
498    const TONEMAP_METHOD_SHIFT_BITS: u32 =
499        Self::PRIMITIVE_TOPOLOGY_SHIFT_BITS - Self::TONEMAP_METHOD_MASK_BITS.count_ones();
500    pub const INDEX_FORMAT_MASK_BITS: u32 = 0b11;
501    pub const INDEX_FORMAT_SHIFT_BITS: u32 =
502        Self::TONEMAP_METHOD_SHIFT_BITS - Self::TONEMAP_METHOD_MASK_BITS.count_ones();
503
504    pub fn from_msaa_samples(msaa_samples: u32) -> Self {
505        let msaa_bits =
506            (msaa_samples.trailing_zeros() & Self::MSAA_MASK_BITS) << Self::MSAA_SHIFT_BITS;
507        Self::from_bits_retain(msaa_bits)
508    }
509
510    /// Create a pipeline key from the view's color target format.
511    #[inline]
512    pub fn from_target_format(format: TextureFormat) -> Self {
513        let code = texture_format_to_code(format)
514            .expect("Texture format is not supported by the pipeline") as u32;
515        Self::from_bits_retain(
516            (code & Self::COLOR_TARGET_FORMAT_MASK_BITS) << Self::COLOR_TARGET_FORMAT_SHIFT_BITS,
517        )
518    }
519
520    /// Color target format of the main pass for this pipeline key.
521    #[inline]
522    pub fn target_format(&self) -> TextureFormat {
523        let code = ((self.bits() >> Self::COLOR_TARGET_FORMAT_SHIFT_BITS)
524            & Self::COLOR_TARGET_FORMAT_MASK_BITS) as u8;
525        texture_format_from_code(code)
526            .expect("Unknown bits in `COLOR_TARGET_FORMAT_MASK_BITS` of the pipeline key")
527    }
528
529    pub fn msaa_samples(&self) -> u32 {
530        1 << ((self.bits() >> Self::MSAA_SHIFT_BITS) & Self::MSAA_MASK_BITS)
531    }
532
533    /// Create a [`Mesh2dPipelineKey`] from mesh primitive topology and index format.
534    ///
535    /// For non-strip topologies, [`Self::STRIP_INDEX_FORMAT_NONE`] is set regardless of the `strip_index_format` argument.
536    pub fn from_primitive_topology_and_strip_index(
537        primitive_topology: PrimitiveTopology,
538        strip_index_format: Option<IndexFormat>,
539    ) -> Self {
540        let index_bits = if primitive_topology.is_strip() {
541            match strip_index_format {
542                None => Self::STRIP_INDEX_FORMAT_NONE,
543                Some(indices) => match indices {
544                    IndexFormat::Uint16 => Self::STRIP_INDEX_FORMAT_U16,
545                    IndexFormat::Uint32 => Self::STRIP_INDEX_FORMAT_U32,
546                },
547            }
548        } else {
549            Self::STRIP_INDEX_FORMAT_NONE
550        }
551        .bits();
552        let primitive_topology_bits = ((primitive_topology as u32)
553            & Self::PRIMITIVE_TOPOLOGY_MASK_BITS)
554            << Self::PRIMITIVE_TOPOLOGY_SHIFT_BITS;
555        Self::from_bits_retain(primitive_topology_bits | index_bits)
556    }
557
558    pub fn primitive_topology(&self) -> PrimitiveTopology {
559        let primitive_topology_bits = (self.bits() >> Self::PRIMITIVE_TOPOLOGY_SHIFT_BITS)
560            & Self::PRIMITIVE_TOPOLOGY_MASK_BITS;
561        match primitive_topology_bits {
562            x if x == PrimitiveTopology::PointList as u32 => PrimitiveTopology::PointList,
563            x if x == PrimitiveTopology::LineList as u32 => PrimitiveTopology::LineList,
564            x if x == PrimitiveTopology::LineStrip as u32 => PrimitiveTopology::LineStrip,
565            x if x == PrimitiveTopology::TriangleList as u32 => PrimitiveTopology::TriangleList,
566            x if x == PrimitiveTopology::TriangleStrip as u32 => PrimitiveTopology::TriangleStrip,
567            _ => PrimitiveTopology::default(),
568        }
569    }
570
571    pub fn strip_index_format(&self) -> Option<IndexFormat> {
572        let index_bits = self.bits() & Self::STRIP_INDEX_FORMAT_RESERVED_BITS.bits();
573        match index_bits {
574            x if x == Self::STRIP_INDEX_FORMAT_U16.bits() => Some(IndexFormat::Uint16),
575            x if x == Self::STRIP_INDEX_FORMAT_U32.bits() => Some(IndexFormat::Uint32),
576            x if x == Self::STRIP_INDEX_FORMAT_NONE.bits() => None,
577            _ => unreachable!(),
578        }
579    }
580}
581
582impl SpecializedMeshPipeline for Mesh2dPipeline {
583    type Key = Mesh2dPipelineKey;
584
585    fn specialize(
586        &self,
587        key: Self::Key,
588        layout: &MeshVertexBufferLayoutRef,
589    ) -> Result<RenderPipelineDescriptor, SpecializedMeshPipelineError> {
590        let mut shader_defs = Vec::new();
591        let mut vertex_attributes = Vec::new();
592
593        if layout.0.contains(Mesh::ATTRIBUTE_POSITION) {
594            shader_defs.push("VERTEX_POSITIONS".into());
595            vertex_attributes.push(Mesh::ATTRIBUTE_POSITION.at_shader_location(0));
596        }
597
598        if layout.0.contains(Mesh::ATTRIBUTE_NORMAL) {
599            shader_defs.push("VERTEX_NORMALS".into());
600            vertex_attributes.push(Mesh::ATTRIBUTE_NORMAL.at_shader_location(1));
601        }
602
603        if layout.0.contains(Mesh::ATTRIBUTE_UV_0) {
604            shader_defs.push("VERTEX_UVS".into());
605            vertex_attributes.push(Mesh::ATTRIBUTE_UV_0.at_shader_location(2));
606        }
607
608        if layout.0.contains(Mesh::ATTRIBUTE_TANGENT) {
609            shader_defs.push("VERTEX_TANGENTS".into());
610            vertex_attributes.push(Mesh::ATTRIBUTE_TANGENT.at_shader_location(3));
611        }
612
613        if layout.0.contains(Mesh::ATTRIBUTE_COLOR) {
614            shader_defs.push("VERTEX_COLORS".into());
615            vertex_attributes.push(Mesh::ATTRIBUTE_COLOR.at_shader_location(4));
616        }
617
618        if key.contains(Mesh2dPipelineKey::TONEMAP_IN_SHADER) {
619            shader_defs.push("TONEMAP_IN_SHADER".into());
620            shader_defs.push(ShaderDefVal::UInt(
621                "TONEMAPPING_LUT_TEXTURE_BINDING_INDEX".into(),
622                2,
623            ));
624            shader_defs.push(ShaderDefVal::UInt(
625                "TONEMAPPING_LUT_SAMPLER_BINDING_INDEX".into(),
626                3,
627            ));
628
629            let method = key.intersection(Mesh2dPipelineKey::TONEMAP_METHOD_RESERVED_BITS);
630
631            match method {
632                Mesh2dPipelineKey::TONEMAP_METHOD_NONE => {
633                    shader_defs.push("TONEMAP_METHOD_NONE".into());
634                }
635                Mesh2dPipelineKey::TONEMAP_METHOD_REINHARD => {
636                    shader_defs.push("TONEMAP_METHOD_REINHARD".into());
637                }
638                Mesh2dPipelineKey::TONEMAP_METHOD_REINHARD_LUMINANCE => {
639                    shader_defs.push("TONEMAP_METHOD_REINHARD_LUMINANCE".into());
640                }
641                Mesh2dPipelineKey::TONEMAP_METHOD_ACES_FITTED => {
642                    shader_defs.push("TONEMAP_METHOD_ACES_FITTED".into());
643                }
644                Mesh2dPipelineKey::TONEMAP_METHOD_AGX => {
645                    shader_defs.push("TONEMAP_METHOD_AGX".into());
646                }
647                Mesh2dPipelineKey::TONEMAP_METHOD_SOMEWHAT_BORING_DISPLAY_TRANSFORM => {
648                    shader_defs.push("TONEMAP_METHOD_SOMEWHAT_BORING_DISPLAY_TRANSFORM".into());
649                }
650                Mesh2dPipelineKey::TONEMAP_METHOD_BLENDER_FILMIC => {
651                    shader_defs.push("TONEMAP_METHOD_BLENDER_FILMIC".into());
652                }
653                Mesh2dPipelineKey::TONEMAP_METHOD_TONY_MC_MAPFACE => {
654                    shader_defs.push("TONEMAP_METHOD_TONY_MC_MAPFACE".into());
655                }
656                Mesh2dPipelineKey::TONEMAP_METHOD_PBR_NEUTRAL => {
657                    shader_defs.push("TONEMAP_METHOD_PBR_NEUTRAL".into());
658                }
659                _ => {}
660            }
661            // Debanding is tied to tonemapping in the shader, cannot run without it.
662            if key.contains(Mesh2dPipelineKey::DEBAND_DITHER) {
663                shader_defs.push("DEBAND_DITHER".into());
664            }
665        }
666
667        if key.contains(Mesh2dPipelineKey::MAY_DISCARD) {
668            shader_defs.push("MAY_DISCARD".into());
669        }
670        if key.contains(Mesh2dPipelineKey::SRGB_COMPOSITING) {
671            shader_defs.push("SRGB_OUTPUT".into());
672        }
673        if key.contains(Mesh2dPipelineKey::OKLAB_COMPOSITING) {
674            shader_defs.push("OKLAB_OUTPUT".into());
675        }
676
677        let vertex_buffer_layout = layout.0.get_layout(&vertex_attributes)?;
678
679        let format = key.target_format();
680
681        let (depth_write_enabled, label, blend);
682        if key.contains(Mesh2dPipelineKey::BLEND_ALPHA) {
683            label = "transparent_mesh2d_pipeline";
684            blend = Some(BlendState::ALPHA_BLENDING);
685            depth_write_enabled = false;
686        } else {
687            label = "opaque_mesh2d_pipeline";
688            blend = None;
689            depth_write_enabled = true;
690        }
691
692        Ok(RenderPipelineDescriptor {
693            vertex: VertexState {
694                shader: self.shader.clone(),
695                shader_defs: shader_defs.clone(),
696                buffers: vec![vertex_buffer_layout],
697                ..default()
698            },
699            fragment: Some(FragmentState {
700                shader: self.shader.clone(),
701                shader_defs,
702                targets: vec![Some(ColorTargetState {
703                    format,
704                    blend,
705                    write_mask: ColorWrites::ALL,
706                })],
707                ..default()
708            }),
709            layout: vec![self.view_layout.clone(), self.mesh_layout.clone()],
710            primitive: PrimitiveState {
711                front_face: FrontFace::Ccw,
712                cull_mode: None,
713                unclipped_depth: false,
714                polygon_mode: PolygonMode::Fill,
715                conservative: false,
716                topology: key.primitive_topology(),
717                strip_index_format: key.strip_index_format(),
718            },
719            depth_stencil: Some(DepthStencilState {
720                format: CORE_2D_DEPTH_FORMAT,
721                depth_write_enabled: Some(depth_write_enabled),
722                depth_compare: Some(CompareFunction::GreaterEqual),
723                stencil: StencilState {
724                    front: StencilFaceState::IGNORE,
725                    back: StencilFaceState::IGNORE,
726                    read_mask: 0,
727                    write_mask: 0,
728                },
729                bias: DepthBiasState {
730                    constant: 0,
731                    slope_scale: 0.0,
732                    clamp: 0.0,
733                },
734            }),
735            multisample: MultisampleState {
736                count: key.msaa_samples(),
737                mask: !0,
738                alpha_to_coverage_enabled: false,
739            },
740            label: Some(label.into()),
741            ..default()
742        })
743    }
744}

769#[derive(Component)]
770pub struct Mesh2dViewBindGroup {
771    pub value: BindGroup,
772}
773
774pub fn prepare_mesh2d_view_bind_groups(
775    mut commands: Commands,
776    render_device: Res<RenderDevice>,
777    pipeline_cache: Res<PipelineCache>,
778    mesh2d_pipeline: Res<Mesh2dPipeline>,
779    view_uniforms: Res<ViewUniforms>,
780    views: Query<(Entity, &Tonemapping), (With<ExtractedView>, With<Camera2d>)>,
781    globals_buffer: Res<GlobalsBuffer>,
782    tonemapping_luts: Res<TonemappingLuts>,
783    images: Res<RenderAssets<GpuImage>>,
784    fallback_image: Res<FallbackImage>,
785) {
786    let (Some(view_binding), Some(globals)) = (
787        view_uniforms.uniforms.binding(),
788        globals_buffer.buffer.binding(),
789    ) else {
790        return;
791    };
792
793    for (entity, tonemapping) in &views {
794        let lut_bindings =
795            get_lut_bindings(&images, &tonemapping_luts, tonemapping, &fallback_image);
796        let view_bind_group = render_device.create_bind_group(
797            "mesh2d_view_bind_group",
798            &pipeline_cache.get_bind_group_layout(&mesh2d_pipeline.view_layout),
799            &BindGroupEntries::sequential((
800                view_binding.clone(),
801                globals.clone(),
802                lut_bindings.0,
803                lut_bindings.1,
804            )),
805        );
806
807        commands.entity(entity).insert(Mesh2dViewBindGroup {
808            value: view_bind_group,
809        });
810    }
811}

833pub struct SetMesh2dBindGroup<const I: usize>;
834impl<P: PhaseItem, const I: usize> RenderCommand<P> for SetMesh2dBindGroup<I> {
835    type Param = SRes<Mesh2dBindGroup>;
836    type ViewQuery = ();
837    type ItemQuery = ();
838
839    #[inline]
840    fn render<'w>(
841        item: &P,
842        _view: (),

844        mesh2d_bind_group: SystemParamItem<'w, '_, Self::Param>,
845        pass: &mut TrackedRenderPass<'w>,
846    ) -> RenderCommandResult {
847        let mut dynamic_offsets: [u32; 1] = Default::default();
848        let mut offset_count = 0;
849        if let PhaseItemExtraIndex::DynamicOffset(dynamic_offset) = item.extra_index() {
850            dynamic_offsets[offset_count] = dynamic_offset;
851            offset_count += 1;
852        }
853        pass.set_bind_group(
854            I,
855            &mesh2d_bind_group.into_inner().value,
856            &dynamic_offsets[..offset_count],
857        );
858        RenderCommandResult::Success
859    }

862pub struct DrawMesh2d;
863impl<P: PhaseItem> RenderCommand<P> for DrawMesh2d {
864    type Param = (
865        SRes<RenderAssets<RenderMesh>>,
866        SRes<RenderMesh2dInstances>,
867        SRes<MeshAllocator>,
868    );
869    type ViewQuery = ();
870    type ItemQuery = ();
871
872    #[inline]
873    fn render<'w>(
874        item: &P,
875        _view: (),
876        _item_query: Option<()>,
877        (meshes, render_mesh2d_instances, mesh_allocator): SystemParamItem<'w, '_, Self::Param>,
878        pass: &mut TrackedRenderPass<'w>,
879    ) -> RenderCommandResult {
880        let meshes = meshes.into_inner();
881        let render_mesh2d_instances = render_mesh2d_instances.into_inner();
882        let mesh_allocator = mesh_allocator.into_inner();
883
884        let Some(RenderMesh2dInstance { mesh_asset_id, .. }) =
885            render_mesh2d_instances.get(&item.main_entity())
886        else {
887            return RenderCommandResult::Skip;
888        };
889        let Some(gpu_mesh) = meshes.get(*mesh_asset_id) else {
890            return RenderCommandResult::Skip;
891        };
892        let Some(vertex_buffer_slice) = mesh_allocator.mesh_vertex_slice(mesh_asset_id) else {
893            return RenderCommandResult::Skip;
894        };
895
896        pass.set_vertex_buffer(0, vertex_buffer_slice.buffer.slice(..));
897
898        let batch_range = item.batch_range();
899        match &gpu_mesh.buffer_info {
900            RenderMeshBufferInfo::Indexed {
901                index_format,
902                count,
903            } => {
904                let Some(index_buffer_slice) = mesh_allocator.mesh_index_slice(mesh_asset_id)
905                else {
906                    return RenderCommandResult::Skip;
907                };
908
909                pass.set_index_buffer(index_buffer_slice.buffer.slice(..), *index_format);
910
911                pass.draw_indexed(
912                    index_buffer_slice.range.start..(index_buffer_slice.range.start + count),
913                    vertex_buffer_slice.range.start as i32,
914                    batch_range.clone(),
915                );
916            }
917            RenderMeshBufferInfo::NonIndexed => {
918                pass.draw(vertex_buffer_slice.range, batch_range.clone());
919            }
920        }
921        RenderCommandResult::Success
922    }
