9/// Add this plugin to enable progress tracking for your states type.
10///
11/// This plugin will set up everything necessary to track progress in
12/// states of the given type.
13///
14/// ```rust
15/// # use bevy::prelude::*;
16/// # use iyes_progress::prelude::*;
17/// #
18/// # #[derive(States, Default, Debug, Clone, PartialEq, Eq, Hash)]
19/// # enum MyStates {
20/// #     #[default]
21/// #     Loading,
22/// #     Done,
23/// # }
24/// #
25/// # fn main_() {
26///     App::new()
27///         .add_plugins(DefaultPlugins)
28///         .init_state::<MyStates>()
29///         .add_plugins(
30///             ProgressPlugin::<MyStates>::new()
31///                 .with_state_transition(MyStates::Loading, MyStates::Done),
32///         )
33///         // ...
34///         .run();
35/// # }
36/// ```
37pub struct ProgressPlugin<S: FreelyMutableState> {
38    transitions: StateTransitionConfig<S>,
39    check_progress_schedule: InternedScheduleLabel,
40    autoclear_on_enter: bool,
41    autoclear_on_exit: bool,
42    #[cfg(feature = "assets")]
43    track_assets: bool,
44    #[cfg(feature = "assets")]
45    autoclear_assets_on_enter: bool,
46    #[cfg(feature = "assets")]
47    autoclear_assets_on_exit: bool,
48}
49
50/// This set represents the "check progress and transition state if ready" step.
51/// It is only useful in the schedule where progress checking occurs (`Last` by
52/// default).
53#[derive(Debug, Default, Clone, Copy, PartialEq, Eq, Hash, SystemSet)]
54pub struct CheckProgressSet;
55
56impl<S: FreelyMutableState> Default for ProgressPlugin<S> {
57    fn default() -> Self {
58        Self {
59            check_progress_schedule: Last.intern(),
60            transitions: Default::default(),
61            autoclear_on_enter: true,
62            autoclear_on_exit: false,
63            #[cfg(feature = "assets")]
64            track_assets: false,
65            #[cfg(feature = "assets")]
66            autoclear_assets_on_enter: false,
67            #[cfg(feature = "assets")]
68            autoclear_assets_on_exit: true,
69        }
70    }
71}

73impl<S: FreelyMutableState> ProgressPlugin<S> {
74    /// Create a new instance of this plugin.
75    pub fn new() -> Self {
76        Self::default()
77    }
78
79    /// Configure progress tracking in a specific state.
80    ///
81    /// (Mutable method variant)
82    ///
83    /// When the `from` state is entered, all values stored in the
84    /// [`ProgressTracker<S>`] resource will be cleared.
85    ///
86    /// When all the progress is complete, a state transition to the
87    /// `to` state will be queued automatically.
88    pub fn add_state_transition(&mut self, from: S, to: S) {
89        self.transitions.map_from_to.insert(from, to);
90    }
91
92    /// Configure progress tracking in a specific state.
93    ///
94    /// (Builder variant)
95    ///
96    /// When the `from` state is entered, all values stored in the
97    /// [`ProgressTracker<S>`] resource will be cleared.
98    ///
99    /// When all the progress is complete, a state transition to the
100    /// `to` state will be queued automatically.
101    pub fn with_state_transition(mut self, from: S, to: S) -> Self {
102        self.add_state_transition(from, to);
103        self
104    }
105
106    /// Configure in which schedule to check the global progress and queue state
107    /// transitions.
108    ///
109    /// Default: `Last`
110    pub fn check_progress_in<L: ScheduleLabel>(mut self, schedule: L) -> Self {
111        self.check_progress_schedule = schedule.intern();
112        self
113    }
114
115    /// Configure whether progress data should be cleared when entering/exiting
116    /// a progress-tracked state.
117    ///
118    /// Default: `on_enter: true, on_exit: false`.
119    pub fn auto_clear(mut self, on_enter: bool, on_exit: bool) -> Self {
120        self.autoclear_on_enter = on_enter;
121        self.autoclear_on_exit = on_exit;
122        self
123    }
124
125    /// Configure whether progress data should be cleared when entering/exiting
126    /// a progress-tracked state.
127    ///
128    /// Default: `on_enter: true, on_exit: false`.
129    pub fn set_auto_clear(&mut self, on_enter: bool, on_exit: bool) {
130        self.autoclear_on_enter = on_enter;
131        self.autoclear_on_exit = on_exit;
132    }
133
134    /// Configure whether assets tracking data should be cleared when
135    /// entering/exiting a progress-tracked state.
136    ///
137    /// Default: `on_enter: false, on_exit: true`.
138    #[cfg(feature = "assets")]
139    pub fn auto_clear_assets(mut self, on_enter: bool, on_exit: bool) -> Self {
140        self.autoclear_assets_on_enter = on_enter;
141        self.autoclear_assets_on_exit = on_exit;
142        self
143    }
144
145    /// Configure whether assets tracking data should be cleared when
146    /// entering/exiting a progress-tracked state.
147    ///
148    /// Default: `on_enter: false, on_exit: true`.
149    #[cfg(feature = "assets")]
150    pub fn set_auto_clear_assets(&mut self, on_enter: bool, on_exit: bool) {
151        self.autoclear_assets_on_enter = on_enter;
152        self.autoclear_assets_on_exit = on_exit;
153    }
154
155    /// Set whether the built-in asset tracking should be enabled.
156    #[cfg(feature = "assets")]
157    pub fn set_asset_tracking(&mut self, asset_tracking: bool) {
158        self.track_assets = asset_tracking;
159    }
160
161    /// Enable the built-in asset tracking feature.
162    #[cfg(feature = "assets")]
163    pub fn with_asset_tracking(mut self) -> Self {
164        self.track_assets = true;
165        self
166    }
167}

169impl<S: FreelyMutableState> Plugin for ProgressPlugin<S> {
170    fn build(&self, app: &mut App) {
171        app.init_resource::<ProgressTracker<S>>();
172        app.insert_resource(self.transitions.clone());
173        app.add_systems(
174            self.check_progress_schedule,
175            transition_if_ready::<S>
176                .run_if(rc_configured_state::<S>)
177                .in_set(CheckProgressSet),
178        );
179        app.add_systems(
180            PostUpdate,
181            apply_progress_from_entities::<S>
182                .run_if(rc_configured_state::<S>)
183                .run_if(any_with_component::<ProgressEntity<S>>),
184        );
185        for s in self.transitions.map_from_to.keys() {
186            if self.autoclear_on_enter {
187                app.add_systems(OnEnter(s.clone()), clear_global_progress::<S>);
188            }
189            if self.autoclear_on_exit {
190                app.add_systems(OnExit(s.clone()), clear_global_progress::<S>);
191            }
192        }
193        #[cfg(feature = "async")]
194        {
195            app.add_systems(
196                PreUpdate,
197                recv_progress_msgs::<S>
198                    .run_if(rc_configured_state::<S>)
199                    .run_if(rc_recv_progress_msgs::<S>),
200            );
201        }
202        #[cfg(feature = "debug")]
203        {
204            use crate::debug::*;
205            app.add_systems(
206                self.check_progress_schedule,
207                debug_progress::<S>
208                    .run_if(rc_debug_progress::<S>)
209                    .in_set(CheckProgressSet)
210                    .before(transition_if_ready::<S>),
211            );
212        }
213        #[cfg(feature = "assets")]
214        if self.track_assets {
215            use crate::assets::*;
216            app.init_resource::<AssetsLoading<S>>();
217            app.add_systems(
218                PostUpdate,
219                assets_progress::<S>
220                    .track_progress::<S>()
221                    .in_set(AssetsTrackProgress)
222                    .run_if(rc_configured_state::<S>),
223            );
224            for s in self.transitions.map_from_to.keys() {
225                if self.autoclear_assets_on_enter {
226                    app.add_systems(
227                        OnEnter(s.clone()),
228                        assets_loading_reset::<S>
229                            .after(clear_global_progress::<S>),
230                    );
231                }
232                if self.autoclear_assets_on_exit {
233                    app.add_systems(
234                        OnExit(s.clone()),
235                        assets_loading_reset::<S>
236                            .after(clear_global_progress::<S>),
237                    );
238                }
239            }
240        }
241    }
