4//! Fundamental text property types.
5//!
6//! This crate is intended as a lightweight, `no_std`-friendly vocabulary layer that can be shared
7//! across style systems, text layout engines, and font tooling. It focuses on small, typed
8//! representations of common “leaf” concepts (weights, widths, OpenType tags, language tags, etc).
9//!
10//! ## Features
11//!
12//! - `std` (enabled by default): This is currently unused and is provided for forward compatibility.
13//! - `bytemuck`: Implement traits from `bytemuck` on [`GenericFamily`].
14//!
15//! ## Example
16//!
17//! ```
18//! use parlance::{Language, Tag};
19//!
20//! let tag = Tag::parse("wght").unwrap();
21//! assert_eq!(tag.to_bytes(), *b"wght");
22//!
23//! let lang = Language::parse("zh-Hans-CN").unwrap();
24//! assert_eq!(lang.as_str(), "zh-Hans-CN");
25//! assert_eq!(lang.language(), "zh");
26//! assert_eq!(lang.script(), Some("Hans"));
27//! assert_eq!(lang.region(), Some("CN"));
28//! ```
