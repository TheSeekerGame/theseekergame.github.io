44/// Creates new windows on the [`winit`] backend for each entity with a newly-added
45/// [`Window`] component.
46///
47/// If any of these entities are missing required components, those will be added with their
48/// default values.
49pub fn create_windows(
50    event_loop: &ActiveEventLoop,
51    (
52        mut commands,
53        mut created_windows,
54        mut window_created_events,
55        mut handlers,
56        accessibility_requested,
57        monitors,
58    ): SystemParamItem<CreateWindowParams>,
59) {
60    WINIT_WINDOWS.with_borrow_mut(|winit_windows| {
61        ACCESS_KIT_ADAPTERS.with_borrow_mut(|adapters| {
62            for (entity, mut window, cursor_options, handle_holder) in &mut created_windows {
63                if winit_windows.get_window(entity).is_some() {
64                    continue;
65                }
66
67                info!("Creating new window {} ({})", window.title.as_str(), entity);
68
69                let winit_window = winit_windows.create_window(
70                    event_loop,
71                    entity,
72                    &window,
73                    cursor_options,
74                    adapters,
75                    &mut handlers,
76                    &accessibility_requested,
77                    &monitors,
78                );
79
80                if let Some(theme) = winit_window.theme() {
81                    window.window_theme = Some(convert_winit_theme(theme));
82                }
83
84                window
85                    .resolution
86                    .set_scale_factor_and_apply_to_physical_size(winit_window.scale_factor() as f32);
87
88                commands.entity(entity).insert((
89                    CachedWindow(window.clone()),
90                    CachedCursorOptions(cursor_options.clone()),
91                    WinitWindowPressedKeys::default(),
92                ));
93
94                if let Ok(handle_wrapper) = RawHandleWrapper::new(winit_window) {
95                    commands.entity(entity).insert(handle_wrapper.clone());
96                    if let Some(handle_holder) = handle_holder {
97                        *handle_holder.0.lock().unwrap() = Some(handle_wrapper);
98                    }
99                }
100
101                #[cfg(target_arch = "wasm32")]
102                {
103                    if window.fit_canvas_to_parent {
104                        let canvas = winit_window
105                            .canvas()
106                            .expect("window.canvas() can only be called in main thread.");
107                        let style = canvas.style();
108                        style.set_property("width", "100%").unwrap();
109                        style.set_property("height", "100%").unwrap();
110                    }
111                }
112
113                #[cfg(target_os = "ios")]
114                {
115                    winit_window.recognize_pinch_gesture(window.recognize_pinch_gesture);
116                    winit_window.recognize_rotation_gesture(window.recognize_rotation_gesture);
117                    winit_window.recognize_doubletap_gesture(window.recognize_doubletap_gesture);
118                    if let Some((min, max)) = window.recognize_pan_gesture {
119                        winit_window.recognize_pan_gesture(true, min, max);
120                    } else {
121                        winit_window.recognize_pan_gesture(false, 0, 0);
122                    }
123                }
124
125                window_created_events.write(WindowCreated { window: entity });
126            }
127        });
128    });
129}

131/// Check whether keyboard focus was lost. This is different from window
132/// focus in that swapping between Bevy windows keeps window focus.
133pub(crate) fn check_keyboard_focus_lost(
134    mut window_focused_reader: MessageReader<WindowFocused>,
135    mut keyboard_focus_lost_writer: MessageWriter<KeyboardFocusLost>,
136    mut keyboard_input_writer: MessageWriter<KeyboardInput>,
137    mut window_event_writer: MessageWriter<WindowEvent>,
138    mut q_windows: Query<&mut WinitWindowPressedKeys>,
139) {
140    let mut focus_lost = vec![];
141    let mut focus_gained = false;
142    for e in window_focused_reader.read() {
143        if e.focused {
144            focus_gained = true;
145        } else {
146            focus_lost.push(e.window);
147        }
148    }
149
150    if !focus_gained {
151        if !focus_lost.is_empty() {
152            window_event_writer.write(WindowEvent::KeyboardFocusLost(KeyboardFocusLost));
153            keyboard_focus_lost_writer.write(KeyboardFocusLost);
154        }
155
156        for window in focus_lost {
157            let Ok(mut pressed_keys) = q_windows.get_mut(window) else {
158                continue;
159            };
160            for (key_code, logical_key) in pressed_keys.0.drain() {
161                let event = KeyboardInput {
162                    key_code,
163                    logical_key,
164                    state: bevy_input::ButtonState::Released,
165                    repeat: false,
166                    window,
167                    text: None,
168                };
169                window_event_writer.write(WindowEvent::KeyboardInput(event.clone()));
170                keyboard_input_writer.write(event);
171            }
172        }
173    }
174}

176/// Synchronize available monitors as reported by [`winit`] with [`Monitor`] entities in the world.
177pub fn create_monitors(
178    event_loop: &ActiveEventLoop,
179    (mut commands, mut monitors): SystemParamItem<CreateMonitorParams>,
180) {
181    let primary_monitor = event_loop.primary_monitor();
182    let mut seen_monitors = vec![false; monitors.monitors.len()];
183
184    'outer: for monitor in event_loop.available_monitors() {
185        for (idx, (m, _)) in monitors.monitors.iter().enumerate() {
186            if &monitor == m {
187                seen_monitors[idx] = true;
188                continue 'outer;
189            }
190        }
191
192        let size = monitor.size();
193        let position = monitor.position();
194
195        let entity = commands
196            .spawn(Monitor {
197                name: monitor.name(),
198                physical_height: size.height,
199                physical_width: size.width,
200                physical_position: IVec2::new(position.x, position.y),
201                refresh_rate_millihertz: monitor.refresh_rate_millihertz(),
202                scale_factor: monitor.scale_factor(),
203                video_modes: monitor
204                    .video_modes()
205                    .map(|v| {
206                        let size = v.size();
207                        VideoMode {
208                            physical_size: UVec2::new(size.width, size.height),
209                            bit_depth: v.bit_depth(),
210                            refresh_rate_millihertz: v.refresh_rate_millihertz(),
211                        }
212                    })
213                    .collect(),
214            })
215            .id();
216
217        if primary_monitor.as_ref() == Some(&monitor) {
218            commands.entity(entity).insert(PrimaryMonitor);
219        }
220
221        seen_monitors.push(true);
222        monitors.monitors.push((monitor, entity));
223    }
224
225    let mut idx = 0;
226    monitors.monitors.retain(|(_m, entity)| {
227        if seen_monitors[idx] {
228            idx += 1;
229            true
230        } else {
231            info!("Monitor removed {}", entity);
232            commands.entity(*entity).despawn();
233            idx += 1;
234            false
235        }
236    });
237}

239pub(crate) fn despawn_windows(
240    closing: Query<Entity, With<ClosingWindow>>,
241    mut closed: RemovedComponents<Window>,
242    window_entities: Query<Entity, With<Window>>,
243    mut closing_event_writer: MessageWriter<WindowClosing>,
244    mut closed_event_writer: MessageWriter<WindowClosed>,
245    mut windows_to_drop: Local<Vec<WindowWrapper<winit::window::Window>>>,
246    mut exit_event_reader: MessageReader<AppExit>,
247    _non_send_marker: NonSendMarker,
248) {
249    // Drop all the windows that are waiting to be closed
250    windows_to_drop.clear();
251    for window in closing.iter() {
252        closing_event_writer.write(WindowClosing { window });
253    }
254    for window in closed.read() {
255        info!("Closing window {}", window);
256        // Guard to verify that the window is in fact actually gone,
257        // rather than having the component added
258        // and removed in the same frame.
259        if !window_entities.contains(window) {
260            WINIT_WINDOWS.with_borrow_mut(|winit_windows| {
261                if let Some(window) = winit_windows.remove_window(window) {
262                    // Keeping WindowWrapper that are dropped for one frame
263                    // Otherwise the last `Arc` of the window could be in the rendering thread, and dropped there
264                    // This would hang on macOS
265                    // Keeping the wrapper and dropping it next frame in this system ensure its dropped in the main thread
266                    windows_to_drop.push(window);
267                }
268            });
269            closed_event_writer.write(WindowClosed { window });
270        }
271    }
272
273    // On macOS, many things need to be dropped on the main thread, or the app will hang:
274    // - notify the rendering thread the windows are about to close
275    // - take the `WindowWrapper`s out of `WINIT_WINDOWS` and into the local `windows_to_drop`
276    if !exit_event_reader.is_empty() {
277        exit_event_reader.clear();
278        WINIT_WINDOWS.with_borrow_mut(|winit_windows| {
279            for window in window_entities.iter() {
280                closing_event_writer.write(WindowClosing { window });
281                if let Some(wrapper) = winit_windows.remove_window(window) {
282                    windows_to_drop.push(wrapper);
283                }
284            }
285        });
286    }
287}

297/// Propagates changes from [`Window`] entities to the [`winit`] backend.
298///
299/// # Notes
300///
301/// - [`Window::present_mode`] and [`Window::composite_alpha_mode`] changes are handled by the `bevy_render` crate.
302/// - [`Window::transparent`] cannot be changed after the window is created.
303/// - [`Window::canvas`] cannot be changed after the window is created.
304/// - [`Window::focused`] cannot be manually changed to `false` after the window is created.
305pub(crate) fn changed_windows(
306    mut commands: Commands,
307    mut changed_windows: Query<
308        (Entity, &mut Window, &mut CachedWindow, Option<&OnMonitor>),
309        Changed<Window>,
310    >,
311    monitors: Res<WinitMonitors>,
312    mut window_resized: MessageWriter<WindowResized>,
313    mut window_event: MessageWriter<WindowEvent>,
314    mut window_rescaled: MessageWriter<WindowScaleFactorChanged>,
315    _non_send_marker: NonSendMarker,
316) {
317    WINIT_WINDOWS.with_borrow(|winit_windows| {
318        for (entity, mut window, mut cache, monitor_relationship) in &mut changed_windows {
319            let Some(winit_window) = winit_windows.get_window(entity) else {
320                continue;
321            };
322
323            if window.title != cache.title {
324                winit_window.set_title(window.title.as_str());
325            }
326
327            if window.mode != cache.mode {
328                let new_mode = match window.mode {
329                    WindowMode::BorderlessFullscreen(monitor_selection) => {
330                        Some(Some(winit::window::Fullscreen::Borderless(select_monitor(
331                            &monitors,
332                            winit_window.primary_monitor(),
333                            winit_window.current_monitor(),
334                            &monitor_selection,
335                        ))))
336                    }
337                    WindowMode::Fullscreen(monitor_selection, video_mode_selection) => {
338                        let monitor = &select_monitor(
339                            &monitors,
340                            winit_window.primary_monitor(),
341                            winit_window.current_monitor(),
342                            &monitor_selection,
343                        )
344                        .unwrap_or_else(|| {
345                            panic!("Could not find monitor for {monitor_selection:?}")
346                        });
347
348                        if let Some(video_mode) = get_selected_videomode(monitor, &video_mode_selection)
349                        {
350                            Some(Some(winit::window::Fullscreen::Exclusive(video_mode)))
351                        } else {
352                            warn!(
353                                "Could not find valid fullscreen video mode for {:?} {:?}",
354                                monitor_selection, video_mode_selection
355                            );
356                            None
357                        }
358                    }
359                    WindowMode::Windowed => Some(None),
360                };
361
362                if let Some(new_mode) = new_mode
363                    && winit_window.fullscreen() != new_mode {
364                        winit_window.set_fullscreen(new_mode);
365                    }
366            }
367
368            // Set position before size so the window is on the correct monitor
369            // (and thus using the correct scale factor) when size is applied.
370            if window.position != cache.position
371                && let Some(position) = crate::winit_window_position(
372                    &window.position,
373                    &window.resolution,
374                    &monitors,
375                    winit_window.primary_monitor(),
376                    winit_window.current_monitor(),
377                ) {
378                    let should_set = match winit_window.outer_position() {
379                        Ok(current_position) => current_position != position,
380                        _ => true,
381                    };
382
383                    if should_set {
384                        winit_window.set_outer_position(position);
385                    }
386                }
387
388            if window.resolution != cache.resolution {
389                let cache_physical_size = PhysicalSize::new(
390                    cache.resolution.physical_width(),
391                    cache.resolution.physical_height(),
392                );
393                let requested_physical_size = PhysicalSize::new(
394                    window.resolution.physical_width(),
395                    window.resolution.physical_height(),
396                );
397
398                if cache_physical_size != requested_physical_size {
399                    // In `None` case, the request will be handled by winit::event::WindowEvent::Resized
400                    if let Some(new_physical_size) = winit_window.request_inner_size(requested_physical_size) {
401                        let event = react_to_resize(entity, &mut window, new_physical_size);
402                        // Need to send two very similar events because different systems rely on those.
403                        window_resized.write(event.clone());
404                        window_event.write(event.into());
405                    }
406                }
407
408                let cache_scale_factor = cache.scale_factor();
409                let requested_scale_factor = window.scale_factor();
410
411                if cache_scale_factor != requested_scale_factor {
412                    // If the scale factor has changed we don't query anything from winit, but send events for camera system to handle.
413                    let event = WindowScaleFactorChanged { scale_factor: requested_scale_factor as f64, window: entity};
414                    // Need to send two very similar events because different systems rely on those.
415                    window_rescaled.write(event.clone());
416                    window_event.write(event.into());
417                }
418            }
419
420            if window.physical_cursor_position() != cache.physical_cursor_position()
421                && let Some(physical_position) = window.physical_cursor_position() {
422                    let position = PhysicalPosition::new(physical_position.x, physical_position.y);
423
424                    if let Err(err) = winit_window.set_cursor_position(position) {
425                        error!("could not set cursor position: {}", err);
426                    }
427                }
428
429            if window.decorations != cache.decorations
430                && window.decorations != winit_window.is_decorated()
431            {
432                winit_window.set_decorations(window.decorations);
433            }
434
435            if window.resizable != cache.resizable
436                && window.resizable != winit_window.is_resizable()
437            {
438                winit_window.set_resizable(window.resizable);
439            }
440
441            if window.enabled_buttons != cache.enabled_buttons {
442                winit_window.set_enabled_buttons(convert_enabled_buttons(window.enabled_buttons));
443            }
444
445            if window.resize_constraints != cache.resize_constraints {
446                let constraints = window.resize_constraints.check_constraints();
447                let min_inner_size = LogicalSize {
448                    width: constraints.min_width,
449                    height: constraints.min_height,
450                };
451                let max_inner_size = LogicalSize {
452                    width: constraints.max_width,
453                    height: constraints.max_height,
454                };
455
456                winit_window.set_min_inner_size(Some(min_inner_size));
457                winit_window.set_max_inner_size(
458                    if constraints.max_width.is_finite() && constraints.max_height.is_finite() {
459                        Some(max_inner_size)
460                    } else {
461                        None
462                    },
463                );
464            }
465
466            if let Some(monitor_link) = monitor_relationship {
467                if let Some(winit_monitor) = winit_window.current_monitor() {
468                    if let Some(linked_monitor) = monitors.find_entity(monitor_link.0) &&
469                        winit_monitor != linked_monitor &&
470                        let Some((_, winit_monitor_entity)) = monitors.monitors.iter().find(|(h, _)| h == &winit_monitor) {
471                        commands.entity(entity).insert(OnMonitor(winit_monitor_entity.to_owned()));
472                    }
473                } else {
474                    commands.entity(entity).remove::<OnMonitor>();
475                }
476            } else {
477                if let Some(winit_monitor) = winit_window.current_monitor()
478                    && let Some((_, winit_monitor_entity)) = monitors.monitors.iter()
479                    .find(|(h, _)| h == &winit_monitor) {
480                    commands.entity(entity).insert(OnMonitor(winit_monitor_entity.to_owned()));
481                }
482            }
483
484            if let Some(maximized) = window.internal.take_maximize_request() {
485                winit_window.set_maximized(maximized);
486            }
487
488            if let Some(minimized) = window.internal.take_minimize_request() {
489                winit_window.set_minimized(minimized);
490            }
491
492            if window.internal.take_move_request()
493                && let Err(e) = winit_window.drag_window() {
494                    warn!("Winit returned an error while attempting to drag the window: {e}");
495                }
496
497            if let Some(resize_direction) = window.internal.take_resize_request()
498                && let Err(e) =
499                    winit_window.drag_resize_window(convert_resize_direction(resize_direction))
500                {
501                    warn!("Winit returned an error while attempting to drag resize the window: {e}");
502                }
503
504            if window.focused != cache.focused && window.focused {
505                winit_window.focus_window();
506            }
507
508            if window.window_level != cache.window_level {
509                winit_window.set_window_level(convert_window_level(window.window_level));
510            }
511
512            // Currently unsupported changes
513            if window.transparent != cache.transparent {
514                window.transparent = cache.transparent;
515                warn!("Winit does not currently support updating transparency after window creation.");
516            }
517
518            #[cfg(target_arch = "wasm32")]
519            if window.canvas != cache.canvas {
520                window.canvas.clone_from(&cache.canvas);
521                warn!(
522                    "Bevy currently doesn't support modifying the window canvas after initialization."
523                );
524            }
525
526            if window.ime_enabled != cache.ime_enabled {
527                winit_window.set_ime_allowed(window.ime_enabled);
528            }
529
530            if window.ime_position != cache.ime_position {
531                winit_window.set_ime_cursor_area(
532                    LogicalPosition::new(window.ime_position.x, window.ime_position.y),
533                    PhysicalSize::new(10, 10),
534                );
535            }
536
537            if window.window_theme != cache.window_theme {
538                winit_window.set_theme(window.window_theme.map(convert_window_theme));
539            }
540
541            if window.visible != cache.visible {
542                winit_window.set_visible(window.visible);
543            }
544
545            #[cfg(target_os = "ios")]
546            {
547                if window.recognize_pinch_gesture != cache.recognize_pinch_gesture {
548                    winit_window.recognize_pinch_gesture(window.recognize_pinch_gesture);
549                }
550                if window.recognize_rotation_gesture != cache.recognize_rotation_gesture {
551                    winit_window.recognize_rotation_gesture(window.recognize_rotation_gesture);
552                }
553                if window.recognize_doubletap_gesture != cache.recognize_doubletap_gesture {
554                    winit_window.recognize_doubletap_gesture(window.recognize_doubletap_gesture);
555                }
556                if window.recognize_pan_gesture != cache.recognize_pan_gesture {
557                    match (
558                        window.recognize_pan_gesture,
559                        cache.recognize_pan_gesture,
560                    ) {
561                        (Some(_), Some(_)) => {
562                            warn!("Bevy currently doesn't support modifying PanGesture number of fingers recognition. Please disable it before re-enabling it with the new number of fingers");
563                        }
564                        (Some((min, max)), _) => winit_window.recognize_pan_gesture(true, min, max),
565                        _ => winit_window.recognize_pan_gesture(false, 0, 0),
566                    }
567                }
568
569                if window.prefers_home_indicator_hidden != cache.prefers_home_indicator_hidden {
570                    winit_window
571                        .set_prefers_home_indicator_hidden(window.prefers_home_indicator_hidden);
572                }
573                if window.prefers_status_bar_hidden != cache.prefers_status_bar_hidden {
574                    winit_window.set_prefers_status_bar_hidden(window.prefers_status_bar_hidden);
575                }
576                if window.preferred_screen_edges_deferring_system_gestures
577                    != cache
578                        .preferred_screen_edges_deferring_system_gestures
579                {
580                    use crate::converters::convert_screen_edge;
581                    let preferred_edge =
582                        convert_screen_edge(window.preferred_screen_edges_deferring_system_gestures);
583                    winit_window.set_preferred_screen_edges_deferring_system_gestures(preferred_edge);
584                }
585            }
586            **cache = window.clone();
587        }
588    });
589}

591pub(crate) fn changed_cursor_options(
592    mut changed_windows: Query<
593        (
594            Entity,
595            &Window,
596            &mut CursorOptions,
597            &mut CachedCursorOptions,
598        ),
599        Changed<CursorOptions>,
600    >,
601    _non_send_marker: NonSendMarker,
602) {
603    WINIT_WINDOWS.with_borrow(|winit_windows| {
604        for (entity, window, mut cursor_options, mut cache) in &mut changed_windows {
605            // This system already only runs when the cursor options change, so we need to bypass change detection or the next frame will also run this system
606            let cursor_options = cursor_options.bypass_change_detection();
607            let Some(winit_window) = winit_windows.get_window(entity) else {
608                continue;
609            };
610            // Don't check the cache for the grab mode. It can change through external means, leaving the cache outdated.
611            if let Err(err) =
612                crate::winit_windows::attempt_grab(winit_window, cursor_options.grab_mode)
613            {
614                warn!(
615                    "Could not set cursor grab mode for window {}: {}",
616                    window.title, err
617                );
618                cursor_options.grab_mode = cache.grab_mode;
619            } else {
620                cache.grab_mode = cursor_options.grab_mode;
621            }
622
623            if cursor_options.visible != cache.visible {
624                winit_window.set_cursor_visible(cursor_options.visible);
625                cache.visible = cursor_options.visible;
626            }
627
628            if cursor_options.hit_test != cache.hit_test {
629                if let Err(err) = winit_window.set_cursor_hittest(cursor_options.hit_test) {
630                    warn!(
631                        "Could not set cursor hit test for window {}: {}",
632                        window.title, err
633                    );
634                    cursor_options.hit_test = cache.hit_test;
635                } else {
636                    cache.hit_test = cursor_options.hit_test;
637                }
638            }
639        }
640    });
641}
