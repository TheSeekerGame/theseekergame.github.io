17/// Converts a [`winit::event::KeyEvent`] and a window [`Entity`] to a Bevy [`KeyboardInput`]
18pub fn convert_keyboard_input(
19    keyboard_input: &winit::event::KeyEvent,
20    window: Entity,
21) -> KeyboardInput {
22    KeyboardInput {
23        state: convert_element_state(keyboard_input.state),
24        key_code: convert_physical_key_code(keyboard_input.physical_key),
25        logical_key: convert_logical_key(&keyboard_input.logical_key),
26        text: keyboard_input.text.clone(),
27        repeat: keyboard_input.repeat,
28        window,
29    }
30}

32/// Converts a [`winit::event::ElementState`] to a Bevy [`ButtonState`]

40/// Converts a [`winit::event::MouseButton`] to a Bevy [`MouseButton`]

52/// Converts a [`winit::event::TouchPhase`] to a Bevy [`TouchPhase`].
53pub fn convert_touch_phase(phase: winit::event::TouchPhase) -> TouchPhase {
54    match phase {

56        winit::event::TouchPhase::Moved => TouchPhase::Moved,

58        winit::event::TouchPhase::Cancelled => TouchPhase::Canceled,

62/// Converts a [`winit::event::Touch`], [`winit::dpi::LogicalPosition<f64>`] and window [`Entity`] to a Bevy [`TouchInput`]
63pub fn convert_touch_input(
64    touch_input: winit::event::Touch,
65    location: winit::dpi::LogicalPosition<f64>,
66    window_entity: Entity,
67) -> TouchInput {
68    TouchInput {
69        phase: convert_touch_phase(touch_input.phase),
70        position: Vec2::new(location.x as f32, location.y as f32),
71        window: window_entity,
72        force: touch_input.force.map(|f| match f {
73            winit::event::Force::Calibrated {
74                force,
75                max_possible_force,
76                altitude_angle,
77            } => ForceTouch::Calibrated {
78                force,
79                max_possible_force,
80                altitude_angle,
81            },
82            winit::event::Force::Normalized(x) => ForceTouch::Normalized(x),
83        }),
84        id: touch_input.id,
85    }
86}

639pub fn convert_native_key(native_key: &NativeKey) -> bevy_input::keyboard::NativeKey {
640    match native_key {
641        NativeKey::Unidentified => bevy_input::keyboard::NativeKey::Unidentified,
642        NativeKey::Android(v) => bevy_input::keyboard::NativeKey::Android(*v),
643        NativeKey::MacOS(v) => bevy_input::keyboard::NativeKey::MacOS(*v),
644        NativeKey::Windows(v) => bevy_input::keyboard::NativeKey::Windows(*v),
645        NativeKey::Xkb(v) => bevy_input::keyboard::NativeKey::Xkb(*v),
646        NativeKey::Web(v) => bevy_input::keyboard::NativeKey::Web(v.clone()),
647    }
648}

715/// Converts a Bevy [`EnabledButtons`] to a [`winit::window::WindowButtons`]
716pub fn convert_enabled_buttons(enabled_buttons: EnabledButtons) -> winit::window::WindowButtons {
717    let mut window_buttons = winit::window::WindowButtons::empty();
718    if enabled_buttons.minimize {
719        window_buttons.insert(winit::window::WindowButtons::MINIMIZE);
720    }
721    if enabled_buttons.maximize {
722        window_buttons.insert(winit::window::WindowButtons::MAXIMIZE);
723    }
724    if enabled_buttons.close {
725        window_buttons.insert(winit::window::WindowButtons::CLOSE);
726    }
727    window_buttons
728}
