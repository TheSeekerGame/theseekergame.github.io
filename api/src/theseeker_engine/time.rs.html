12impl Plugin for GameTimePlugin {
13    fn build(&self, app: &mut App) {
14        for schedule in [Schedule::new(GameTickPre), Schedule::new(GameTickUpdate), Schedule::new(GameTickPost)] {
15            app.add_schedule(schedule);
16        }
17        app.edit_schedule(GameTickPre, |schedule| { schedule.set_executor(SingleThreadedExecutor::new()); });
18        app.edit_schedule(GameTickUpdate, |schedule| { schedule.set_executor(SingleThreadedExecutor::new()); });
19        app.edit_schedule(GameTickPost, |schedule| { schedule.set_executor(SingleThreadedExecutor::new()); });
20        app.init_resource::<MasterClock>().init_resource::<ClockTime>()
21            .init_resource::<SimulationWork>()
22            .init_resource::<PresentationTime>().init_resource::<GameTransport>()
23            .init_resource::<RenderInterpolation>().init_resource::<GameTime>()
24            .init_resource::<ClockSource>().init_resource::<TimingSettings>();
25        app.configure_sets(Update, MasterClockSet::Update.after(AudioSet::Update));
26        app.add_systems(Update, (
27            update_master_clock.in_set(MasterClockSet::Update),
28            run_gametickupdate_schedule.after(MasterClockSet::Update),
29            log_diagnostics.in_set(GameTickSet::Post),
30        ));
31        app.configure_sets(Update, GameTickSet::Pre.after(MasterClockSet::Update).before(run_gametickupdate_schedule));
32        app.configure_sets(Update, GameTickSet::Post.after(run_gametickupdate_schedule));
33    }
34}

99/// Simulation time; updated for each fixed tick, never from the monitor refresh rate.
100#[derive(Resource, Debug, Clone, Default)]
101pub struct ClockTime { elapsed: Duration, delta: Duration }
102impl ClockTime {
103    pub fn elapsed(&self) -> Duration { self.elapsed }
104    pub fn delta(&self) -> Duration { self.delta }
105    pub fn elapsed_secs(&self) -> f32 { self.elapsed.as_secs_f32() }
106    pub fn delta_secs(&self) -> f32 { self.delta.as_secs_f32() }
107    fn advance_to(&mut self, seconds: f64) {
108        let elapsed = Duration::from_secs_f64(seconds.max(0.0));
109        self.delta = elapsed.saturating_sub(self.elapsed);
110        self.elapsed = elapsed;
111    }
112}

140#[derive(Default)]
141struct FrameIntervals { samples: [f64; 32], count: usize, next: usize }
142impl FrameIntervals {
143    fn record(&mut self, seconds: f64) -> Option<f64> {
144        if !(0.0..0.1).contains(&seconds) || seconds == 0.0 { return None; }
145        self.samples[self.next] = seconds;
146        self.next = (self.next + 1) % self.samples.len();
147        self.count = (self.count + 1).min(self.samples.len());
148        if self.count < self.samples.len() { return None; }
149        let mut ordered = self.samples;
150        ordered[..self.count].sort_unstable_by(f64::total_cmp);
151        Some(ordered[(self.count - 1) * 9 / 10])
152    }
153}

173fn update_master_clock(inputs: ClockInputs, mut out: ClockOutputs, mut engine: ResMut<AudioEngine>, mut state: Local<ClockState>) {
174    advance_master_clock(inputs, &mut out, &mut engine, &mut state, Instant::now());
175}

177fn advance_master_clock(inputs: ClockInputs, out: &mut ClockOutputs, engine: &mut AudioEngine, state: &mut ClockState, now: Instant) {
178    let delta = state.host.replace(now).map_or(0.0, |previous| now.saturating_duration_since(previous).as_secs_f64());
179    let rate = f64::from(inputs.config.tick_rate_hz.max(1));
180    if let Some(budget) = state.frame_intervals.record(delta) { state.frame_budget_seconds = budget; }
181    let healthy = inputs.clock.healthy(now);
182    let recovered = healthy && (!state.audio_active || state.device_generation != inputs.clock.generation);
183    // Lead controls how far ahead we simulate. Growing it does not move the
184    // sample timeline or invalidate voices. Do not shrink it during playback.
185    out.transport.lead_seconds = out.transport.lead_seconds.max(
186        scheduling_lead(&inputs.config, &inputs.clock, &inputs.settings, state.frame_budget_seconds));
187    let lead = out.transport.lead_seconds;
188    if out.transport.paused != state.paused {
189        state.paused = out.transport.paused;
190        state.freeze_seconds = out.game.tick() as f64 / rate;
191        if state.paused { engine.pause_at_tick(out.game.tick()); }
192        else { engine.resume_after(Duration::from_secs_f64(lead)); }
193    }
194    let forced = engine.take_resync_request();
195    if !healthy || state.awaiting_generation.is_some_and(|expected| inputs.clock.generation > expected) {
196        state.awaiting_generation = None;
197    }
198    if healthy && (forced || recovered) && out.game.tick() > 0 && state.awaiting_generation.is_none() {
199        resynchronize(out, state, engine);
200    }
201    if state.awaiting_generation.is_some_and(|generation| inputs.clock.generation == generation) {
202        state.awaiting_generation = None;
203    }
204    state.audio_active = healthy;
205    state.device_generation = inputs.clock.generation;
206    let mut seconds = if healthy && state.awaiting_generation.is_none() {
207        inputs.clock.transport_position(now) / f64::from(inputs.clock.sample_rate)
208    } else if healthy { state.seconds }
209    else { state.seconds + delta };
210    if state.paused { seconds = seconds.min(state.freeze_seconds); }
211    seconds = seconds.max(state.seconds);
212    state.seconds = seconds;
213    let mut target = if state.paused || state.awaiting_generation.is_some() { out.game.tick() }
214        else { ((seconds + lead) * rate).floor() as u64 };
215    if target.saturating_sub(out.game.tick()) > (lead * rate).ceil() as u64 + 8 {
216        if healthy { resynchronize(out, state, engine); }
217        else { state.seconds = (out.game.tick() as f64 / rate - lead).max(0.0); out.transport.resynchronizations += 1; }
218        target = out.game.tick();
219    }
220    out.master.tick_rate_hz = inputs.config.tick_rate_hz;
221    out.master.sample_rate = inputs.clock.sample_rate;
222    out.master.tick = target.max(out.game.tick());
223    out.master.tick_alpha = ((state.seconds + lead) * rate).fract();
224    let elapsed = Duration::from_secs_f64(out.master.tick as f64 / rate);
225    out.master.delta = elapsed.saturating_sub(out.master.elapsed);
226    out.master.elapsed = elapsed;
227    out.simulation_time.advance_to(elapsed.as_secs_f64());
228    out.game.update_from_master(&out.master);
229    *out.source = if healthy { ClockSource::Audio } else { ClockSource::Time };
230    let offset = (inputs.settings.display_latency_ms + inputs.settings.av_offset_ms) / 1000.0;
231    let present = if healthy && state.awaiting_generation.is_none() {
232        let instant = if offset >= 0.0 { now + Duration::from_secs_f64(offset) }
233            else { now - Duration::from_secs_f64(-offset) };
234        inputs.clock.transport_position(instant) / f64::from(inputs.clock.sample_rate)
235    } else { state.seconds + offset };
236    let present = present.max(out.presentation.elapsed().as_secs_f64()).min(out.master.tick as f64 / rate);
237    out.interpolation.position = present.max(0.0) * rate;
238    out.presentation.advance_to(present);
239    out.transport.generation = inputs.clock.generation;
240}

242fn scheduling_lead(config: &AudioEngineConfig, clock: &AudioClock, settings: &TimingSettings, frame_budget: f64) -> f64 {
243    let tick = 1.0 / f64::from(config.tick_rate_hz.max(1));
244    let minimum = clock.latency().as_secs_f64() + f64::from(clock.block_frames) / f64::from(clock.sample_rate) + frame_budget.max(tick) + tick;
245    let display = (settings.display_latency_ms + settings.av_offset_ms).max(0.0) / 1000.0 + tick;
246    ((f64::from(config.lead_ticks) * tick).max(minimum).max(display) / tick).ceil() * tick
247}
248
249fn resynchronize(out: &mut ClockOutputs, state: &mut ClockState, engine: &mut AudioEngine) {
250    let generation = engine.resynchronize(out.game.tick());
251    state.awaiting_generation = Some(generation);
252    state.seconds = out.game.tick() as f64 / out.game.hz;
253    out.transport.resynchronizations += 1;
254    if out.transport.paused { engine.pause_at_tick(out.game.tick()); }
255    else { engine.resume_after(Duration::from_secs_f64(out.transport.lead_seconds)); }
256}

261impl GameTimeAppExt for App {
262    fn add_gametick_event<T: Message>(&mut self) -> &mut Self {
263        if !self.world().contains_resource::<Messages<T>>() {
264            self.init_resource::<Messages<T>>();
265            self.add_systems(
266                GameTickUpdate,
267                minimal_event_update_system::<T>.in_set(GameTickEventClearSet),
268            );
269        } else {
270            warn!("Attempted to add a Game Tick event type that had already been added as an event before!");
271        }
272        self
273    }
274}

294pub struct GameTime {

296    pub hz: f64,
297    tick: u64,
298    new_ticks: u64,
299    total_ticks: u64,
300    overstep: f64,
301    last_update: Duration,

304impl Default for GameTime {
305    fn default() -> Self {
306        Self {
307            hz: f64::from(TICK_RATE_HZ),
308            tick: 0,
309            new_ticks: 0,
310            total_ticks: 0,
311            overstep: 0.0,
312            last_update: Duration::new(0, 0),
313        }
314    }
315}
316
317impl GameTime {
318    /// Create with a non-default tick rate
319    pub fn new(hz: f64) -> Self {
320        Self {
321            hz,
322            ..Default::default()
323        }
324    }
325
326    /// Get the current tick number to be simulated
327    ///
328    /// Increments with every run of the `GameTickUpdate` schedule, until it reaches
329    /// the `total_ticks` target value.
330    pub fn tick(&self) -> u64 {
331        self.tick
332    }
333
334    /// Get the number of new ticks to be simulated this Bevy frame update
335    pub fn new_ticks(&self) -> u64 {
336        self.new_ticks
337    }
338
339    /// Get the total target number of ticks as of this Bevy frame update
340    pub fn total_ticks(&self) -> u64 {
341        self.total_ticks
342    }
343
344    /// Get the leftover partial tick to be carried over to the next Bevy frame update
345    pub fn overstep(&self) -> f64 {
346        self.overstep
347    }
348
349    /// Get the "elapsed" time when we last updated
350    pub fn last_update(&self) -> Duration {
351        self.last_update
352    }
353
354    /// Reset tick counters to zero, set the last update to now, keep the `hz` value
355    pub fn reset(&mut self, now: Duration) {
356        *self = Self {
357            hz: self.hz,
358            last_update: now,
359            ..Default::default()
360        };
361    }
362
363    /// Every frame, this gets called to advance the tick counters.
364    pub fn update_from_master(&mut self, master: &MasterClock) {
365        self.hz = master.tick_rate_hz as f64;
366        self.last_update = master.elapsed;
367        self.total_ticks = master.tick;
368        self.overstep = master.tick_alpha;
369        self.new_ticks = self.total_ticks.saturating_sub(self.tick);
370    }
371
372    /// returns the amount of time since start; ie: ticks * tick length
373    pub fn time_in_seconds(&self) -> f64 {
374        self.tick() as f64 * self.seconds_per_tick()
375    }
376
377    /// Convenience function to save you the math
378    pub fn seconds_per_tick(&self) -> f64 {
379        1.0 / self.hz
380    }
381}

384pub fn run_gametickupdate_schedule(world: &mut World) {
385    let started = Instant::now();
386    let first_tick = world.resource::<GameTime>().tick;
387    let frame_clock = world.get_resource::<ClockTime>().cloned();
388    while world.resource::<GameTime>().tick < world.resource::<GameTime>().total_ticks {
389        let tick = world.resource::<GameTime>().tick;
390        let seconds = world.resource::<GameTime>().time_in_seconds();
391        let tick_duration = Duration::from_secs_f64(world.resource::<GameTime>().seconds_per_tick());
392        if let Some(mut clock) = world.get_resource_mut::<ClockTime>() {
393            *clock = ClockTime { elapsed: Duration::from_secs_f64(seconds), delta: tick_duration };
394        }
395        if let Some(mut engine) = world.get_resource_mut::<AudioEngine>() { engine.set_command_time(AudioStartTime::Tick(tick)); }
396        world.run_schedule(GameTickPre);
397        world.run_schedule(GameTickUpdate);
398        world.run_schedule(GameTickPost);
399        if let Some(mut engine) = world.get_resource_mut::<AudioEngine>() { engine.flush(); }
400        world.resource_mut::<GameTime>().tick += 1;
401    }
402    if let Some(clock) = frame_clock { world.insert_resource(clock); }
403    let ticks = world.resource::<GameTime>().tick - first_tick;
404    if let Some(mut work) = world.get_resource_mut::<SimulationWork>() {
405        work.seconds = started.elapsed().as_secs_f64();
406        work.ticks = ticks;
407    }
408}

414fn log_diagnostics(
415    settings: Res<TimingSettings>, transport: Res<GameTransport>, engine: Res<AudioEngine>,
416    presentation: Res<RenderInterpolation>, time: Res<Time<Real>>, mut elapsed: Local<f64>,
417) {
418    if !settings.diagnostics { return; }
419    *elapsed += time.delta_secs_f64();
420    if *elapsed < 1.0 { return; }
421    *elapsed = 0.0;
422    info!("Transport: lead_ms={:.2} display_ms={:.2} av_ms={:.2} presentation_tick={:.3} resyncs={} late={} (plays={} controls={} max_frames={}) rejected={} retries={} callback_max_us={}",
423        transport.lead_seconds * 1000.0, settings.display_latency_ms, settings.av_offset_ms,
424        presentation.position(), transport.resynchronizations, engine.diagnostics.late_events,
425        engine.diagnostics.late_plays, engine.diagnostics.late_controls, engine.diagnostics.max_lateness_frames,
426        engine.diagnostics.rejected_events, engine.dropped_commands, engine.diagnostics.max_callback_micros);
