5/// Arbitrary config asset type
6///
7/// Would typically be loaded from TOML files.
8#[derive(Asset, Debug, Clone)]
9#[derive(Serialize, Deserialize)]
10#[derive(TypePath)]
11#[serde(transparent)]
12pub struct DynamicConfig(pub HashMap<String, DynamicConfigValue>);
13
14#[derive(Debug, Clone, PartialEq)]
15#[derive(Serialize, Deserialize)]
16#[serde(untagged)]
17pub enum DynamicConfigValue {
18    Int(i64),
19    Float(f64),
20    String(String),
21}
22impl DynamicConfigValue {
23    pub fn as_float(&self) -> Result<f64, String> {
24        match self {
25            DynamicConfigValue::Float(value) => Ok(*value),
26            other => Err(format!(
27                "Expected float value, but found {other:?}"
28            )),
29        }
30    }
31
32}

34/// A utility type for loading config values. Use like:
35///
36/// ```
37/// use std::collections::HashMap;
38/// use theseeker_engine::assets::config::{DynamicConfigValue, update_field};
39///
40/// struct PlayerConfig {
41///     jump_vel_init: f32,
42/// }
43///
44/// let mut errors = Vec::new();
45/// let mut dynamic_cfg = HashMap::new();
46/// dynamic_cfg.insert("jump_vel_init".to_owned(), DynamicConfigValue::Float(12.0));
47/// let mut struct_config = PlayerConfig { jump_vel_init: 0.0 };
48///
49/// update_field(&mut errors, &dynamic_cfg, "jump_vel_init", |val| {
50///     struct_config.jump_vel_init = val;
51/// });
52///
53/// assert!(errors.is_empty());
54/// assert_eq!(struct_config.jump_vel_init, 12.0);
55/// ```
56pub fn update_field<F>(
57    errors: &mut Vec<String>,
58    config: &HashMap<String, DynamicConfigValue>,
59    field: &str,
60    setter: F,
61) where
62    F: FnOnce(f32),
63{
64    if let Some(value) = config.get(field) {
65        match value.as_float() {
66            Ok(val) => setter(val as f32),
67            Err(err) => errors.push(format!(
68                "Error parsing '{field}': {err}"
69            )),
70        }
71    } else {
72        errors.push(format!(
73            "Missing '{field}' field in the config"
74        ));
75    }
76}
