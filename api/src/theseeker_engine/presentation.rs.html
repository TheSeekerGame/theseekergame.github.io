58#[derive(Debug, Clone)]
59pub struct TickHistory<T> { samples: VecDeque<(u64, T)>, capacity: usize }
60impl<T> Default for TickHistory<T> {
61    fn default() -> Self { Self { samples: VecDeque::with_capacity(16), capacity: 64 } }
62}
63impl<T> TickHistory<T> {
64    pub fn push(&mut self, tick: u64, value: T) {
65        if self.samples.back().is_some_and(|sample| sample.0 >= tick) { self.samples.pop_back(); }
66        if self.samples.len() == self.capacity { self.samples.pop_front(); }
67        self.samples.push_back((tick, value));
68    }
69    pub fn pair(&self, position: f64) -> Option<(&T, &T, f32)> {
70        let index = self.samples.partition_point(|sample| sample.0 as f64 <= position);
71        let (tick, previous) = self.samples.get(index.checked_sub(1)?)?;
72        let Some((next_tick, next)) = self.samples.get(index) else { return Some((previous, previous, 0.0)); };
73        let alpha = ((position - *tick as f64) / (*next_tick - *tick) as f64).clamp(0.0, 1.0) as f32;
74        Some((previous, next, alpha))
75    }
76    pub fn discard_before(&mut self, position: f64) {
77        let keep = self.samples.partition_point(|sample| sample.0 as f64 <= position).saturating_sub(1);
78        self.samples.drain(..keep);
79    }
80    pub fn clear(&mut self) { self.samples.clear(); }
81    pub fn len(&self) -> usize { self.samples.len() }
82    pub fn is_empty(&self) -> bool { self.samples.is_empty() }
83}

111pub struct PresentationPlugin;
112impl Plugin for PresentationPlugin {
113    fn build(&self, app: &mut App) {
114        app.init_resource::<SpriteTracks>().init_resource::<PresentationDebug>();
115        app.configure_sets(GameTickPost, (PresentationSet::Prepare, PresentationSet::Capture).chain());
116        app.add_systems(GameTickPost, mark_attached_sprites.in_set(PresentationSet::Prepare));
117        app.add_systems(GameTickPost, capture_sprites.in_set(PresentationSet::Capture));
118        app.add_systems(PostUpdate, present_sprites.in_set(PresentationSet::Present).before(TransformSystems::Propagate));
119        if let Some(render) = app.get_sub_app_mut(RenderApp) {
120            render.add_systems(ExtractSchedule, suppress_simulation_sprites.after(SpriteSystems::ExtractSprites));
121        }
122    }
123}

125fn mark_attached_sprites(
126    mut commands: Commands,
127    sprites: Query<Entity, (With<Sprite>, Without<PresentationSource>, Or<(Added<Sprite>, Changed<ChildOf>)>)>,
128    parents: Query<&ChildOf>, sources: Query<(), Or<(With<crate::gent::Gent>, SourceFilter)>>,
129) {
130    for entity in &sprites {
131        if attached_to_source(entity, &parents, &sources) { commands.entity(entity).insert(PresentationSource); }
132    }
133}

135fn attached_to_source(mut entity: Entity, parents: &Query<&ChildOf>, sources: &Query<(), Or<(With<crate::gent::Gent>, SourceFilter)>>) -> bool {
136    while let Ok(parent) = parents.get(entity) {
137        entity = parent.parent();
138        if sources.contains(entity) { return true; }
139    }
140    false
141}

144struct CaptureQueries<'w, 's> {
145    sprites: Query<'w, 's, (Entity, &'static Sprite, &'static Anchor, Option<&'static RenderLayers>, Option<&'static TransformGfxFromGent>), SourceFilter>,
146    transforms: TransformHelper<'w, 's>,
147    visibility: Query<'w, 's, (Option<&'static Visibility>, Option<&'static ChildOf>)>,
148    policies: Query<'w, 's, &'static PresentationPolicy>,
149    alignments: Query<'w, 's, &'static PresentationAlignment>,
150    jumps: Query<'w, 's, &'static PresentationJump>,
151    anchors: Query<'w, 's, &'static PresentationAnchor>,
152    bindings: Query<'w, 's, &'static TransformGfxFromGent>,
153    stretches: Query<'w, 's, (&'static crate::effects::SpriteScale, Option<&'static crate::effects::sprite_stretch::SpriteStretch>)>,
154}

156fn capture_sprites(mut commands: Commands, time: Res<GameTime>, queries: CaptureQueries, mut tracks: ResMut<SpriteTracks>) {
157    for track in tracks.tracks.values_mut() {
158        if track.dead_at.is_none() { track.dead_at = Some(time.tick()); }
159    }
160    for (entity, sprite, sprite_anchor, layers, gent) in &queries.sprites {
161        let Ok(mut transform) = queries.transforms.compute_global_transform(entity) else { continue; };
162        if let Ok((scale, stretch)) = queries.stretches.get(entity) {
163            crate::effects::sprite_stretch::apply_render_stretch(scale, stretch, &mut transform);
164        }
165        let source = gent.map_or(entity, |binding| binding.gent);
166        let policy = queries.policies.get(entity).or_else(|_| queries.policies.get(source)).copied().unwrap_or_default();
167        let pixel_aligned = policy.pixel_aligned.unwrap_or_else(|| gent.is_some_and(|binding| binding.pixel_aligned));
168        let pose = SpritePose {
169            sprite: sprite.clone(), sprite_anchor: *sprite_anchor, transform: transform.compute_transform(),
170            visible: source_visible(entity, &queries.visibility), layers: layers.cloned().unwrap_or_default(),
171            alignment: queries.alignments.get(entity).or_else(|_| queries.alignments.get(source)).copied().unwrap_or_default(),
172            pixel_aligned, anchor: attachment_anchor(entity, &queries), stepped: policy.stepped,
173            jump: attachment_jumped(entity, time.tick(), &queries),
174        };
175        capture_sprite(&mut commands, &mut tracks, entity, time.tick(), pose);
176    }
177}

179fn attachment_parent(entity: Entity, queries: &CaptureQueries) -> Option<Entity> {
180    queries.anchors.get(entity).ok().map(|anchor| anchor.0)
181        .or_else(|| queries.visibility.get(entity).ok().and_then(|(_, parent)| parent.map(ChildOf::parent)))
182}

184fn attachment_anchor(mut entity: Entity, queries: &CaptureQueries) -> Option<Vec3> {
185    while let Some(parent) = attachment_parent(entity, queries) {
186        entity = parent;
187        if anchor_is_aligned(entity, queries) {
188            return queries.transforms.compute_global_transform(entity).ok().map(|transform| transform.translation());
189        }
190    }
191    None
192}
193
194fn anchor_is_aligned(entity: Entity, queries: &CaptureQueries) -> bool {
195    let Ok(binding) = queries.bindings.get(entity) else { return false; };
196    queries.policies.get(entity).or_else(|_| queries.policies.get(binding.gent)).ok()
197        .and_then(|policy| policy.pixel_aligned).unwrap_or(binding.pixel_aligned)
198}

200fn attachment_jumped(mut entity: Entity, tick: u64, queries: &CaptureQueries) -> bool {
201    loop {
202        let source = queries.bindings.get(entity).map_or(entity, |binding| binding.gent);
203        if queries.jumps.get(entity).or_else(|_| queries.jumps.get(source)).is_ok_and(|jump| jump.0 == tick) { return true; }
204        let Some(parent) = attachment_parent(entity, queries) else { return false; };
205        entity = parent;
206    }
207}

209fn align_pose_pixels(transform: &mut Transform, previous: &SpritePose, next: &SpritePose, alpha: f32) {
210    let anchor = if previous.pixel_aligned { Some(transform.translation) }
211        else { previous.anchor.map(|anchor| anchor.lerp(next.anchor.unwrap_or(anchor), alpha)) };
212    if let Some(anchor) = anchor {
213        transform.translation.x += anchor.x.round() - anchor.x;
214        transform.translation.y += anchor.y.round() - anchor.y;
215    }
216}

218fn capture_sprite(commands: &mut Commands, tracks: &mut SpriteTracks, entity: Entity, tick: u64, pose: SpritePose) {
219    let track = tracks.tracks.entry(entity).or_insert_with(|| {
220        commands.entity(entity).insert(PresentationSource);
221        SpriteTrack {
222        proxy: commands.spawn((PresentationProxy, PresentationProxySource(entity), Sprite::default(), Visibility::Hidden, Transform::default())).id(),
223        history: TickHistory::default(), dead_at: None,
224        }
225    });
226    track.dead_at = None;
227    track.history.push(tick, pose);
228}

230fn source_visible(mut entity: Entity, query: &Query<(Option<&Visibility>, Option<&ChildOf>)>) -> bool {
231    // Bevy rejects hierarchy cycles; a finite walk also tolerates a just-despawned parent.
232    loop {
233        let Ok((visibility, parent)) = query.get(entity) else { return false; };
234        if visibility == Some(&Visibility::Hidden) { return false; }
235        if visibility == Some(&Visibility::Visible) { return true; }
236        let Some(parent) = parent else { return true; };
237        entity = parent.parent();
238    }
239}

241fn present_sprites(
242    mut commands: Commands, mut tracks: ResMut<SpriteTracks>, cursor: Res<RenderInterpolation>,
243    mut debug: ResMut<PresentationDebug>,
244    mut proxies: ProxySprites,
245) {
246    let position = cursor.position();
247    tracks.tracks.retain(|_, track| {
248        if track.dead_at.is_some_and(|tick| position >= tick as f64) {
249            commands.entity(track.proxy).try_despawn();
250            false
251        } else { true }
252    });
253    debug.retained_samples = tracks.tracks.values().map(|track| track.history.len()).sum();
254    debug.proxies = tracks.tracks.len();
255    for track in tracks.tracks.values_mut() {
256        present_sprite(&mut proxies, track, position, &debug);
257        track.history.discard_before(position);
258    }
259}

261fn present_sprite(
262    proxies: &mut ProxySprites,
263    track: &SpriteTrack, position: f64, debug: &PresentationDebug,
264) {
265    let Ok((mut sprite, mut sprite_anchor, mut transform, mut visibility, mut layers, mut alignment)) = proxies.get_mut(track.proxy) else { return; };
266    let Some((previous, next, alpha)) = track.history.pair(position) else { visibility.set_if_neq(Visibility::Hidden); return; };
267    let alpha = if previous.stepped || next.jump || debug.interpolation_disabled { 0.0 } else { alpha };
268    let mut pose = interpolate_transform(&previous.transform, &next.transform, alpha);
269    align_pose_pixels(&mut pose, previous, next, alpha);
270    transform.set_if_neq(pose);
271    *sprite = previous.sprite.clone();
272    sprite_anchor.set_if_neq(previous.sprite_anchor);
273    visibility.set_if_neq(if previous.visible { Visibility::Visible } else { Visibility::Hidden });
274    layers.set_if_neq(previous.layers.clone());
275    alignment.set_if_neq(previous.alignment);
276}

278pub fn interpolate_transform(previous: &Transform, next: &Transform, alpha: f32) -> Transform {
279    Transform {
280        translation: previous.translation.lerp(next.translation, alpha),
281        rotation: previous.rotation.slerp(next.rotation, alpha),
282        scale: previous.scale.lerp(next.scale, alpha),
283    }
284}
