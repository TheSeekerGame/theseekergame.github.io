1//! Immediate stretch followed by cubic recovery starting on tick one.
2use crate::prelude::*;
3
4#[derive(Debug, Clone, Copy)]
5pub struct ScaleRecovery {
6    current_tick: u32,
7    base_scale: Vec3,
8}
9impl ScaleRecovery {
10    pub fn start(transform: &mut Transform, existing: Option<Self>, peak: Vec2) -> Self {
11        let base_scale = existing.map_or(transform.scale, |recovery| recovery.base_scale);
12        transform.scale = Vec3::new(base_scale.x * peak.x, base_scale.y * peak.y, base_scale.z);
13        Self { current_tick: 1, base_scale }
14    }
15    /// Returns true on the same tick that restores the base scale.
16    pub fn tick(&mut self, transform: &mut Transform, duration: u32, peak: Vec2) -> bool {
17        let t = (self.current_tick as f32 / duration as f32).clamp(0.0, 1.0);
18        let eased = 1.0 - (1.0 - t).powi(3);
19        let strength = 1.0 - eased;
20        transform.scale = Vec3::new(
21            self.base_scale.x * (1.0 - (1.0 - peak.x) * strength),
22            self.base_scale.y * (1.0 + (peak.y - 1.0) * strength), self.base_scale.z,
23        );
24        self.current_tick += 1;
25        if self.current_tick > duration {
26            transform.scale = self.base_scale;
27            return true;
28        }
29        false
30    }
31}

33#[cfg(test)]
34mod tests {
35    use super::*;
36    #[test]
37    fn retrigger_uses_original_scale_and_finishes_on_the_last_tick() {
38        let base = Vec3::new(2.0, 3.0, 4.0);
39        let peak = Vec2::new(0.5, 1.5);
40        let mut transform = Transform::from_scale(base);
41        let mut recovery = ScaleRecovery::start(&mut transform, None, peak);
42        assert!(!recovery.tick(&mut transform, 3, peak));
43        recovery = ScaleRecovery::start(&mut transform, Some(recovery), peak);
44        assert_eq!(transform.scale, Vec3::new(1.0, 4.5, 4.0));
45        assert!(!recovery.tick(&mut transform, 3, peak));
46        assert!(!recovery.tick(&mut transform, 3, peak));
47        assert!(recovery.tick(&mut transform, 3, peak));
48        assert_eq!(transform.scale, base);
49    }
50}
