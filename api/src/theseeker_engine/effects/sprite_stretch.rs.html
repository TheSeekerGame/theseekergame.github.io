5/// Anchor point for the stretch/squeeze effect
6#[derive(Debug, Clone, Copy, PartialEq)]
7pub enum StretchAnchor {
8    /// Stretch from center (default sprite behavior)
9    Center,
10    /// Keep bottom edge fixed, stretch upward only
11    Bottom,
12}
13
14/// Component that drives sprite stretching/squeezing animation
15#[derive(Component, Debug)]
16pub struct SpriteStretch {
17    /// Current tick in the animation
18    pub current_tick: u32,
19    /// Ticks to reach maximum deformation
20    pub squeeze_ticks: u32,
21    /// Ticks to hold at maximum deformation
22    pub hold_ticks: u32,
23    /// Ticks to return to normal from maximum deformation
24    pub release_ticks: u32,
25    /// Scale factor at maximum deformation (horizontal)
26    pub max_squeeze_x: f32,
27    /// Scale factor at maximum deformation (vertical)
28    pub max_stretch_y: f32,
29    /// How the stretch should be anchored
30    pub anchor: StretchAnchor,
31    /// Estimated sprite height in pixels (used for bottom-anchor offset calculation)
32    pub sprite_height: f32,
33}
34
35impl SpriteStretch {
36    /// Total duration of the animation in ticks
37    pub fn total_ticks(&self) -> u32 {
38        self.squeeze_ticks + self.hold_ticks + self.release_ticks
39    }
40}
41
42impl Default for SpriteStretch {
43    fn default() -> Self {
44        Self {
45            current_tick: 0,
46            squeeze_ticks: 0,
47            hold_ticks: 0,
48            release_ticks: 8,
49            max_squeeze_x: 0.7,
50            max_stretch_y: 1.3,
51            anchor: StretchAnchor::Bottom,
52            sprite_height: 24.0, // Reasonable default
53        }
54    }
55}

72/// Updates the stretch animation state each tick
73fn update_sprite_stretch(
74    mut query: Query<(
75        Entity,
76        &mut SpriteStretch,
77        &mut SpriteScale,
78    )>,
79    mut commands: Commands,
80) {
81    for (entity, mut stretch, mut scale) in query.iter_mut() {
82        let squeeze_end = stretch.squeeze_ticks;
83        let hold_end = squeeze_end + stretch.hold_ticks;
84        let total_ticks = stretch.total_ticks();
85
86        // Interpolation progress based on animation phase
87        let progress = if stretch.current_tick < squeeze_end {
88            // Phase 1: Squeeze phase (0 to squeeze_ticks)
89            // Interpolate from 0.0 to 1.0 (no deformation to max deformation)
90            stretch.current_tick as f32 / stretch.squeeze_ticks as f32
91        } else if stretch.current_tick < hold_end {
92            // Phase 2: Hold phase (squeeze_ticks to squeeze_ticks + hold_ticks)
93            // Stay at maximum deformation (progress = 1.0)
94            1.0
95        } else if stretch.current_tick < total_ticks {
96            // Phase 3: Release phase (hold_end to total_ticks)
97            // Interpolate from 1.0 to 0.0 (max deformation back to normal)
98            let release_progress = (stretch.current_tick - hold_end) as f32
99                / stretch.release_ticks as f32;
100            1.0 - release_progress
101        } else {
102            // Animation complete
103            0.0
104        };
105
106        // Easing curve (smooth in/out) for squeeze and release phases
107        let eased_progress = if stretch.current_tick < squeeze_end
108            || stretch.current_tick >= hold_end
109        {
110            ease_in_out_cubic(progress)
111        } else {
112            progress // No easing during hold phase
113        };
114
115        // Apply scale values
116        scale.x = 1.0 - (1.0 - stretch.max_squeeze_x) * eased_progress;
117        scale.y = 1.0 + (stretch.max_stretch_y - 1.0) * eased_progress;
118
119        // Advance animation
120        stretch.current_tick += 1;
121
122        // Remove component when animation completes
123        if stretch.current_tick >= total_ticks {
124            commands.entity(entity).remove::<SpriteStretch>();
125            scale.reset();
126        }
127    }
128}

130pub(crate) fn apply_render_stretch(
131    scale: &SpriteScale,
132    stretch_opt: Option<&SpriteStretch>,
133    global_transform: &mut GlobalTransform,
134) {
135    let mut transform = global_transform.compute_transform();
136
137    let x_sign = transform.scale.x.signum();
138    let y_sign = transform.scale.y.signum();
139    transform.scale.x = x_sign * scale.x.abs();
140    transform.scale.y = y_sign * scale.y.abs();
141    transform.scale.z = 1.0;
142
143    apply_bottom_stretch_offset(scale, stretch_opt, &mut transform);
144    *global_transform = transform.into();
145}
146
147fn apply_bottom_stretch_offset(
148    scale: &SpriteScale,
149    stretch_opt: Option<&SpriteStretch>,
150    transform: &mut Transform,
151) {
152    let Some(stretch) = stretch_opt else {
153        return;
154    };
155    if stretch.anchor != StretchAnchor::Bottom {
156        return;
157    }
158
159    let vertical_offset = (scale.y - 1.0) * stretch.sprite_height * 0.5;
160    transform.translation.y += vertical_offset;
161}
162
163/// Cubic easing function for smooth animation
164fn ease_in_out_cubic(t: f32) -> f32 {
165    if t < 0.5 {
166        4.0 * t * t * t
167    } else {
168        1.0 - (-2.0 * t + 2.0).powi(3) / 2.0
169    }
