20/// Fade curve for ghost opacity over time
21#[derive(Debug, Clone, Copy, PartialEq)]
22pub enum FadeCurve {
23    /// Linear fade from initial_alpha to 0
24    Linear,
25    /// Ease-out curve (fast fade at start, slow at end)
26    EaseOut,
27    /// Exponential decay
28    Exponential,
29    /// Custom cubic bezier (t^3 interpolation)
30    Cubic,
31    /// Keep alpha constant for full-lifetime visibility
32    Constant,
33}

35/// Scale curve for ghost size over time
36#[derive(Debug, Clone, Copy, PartialEq)]
37pub enum ScaleCurve {
38    /// Maintain original scale
39    Constant,
40    /// Shrink to a percentage of original (0.0 to 1.0)
41    Shrink(f32),
42    /// Grow to a percentage of original (> 1.0)
43    Grow(f32),
44    /// Shrink horizontally, maintain vertical
45    ShrinkHorizontal(f32),
46    /// Scale each axis independently to a target percentage of original
47    Axes { x: f32, y: f32 },
48}

50/// Component that marks an entity to produce ghost trail effects.
51/// Attach to an entity with a `Sprite` (or a `Gent` with a sprite) to emit ghosts.
52#[derive(Component, Debug)]
53pub struct GhostingSource {
54    /// How many ticks between spawning ghosts (1 = every tick, 2 = every other, etc.)
55    pub spawn_interval_ticks: u32,
56    /// How many ticks each ghost lives
57    pub ghost_lifetime_ticks: u32,
58    /// Starting opacity for ghosts (0.0 to 1.0)
59    pub initial_alpha: f32,
60    /// How the ghost fades over time
61    pub fade_curve: FadeCurve,
62    /// Color mode for the ghosts
63    pub color_mode: GhostColorMode,
64    /// How the ghost scales over time
65    pub scale_over_time: ScaleCurve,
66    /// Position offset from source entity
67    pub offset: Vec2,
68    /// Motion to apply to spawned ghosts over their lifetime
69    pub movement: GhostMovement,
70    /// Render layers for spawned ghosts. If None, uses GhostingDefaults.
71    pub render_layers: Option<RenderLayers>,
72
73    /// Internal counter for spawn timing
74    pub ticks_since_last_spawn: u32,
75}

87            movement: GhostMovement::Static,

102/// Motion model for spawned ghosts
103#[derive(Debug, Clone, Copy, PartialEq)]
104pub enum GhostMovement {

107    /// Follow the source velocity, retaining the spawn velocity if the source disappears

139impl Plugin for GhostingPlugin {
140    fn build(&self, app: &mut App) {
141        // Configure dedicated ghosting sets so gameplay systems can order around them
142        app.configure_sets(
143            GameTickUpdate,
144            (
145                GhostingSet::Spawn
146                    .after(crate::animation::AnimationSet::Signals)
147                    .after(
148                        bevy::transform::TransformSystems::Propagate,
149                    ),
150                GhostingSet::Update.after(GhostingSet::Spawn),
151            ),
152        );
153        // Spawn ghosts from source entities
154        app.add_systems(
155            GameTickUpdate,
156            spawn_ghosts.in_set(GhostingSet::Spawn),
157        );
158
159        // Update ghost opacity/scale and cleanup
160        app.add_systems(
161            GameTickUpdate,
162            update_ghosts.in_set(GhostingSet::Update),
163        );
164    }

174/// System that spawns ghost entities from sources
175fn spawn_ghosts(
176    mut query: Query<(
177        Entity,
178        &mut GhostingSource,
179        &GlobalTransform,
180        Option<&Sprite>,
181        Option<&Anchor>,
182        Option<&crate::gent::Gent>,
183        Option<&LinearVelocity>,
184    )>,
185    gfx_query: Query<(&GlobalTransform, &Sprite, &Anchor), Without<GhostingSource>>,
186    vel_query: Query<&LinearVelocity>,
187    any_tf_query: Query<&GlobalTransform>,
188    defaults: Option<Res<GhostingDefaults>>,
189    mut commands: Commands,
190) {
191    for (
192        source_entity,
193        mut source,
194        transform,
195        sprite_opt,
196        anchor_opt,
197        gent_opt,
198        self_vel_opt,
199    ) in query.iter_mut()
200    {
201        if !source_ready_to_spawn(&mut source) {
202            continue;
203        }
204
205        let Some(visual) =
206            source_visual(transform, sprite_opt, anchor_opt, gent_opt, &gfx_query)
207        else {
208            continue;
209        };
210
211        let ghost_pos = ghost_spawn_position(visual.transform, source.offset);
212        let base_color_rgb =
213            ghost_base_color(visual.sprite, source.color_mode);
214        let ghost_color = base_color_rgb.with_alpha(source.initial_alpha);
215        let ghost_sprite = ghost_sprite_from_source(visual.sprite, ghost_color);
216        let source_world_transform = visual.transform.compute_transform();
217        let base_scale = source_world_transform.scale;
218        let base_rotation = source_world_transform.rotation;
219        let motion = ghost_spawn_motion(
220            source_entity,
221            &source,
222            transform,
223            ghost_pos,
224            self_vel_opt,
225            &vel_query,
226            &any_tf_query,
227        );
228        let spawn_transform =
229            ghost_spawn_transform(ghost_pos, base_rotation, base_scale, motion);
230
231        spawn_ghost_entity(
232            &mut commands,
233            &source,
234            GhostSpawnBundle {
235                sprite: ghost_sprite,
236                anchor: visual.anchor,
237                transform: spawn_transform,
238                base_scale,
239                base_color: base_color_rgb,
240                motion,
241            },
242            defaults.as_deref(),
243        );
244    }
245}

270fn source_ready_to_spawn(source: &mut GhostingSource) -> bool {
271    source.ticks_since_last_spawn += 1;
272    let interval = source.spawn_interval_ticks.max(1);
273    if source.ticks_since_last_spawn < interval {
274        return false;
275    }
276    source.ticks_since_last_spawn = 0;
277    true
278}

280fn source_visual<'a>(
281    transform: &'a GlobalTransform,
282    sprite_opt: Option<&'a Sprite>,
283    anchor_opt: Option<&Anchor>,
284    gent_opt: Option<&crate::gent::Gent>,
285    gfx_query: &'a Query<(&GlobalTransform, &Sprite, &Anchor), Without<GhostingSource>>,
286) -> Option<SourceVisual<'a>> {
287    if let Some(gent) = gent_opt {
288        let (gfx_transform, gfx_sprite, gfx_anchor) = gfx_query.get(gent.e_gfx).ok()?;
289        return Some(SourceVisual {
290            transform: gfx_transform,
291            sprite: gfx_sprite,
292            anchor: *gfx_anchor,
293        });
294    }
295
296    sprite_opt.map(|sprite| SourceVisual { transform, sprite, anchor: anchor_opt.copied().unwrap_or_default() })
297}

299fn ghost_spawn_position(transform: &GlobalTransform, offset: Vec2) -> Vec3 {
300    let source_translation = transform.translation();
301    Vec3::new(
302        (source_translation.x + offset.x).round(),
303        (source_translation.y + offset.y).round(),
304        source_translation.z - GHOST_Z_EPSILON,
305    )
306}

308fn ghost_base_color(sprite: &Sprite, color_mode: GhostColorMode) -> Color {
309    let base = sprite.color.to_srgba();
310    match color_mode {
311        GhostColorMode::Original => Color::srgb(base.red, base.green, base.blue),
312        GhostColorMode::Tint(tint) => {
313            let tint = tint.to_srgba();
314            Color::srgb(
315                base.red * tint.red,
316                base.green * tint.green,
317                base.blue * tint.blue,
318            )
319        },
320    }
321}

336fn ghost_spawn_motion(
337    source_entity: Entity,
338    source: &GhostingSource,
339    transform: &GlobalTransform,
340    ghost_pos: Vec3,
341    self_vel_opt: Option<&LinearVelocity>,
342    vel_query: &Query<&LinearVelocity>,
343    any_tf_query: &Query<&GlobalTransform>,
344) -> GhostMotion {
345    match source.movement {
346        GhostMovement::Static => GhostMotion::default(),
347        GhostMovement::CopyLinearVelocityFromSelf => GhostMotion::Velocity {
348            source: source_entity,
349            fallback: self_vel_opt.map(|v| v.0).unwrap_or(Vec2::ZERO),
350        },
351        GhostMovement::CopyLinearVelocityFrom(entity) => GhostMotion::Velocity {
352            source: entity,
353            fallback: vel_query.get(entity).map(|v| v.0).unwrap_or(Vec2::ZERO),
354        },
355        GhostMovement::CopyDisplacementFromSelf => {
356            displacement_follow_motion(source_entity, transform.translation(), ghost_pos)
357        },
358        GhostMovement::CopyDisplacementFrom(entity) => {
359            let parent_world = any_tf_query
360                .get(entity)
361                .map(bevy::prelude::GlobalTransform::translation)
362                .unwrap_or_else(|_| transform.translation());
363            displacement_follow_motion(entity, parent_world, ghost_pos)
364        },
365    }
366}

369fn displacement_follow_motion(parent: Entity, parent_world: Vec3, ghost_pos: Vec3) -> GhostMotion {
370    GhostMotion::Parent { entity: parent, local: ghost_pos - parent_world }
371}

373fn ghost_spawn_transform(
374    ghost_pos: Vec3,
375    base_rotation: Quat,
376    base_scale: Vec3,
377    motion: GhostMotion,
378) -> Transform {
379    if let GhostMotion::Parent { local, .. } = motion {
380        return Transform::from_translation(local).with_scale(base_scale);
381    }
382
383    Transform::from_translation(ghost_pos)
384        .with_rotation(base_rotation)
385        .with_scale(base_scale)
386}

388fn spawn_ghost_entity(
389    commands: &mut Commands,
390    source: &GhostingSource,
391    spawn: GhostSpawnBundle,
392    defaults: Option<&GhostingDefaults>,
393) {
394    let mut entity_commands = commands.spawn((
395        spawn.sprite,
396        spawn.anchor,
397        crate::presentation::PresentationSource,
398        spawn.transform,
399        GlobalTransform::default(),
400        Visibility::default(),
401        InheritedVisibility::default(),
402        ViewVisibility::default(),
403        Ghost {
404            ticks_remaining: source.ghost_lifetime_ticks,
405            max_ticks: source.ghost_lifetime_ticks,
406            initial_alpha: source.initial_alpha,
407            fade_curve: source.fade_curve,
408            scale_curve: source.scale_over_time,
409            base_scale: spawn.base_scale,
410            base_color: spawn.base_color,
411            motion: spawn.motion,
412        },
413    ));
414
415    let layers = source
416        .render_layers
417        .clone()
418        .or_else(|| defaults.and_then(|d| d.render_layers.clone()));
419    if let Some(layers) = layers {
420        entity_commands.insert(layers);
421    }
422
423    if let GhostMotion::Parent { entity: parent, .. } = spawn.motion {
424        entity_commands.insert((ChildOf(parent),));
425    }
426}

428/// System that updates ghost appearance over time
429fn update_ghosts(
430    mut commands: Commands,
431    mut query: Query<(Entity, &mut Ghost, &mut Sprite, &mut Transform)>,
432    vel_query: Query<&LinearVelocity>,
433) {
434    for (entity, mut ghost, mut sprite, mut transform) in query.iter_mut() {
435        // Update lifetime
436        if ghost.ticks_remaining > 0 {
437            ghost.ticks_remaining = ghost.ticks_remaining.saturating_sub(1);
438        }
439
440        apply_ghost_movement(&ghost, &vel_query, &mut transform);
441
442        // Calculate normalized progress (0 = just spawned, 1 = expired)
443        let ticks_elapsed =
444            ghost.max_ticks.saturating_sub(ghost.ticks_remaining) as f32;
445        let progress = ticks_elapsed / ghost.max_ticks as f32;
446        let progress = progress.clamp(0.0, 1.0);
447
448        // Apply fade curve
449        let alpha = match ghost.fade_curve {
450            FadeCurve::Linear => ghost.initial_alpha * (1.0 - progress),
451            FadeCurve::EaseOut => {
452                // Quadratic ease out: 1 - (1 - x)^2
453                let t = 1.0 - progress;
454                ghost.initial_alpha * t * t
455            },
456            FadeCurve::Exponential => {
457                // Exponential decay
458                ghost.initial_alpha * (1.0 - progress).powf(2.5)
459            },
460            FadeCurve::Cubic => {
461                // Cubic curve for smoother fade
462                let t = 1.0 - progress;
463                ghost.initial_alpha * t * t * t
464            },
465            FadeCurve::Constant => ghost.initial_alpha,
466        };
467
468        // Update sprite alpha while preserving the base color
469        sprite.color = ghost.base_color.with_alpha(alpha);
470
471        if let Some(scale) = ghost_scale(ghost.scale_curve, progress) {
472            transform.scale = ghost.base_scale * scale;
473        }
474        if ghost.ticks_remaining == 0 {
475            commands.entity(entity).despawn();
476        }
477    }
478}

480fn ghost_scale(curve: ScaleCurve, progress: f32) -> Option<Vec3> {
481    Some(match curve {
482        ScaleCurve::Constant => return None,
483        ScaleCurve::Shrink(target) => Vec3::splat(1.0 - (1.0 - target) * progress),
484        ScaleCurve::Grow(target) => Vec3::splat(1.0 + (target - 1.0) * progress),
485        ScaleCurve::ShrinkHorizontal(target) => Vec3::new(1.0 - (1.0 - target) * progress, 1.0, 1.0),
486        ScaleCurve::Axes { x, y } => Vec3::new(1.0 - (1.0 - x) * progress, 1.0 - (1.0 - y) * progress, 1.0),
487    })
488}

490fn apply_ghost_movement(
491    ghost: &Ghost,
492    vel_query: &Query<&LinearVelocity>,
493    transform: &mut Transform,
494) {
495    let vel = ghost_velocity(ghost, vel_query);
496    transform.translation.x += vel.x;
497    transform.translation.y += vel.y;
498}
499
500fn ghost_velocity(ghost: &Ghost, vel_query: &Query<&LinearVelocity>) -> Vec2 {
501    match ghost.motion {
502        GhostMotion::Velocity { source, fallback } => vel_query.get(source).map_or(fallback, |velocity| velocity.0),
503        GhostMotion::Static | GhostMotion::Parent { .. } => Vec2::ZERO,
504    }
505}

509mod tests {
510    use super::*;
511    fn ghost(ticks: u32, motion: GhostMotion) -> Ghost {
512        Ghost { ticks_remaining: ticks, max_ticks: ticks, initial_alpha: 1.0,

516    #[test]
517    fn velocity_tracks_live_source_then_reverts_to_spawn_velocity_without_resetting_scale() {
518        let mut app = App::new();
519        app.add_systems(Update, update_ghosts);
520        let source = app.world_mut().spawn(LinearVelocity(Vec2::new(3.0, 4.0))).id();
521        let entity = app.world_mut().spawn((ghost(4, GhostMotion::Velocity { source, fallback: Vec2::X }),
522            Sprite::default(), Transform::from_scale(Vec3::splat(2.0)))).id();
523        app.update();
524        assert_eq!(app.world().get::<Transform>(entity).unwrap().translation.xy(), Vec2::new(3.0, 4.0));
525        app.world_mut().get_mut::<LinearVelocity>(source).unwrap().0 = Vec2::new(2.0, 1.0);
526        app.update();
527        app.world_mut().despawn(source);
528        app.update();
529        let transform = app.world().get::<Transform>(entity).unwrap();
530        assert_eq!(transform.translation.xy(), Vec2::new(6.0, 5.0));
531        assert_eq!(transform.scale, Vec3::splat(2.0));
532        app.update();
533        assert!(app.world().get_entity(entity).is_err());
534    }

536    fn zero_and_one_tick_ghosts_expire_on_the_first_update() {
537        let mut app = App::new();

539        let entities = [0, 1].map(|ticks| app.world_mut().spawn((ghost(ticks, GhostMotion::Static),
540            Sprite::default(), Transform::default())).id());

542        for entity in entities { assert!(app.world().get_entity(entity).is_err()); }
543    }
