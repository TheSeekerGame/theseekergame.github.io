74    pub(crate) fn test_running(now: Instant, frame: i64, generation: u64) -> Self {

80    pub fn healthy(&self, now: Instant) -> bool {
81        self.running && self.snapshot.is_some_and(|snapshot| {
82            let deadline = Duration::from_secs_f64((f64::from(self.block_frames) / f64::from(self.sample_rate) * 8.0).max(0.25));
83            now.saturating_duration_since(snapshot.observed) < deadline

88        self.snapshot.map_or(0.0, |snapshot| {

94        self.snapshot.map_or(0.0, |snapshot| snapshot.mapping.position(self.device_position(at)))

101fn signed_seconds(a: Instant, b: Instant) -> f64 {
102    if a >= b { a.duration_since(b).as_secs_f64() } else { -b.duration_since(a).as_secs_f64() }

191        let (command_tx, command_rx) = bounded(config.command_capacity.max(1));

256    fn play(&mut self, bus: AudioBus, label: Option<&str>, handle: &Handle<AudioSource>, config: AudioPlaybackConfig) -> Option<AudioInstanceId> {
257        if !self.running() { return None; }
258        if !config.volume.is_finite() || !config.pan.is_finite() || self.pending.len() >= self.config.command_capacity {
259            self.dropped_commands += 1;
260            return None;
261        }
262        if bus != AudioBus::Sfx && self.instances.values().filter(|instance| instance.bus != AudioBus::Sfx).count() >= self.config.max_background_voices.max(2) {
263            return None;
264        }
265        let asset = handle.id().untyped();
266        if self.failed.contains(&asset) { return None; }
267        self.prepare(handle);
268        let id = AudioInstanceId(self.next_id);
269        self.next_id += 1;
270        let at = self.resolve_time(config.timing);
271        let order = self.order();
272        let ended = self.completion();
273        self.instances.insert(id, AudioInstance { bus, label: label.map(str::to_owned), asset, _handle: handle.clone(), ended });
274        self.pending.push(PendingPlay { id, asset, config, at, order });
275        Some(id)
276    }

385    pub(super) fn update(&mut self) {
386        while let Ok(snapshot) = self.clock_rx.try_recv() {
387            self.resync_requested |= snapshot.stats.resync_requests > self.diagnostics.resync_requests;
388            self.diagnostics = snapshot.stats;
389            self.clock.block_frames = snapshot.block_frames;
390            self.clock.generation = snapshot.epoch + self.session_generation;
391            self.clock.snapshot = Some(snapshot);
392        }
393        self.clock.running = self.running();
394        let completed = self.diagnostics.callbacks;
395        self.instances.retain(|_, instance| instance.ended.load(Ordering::Acquire) > completed);
396        {
397            let mut pins = self.pins.lock().expect("Audio pins lock poisoned");
398            pins.samples.retain(|sample| Arc::strong_count(sample) > 1);
399            pins.completions.retain(|ended| Arc::strong_count(ended) > 1);
400        }
401        self.flush();
402    }

404    pub(crate) fn flush(&mut self) {
405        let pending = std::mem::take(&mut self.pending);
406        for play in pending { self.submit_play(play); }
407        flush_controls(&self.control_tx, &mut self.controls);
408    }

410    fn submit_play(&mut self, pending: PendingPlay) {
411        let Some(instance) = self.instances.get(&pending.id) else { return; };
412        if self.failed.contains(&pending.asset) { self.instances.remove(&pending.id); return; }
413        let Some(prepared) = self.samples.get(&pending.asset) else { self.pending.push(pending); return; };
414        let sample = &prepared.sample;
415        let region = resolve_loop(&pending.config, sample.frames, self.sample_rate());
416        if pending.config.looped && region.is_none() { self.instances.remove(&pending.id); return; }
417        let command = Command { at: pending.at, epoch: self.epoch, order: pending.order, kind: CommandKind::Play(Play {
418            id: pending.id.0, sample: sample.clone(), ended: instance.ended.clone(), bus: instance.bus,
419            volume: pending.config.volume.max(0.0), pan: pending.config.pan,
420            loop_range: region, position: duration_sample(pending.config.position, self.sample_rate()) as usize,
421            fade_frames: duration_sample(pending.config.fade_in, self.sample_rate()) as usize,
422        }) };
423        if self.command_tx.try_send(command).is_err() { self.pending.push(pending); self.dropped_commands += 1; }
424    }

427fn resolve_loop(config: &AudioPlaybackConfig, frames: usize, rate: u32) -> Option<(usize, usize)> {
428    if !config.looped || frames == 0 { return None; }
429    let Some((start, end, source_rate)) = config.loop_region else { return Some((0, frames)); };
430    if source_rate == 0 || end <= start { return None; }
431    let start = tick_sample(start, rate, source_rate) as usize;
432    let end = (tick_sample(end, rate, source_rate) as usize).min(frames);
433    (end > start).then_some((start, end))
434}

436fn decode(bytes: Arc<[u8]>, rate: u32) -> Result<Arc<SampleData>, String> {
437    let decoded = SymphoniumLoader::new().load_f32_from_source(
438        Box::new(std::io::Cursor::new(bytes)), None, Some(rate), ResampleQuality::High, None,
439    ).map_err(|error| error.to_string())?;
440    let channels = decoded.channels().min(2);
441    let frames = decoded.frames();
442    if frames == 0 || channels == 0 { return Err("Audio contains no frames".to_owned()); }
443    let mut data = Vec::with_capacity(frames * channels);
444    for frame in 0..frames { for channel in 0..channels { data.push(decoded.data[channel][frame]); } }
445    Ok(Arc::new(SampleData { channels, frames, data }))
446}

448pub(super) fn prepare_audio(
449    mut engine: ResMut<AudioEngine>, assets: Res<Assets<AudioSource>>, server: Res<AssetServer>,
450    mut events: MessageReader<AssetEvent<AudioSource>>,
451) {
452    for event in events.read() { invalidate_prepared(&mut engine, event); }
453    let mut completed = Vec::new();
454    for (&id, (task, deferred)) in &mut engine.tasks {
455        if let Some(result) = block_on(future::poll_once(task)) { completed.push((id, *deferred, result)); }
456    }
457    for (id, deferred, result) in completed {
458        engine.tasks.remove(&id);
459        match result {
460            Ok(sample) => {
461                engine.pins.lock().expect("Audio pins lock poisoned").samples.push(sample.clone());
462                engine.samples.insert(id, PreparedSample { sample, deferred });
463            },
464            Err(error) => { warn!("Audio preparation failed for {id:?}: {error}"); engine.failed.insert(id); },
465        }
466    }
467    for (id, source) in assets.iter() {
468        let id = id.untyped();
469        let deferred = server.get_path(id).is_some_and(|path| path.path().starts_with("audio/music"));
470        let wanted = !deferred || engine.requested.contains(&id);
471        if wanted && !engine.samples.contains_key(&id) && !engine.tasks.contains_key(&id) && !engine.failed.contains(&id) {
472            let bytes = source.bytes.clone();
473            let rate = engine.sample_rate();
474            engine.tasks.insert(id, (spawn_decode(bytes, rate), deferred));
475        }
476    }
477    let evict: Vec<_> = engine.samples.iter().filter(|(id, sample)| sample.deferred
478        && !engine.retained_music.contains(id) && !engine.requested.contains(id)
479        && !engine.instances.values().any(|instance| &instance.asset == *id)).map(|(&id, _)| id).collect();
480    for id in evict { engine.samples.remove(&id); }
481    engine.requested.clear();
482}

484fn invalidate_prepared(engine: &mut AudioEngine, event: &AssetEvent<AudioSource>) {
485    let (AssetEvent::Modified { id } | AssetEvent::Removed { id }) = event else { return; };
486    let id = id.untyped();
487    engine.samples.remove(&id);
488    engine.failed.remove(&id);
489    engine.tasks.remove(&id);
490}

492pub(super) fn update_audio_engine(mut engine: ResMut<AudioEngine>, mut clock: ResMut<AudioClock>, mut stream: NonSendMut<AudioStream>) {
493    engine.update();
494    let now = Instant::now();
495    let disconnected = !engine.running() || engine.clock.snapshot.is_some_and(|_| !engine.clock.healthy(now));
496    if engine.config.output_enabled && disconnected && now >= engine.reconnect_at {
497        stream.stream = None;
498        let config = engine.config.clone();
499        let generation = engine.session_generation + 1_000_000;
500        let (mut replacement, replacement_stream) = AudioEngine::build(&config);
501        replacement.session_generation = generation;
502        replacement.next_id = engine.next_id;
503        replacement.next_order = engine.next_order;
504        *engine = replacement;
505        *stream = replacement_stream;
506    }
507    *clock = engine.clock.clone();
508}

512fn flush_controls(sender: &Sender<Control>, controls: &mut VecDeque<Control>) {
513    while let Some(control) = controls.pop_front() {
514        if let Err(error) = sender.try_send(control) {
515            controls.push_front(error.into_inner());
516            break;
517        }
518    }
519}
