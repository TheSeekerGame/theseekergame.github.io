13fn select_output_config(
14    device: &cpal::Device,
15    desired_sample_rate: u32,
16    desired_block_frames: u32,
17) -> Result<SelectedConfig, String> {
18    let mut best: Option<(i32, SelectedConfig)> = None;
19    let configs = device
20        .supported_output_configs()
21        .map_err(|e| e.to_string())?;
22
23    for range in configs {
24        let format_score = match range.sample_format() {
25            SampleFormat::F32 => 3,
26            SampleFormat::I16 => 2,
27            SampleFormat::U16 => 1,
28            _ => continue,
29        };
30        let channels = range.channels();
31        let channel_score = if channels == 2 {
32            2
33        } else { i32::from(channels > 2) };
34        let min_rate = range.min_sample_rate().0;
35        let max_rate = range.max_sample_rate().0;
36        let chosen_rate = desired_sample_rate.clamp(min_rate, max_rate);
37        let rate_score = if chosen_rate == desired_sample_rate {
38            2
39        } else {
40            0
41        };
42
43        let supported = range
44            .try_with_sample_rate(SampleRate(chosen_rate))
45            .ok_or_else(|| "Unsupported sample rate".to_string())?;
46        let buffer_size =
47            selected_buffer_size(range.buffer_size(), desired_block_frames);
48
49        let mut config = supported.config();
50        config.buffer_size = buffer_size;
51        let block_frames = match config.buffer_size {
52            BufferSize::Fixed(frames) => frames,
53            BufferSize::Default => desired_block_frames,
54        };
55
56        let score = rate_score * 100 + channel_score * 10 + format_score;
57        let selected = SelectedConfig {
58            config,
59            sample_format: range.sample_format(),
60            sample_rate: chosen_rate,
61            block_frames,
62        };
63
64        match best {
65            Some((best_score, _)) if best_score >= score => {},
66            _ => {
67                best = Some((score, selected));
68            },
69        }
70    }
71
72    if let Some((_score, selected)) = best {
73        Ok(selected)
74    } else {
75        Err("No supported output configs".to_string())
76    }
77}

79fn selected_buffer_size(
80    supported: &SupportedBufferSize,
81    desired_block_frames: u32,
82) -> BufferSize {
83    let SupportedBufferSize::Range { min, max } = supported else {
84        return BufferSize::Default;
85    };
86
87    if desired_block_frames >= *min && desired_block_frames <= *max {
88        return BufferSize::Fixed(desired_block_frames);
89    }
90
91    BufferSize::Default
92}

95pub(super) fn start(
96    config: &AudioEngineConfig,
97    commands: crossbeam_channel::Receiver<super::mixer::Command>,
98    controls: crossbeam_channel::Receiver<super::mixer::Control>,
99    snapshots: crossbeam_channel::Sender<ClockSnapshot>,
100    obsolete_snapshots: crossbeam_channel::Receiver<ClockSnapshot>,
101    running: std::sync::Arc<std::sync::atomic::AtomicBool>,
102) -> Result<(cpal::Stream, u32, u32), String> {
103    let device = cpal::default_host().default_output_device().ok_or("No output device")?;
104    let mut selected = select_output_config(&device, config.desired_sample_rate, config.desired_block_frames)?;
105    // A backend can advertise a range but reject a particular buffer size.
106    let make_state = || Callback {
107        mixer: Mixer::new(commands.clone(), controls.clone(), config.max_sfx_voices.max(1),
108            config.max_background_voices.max(2), config.command_capacity.max(1), selected.sample_rate, config.tick_rate_hz),
109        snapshots: snapshots.clone(), obsolete_snapshots: obsolete_snapshots.clone(), previous_arrival: None, channels: selected.config.channels as usize,
110        rate: selected.sample_rate, timestamps: TimestampAnchor::default(),
111    };
112    let first = build_stream(&device, &selected, make_state(), running.clone());
113    let stream = if let Ok(stream) = first { stream } else {
114        let state = make_state();
115        selected.config.buffer_size = BufferSize::Default;
116        build_stream(&device, &selected, state, running.clone())?
117    };
118    stream.play().map_err(|error| error.to_string())?;
119    running.store(true, std::sync::atomic::Ordering::Release);
120    bevy::log::info!("Custom audio: {} Hz, {} channels, {:?}", selected.sample_rate, selected.config.channels, selected.config.buffer_size);
121    Ok((stream, selected.sample_rate, selected.block_frames))
122}

124fn build_stream(
125    device: &cpal::Device, selected: &SelectedConfig, mut state: Callback,
126    running: std::sync::Arc<std::sync::atomic::AtomicBool>,
127) -> Result<cpal::Stream, String> {
128    let on_error = move |_error| { running.store(false, std::sync::atomic::Ordering::Release); };
129    match selected.sample_format {
130        SampleFormat::F32 => device.build_output_stream(&selected.config, move |data: &mut [f32], info| state.process(data, info), on_error, None),
131        SampleFormat::I16 => device.build_output_stream(&selected.config, move |data: &mut [i16], info| state.process(data, info), on_error, None),
132        SampleFormat::U16 => device.build_output_stream(&selected.config, move |data: &mut [u16], info| state.process(data, info), on_error, None),
133        _ => return Err("Unsupported sample format".to_owned()),
134    }.map_err(|error| error.to_string())
135}

137#[derive(Default)]
138struct TimestampAnchor {
139    first: Option<(cpal::StreamInstant, std::time::Instant)>,
140    previous: Option<cpal::StreamInstant>,
141    fallbacks: u64,
142}
143
144impl TimestampAnchor {
145    fn playback(&mut self, info: &cpal::OutputCallbackInfo, now: std::time::Instant, block: std::time::Duration) -> std::time::Instant {
146        let timestamp = info.timestamp();
147        let chronological = self.previous.is_none_or(|previous| timestamp.callback.duration_since(&previous).is_some_and(|duration| !duration.is_zero()));
148        self.previous = Some(timestamp.callback);
149        let latency = timestamp.playback.duration_since(&timestamp.callback).filter(|latency| *latency <= std::time::Duration::from_millis(500));
150        let Some(latency) = latency.filter(|_| chronological) else {
151            self.first = None;
152            self.fallbacks += 1;
153            return now + block;
154        };
155        let (first, host) = *self.first.get_or_insert((timestamp.callback, now));
156        let candidate = timestamp.callback.duration_since(&first).and_then(|duration| host.checked_add(duration));
157        let Some(playback) = validated_playback(candidate, latency, now) else {
158            self.first = Some((timestamp.callback, now));
159            self.fallbacks += 1;
160            return now + block;
161        };
162        playback
163    }
164}

166struct Callback {
167    mixer: Mixer,
168    snapshots: crossbeam_channel::Sender<ClockSnapshot>,
169    obsolete_snapshots: crossbeam_channel::Receiver<ClockSnapshot>,
170    previous_arrival: Option<std::time::Instant>,
171    channels: usize,
172    rate: u32,
173    timestamps: TimestampAnchor,
174}
175
176impl Callback {
177    fn process<T: cpal::Sample + cpal::FromSample<f32>>(&mut self, output: &mut [T], info: &cpal::OutputCallbackInfo) {
178        let now = std::time::Instant::now();
179        let frames = output.len() / self.channels;
180        if frames == 0 { return; }
181        let duration = std::time::Duration::from_secs_f64(frames as f64 / f64::from(self.rate));
182        let playback = self.timestamps.playback(info, now, duration);
183        self.mixer.begin_callback();
184        if self.previous_arrival.replace(now).is_some_and(|previous| now.saturating_duration_since(previous) > duration * 2) {
185            self.mixer.stats.delayed_callbacks += 1;
186        }
187        self.mixer.stats.timestamp_fallbacks = self.timestamps.fallbacks;
188        let mut scratch = [0.0f32; 512];
189        for chunk in output.chunks_mut(256 * self.channels) {
190            let count = chunk.len() / self.channels;
191            self.mixer.render(&mut scratch[..count * 2]);
192            self.write_output(chunk, &scratch[..count * 2]);
193        }
194        self.mixer.stats.max_callback_micros = self.mixer.stats.max_callback_micros.max(now.elapsed().as_micros() as u64);
195        let snapshot = ClockSnapshot {
196            device_end: self.mixer.device_frame, playback_end: playback + duration,
197            observed: now, mapping: self.mixer.mapping, epoch: self.mixer.epoch,
198            block_frames: frames as u32, stats: self.mixer.stats,
199            latency: playback.saturating_duration_since(now),
200        };
201        // Dropping an obsolete Copy snapshot cannot allocate or destroy audio resources.
202        if let Err(error) = self.snapshots.try_send(snapshot) {
203            let _ = self.obsolete_snapshots.try_recv();
204            let _ = self.snapshots.try_send(error.into_inner());
205        }
206    }
207
208    fn write_output<T: cpal::Sample + cpal::FromSample<f32>>(&mut self, output: &mut [T], stereo: &[f32]) {
209        for (frame, pair) in output.chunks_mut(self.channels).zip(stereo.as_chunks::<2>().0.iter()) {
210            self.write_frame(frame, pair);
211        }
212    }
213
214    fn write_frame<T: cpal::Sample + cpal::FromSample<f32>>(&mut self, frame: &mut [T], pair: &[f32]) {
215            frame.fill(T::from_sample(0.0));
216            let left = if self.channels == 1 { (pair[0] + pair[1]) * 0.5 } else { pair[0] };
217            frame[0] = self.output_sample(left);
218            if self.channels > 1 { frame[1] = self.output_sample(pair[1]); }
219    }
220
221    fn output_sample<T: cpal::Sample + cpal::FromSample<f32>>(&mut self, value: f32) -> T {
222        if value.abs() > 1.0 { self.mixer.stats.clipped_samples += 1; }
223        T::from_sample(value.clamp(-1.0, 1.0))
224    }
225}

227fn validated_playback(candidate: Option<std::time::Instant>, latency: std::time::Duration, now: std::time::Instant) -> Option<std::time::Instant> {
228    let candidate = candidate?;
229    let distance = candidate.saturating_duration_since(now).max(now.saturating_duration_since(candidate));
230    if distance > std::time::Duration::from_millis(100) || latency > std::time::Duration::from_millis(500) { return None; }
231    candidate.checked_add(latency)
232}

237    use std::time::{Duration, Instant};

239    #[test]
240    fn timestamp_validation_rejects_missing_implausible_and_discontinuous_estimates() {
241        let now = Instant::now();
242        assert_eq!(validated_playback(None, Duration::ZERO, now), None);
243        assert_eq!(validated_playback(Some(now), Duration::from_secs(1), now), None);
244        assert_eq!(validated_playback(Some(now + Duration::from_secs(1)), Duration::ZERO, now), None);
245        assert_eq!(validated_playback(Some(now - Duration::from_secs(1)), Duration::ZERO, now), None);
246        assert_eq!(validated_playback(Some(now - Duration::from_millis(1)), Duration::from_millis(5), now), Some(now + Duration::from_millis(4)));
247    }
