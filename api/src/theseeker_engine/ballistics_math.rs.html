1// Stationary trajectory solver translated from:
2// https://github.com/forrestthewoods/lib_fts/blob/master/code/fts_ballistic_trajectory.cs
3
4use glam::Vec2;
5
6/// For a stationary target
7pub fn solve_ballistic_arc(
8    proj_pos: Vec2,
9    proj_speed: f32,
10    target: Vec2,
11    gravity: f32,
12) -> (Vec2, Vec2, i32) {
13    if proj_pos != target && proj_speed > 0.0 && gravity > 0.0 {
14    } else {
15        return (Vec2::ZERO, Vec2::ZERO, 0);
16    }
17
18    let mut s0 = Vec2::ZERO;
19    let mut s1 = Vec2::ZERO;
20
21    let diff = target - proj_pos;
22    let speed2 = proj_speed * proj_speed;
23    let speed4 = proj_speed * proj_speed * proj_speed * proj_speed;
24    let y = diff.y;
25    let x = diff.x.abs();
26    let gx = gravity * x;
27    let root = speed4 - gravity * (gravity * x * x + 2.0 * y * speed2);
28
29    // No solution
30    if root < 0.0 {
31        return (s0, s1, 0);
32    }
33
34    let root = root.sqrt();
35    let low_ang = f32::atan2(speed2 - root, gx);
36    let high_ang = f32::atan2(speed2 + root, gx);
37    let num_solutions = if low_ang != high_ang { 2 } else { 1 };
38
39    let ground_dir = Vec2::X * diff.x.signum();
40    s0 = ground_dir * f32::cos(low_ang) * proj_speed
41        + Vec2::new(0.0, f32::sin(low_ang) * proj_speed);
42    if num_solutions > 1 {
43        s1 = ground_dir * f32::cos(high_ang) * proj_speed
44            + Vec2::new(0.0, f32::sin(high_ang) * proj_speed);
45    }
46
47    (s0, s1, num_solutions)
48}
