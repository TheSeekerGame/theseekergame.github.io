10/// Wraps the given assertions in a `const` block.
11///
12/// This also handles the removal of nested `assert_type_match` attributes
13/// and the presence of the `test_only` attribute.
14pub(crate) fn wrap_assertions<F>(mut input: DeriveInput, args: Args, f: F) -> TokenStream
15where
16    F: FnOnce(&DeriveInput, &Args) -> syn::Result<TokenStream>,
17{
18    if let Err(error) = check_name(&mut input, &args) {
19        return error;
20    }
21
22    if args.test_only() {
23        input.ident = format_ident!("__Assert{}", input.ident);
24    }
25
26    let assertions = match f(&input, &args) {
27        Ok(assertions) => assertions,
28        Err(err) => return err.to_compile_error(),
29    };
30
31    let from_impl = match generate_from(&input, &args) {
32        Ok(from_impl) => from_impl,
33        Err(err) => return err.to_compile_error(),
34    };
35
36    let mut output = if args.test_only() {
37        TokenStream::new()
38    } else {
39        // Remove all nested `assert_type_match` attributes
40        strip_assertion_attributes(&mut input);
41
42        input.to_token_stream()
43    };
44
45    if args.test_only() {
46        output.extend(quote! {
47            const _: () = {
48                #input
49                #assertions
50                #from_impl
51            };
52        });
53    } else {
54        output.extend(quote! {
55            const _: () = {
56                #assertions
57                #from_impl
58            };
59        });
60    }
61
62    output
63}

65fn check_name(input: &mut DeriveInput, args: &Args) -> Result<(), TokenStream> {
66    let ident = &input.ident;
67    let foreign_ty = args.foreign_ty();
68
69    if !args.skip_name() {
70        let Some(segment) = foreign_ty.path.segments.last() else {
71            return Err(
72                syn::Error::new(foreign_ty.span(), "expected a type path").to_compile_error()
73            );
74        };
75
76        if &segment.ident != ident {
77            return Err(syn::Error::new(
78                ident.span(),
79                format_args!("type name does not match: expected `{}`", segment.ident),
80            )
81            .to_compile_error());
82        }
83    }
84
85    Ok(())
86}

88/// Removes all `assert_type_match` attributes from the given input.
89fn strip_assertion_attributes(input: &mut DeriveInput) {
90    match &mut input.data {
91        Data::Struct(data) => {
92            for field in data.fields.iter_mut() {
93                field.attrs.retain(|attr| !attr.path().is_ident(ATTRIBUTE));
94            }
95        }
96        Data::Enum(data) => {
97            for variant in data.variants.iter_mut() {
98                variant
99                    .attrs
100                    .retain(|attr| !attr.path().is_ident(ATTRIBUTE));
101
102                for field in variant.fields.iter_mut() {
103                    field.attrs.retain(|attr| !attr.path().is_ident(ATTRIBUTE));
104                }
105            }
106        }
107        _ => {}
108    }
109}

111fn generate_from(input: &DeriveInput, args: &Args) -> syn::Result<TokenStream> {
112    if !args.from() {
113        return Ok(TokenStream::new());
114    }
115
116    match &input.data {
117        Data::Struct(data) => struct_from(data, input, args),
118        Data::Enum(data) => enum_from(data, input, args),
119        Data::Union(data) => Err(syn::Error::new(
120            data.union_token.span,
121            "unions are not supported",
122        )),
123    }
124}
