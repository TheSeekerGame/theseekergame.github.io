19pub fn main_transparent_pass_3d(
20    world: &World,
21    view: ViewQuery<(
22        &ExtractedCamera,
23        &ExtractedView,
24        &ViewTarget,
25        &ViewDepthTexture,
26        Option<&MainPassResolutionOverride>,
27        Has<OrderIndependentTransparencySettings>,
28        Option<&OitResolvePipelineId>,
29    )>,
30    transparent_phases: Res<ViewSortedRenderPhases<Transparent3d>>,
31    mut ctx: RenderContext,
32) {
33    let view_entity = view.entity();
34
35    let (
36        camera,
37        extracted_view,
38        target,
39        depth,
40        resolution_override,
41        has_oit,
42        oit_resolve_pipeline_id,
43    ) = view.into_inner();
44
45    let Some(transparent_phase) = transparent_phases.get(&extracted_view.retained_view_entity)
46    else {
47        return;
48    };
49
50    if !transparent_phase.items.is_empty() {
51        #[cfg(feature = "trace")]
52        let _main_transparent_pass_3d_span = info_span!("main_transparent_pass_3d").entered();
53
54        let diagnostics = ctx.diagnostic_recorder();
55        let diagnostics = diagnostics.as_deref();
56
57        if has_oit {
58            // We can't run transparent phase if OitResolvePipelineId is not ready
59            // Otherwise we will write to `oit_atomic_counter` and `oit_heads` buffer without resetting them
60            // which causes corrupted linked list(can have circular references) on the next pass
61            let Some(oit_resolve_pipeline_id) = oit_resolve_pipeline_id else {
62                return;
63            };
64            let pipeline_cache = world.resource::<PipelineCache>();
65            if pipeline_cache
66                .get_render_pipeline(oit_resolve_pipeline_id.0)
67                .is_none()
68            {
69                return;
70            }
71        }
72
73        let mut render_pass = ctx.begin_tracked_render_pass(RenderPassDescriptor {
74            label: Some("main_transparent_pass_3d"),
75            color_attachments: &[Some(target.get_color_attachment())],
76            // NOTE: For the transparent pass we load the depth buffer. There should be no
77            // need to write to it, but store is set to `true` as a workaround for issue #3776,
78            // https://github.com/bevyengine/bevy/issues/3776
79            // so that wgpu does not clear the depth buffer.
80            // As the opaque and alpha mask passes run first, opaque meshes can occlude
81            // transparent ones.
82            depth_stencil_attachment: Some(depth.get_attachment(StoreOp::Store)),
83            timestamp_writes: None,
84            occlusion_query_set: None,
85            multiview_mask: None,
86        });
87        let pass_span = diagnostics.pass_span(&mut render_pass, "main_transparent_pass_3d");
88
89        if let Some(viewport) =
90            Viewport::from_viewport_and_override(camera.viewport.as_ref(), resolution_override)
91        {
92            render_pass.set_camera_viewport(&viewport);
93        }
94
95        if let Err(err) = transparent_phase.render(&mut render_pass, world, view_entity) {
96            error!("Error encountered while rendering the transparent phase {err:?}");
97        }
98
99        pass_span.end(&mut render_pass);
100    }
101
102    // WebGL2 quirk: if ending with a render pass with a custom viewport, the viewport isn't
103    // reset for the next render pass so add an empty render pass without a custom viewport
104    #[cfg(all(feature = "webgl", target_arch = "wasm32", not(feature = "webgpu")))]
105    if camera.viewport.is_some() {
106        #[cfg(feature = "trace")]
107        let _reset_viewport_pass_3d = info_span!("reset_viewport_pass_3d").entered();
108        let pass_descriptor = RenderPassDescriptor {
109            label: Some("reset_viewport_pass_3d"),
110            color_attachments: &[Some(target.get_color_attachment())],
111            depth_stencil_attachment: None,
112            timestamp_writes: None,
113            occlusion_query_set: None,
114            multiview_mask: None,
115        };
116
117        ctx.command_encoder().begin_render_pass(&pass_descriptor);
118    }
119}
