19/// Functionally identical to [`Punctuated`](syn::punctuated::Punctuated), but fills the given `$list` Vec instead
20/// of allocating a new one inside [`Punctuated`](syn::punctuated::Punctuated). This exists to avoid allocating an intermediate Vec.
21///
22/// This also attempts to parse $parse a second time _before_ parsing $separator, as this enables autocomplete to work in cases where
23/// it is being typed in the middle of a list
24macro_rules! parse_punctuated_vec_autocomplete_friendly {
25    ($list:ident, $input:ident, $parse:ident, $separator:ident) => {
26        loop {
27            if $input.is_empty() {
28                break;
29            }
30            let value = $input.parse::<$parse>()?;
31            $list.push(value);
32            if $input.is_empty() {
33                break;
34            }
35
36            // Try parsing without a comma separator first. This makes autocomplete
37            // work in more places
38            if !$input.is_empty() && !$input.peek($separator) {
39                let value = $input.parse::<$parse>()?;
40                $list.push(value);
41            }
42            $input.parse::<$separator>()?;
43        }
44    };
45}

59impl<const ALLOW_FLAT: bool> Parse for Bsn<ALLOW_FLAT> {
60    fn parse(input: ParseStream) -> Result<Self> {
61        let mut entries = Vec::new();
62        if input.peek(Paren) {
63            let content;
64            parenthesized![content in input];
65            while !content.is_empty() {
66                let entry = BsnEntry::parse(&content)?;
67                if matches!(entry, BsnEntry::CachedScene(_)) && !entries.is_empty() {
68                    return Err(syn::Error::new(
69                        content.span(),
70                        "Caching entries after the first is not supported, remove the ':' prefix or make this the first entry.",
71                    ));
72                }
73                entries.push(entry);
74            }
75        } else if ALLOW_FLAT {
76            while !input.is_empty() {
77                let entry = BsnEntry::parse(input)?;
78                if matches!(entry, BsnEntry::CachedScene(_)) && !entries.is_empty() {
79                    return Err(syn::Error::new(
80                        input.span(),
81                        "Caching entries after the first is not supported, remove the ':' prefix or make this the first entry.",
82                    ));
83                }
84                entries.push(entry);
85                if input.peek(Comma) {
86                    // Not ideal, but this anticipatory break allows us to parse non-parenthesized
87                    // flat Bsn entries in SceneLists
88                    break;
89                }
90            }
91        } else {
92            entries.push(BsnEntry::parse(input)?);
93        }
94
95        Ok(Self { entries })
96    }
97}

99impl BsnEntry {
100    fn parse(input: ParseStream) -> Result<Self> {
101        Ok(if input.peek(Token![:]) && !input.peek(Token![::]) {
102            BsnEntry::CachedScene(BsnScene::parse(input)?)
103        } else if input.peek(Token![#]) {
104            input.parse::<Token![#]>()?;
105            BsnEntry::Name(input.parse::<Ident>()?)
106        } else if input.peek(Brace) || input.peek(At) {
107            BsnEntry::UncachedScene(BsnScene::parse(input)?)
108        } else {
109            let is_template = input.peek(Tilde);
110            if is_template {
111                input.parse::<Tilde>()?;
112            }
113            let mut path = input.parse::<Path>()?;
114            let path_type = PathType::new(&path);
115            match path_type {
116                PathType::Type | PathType::Enum => {
117                    let enum_variant = if matches!(path_type, PathType::Enum) {
118                        take_last_path_ident(&mut path)
119                    } else {
120                        None
121                    };
122                    if input.peek(Bracket) {
123                        // TODO: fail if this is an enum variant
124                        BsnEntry::RelatedSceneList(BsnRelatedSceneList {
125                            relationship_path: path,
126                            scene_list: input.parse::<BsnSceneList>()?,
127                        })
128                    } else {
129                        let fields = input.parse::<BsnFields>()?;
130                        let bsn_type = BsnType {
131                            path,
132                            enum_variant,
133                            fields,
134                        };
135                        if is_template {
136                            BsnEntry::TemplatePatch(bsn_type)
137                        } else {
138                            BsnEntry::FromTemplatePatch(bsn_type)
139                        }
140                    }
141                }
142                PathType::TypeConst => {
143                    let const_ident = take_last_path_ident(&mut path).unwrap();
144                    BsnEntry::TemplateConst {
145                        type_path: path,
146                        const_ident,
147                    }
148                }
149                PathType::Const => {
150                    return Err(syn::Error::new(
151                        path.span(),
152                        "Consts are not currently supported in this position",
153                    ))
154                }
155                PathType::TypeFunction => {
156                    let function = take_last_path_ident(&mut path).unwrap();
157                    let args = input.parse::<BsnFnArgs>()?;
158                    let bsn_constructor = BsnConstructor {
159                        type_path: path,
160                        function,
161                        args,
162                    };
163                    if is_template {
164                        BsnEntry::TemplateConstructor(bsn_constructor)
165                    } else {
166                        BsnEntry::FromTemplateConstructor(bsn_constructor)
167                    }
168                }
169                PathType::Function => {
170                    if input.peek(Paren) {
171                        let args = input.parse::<BsnFnArgs>()?;
172                        BsnEntry::UncachedScene(BsnScene::Fn(BsnSceneFn { path, args }))
173                    } else {
174                        BsnEntry::UncachedScene(BsnScene::Expression(quote! {#path}))
175                    }
176                }
177            }
178        })
179    }
180}

208impl BsnScene {
209    fn parse(input: ParseStream) -> Result<Self> {
210        let cached = if input.peek(Token![:]) {
211            Some(input.parse::<Token![:]>()?)
212        } else {
213            None
214        };
215
216        let err_if_cached = |msg: &str| {
217            if let Some(colon) = cached {
218                Err(syn::Error::new(colon.span(), msg))
219            } else {
220                Ok(())
221            }
222        };
223
224        // It may seem odd how this is checking LitStr again
225        // and how there doesn't seem to be a need for all the specific `err_if_cached`
226        // in later code. But since caching is planned, and will very likely
227        // have the limitations which are ensured by the other errors below,
228        // this is its own block so its very simple to remove once caching is implemented.
229        if !input.peek(LitStr) {
230            err_if_cached("Currently, caching is only supported for scene assets. Please remove the ':' prefix for now.")?;
231        }
232
233        Ok(if input.peek(LitStr) {
234            let path = input.parse::<LitStr>()?;
235            if cached.is_none() {
236                return Err(syn::Error::new(
237                    path.span(),
238                    "Cannot use scene assets without caching, please add the ':' prefix.",
239                ));
240            }
241            BsnScene::Asset(path)
242        } else if input.peek(Brace) {
243            err_if_cached("Cannot cache scene expressions")?;
244            BsnScene::Expression(braced_tokens(input)?)
245        } else if input.peek(At) {
246            input.parse::<At>()?;
247            let sc = input.parse::<BsnType>()?;
248            if sc.fields.len() > 0 {
249                err_if_cached("Cannot cache Scene Components with props/fields")?;
250            }
251            BsnScene::SceneComponent(sc)
252        } else {
253            // PERF: do we really need this fork here?
254            let path = input.fork().parse::<Path>()?;
255            match PathType::new(&path) {
256                PathType::Type | PathType::Enum => {
257                    // Scene components are parsed before this if an @ is found.
258                    // If this path is hit, that means it wasn't prefixed by @
259                    return Err(syn::Error::new(
260                        path.span(),
261                        format!(
262                            "Scene component {} needs to be prefixed by '@'",
263                            path_to_string(&path),
264                        ),
265                    ));
266                }
267                PathType::Function | PathType::TypeFunction => {
268                    let path = input.parse::<Path>()?;
269                    let args = input.parse::<BsnFnArgs>()?;
270                    if !args.0.is_empty() {
271                        err_if_cached("Cannot cache Scene function with arguments")?;
272                    }
273                    BsnScene::Fn(BsnSceneFn { path, args })
274                }
275                path_type => {
276                    return Err(syn::Error::new(
277                        path.span(),
278                        format!(
279                            "Cannot cache path {} of type {:?}",
280                            path_to_string(&path),
281                            path_type,
282                        ),
283                    ))
284                }
285            }
286        })
287    }
288}

290impl Parse for BsnType {
291    fn parse(input: ParseStream) -> Result<Self> {
292        let mut path = input.parse::<Path>()?;
293        let enum_variant = match PathType::new(&path) {
294            PathType::Type => None,
295            PathType::Enum => take_last_path_ident(&mut path),
296            PathType::Function | PathType::TypeFunction => {
297                return Err(syn::Error::new(
298                    path.span(),
299                    "Expected a path to a BSN type but encountered a path to a function.",
300                ))
301            }
302            PathType::Const | PathType::TypeConst => {
303                return Err(syn::Error::new(
304                    path.span(),
305                    "Expected a path to a BSN type but encountered a path to a const.",
306                ))
307            }
308        };
309        let fields = input.parse::<BsnFields>()?;
310        Ok(BsnType {
311            path,
312            enum_variant,
313            fields,
314        })
315    }

354        let is_prop = if input.peek(At) {

356            true

373        };

375            name,
376            value,
377            is_prop,
378            is_name_shorthand,
379        })
380    }

390/// Parses tuple arguments into a list of [`TokenStream`]s. This avoids
391/// fully parsing Rust expressions, which makes this less strict and cheaper to parse.
392/// This also allows autocomplete to work, even if the tokens aren't a valid rust expression.
393///
394/// This will accept anything "tuple-like" in the form (X1, ..., XY), where XY is a TokenStream.
395fn parse_tuple_loose(input: &ParseBuffer) -> Result<Vec<TokenStream>> {
396    let content;
397    parenthesized!(content in input);
398    let mut args = Vec::new();
399    let mut current_tokens = Vec::new();
400    let mut in_closure_args = false;
401    let mut generic_scope = 0;
402    while !content.is_empty() {
403        let tt = content.parse::<TokenTree>()?;
404        match &tt {
405            TokenTree::Punct(punct) => match punct.as_char() {
406                ',' if !in_closure_args && generic_scope == 0 => {
407                    args.push(TokenStream::from_iter(current_tokens.drain(..)));
408                }
409                '|' => {
410                    in_closure_args = !in_closure_args;
411                    current_tokens.push(tt);
412                }
413                '<' => {
414                    generic_scope += 1;
415                    current_tokens.push(tt);
416                }
417                '>' => {
418                    generic_scope -= 1;
419                    current_tokens.push(tt);
420                }
421                _ => current_tokens.push(tt),
422            },
423            _ => current_tokens.push(tt),
424        }
425    }
426
427    if !current_tokens.is_empty() {
428        args.push(TokenStream::from_iter(current_tokens));
429    }
430    Ok(args)
431}

433/// Parse a closure "loosely" without caring about the tokens between `|...|` and `{...}`. This ensures autocomplete works.
434fn parse_closure_loose(input: &ParseBuffer) -> Result<TokenStream> {
435    let start = input.cursor();
436    input.parse::<Token![|]>()?;
437    let tokens = input.step(|cursor| {
438        let mut rest = *cursor;
439        while let Some((tt, next)) = rest.token_tree() {
440            match &tt {
441                TokenTree::Punct(punct) if punct.as_char() == '|' => {
442                    if let Some((TokenTree::Group(group), next)) = next.token_tree()
443                        && group.delimiter() == Delimiter::Brace
444                    {
445                        return Ok((tokens_between(start, next), next));
446                    } else {
447                        return Err(cursor.error("closures expect '{' to follow '|'"));
448                    }
449                }
450                _ => rest = next,
451            }
452        }
453        Err(cursor.error("no matching `|` was found after this point"))
454    })?;
455    Ok(tokens)
456}

479fn tokens_between(begin: Cursor, end: Cursor) -> TokenStream {
480    assert!(begin <= end);
481    let mut cursor = begin;
482    let mut tokens = TokenStream::new();
483    while cursor < end {
484        let (token, next) = cursor.token_tree().unwrap();
485        tokens.extend(std::iter::once(token));
486        cursor = next;
487    }
488    tokens
489}

491impl Parse for BsnValue {
492    fn parse(input: ParseStream) -> Result<Self> {
493        Ok(if input.peek(Brace) {
494            BsnValue::Expr(braced_tokens(input)?)
495        } else if input.peek(Token![const]) && input.peek2(Brace) {
496            let const_token = input.parse::<Token![const]>()?;
497            let braced = braced_tokens(input)?;
498
499            BsnValue::Expr(quote! {#const_token {#braced}})
500        } else if input.peek(Token![unsafe]) && input.peek2(Brace) {
501            let unsafe_token = input.parse::<Token![unsafe]>()?;
502            let braced = braced_tokens(input)?;
503
504            BsnValue::Expr(quote! {#unsafe_token {#braced}})
505        } else if input.peek(Token![|]) {
506            let tokens = parse_closure_loose(input)?;
507            BsnValue::Closure(tokens)
508        } else if input.peek(Ident) {
509            let forked = input.fork();
510            let path = forked.parse::<Path>()?;
511            if path.segments.len() == 1 && (forked.is_empty() || forked.peek(Comma)) {
512                return Ok(BsnValue::Ident(input.parse::<Ident>()?));
513            }
514            match PathType::new(&path) {
515                PathType::TypeFunction | PathType::Function => {
516                    input.parse::<Path>()?;
517                    let maybe_macro = input.parse::<Token![!]>().ok();
518                    if input.peek(Paren) {
519                        let token_stream = parenthesized_tokens(input)?;
520                        BsnValue::Expr(quote! { #path #maybe_macro (#token_stream) })
521                    } else if input.peek(Bracket) {
522                        let token_stream = bracketed_tokens(input)?;
523                        BsnValue::Expr(quote! { #path #maybe_macro [#token_stream] })
524                    } else if input.peek(Brace) {
525                        let token_stream = braced_tokens(input)?;
526                        BsnValue::Expr(quote! { #path #maybe_macro { #token_stream } })
527                    } else {
528                        return Err(input.error("Unexpected input after function name"));
529                    }
530                }
531                PathType::Const | PathType::TypeConst => {
532                    input.parse::<Path>()?;
533                    BsnValue::Expr(quote! { #path })
534                }
535                PathType::Type | PathType::Enum => BsnValue::Type(input.parse::<BsnType>()?),
536            }
537        } else if input.peek(Lit) {
538            BsnValue::Lit(input.parse::<Lit>()?)
539        } else if input.peek(Paren) {
540            BsnValue::Tuple(input.parse::<BsnTuple>()?)
541        } else if input.peek(Token![#]) {
542            input.parse::<Token![#]>()?;
543            BsnValue::Name(input.parse::<Ident>()?)
544        } else {
545            return Err(input.error("Unexpected input: Invalid BsnValue. This does not match any expected BSN value type."));
546        })
547    }
548}

585fn take_last_path_ident(path: &mut Path) -> Option<Ident> {
586    let ident = path.segments.pop().map(|s| s.into_value().ident);
587    path.segments.pop_punct();
588    ident
