1//! This crate provides the tools for positioning and rendering text in Bevy.
2//!
3//! # `Font`
4//!
5//! Fonts contain information for drawing glyphs, which are shapes that typically represent a single character,
6//! but in some cases part of a "character" (grapheme clusters) or more than one character (ligatures).
7//!
8//! A font *face* is part of a font family,
9//! and is distinguished by its style (e.g. italic), its weight (e.g. bold) and its stretch (e.g. condensed).
10//!
11//! In Bevy, [`Font`]s are loaded by the [`FontLoader`] as [assets](bevy_asset::AssetPlugin).
12//!
13//! # `TextPipeline`
14//!
15//! The [`TextPipeline`] resource does all of the heavy lifting for rendering text.
16//!
17//! UI `Text` is first measured by creating a [`TextMeasureInfo`] in [`TextPipeline::create_text_measure`],
18//! which is called by the `measure_text_system` system of `bevy_ui`.
19//!
20//! Note that text measurement is only relevant in a UI context.
21//!
22//! With the actual text bounds defined, the `bevy_ui::widget::text::text_system` system (in a UI context)
23//! or `bevy_sprite::text2d::update_text2d_layout` system (in a 2d world space context)
24//! passes it into [`TextPipeline::update_text_layout_info`], which:
25//!
26//! 1. updates a [`Layout`](parley::Layout) from the [`TextSpan`]s, generating new [`FontAtlas`]es if necessary.
27//! 2. iterates over each glyph in the [`Layout`](parley::Layout) to create a [`PositionedGlyph`],
28//!    retrieving glyphs from the cache, or rasterizing to a [`FontAtlas`] if necessary.
29//! 3. [`PositionedGlyph`]s are stored in a [`TextLayoutInfo`],
30//!    which contains all the information that downstream systems need for rendering.

80/// The raw data for the default font used by `bevy_text`

84/// Adds text rendering support to an app.
85///
86/// When the `bevy_text` feature is enabled with the `bevy` crate, this
87/// plugin is included by default in the `DefaultPlugins`.
88///
89/// ## Clipboard support
90///
91/// [`EditableText`] supports copy, cut, and paste via [`bevy_clipboard::Clipboard`].
92/// By default, clipboard operations use an in-process fallback buffer rather than the OS clipboard.
93/// To enable OS clipboard integration, activate the `system_clipboard` feature on this crate:
94///
95/// ```toml
96/// bevy_text = { version = "...", features = ["system_clipboard"] }
97/// ```
98///
99/// When using the top-level `bevy` crate, `bevy/system_clipboard` enables this transitively.
100#[derive(Default)]
101pub struct TextPlugin;
102
103/// System set in [`PostUpdate`] where all 2d text update systems are executed.
104#[derive(Debug, Hash, PartialEq, Eq, Clone, SystemSet)]
105pub struct Text2dUpdateSystems;
106
107/// System set where [`EditableText::pending_edits`] are applied.
108#[derive(Debug, Hash, PartialEq, Eq, Clone, SystemSet)]
109pub struct EditableTextSystems;
110
111impl Plugin for TextPlugin {
112    fn build(&self, app: &mut App) {
113        if !app.is_plugin_added::<bevy_clipboard::ClipboardPlugin>() {
114            app.add_plugins(bevy_clipboard::ClipboardPlugin);
115        }
116        app.init_asset::<Font>()
117            .init_asset_loader::<FontLoader>()
118            .init_resource::<FontAtlasSet>()
119            .init_resource::<TextPipeline>()
120            .init_resource::<FontCx>()
121            .init_resource::<LayoutCx>()
122            .init_resource::<ScaleCx>()
123            .init_resource::<TextIterScratch>()
124            .init_resource::<RemSize>()
125            .add_systems(
126                PostUpdate,
127                (
128                    load_font_assets_into_font_collection,
129                    detect_text_needs_rerender,
130                )
131                    .chain(),
132            )
133            .add_systems(Last, trim_source_cache)
134            .add_systems(
135                PostUpdate,
136                apply_text_edits
137                    .after(load_font_assets_into_font_collection)
138                    .in_set(EditableTextSystems),
139            );
140
141        #[cfg(feature = "default_font")]
142        {
143            use bevy_asset::{AssetId, Assets};
144            let mut assets = app.world_mut().resource_mut::<Assets<Font>>();
145            let asset = Font::from_bytes(DEFAULT_FONT_DATA.to_vec());
146            assets.insert(AssetId::default(), asset).unwrap();
147        };
148    }
149}
