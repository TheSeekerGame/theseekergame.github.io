35/// Options for a [font collection](Collection).
36#[derive(Copy, Clone, Debug)]
37pub struct CollectionOptions {
38    /// If true, the font collection will use a secondary shared store
39    /// guaranteeing that any changes to the collection will be
40    /// visible to all clones.
41    ///
42    /// If the font collection will be used by a single thread, this is
43    /// pure overhead and should be disabled.
44    ///
45    /// The default value is `false`.
46    pub shared: bool,
47
48    /// If true, the font collection will provide access to system fonts
49    /// using platform specific APIs.
50    ///
51    /// The default value is `true`.
52    pub system_fonts: bool,
53}
54
55impl Default for CollectionOptions {
56    fn default() -> Self {
57        Self {
58            shared: false,
59            system_fonts: true,
60        }
61    }
62}

250    /// Creates a new collection with the given options.
251    pub fn new(options: CollectionOptions) -> Self {
252        let system = options.system_fonts.then(System::new);
253        let shared = options.shared.then(|| Arc::new(Shared::default()));
254        Self {
255            system,
256            data: CommonData::default(),
257            shared,
258            shared_version: 0,
259            fallback_cache: FallbackCache::default(),
260        }

263    #[cfg(feature = "std")]
264    pub fn make_shared(&mut self) {
265        if self.shared.is_none() {
266            self.shared = Some(Arc::new(Shared {
267                data: Mutex::new(self.data.clone()),
268                version: AtomicCounter::new(self.shared_version),
269            }));
270        }
271    }

274    pub fn load_system_fonts(&mut self) {
275        self.system = Some(System::new());
276    }

278    /// Returns an iterator over all available family names in the collection.
279    ///
280    /// This includes both system and registered fonts.
281    pub fn family_names(&mut self) -> impl Iterator<Item = &str> + '_ + Clone {
282        self.sync_shared();
283        FamilyNames {
284            ours: self.data.family_names.iter(),
285            system: self.system.as_ref().map(|sys| sys.family_names.iter()),
286        }
287        .map(|name| name.name())
288    }
289
290    /// Returns the family identifier for the given family name.
291    pub fn family_id(&mut self, name: &str) -> Option<FamilyId> {
292        self.sync_shared();
293        self.data
294            .family_names
295            .get(name)
296            .or_else(|| {
297                self.system
298                    .as_ref()
299                    .and_then(|sys| sys.family_names.get(name))
300            })
301            .map(|n| n.id())
302    }
303
304    /// Returns the family name for the given family identifier.
305    pub fn family_name(&mut self, id: FamilyId) -> Option<&str> {
306        self.sync_shared();
307        self.data
308            .family_names
309            .get_by_id(id)
310            .or_else(|| {
311                self.system
312                    .as_ref()
313                    .and_then(|sys| sys.family_names.get_by_id(id))
314            })
315            .map(|name| name.name())
316    }

318    /// Returns the family object for the given family identifier.
319    pub fn family(&mut self, id: FamilyId) -> Option<FamilyInfo> {
320        self.sync_shared();
321        if let Some(family) = self.data.families.get(&id) {
322            family.as_ref().cloned()
323        } else {
324            #[cfg(feature = "system")]
325            if let Some(system) = &self.system {
326                let family = system.fonts.lock().unwrap().family(id);
327                self.data.families.insert(id, family.clone());
328                family
329            } else {
330                None
331            }
332            #[cfg(not(feature = "system"))]
333            {
334                None
335            }
336        }
337    }
338
339    /// Returns the family object for the given name.
340    pub fn family_by_name(&mut self, name: &str) -> Option<FamilyInfo> {
341        if let Some(id) = self.family_id(name) {
342            self.family(id)
343        } else {
344            None
345        }
346    }

364    /// Replaces the set of family identifiers associated with the given generic
365    /// family.
366    pub fn set_generic_families(
367        &mut self,
368        generic: GenericFamily,
369        families: impl Iterator<Item = FamilyId>,

372        #[cfg(feature = "std")]

411    /// Returns an iterator over the fallback families for the given
412    /// key.
413    pub fn fallback_families(
414        &mut self,
415        key: impl Into<FallbackKey>,
416    ) -> impl Iterator<Item = FamilyId> + '_ + Clone {
417        let selector = key.into();
418        let script = selector.script();
419        let lang_key = selector.locale();
420        if self.fallback_cache.script != Some(script) || self.fallback_cache.language != lang_key {
421            self.sync_shared();
422            self.fallback_cache.reset();
423            #[cfg(feature = "system")]
424            if let Some(families) = self.data.fallbacks.get(selector) {
425                self.fallback_cache.set(script, lang_key, families);
426            } else if let Some(system) = self.system.as_ref() {
427                // Some platforms don't need mut System
428                #[allow(unused_mut)]
429                let mut system = system.fonts.lock().unwrap();
430                if let Some(family) = system.fallback(selector) {
431                    self.data.fallbacks.set(selector, core::iter::once(family));
432                    self.fallback_cache.set(script, lang_key, &[family]);
433                }
434            }
435            #[cfg(not(feature = "system"))]
436            if let Some(families) = self.data.fallbacks.get(selector) {
437                self.fallback_cache.set(script, lang_key, families);
438            }
439        }
440        self.fallback_cache.families.iter().copied()
441    }

443    /// Replaces the set of family identifiers associated with the fallback
444    /// key.
445    pub fn set_fallbacks(
446        &mut self,
447        key: impl Into<FallbackKey>,
448        families: impl Iterator<Item = FamilyId>,
449    ) -> bool {
450        self.sync_shared();
451        self.fallback_cache.reset();
452        #[cfg(feature = "std")]
453        if let Some(shared) = &self.shared {
454            let result = shared.data.lock().unwrap().fallbacks.set(key, families);
455            shared.bump_version();
456            result
457        } else {
458            self.data.fallbacks.set(key, families)
459        }
460        #[cfg(not(feature = "std"))]
461        self.data.fallbacks.set(key, families)
462    }

523    /// Unregisters the font with the given attributes from the given family.
524    ///
525    /// Returns true if a font was removed from the family, false otherwise.
526    pub fn unregister_font(
527        &mut self,
528        family: FamilyId,
529        width: FontWidth,
530        style: FontStyle,
531        weight: FontWeight,
532    ) -> bool {
533        #[cfg(feature = "std")]
534        if let Some(shared) = &self.shared {
535            let result = shared
536                .data
537                .lock()
538                .unwrap()
539                .unregister_font(family, width, style, weight);
540            shared.bump_version();
541            result.is_some()
542        } else {
543            self.data
544                .unregister_font(family, width, style, weight)
545                .is_some()
546        }
547        #[cfg(not(feature = "std"))]
548        self.data
549            .unregister_font(family, width, style, weight)
550            .is_some()
551    }
552
553    /// Clears this collection. Un-registers all fonts previously registered via
554    /// [`Self::register_fonts`], and unsets all previously-set generic families
555    /// and fallbacks. This will not remove any system fonts.
556    pub fn clear(&mut self) {
557        #[cfg(feature = "std")]
558        if let Some(shared) = &self.shared {
559            shared.data.lock().unwrap().clear();
560            shared.bump_version();
561        } else {
562            self.data.clear();
563        }
564        self.data.clear();
565    }
566
567    fn sync_shared(&mut self) {
568        #[cfg(feature = "std")]
569        if let Some(shared) = &self.shared {
570            let version = shared.version.load(Ordering::Acquire);
571            if self.shared_version != version {
572                // This is an ugly deep copy, but the assumption is that
573                // modifications to font collections are fairly rare.
574                // If this becomes a problem, do more fine grained tracking
575                // of changes.
576                self.data = shared.data.lock().unwrap().clone();
577                self.shared_version = version;
578                self.fallback_cache.reset();
579            }
580        }
581    }

624#[derive(Clone, Default)]
625struct FallbackCache {
626    script: Option<Script>,
627    language: Option<Language>,
628    families: Vec<FamilyId>,
629}
630
631impl FallbackCache {
632    fn reset(&mut self) {
633        self.script = None;
634        self.language = None;
635        self.families.clear();
636    }
637
638    fn set(&mut self, script: Script, language: Option<Language>, families: &[FamilyId]) {
639        self.script = Some(script);
640        self.language = language;
641        self.families.clear();
642        self.families.extend_from_slice(families);
643    }

671/// Common data for base and shared collections.
672#[derive(Clone, Default)]
673struct CommonData {
674    family_names: FamilyNameMap,
675    families: FamilyMap,
676    generic_families: GenericFamilyMap,
677    fallbacks: FallbackMap,
678}
679
680impl CommonData {
681    #[cfg(feature = "std")]
682    fn load_fonts_from_paths(&mut self, paths: impl IntoIterator<Item = impl AsRef<Path>>) {
683        let mut families: HashMap<FamilyId, (FamilyName, Vec<FontInfo>)> = HashMap::default();
684        let mut scratch_family_name = String::default();
685        crate::scan::scan_paths(paths, 16, |scanned_font| {
686            let source = SourceInfo {
687                id: SourceId::new(),
688                kind: SourceKind::Path(Arc::from(scanned_font.path.unwrap())),
689            };
690
691            let font_data = scanned_font.font.data().as_bytes();
692            self.register_font_impl(
693                font_data,
694                source,
695                None,
696                &mut scratch_family_name,
697                &mut families,
698            );
699        });
700    }
701
702    fn register_fonts(
703        &mut self,
704        data: Blob<u8>,
705        info_override: Option<FontInfoOverride<'_>>,
706    ) -> Vec<(FamilyId, Vec<FontInfo>)> {
707        let mut families: HashMap<FamilyId, (FamilyName, Vec<FontInfo>)> = HashMap::default();
708        let mut scratch_family_name = String::default();
709
710        let source = SourceInfo {
711            id: SourceId::new(),
712            kind: SourceKind::Memory(data.clone()),
713        };
714
715        self.register_font_impl(
716            data.as_ref(),
717            source,
718            info_override,
719            &mut scratch_family_name,
720            &mut families,
721        );
722
723        families
724            .into_iter()
725            .map(|(id, (_, fonts))| (id, fonts))
726            .collect()
727    }
728
729    fn register_font_impl(
730        &mut self,
731        font_data: &[u8],
732        source: SourceInfo,
733        info_override: Option<FontInfoOverride<'_>>,
734        scratch_family_name: &mut String,
735        families: &mut HashMap<FamilyId, (FamilyName, Vec<FontInfo>)>,
736    ) {
737        super::scan::scan_memory(font_data, |scanned_font| {
738            scratch_family_name.clear();
739
740            let family_name =
741                if let Some(override_family_name) = info_override.and_then(|o| o.family_name) {
742                    override_family_name
743                } else {
744                    let family_chars = scanned_font
745                        .english_or_first_name(NameId::TYPOGRAPHIC_FAMILY_NAME)
746                        .or_else(|| scanned_font.english_or_first_name(NameId::FAMILY_NAME))
747                        .map(|name| name.chars());
748                    let Some(family_chars) = family_chars else {
749                        return;
750                    };
751                    scratch_family_name.extend(family_chars);
752
753                    #[allow(clippy::needless_borrow)] // false positive
754                    &scratch_family_name
755                };
756
757            if family_name.is_empty() {
758                return;
759            }
760
761            let Some(mut font) =
762                FontInfo::from_font_ref(&scanned_font.font, source.clone(), scanned_font.index)
763            else {
764                return;
765            };
766
767            if let Some(info_override) = info_override.as_ref() {
768                font.apply_override(info_override);
769            }
770
771            let name = self.family_names.get_or_insert(family_name);
772            families
773                .entry(name.id())
774                .or_insert_with(|| (name, Vec::default()))
775                .1
776                .push(font);
777        });
778        for (id, (name, fonts)) in families.iter() {
779            if let Some(Some(family)) = self.families.get_mut(id) {
780                let new_fonts = family.fonts().iter().chain(fonts).cloned();
781                *family = FamilyInfo::new(name.clone(), new_fonts);
782            } else {
783                let family = FamilyInfo::new(name.clone(), fonts.iter().cloned());
784                self.families.insert(*id, Some(family));
785            }
786        }
787    }
788
789    fn unregister_font(
790        &mut self,
791        family: FamilyId,
792        width: FontWidth,
793        style: FontStyle,
794        weight: FontWeight,
795    ) -> Option<()> {
796        let family_name = self.family_names.get_by_id(family)?;
797        let family = self.families.get_mut(&family)?.as_mut()?;
798
799        let new_fonts = family
800            .fonts()
801            .iter()
802            .filter(|f| f.width() != width || f.style() != style || f.weight() != weight)
803            .cloned();
804        *family = FamilyInfo::new(family_name.clone(), new_fonts);
805
806        Some(())
807    }
808
809    fn clear(&mut self) {
810        *self = Self::default();
811    }

831#[test]
832#[cfg(all(
833    feature = "std",
834    any(target_os = "linux", target_os = "macos", target_os = "windows")
835))]
836fn make_shared_matches_local() {
837    use crate::{Collection, CollectionOptions};
838
839    let mut collection = Collection::new(CollectionOptions {
840        shared: false,
841        system_fonts: false,
842    });
843
844    let font_dirs: Vec<std::path::PathBuf> = [
845        #[cfg(target_os = "macos")]
846        "/Library/Fonts",
847        #[cfg(target_os = "linux")]
848        "/usr/share/fonts",
849        #[cfg(target_os = "windows")]
850        "C:\\Windows\\Fonts",
851    ]
852    .iter()
853    .map(std::path::PathBuf::from)
854    .filter(|p| p.is_dir())
855    .collect();
856
857    if font_dirs.is_empty() {
858        return;
859    }
860
861    collection.load_fonts_from_paths(&font_dirs);
862
863    let names_before: Vec<String> = collection.family_names().map(String::from).collect();
864
865    collection.make_shared();
866
867    let names_after: Vec<String> = collection.family_names().map(String::from).collect();
868
869    assert_eq!(names_before.len(), names_after.len());
870}
