30impl FontInfo {
31    /// Creates a new font object from the given source and index.
32    pub fn from_source(source: SourceInfo, index: u32) -> Option<Self> {
33        match &source.kind {
34            #[cfg(feature = "std")]
35            SourceKind::Path(path) => {
36                let file = std::fs::File::open(&**path).ok()?;
37                let mapped = unsafe { memmap2::Mmap::map(&file).ok()? };
38                let font = FontRef::from_index(&mapped, index).ok()?;
39                Self::from_font_ref(&font, source.clone(), index)
40            }
41            SourceKind::Memory(memory) => {
42                let font = FontRef::from_index(memory.as_ref(), index).ok()?;
43                Self::from_font_ref(&font, source.clone(), index)
44            }
45        }
46    }

59    /// Attempts to load the font, optionally from a source cache.
60    pub fn load(&self, source_cache: Option<&mut SourceCache>) -> Option<Blob<u8>> {
61        if let Some(source_cache) = source_cache {
62            source_cache.get(&self.source)
63        } else {
64            match &self.source.kind {
65                SourceKind::Memory(blob) => Some(blob.clone()),
66                #[cfg(feature = "std")]
67                SourceKind::Path(path) => super::source_cache::load_blob(path),
68            }
69        }
70    }

89    /// Returns synthesis suggestions for this font with the given attributes.
90    pub fn synthesis(&self, width: FontWidth, style: FontStyle, weight: FontWeight) -> Synthesis {
91        let mut synth = Synthesis::default();
92        let mut len = 0_usize;
93        if self.has_width_axis() && self.width != width {
94            synth.vars[len] = (Tag::new(b"wdth"), width.percentage());
95            len += 1;
96        }
97        if self.weight != weight {
98            if self.has_weight_axis() {
99                synth.vars[len] = (Tag::new(b"wght"), weight.value());
100                len += 1;
101            } else if weight.value() > self.weight.value() {
102                synth.embolden = true;
103            }
104        }
105        if self.style != style {
106            match style {
107                FontStyle::Normal => {}
108                FontStyle::Italic => {
109                    if self.style == FontStyle::Normal {
110                        if self.has_italic_axis() {
111                            synth.vars[len] = (Tag::new(b"ital"), 1.0);
112                            len += 1;
113                        } else if self.has_slant_axis() {
114                            synth.vars[len] = (Tag::new(b"slnt"), 14.0);
115                            len += 1;
116                        } else {
117                            synth.skew = 14;
118                        }
119                    }
120                }
121                FontStyle::Oblique(angle) => {
122                    if self.style == FontStyle::Normal {
123                        let degrees = angle.unwrap_or(14.0);
124                        if self.has_slant_axis() {
125                            synth.vars[len] = (Tag::new(b"slnt"), degrees);
126                            len += 1;
127                        } else if self.has_italic_axis() && degrees > 0. {
128                            synth.vars[len] = (Tag::new(b"ital"), 1.0);
129                            len += 1;
130                        } else {
131                            synth.skew = degrees as i8;
132                        }
133                    }
134                }
135            }
136        }
137        synth.len = len as u8;
138        synth
139    }

206impl FontInfo {
207    pub(crate) fn from_font_ref(
208        font: &FontRef<'_>,
209        source: SourceInfo,
210        index: u32,
211    ) -> Option<Self> {
212        // It's probably not useful to retain fonts that don't have
213        // a valid cmap so just bail here if we fail.
214        let charmap_index = CharmapIndex::new(font)?;
215        let (width, style, weight) = read_attributes(font);
216        let (axes, attr_axes) = if let Ok(fvar_axes) = font.fvar().and_then(|fvar| fvar.axes()) {
217            let mut axes = SmallVec::<[AxisInfo; 1]>::with_capacity(fvar_axes.len());
218            let mut attrs_axes = 0_u8;
219            for fvar_axis in fvar_axes {
220                let axis = AxisInfo {
221                    tag: fvar_axis.axis_tag(),
222                    min: fvar_axis.min_value().to_f32(),
223                    max: fvar_axis.max_value().to_f32(),
224                    default: fvar_axis.default_value().to_f32(),
225                };
226                axes.push(axis);
227                match &axis.tag.to_be_bytes() {
228                    b"wght" => attrs_axes |= WEIGHT_AXIS,
229                    b"wdth" => attrs_axes |= WIDTH_AXIS,
230                    b"slnt" => attrs_axes |= SLANT_AXIS,
231                    b"ital" => attrs_axes |= ITALIC_AXIS,
232                    b"opsz" => attrs_axes |= OPTICAL_SIZE_AXIS,
233                    _ => {}
234                }
235            }
236            (axes, attrs_axes)
237        } else {
238            (SmallVec::default(), 0)
239        };
240        Some(Self {
241            source,
242            index,
243            width,
244            style,
245            weight,
246            axes,
247            attr_axes,
248            charmap_index,
249        })
250    }

252    #[allow(unused)]
253    pub(crate) fn maybe_override_attributes(
254        &mut self,
255        width: FontWidth,
256        style: FontStyle,
257        weight: FontWeight,
258    ) {
259        if self.width == FontWidth::default() {
260            self.width = width;
261        }
262        if self.style == FontStyle::default() {
263            self.style = style;
264        }
265        if self.weight == FontWeight::default() {
266            self.weight = weight;
267        }
268    }
269
270    pub(crate) fn apply_override(&mut self, info_override: &FontInfoOverride<'_>) {
271        if let Some(width) = info_override.width {
272            self.width = width;
273        }
274        if let Some(style) = info_override.style {
275            self.style = style;
276        }
277        if let Some(weight) = info_override.weight {
278            self.weight = weight;
279        }
280        if let Some(axes) = info_override.axes {
281            // This is O(n^2) but no font should have enough axes for that to
282            // matter (read-fonts does the same thing)
283            for (tag, value) in axes {
284                if let Some(axis) = self.axes.iter_mut().find(|axis| axis.tag == *tag) {
285                    axis.default = *value;
286                }
287            }
288        }
289    }
290}

292const WEIGHT_AXIS: u8 = 0x01;
293const WIDTH_AXIS: u8 = 0x02;
294const SLANT_AXIS: u8 = 0x04;
295const ITALIC_AXIS: u8 = 0x08;
296const OPTICAL_SIZE_AXIS: u8 = 0x10;

298/// An axis of variation for a variable font.
299///
300/// Instances of this can be obtained from [`FontInfo::axes`].
301///
302/// These give the [`Tag`] and range of valid values for a given font
303/// variation. In `parley`, these values are used to create a
304/// `FontVariation`.
305///
306/// OpenType defines some common axes:
307///
308/// * [Italic](https://fonts.google.com/knowledge/glossary/italic_axis) or `ital`
309/// * [Optical Size](https://fonts.google.com/knowledge/glossary/optical_size_axis) or `opsz`
310/// * [Slant](https://fonts.google.com/knowledge/glossary/slant_axis) or `slnt`
311/// * [FontWeight](https://fonts.google.com/knowledge/glossary/weight_axis) or `wght`
312/// * [Width](https://fonts.google.com/knowledge/glossary/width_axis) or `wdth`
313///
314/// For a broader explanation of this, see
315/// [Axis in Variable Fonts](https://fonts.google.com/knowledge/glossary/axis_in_variable_fonts)
316/// from Google Fonts.
317#[derive(Copy, Clone, Default, Debug)]
318pub struct AxisInfo {
319    /// The tag that identifies the axis.
320    pub tag: Tag,
321    /// The inclusive minimum value of the axis.
322    pub min: f32,
323    /// The inclusive maximum value of the axis.
324    pub max: f32,
325    /// The default value of the axis.
326    pub default: f32,
327}

329/// Suggestions for synthesizing a set of font attributes for a given
330/// font.
331///
332/// Instances of this can be obtained from [`FontInfo::synthesis`]
333/// as well as [`QueryFont::synthesis`].
334///
335/// [`QueryFont::synthesis`]: crate::QueryFont::synthesis
336#[derive(Copy, Clone, Default, PartialEq)]
337pub struct Synthesis {
338    vars: [(Tag, f32); 3],
339    len: u8,
340    embolden: bool,
341    skew: i8,
342}
343
344impl Synthesis {
345    /// Returns `true` if any synthesis suggestions are available.
346    pub fn any(&self) -> bool {
347        self.len != 0 || self.embolden || self.skew != 0
348    }
349
350    /// Returns the variation settings that should be applied to match the
351    /// requested attributes.
352    ///
353    /// When using `parley`, these can be used to create `FontVariation`
354    /// settings.
355    pub fn variation_settings(&self) -> &[(Tag, f32)] {
356        &self.vars[..self.len as usize]
357    }
358
359    /// Returns `true` if the scaler should apply a faux bold.
360    pub fn embolden(&self) -> bool {
361        self.embolden
362    }
363
364    /// Returns a skew angle for faux italic/oblique, if requested.
365    pub fn skew(&self) -> Option<f32> {
366        if self.skew != 0 {
367            Some(self.skew as f32)
368        } else {
369            None
370        }
371    }
372}

374#[allow(clippy::missing_fields_in_debug)]
375impl fmt::Debug for Synthesis {
376    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
377        f.debug_struct("Synthesis")
378            .field("vars", &self.variation_settings())
379            .field("embolden", &self.embolden)

382    }

385fn read_attributes(font: &FontRef<'_>) -> (FontWidth, FontStyle, FontWeight) {
386    use read_fonts::{
387        TableProvider,
388        tables::{
389            head::{Head, MacStyle},
390            os2::{Os2, SelectionFlags},
391            post::Post,
392        },
393    };
394
395    fn width_from_width_class(width_class: u16) -> FontWidth {
396        FontWidth::from_ratio(match width_class {
397            0..=1 => 0.5,
398            2 => 0.625,
399            3 => 0.75,
400            4 => 0.875,
401            5 => 1.0,
402            6 => 1.125,
403            7 => 1.25,
404            8 => 1.5,
405            _ => 2.0,
406        })
407    }
408
409    fn from_os2_post(os2: Os2<'_>, post: Option<Post<'_>>) -> (FontWidth, FontStyle, FontWeight) {
410        let width = width_from_width_class(os2.us_width_class());
411        // Bits 1 and 9 of the fsSelection field signify italic and
412        // oblique, respectively.
413        // See: <https://learn.microsoft.com/en-us/typography/opentype/spec/os2#fsselection>
414        let fs_selection = os2.fs_selection();
415        let style = if fs_selection.contains(SelectionFlags::ITALIC) {
416            FontStyle::Italic
417        } else if fs_selection.contains(SelectionFlags::OBLIQUE) {
418            let angle = post.map(|post| post.italic_angle().to_f64() as f32);
419            FontStyle::Oblique(angle)
420        } else {
421            FontStyle::Normal
422        };
423        // The usFontWeightClass field is specified with a 1-1000 range, but
424        // we don't clamp here because variable fonts could potentially
425        // have a value outside of that range.
426        // See <https://learn.microsoft.com/en-us/typography/opentype/spec/os2#usweightclass>
427        let weight = FontWeight::new(os2.us_weight_class() as f32);
428        (width, style, weight)
429    }
430
431    fn from_head(head: Head<'_>) -> (FontWidth, FontStyle, FontWeight) {
432        let mac_style = head.mac_style();
433        let style = if mac_style.contains(MacStyle::ITALIC) {
434            FontStyle::Italic
435        } else {
436            FontStyle::default()
437        };
438        let weight = if mac_style.contains(MacStyle::BOLD) {
439            FontWeight::BOLD
440        } else {
441            FontWeight::default()
442        };
443        (FontWidth::default(), style, weight)
444    }
445
446    if let Ok(os2) = font.os2() {
447        // Prefer values from the OS/2 table if it exists. We also use
448        // the post table to extract the angle for oblique styles.
449        from_os2_post(os2, font.post().ok())
450    } else if let Ok(head) = font.head() {
451        // Otherwise, fall back to the macStyle field of the head table.
452        from_head(head)
453    } else {
454        (
455            FontWidth::default(),
456            FontStyle::Normal,
457            FontWeight::default(),
458        )
459    }
460}

462/// Helper for specifying aspects of a font's metadata to be overridden when the
463/// font is registered. Helpful when implementing a `@font-face`-like API, which
464/// allows those defining the fonts to specify certain font properties manually.
465#[derive(Copy, Clone, Debug, Default)]
466pub struct FontInfoOverride<'a> {
467    /// Font family name to be used instead of the one specified in the font
468    /// itself.
469    pub family_name: Option<&'a str>,
470    /// Font width to be used instead of the one specified in the font itself.
471    pub width: Option<FontWidth>,
472    /// Font's visual style / "slope" to be used instead of the one specified in
473    /// the font itself.
474    pub style: Option<FontStyle>,
475    /// Font weight to be used instead of the one specified in the font itself.
476    pub weight: Option<FontWeight>,
477    /// Default values for the font's variation axes. Axes not included within
478    /// the font will be ignored.
479    pub axes: Option<&'a [(Tag, f32)]>,
480}
