4//! Support for working with Unicode scripts.
5
6use parlance::Script;
7
8pub trait ScriptExt {
9    /// Returns a mapping of scripts to sample text.
10    fn all_samples() -> &'static [(Script, &'static str)];
11
12    /// Returns a string containing sample characters for this script.
13    fn sample(&self) -> Option<&'static str>;
14}
15
16impl ScriptExt for Script {
17    fn all_samples() -> &'static [(Self, &'static str)] {
18        SCRIPT_SAMPLES
19    }
20
21    fn sample(&self) -> Option<&'static str> {
22        let ix = SCRIPT_SAMPLES
23            .binary_search_by(|entry| entry.0.cmp(self))
24            .ok()?;
25        SCRIPT_SAMPLES.get(ix).map(|entry| entry.1)
26    }
27}
28
29#[test]
30fn assert_sorted() {
31    for w in SCRIPT_SAMPLES.windows(2) {
32        let &[prev, curr] = w else { unreachable!() };
33        assert!(prev.0 < curr.0);
34    }
35}
