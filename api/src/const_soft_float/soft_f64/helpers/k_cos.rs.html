13use crate::soft_f64::SoftF64;
14
15const C1: SoftF64 = SoftF64(4.16666666666666019037e-02); /* 0x3FA55555, 0x5555554C */
16const C2: SoftF64 = SoftF64(-1.38888888888741095749e-03); /* 0xBF56C16C, 0x16C15177 */
17const C3: SoftF64 = SoftF64(2.48015872894767294178e-05); /* 0x3EFA01A0, 0x19CB1590 */
18const C4: SoftF64 = SoftF64(-2.75573143513906633035e-07); /* 0xBE927E4F, 0x809C52AD */
19const C5: SoftF64 = SoftF64(2.08757232129817482790e-09); /* 0x3E21EE9E, 0xBDB4B1C4 */
20const C6: SoftF64 = SoftF64(-1.13596475577881948265e-11); /* 0xBDA8FAE9, 0xBE8838D4 */

22// kernel cos function on [-pi/4, pi/4], pi/4 ~ 0.785398164
23// Input x is assumed to be bounded by ~pi/4 in magnitude.
24// Input y is the tail of x.
25//
26// Algorithm
27//      1. Since cos(-x) = cos(x), we need only to consider positive x.
28//      2. if x < 2^-27 (hx<0x3e400000 0), return 1 with inexact if x!=0.
29//      3. cos(x) is approximated by a polynomial of degree 14 on
30//         [0,pi/4]
31//                                       4            14
32//              cos(x) ~ 1 - x*x/2 + C1*x + ... + C6*x
33//         where the remez error is
34//
35//      |              2     4     6     8     10    12     14 |     -58
36//      |cos(x)-(1-.5*x +C1*x +C2*x +C3*x +C4*x +C5*x  +C6*x  )| <= 2
37//      |                                                      |
38//
39//                     4     6     8     10    12     14
40//      4. let r = C1*x +C2*x +C3*x +C4*x +C5*x  +C6*x  , then
41//             cos(x) ~ 1 - x*x/2 + r
42//         since cos(x+y) ~ cos(x) - sin(x)*y
43//                        ~ cos(x) - x*y,
44//         a correction term is necessary in cos(x) and hence
45//              cos(x+y) = 1 - (x*x/2 - (r - x*y))
46//         For better accuracy, rearrange to
47//              cos(x+y) ~ w + (tmp + (r-x*y))
48//         where w = 1 - x*x/2 and tmp is a tiny correction term
49//         (1 - x*x/2 == w + tmp exactly in infinite precision).
50//         The exactness of w + tmp in infinite precision depends on w
51//         and tmp having the same precision as x.  If they have extra
52//         precision due to compiler bugs, then the extra precision is
53//         only good provided it is retained in all terms of the final
54//         expression for cos().  Retention happens in all cases tested
55//         under FreeBSD, so don't pessimize things by forcibly clipping
56//         any extra precision in w.
57pub(crate) const fn k_cos(x: SoftF64, y: SoftF64) -> SoftF64 {
58    let z = x.mul(x);
59    let w = z.mul(z);
60    let r = z
61        .mul(C1.add(z.mul(C2.add(z.mul(C3)))))
62        .add(w.mul(w.mul(C4.add(z.mul(C5.add(z.mul(C6)))))));
63    let hz = SoftF64(0.5).mul(z);
64    let w = SoftF64::ZERO.sub(hz);
65    w.add(((SoftF64::ONE.sub(w)).sub(hz)).add(z.mul(r).sub(x.mul(y))))
66}
