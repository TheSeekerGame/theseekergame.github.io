63pub fn derive_as_bind_group(ast: syn::DeriveInput) -> Result<TokenStream> {
64    let (render_path, image_path, asset_path, ecs_path) = BevyManifest::shared(|manifest| {
65        let render_path = manifest.get_path("bevy_render");
66        let image_path = manifest.get_path("bevy_image");
67        let asset_path = manifest.get_path("bevy_asset");
68        let ecs_path = manifest.get_path("bevy_ecs");
69
70        (render_path, image_path, asset_path, ecs_path)
71    });
72
73    let mut binding_states: Vec<BindingState> = Vec::new();
74    let mut binding_impls = Vec::new();
75    let mut bindless_binding_layouts = Vec::new();
76    let mut non_bindless_binding_layouts = Vec::new();
77    let mut bindless_resource_types = Vec::new();
78    let mut bindless_buffer_descriptors = Vec::new();
79    // Whether this material needs buffer binding arrays (BUFFER_BINDING_ARRAY feature).
80    // False when only #[data(...)], textures, and samplers are used.
81    // NOTE: When wgpu adds support for buffer binding arrays on Metal
82    // (see https://github.com/gfx-rs/wgpu/pull/9081), this conditional
83    // can be removed and BUFFER_BINDING_ARRAY required unconditionally.
84    let mut has_buffer_binding_arrays = false;
85    let mut attr_prepared_data_ident = None;
86    // After the first attribute pass, this will be `None` if the object isn't
87    // bindless and `Some` if it is.
88    let mut attr_bindless_count = None;
89    let mut attr_bindless_index_table_range = None;
90    let mut attr_bindless_index_table_binding = None;
91
92    // `actual_bindless_slot_count` holds the actual number of bindless slots
93    // per bind group, taking into account whether the current platform supports
94    // bindless resources.
95    let actual_bindless_slot_count = Ident::new("actual_bindless_slot_count", Span::call_site());
96    let bind_group_layout_entries = Ident::new("bind_group_layout_entries", Span::call_site());
97
98    // The `BufferBindingType` and corresponding `BufferUsages` used for
99    // uniforms. We need this because bindless uniforms don't exist, so in
100    // bindless mode we must promote uniforms to storage buffers.
101    let uniform_binding_type = Ident::new("uniform_binding_type", Span::call_site());
102    let uniform_buffer_usages = Ident::new("uniform_buffer_usages", Span::call_site());
103
104    // Read struct-level attributes, first pass.
105    for attr in &ast.attrs {
106        if let Some(attr_ident) = attr.path().get_ident() {
107            if attr_ident == BIND_GROUP_DATA_ATTRIBUTE_NAME {
108                if let Ok(prepared_data_ident) =
109                    attr.parse_args_with(|input: ParseStream| input.parse::<Ident>())
110                {
111                    attr_prepared_data_ident = Some(prepared_data_ident);
112                }
113            } else if attr_ident == BINDLESS_ATTRIBUTE_NAME {
114                attr_bindless_count = Some(BindlessSlabResourceLimitAttr::Auto);
115                if let Meta::List(_) = attr.meta {
116                    // Parse bindless features.
117                    attr.parse_nested_meta(|submeta| {
118                        if submeta.path.is_ident(&LIMIT_MODIFIER_NAME) {
119                            let content;
120                            parenthesized!(content in submeta.input);
121                            let lit: LitInt = content.parse()?;
122
123                            attr_bindless_count = Some(BindlessSlabResourceLimitAttr::Limit(lit));
124                            return Ok(());
125                        }
126
127                        if submeta.path.is_ident(&INDEX_TABLE_MODIFIER_NAME) {
128                            submeta.parse_nested_meta(|subsubmeta| {
129                                if subsubmeta.path.is_ident(&RANGE_MODIFIER_NAME) {
130                                    let content;
131                                    parenthesized!(content in subsubmeta.input);
132                                    let start: LitInt = content.parse()?;
133                                    content.parse::<DotDot>()?;
134                                    let end: LitInt = content.parse()?;
135                                    attr_bindless_index_table_range =
136                                        Some(BindlessIndexTableRangeAttr { start, end });
137                                    return Ok(());
138                                }
139
140                                if subsubmeta.path.is_ident(&BINDING_MODIFIER_NAME) {
141                                    let content;
142                                    parenthesized!(content in subsubmeta.input);
143                                    let lit: LitInt = content.parse()?;
144
145                                    attr_bindless_index_table_binding = Some(lit);
146                                    return Ok(());
147                                }
148
149                                Err(Error::new_spanned(
150                                    attr,
151                                    "Expected `range(M..N)` or `binding(N)`",
152                                ))
153                            })?;
154                            return Ok(());
155                        }
156
157                        Err(Error::new_spanned(
158                            attr,
159                            "Expected `limit` or `index_table`",
160                        ))
161                    })?;
162                }
163            }
164        }
165    }
166
167    // Read struct-level attributes, second pass.
168    for attr in &ast.attrs {
169        if let Some(attr_ident) = attr.path().get_ident()
170            && (attr_ident == UNIFORM_ATTRIBUTE_NAME || attr_ident == DATA_ATTRIBUTE_NAME)
171        {
172            let UniformBindingAttr {
173                binding_type,
174                binding_index,
175                converted_shader_type,
176                binding_array: binding_array_binding,
177            } = get_uniform_binding_attr(attr)?;
178            match binding_type {
179                UniformBindingAttrType::Uniform => {
180                    binding_impls.push(quote! {{
181                            use #render_path::render_resource::AsBindGroupShaderType;
182                            let mut buffer = #render_path::render_resource::encase::UniformBuffer::new(::std::vec::Vec::new());
183                            let converted: #converted_shader_type = self.as_bind_group_shader_type(&images);
184                            buffer.write(&converted).unwrap();
185                            (
186                                #binding_index,
187                                #render_path::render_resource::OwnedBindingResource::Buffer(render_device.create_buffer_with_data(
188                                    &#render_path::render_resource::BufferInitDescriptor {
189                                        label: #FQOption::None,
190                                        usage: #uniform_buffer_usages,
191                                        contents: buffer.as_ref(),
192                                    },
193                                ))
194                            )
195                        }});
196
197                    match (&binding_array_binding, &attr_bindless_count) {
198                        (&None, &Some(_)) => {
199                            return Err(Error::new_spanned(
200                                attr,
201                                "Must specify `binding_array(...)` with `#[uniform]` if the \
202                                    object is bindless",
203                            ));
204                        }
205                        (&Some(_), &None) => {
206                            return Err(Error::new_spanned(
207                                attr,
208                                "`binding_array(...)` with `#[uniform]` requires the object to \
209                                    be bindless",
210                            ));
211                        }
212                        _ => {}
213                    }
214
215                    let binding_array_binding = binding_array_binding.unwrap_or(0);
216                    bindless_binding_layouts.push(quote! {
217                            #bind_group_layout_entries.push(
218                                #render_path::render_resource::BindGroupLayoutEntry {
219                                    binding: #binding_array_binding,
220                                    visibility: #render_path::render_resource::ShaderStages::FRAGMENT | #render_path::render_resource::ShaderStages::VERTEX | #render_path::render_resource::ShaderStages::COMPUTE,
221                                    ty: #render_path::render_resource::BindingType::Buffer {
222                                        ty: #uniform_binding_type,
223                                        has_dynamic_offset: false,
224                                        min_binding_size: #FQOption::Some(<#converted_shader_type as #render_path::render_resource::ShaderType>::min_size()),
225                                    },
226                                    count: #actual_bindless_slot_count,
227                                }
228                            );
229                        });
230
231                    add_bindless_resource_type(
232                        &render_path,
233                        &mut bindless_resource_types,
234                        binding_index,
235                        quote! { #render_path::render_resource::BindlessResourceType::Buffer },
236                    );
237                    has_buffer_binding_arrays = true;
238                }
239
240                UniformBindingAttrType::Data => {
241                    binding_impls.push(quote! {{
242                            use #render_path::render_resource::AsBindGroupShaderType;
243                            use #render_path::render_resource::encase::{ShaderType, internal::WriteInto};
244                            let mut buffer: ::std::vec::Vec<u8> = ::std::vec::Vec::new();
245                            let converted: #converted_shader_type = self.as_bind_group_shader_type(&images);
246                            converted.write_into(
247                                &mut #render_path::render_resource::encase::internal::Writer::new(
248                                    &converted,
249                                    &mut buffer,
250                                    0,
251                                ).unwrap(),
252                            );
253                            let min_size = <#converted_shader_type as #render_path::render_resource::ShaderType>::min_size().get() as usize;
254                            while buffer.len() < min_size {
255                                buffer.push(0);
256                            }
257                            (
258                                #binding_index,
259                                #render_path::render_resource::OwnedBindingResource::Data(
260                                    #render_path::render_resource::OwnedData(buffer)
261                                )
262                            )
263                        }});
264
265                    let binding_array_binding = binding_array_binding.unwrap_or(0);
266                    bindless_binding_layouts.push(quote! {
267                            #bind_group_layout_entries.push(
268                                #render_path::render_resource::BindGroupLayoutEntry {
269                                    binding: #binding_array_binding,
270                                    visibility: #render_path::render_resource::ShaderStages::FRAGMENT | #render_path::render_resource::ShaderStages::VERTEX | #render_path::render_resource::ShaderStages::COMPUTE,
271                                    ty: #render_path::render_resource::BindingType::Buffer {
272                                        ty: #uniform_binding_type,
273                                        has_dynamic_offset: false,
274                                        min_binding_size: #FQOption::Some(<#converted_shader_type as #render_path::render_resource::ShaderType>::min_size()),
275                                    },
276                                    count: #FQOption::None,
277                                }
278                            );
279                        });
280
281                    add_bindless_resource_type(
282                        &render_path,
283                        &mut bindless_resource_types,
284                        binding_index,
285                        quote! { #render_path::render_resource::BindlessResourceType::DataBuffer },
286                    );
287                }
288            }
289
290            // Push the non-bindless binding layout.
291
292            non_bindless_binding_layouts.push(quote!{
293                    #bind_group_layout_entries.push(
294                        #render_path::render_resource::BindGroupLayoutEntry {
295                            binding: #binding_index,
296                            visibility: #render_path::render_resource::ShaderStages::FRAGMENT | #render_path::render_resource::ShaderStages::VERTEX | #render_path::render_resource::ShaderStages::COMPUTE,
297                            ty: #render_path::render_resource::BindingType::Buffer {
298                                ty: #uniform_binding_type,
299                                has_dynamic_offset: false,
300                                min_binding_size: #FQOption::Some(<#converted_shader_type as #render_path::render_resource::ShaderType>::min_size()),
301                            },
302                            count: #FQOption::None,
303                        }
304                    );
305                });
306
307            bindless_buffer_descriptors.push(quote! {
308                #render_path::render_resource::BindlessBufferDescriptor {
309                    // Note that, because this is bindless, *binding
310                    // index* here refers to the index in the
311                    // bindless index table (`bindless_index`), and
312                    // the actual binding number is the *binding
313                    // array binding*.
314                    binding_number: #render_path::render_resource::BindingNumber(
315                        #binding_array_binding
316                    ),
317                    bindless_index:
318                        #render_path::render_resource::BindlessIndex(#binding_index),
319                            size: #FQOption::Some(
320                                <
321                                    #converted_shader_type as
322                                    #render_path::render_resource::ShaderType
323                                >::min_size().get() as usize
324                            ),
325                }
326            });
327
328            let required_len = binding_index as usize + 1;
329            if required_len > binding_states.len() {
330                binding_states.resize(required_len, BindingState::Free);
331            }
332            binding_states[binding_index as usize] = BindingState::OccupiedConvertedUniform;
333        }
334    }
335
336    let fields = match &ast.data {
337        Data::Struct(DataStruct {
338            fields: Fields::Named(fields),
339            ..
340        }) => &fields.named,
341        _ => {
342            return Err(Error::new_spanned(
343                ast,
344                "Expected a struct with named fields",
345            ));
346        }
347    };
348
349    // Count the number of sampler fields needed. We might have to disable
350    // bindless if bindless arrays take the GPU over the maximum number of
351    // samplers.
352    let mut sampler_binding_count: u32 = 0;
353
354    // Read field-level attributes
355    for field in fields {
356        // Search ahead for texture attributes so we can use them with any
357        // corresponding sampler attribute.
358        let mut tex_attrs = None;
359        for attr in &field.attrs {
360            let Some(attr_ident) = attr.path().get_ident() else {
361                continue;
362            };
363            if attr_ident == TEXTURE_ATTRIBUTE_NAME {
364                let (_binding_index, nested_meta_items) = get_binding_nested_attr(attr)?;
365                tex_attrs = Some(get_texture_attrs(nested_meta_items)?);
366            }
367        }
368
369        for attr in &field.attrs {
370            let Some(attr_ident) = attr.path().get_ident() else {
371                continue;
372            };
373
374            let binding_type = if attr_ident == UNIFORM_ATTRIBUTE_NAME {
375                BindingType::Uniform
376            } else if attr_ident == TEXTURE_ATTRIBUTE_NAME {
377                BindingType::Texture
378            } else if attr_ident == STORAGE_TEXTURE_ATTRIBUTE_NAME {
379                BindingType::StorageTexture
380            } else if attr_ident == SAMPLER_ATTRIBUTE_NAME {
381                BindingType::Sampler
382            } else if attr_ident == STORAGE_ATTRIBUTE_NAME {
383                BindingType::Storage
384            } else {
385                continue;
386            };
387
388            let (binding_index, nested_meta_items) = get_binding_nested_attr(attr)?;
389
390            let field_name = field.ident.as_ref().unwrap();
391            let required_len = binding_index as usize + 1;
392            if required_len > binding_states.len() {
393                binding_states.resize(required_len, BindingState::Free);
394            }
395
396            match &mut binding_states[binding_index as usize] {
397                value @ BindingState::Free => {
398                    *value = match binding_type {
399                        BindingType::Uniform => BindingState::OccupiedMergeableUniform {
400                            uniform_fields: vec![field],
401                        },
402                        _ => {
403                            // only populate bind group entries for non-uniforms
404                            // uniform entries are deferred until the end
405                            BindingState::Occupied {
406                                binding_type,
407                                ident: field_name,
408                            }
409                        }
410                    }
411                }
412                BindingState::Occupied {
413                    binding_type,
414                    ident: occupied_ident,
415                } => {
416                    return Err(Error::new_spanned(
417                        attr,
418                        format!("The '{field_name}' field cannot be assigned to binding {binding_index} because it is already occupied by the field '{occupied_ident}' of type {binding_type:?}.")
419                    ));
420                }
421                BindingState::OccupiedConvertedUniform => {
422                    return Err(Error::new_spanned(
423                        attr,
424                        format!("The '{field_name}' field cannot be assigned to binding {binding_index} because it is already occupied by a struct-level uniform binding at the same index.")
425                    ));
426                }
427                BindingState::OccupiedMergeableUniform { uniform_fields } => match binding_type {
428                    BindingType::Uniform => {
429                        uniform_fields.push(field);
430                    }
431                    _ => {
432                        return Err(Error::new_spanned(
433                                attr,
434                                format!("The '{field_name}' field cannot be assigned to binding {binding_index} because it is already occupied by a {:?}.", BindingType::Uniform)
435                            ));
436                    }
437                },
438            }
439
440            match binding_type {
441                BindingType::Uniform => {
442                    if attr_bindless_count.is_some() {
443                        return Err(Error::new_spanned(
444                            attr,
445                            "Only structure-level `#[uniform]` attributes are supported in \
446                             bindless mode",
447                        ));
448                    }
449
450                    // uniform codegen is deferred to account for combined uniform bindings
451                }
452
453                BindingType::Storage => {
454                    let StorageAttrs {
455                        visibility,
456                        binding_array: binding_array_binding,
457                        read_only,
458                        buffer,
459                    } = get_storage_binding_attr(nested_meta_items)?;
460                    let visibility =
461                        visibility.hygienic_quote(&quote! { #render_path::render_resource });
462
463                    let field_name = field.ident.as_ref().unwrap();
464
465                    if buffer {
466                        binding_impls.push(quote! {
467                            (
468                                #binding_index,
469                                #render_path::render_resource::OwnedBindingResource::Buffer({
470                                    self.#field_name.clone()
471                                })
472                            )
473                        });
474                    } else {
475                        binding_impls.push(quote! {
476                        (
477                            #binding_index,
478                            #render_path::render_resource::OwnedBindingResource::Buffer({
479                                let handle: &#asset_path::Handle<#render_path::storage::ShaderBuffer> = (&self.#field_name);
480                                storage_buffers.get(handle).ok_or_else(|| #render_path::render_resource::AsBindGroupError::RetryNextUpdate)?.buffer.clone()
481                            })
482                        )
483                        });
484                    }
485
486                    non_bindless_binding_layouts.push(quote! {
487                        #bind_group_layout_entries.push(
488                            #render_path::render_resource::BindGroupLayoutEntry {
489                                binding: #binding_index,
490                                visibility: #visibility,
491                                ty: #render_path::render_resource::BindingType::Buffer {
492                                    ty: #render_path::render_resource::BufferBindingType::Storage { read_only: #read_only },
493                                    has_dynamic_offset: false,
494                                    min_binding_size: #FQOption::None,
495                                },
496                                count: #actual_bindless_slot_count,
497                            }
498                        );
499                    });
500
501                    if let Some(binding_array_binding) = binding_array_binding {
502                        // Add the storage buffer to the `BindlessResourceType` list
503                        // in the bindless descriptor.
504                        let bindless_resource_type = quote! {
505                            #render_path::render_resource::BindlessResourceType::Buffer
506                        };
507                        add_bindless_resource_type(
508                            &render_path,
509                            &mut bindless_resource_types,
510                            binding_index,
511                            bindless_resource_type,
512                        );
513
514                        has_buffer_binding_arrays = true;
515
516                        // Push the buffer descriptor.
517                        bindless_buffer_descriptors.push(quote! {
518                            #render_path::render_resource::BindlessBufferDescriptor {
519                                // Note that, because this is bindless, *binding
520                                // index* here refers to the index in the bindless
521                                // index table (`bindless_index`), and the actual
522                                // binding number is the *binding array binding*.
523                                binding_number: #render_path::render_resource::BindingNumber(
524                                    #binding_array_binding
525                                ),
526                                bindless_index:
527                                    #render_path::render_resource::BindlessIndex(#binding_index),
528                                size: #FQOption::None,
529                            }
530                        });
531
532                        // Declare the binding array.
533                        bindless_binding_layouts.push(quote!{
534                            #bind_group_layout_entries.push(
535                                #render_path::render_resource::BindGroupLayoutEntry {
536                                    binding: #binding_array_binding,
537                                    visibility: #render_path::render_resource::ShaderStages::FRAGMENT | #render_path::render_resource::ShaderStages::VERTEX | #render_path::render_resource::ShaderStages::COMPUTE,
538                                    ty: #render_path::render_resource::BindingType::Buffer {
539                                        ty: #render_path::render_resource::BufferBindingType::Storage {
540                                            read_only: #read_only
541                                        },
542                                        has_dynamic_offset: false,
543                                        min_binding_size: #FQOption::None,
544                                    },
545                                    count: #actual_bindless_slot_count,
546                                }
547                            );
548                        });
549                    }
550                }
551
552                BindingType::StorageTexture => {
553                    if attr_bindless_count.is_some() {
554                        return Err(Error::new_spanned(
555                            attr,
556                            "Storage textures are unsupported in bindless mode",
557                        ));
558                    }
559
560                    let StorageTextureAttrs {
561                        dimension,
562                        image_format,
563                        access,
564                        visibility,
565                    } = get_storage_texture_binding_attr(nested_meta_items)?;
566
567                    let visibility =
568                        visibility.hygienic_quote(&quote! { #render_path::render_resource });
569
570                    let fallback_image = get_fallback_image(&render_path, dimension);
571
572                    // insert fallible texture-based entries at 0 so that if we fail here, we exit before allocating any buffers
573                    binding_impls.insert(0, quote! {
574                        ( #binding_index,
575                          #render_path::render_resource::OwnedBindingResource::TextureView(
576                                #render_path::render_resource::#dimension,
577                                {
578                                    let handle: #FQOption<&#asset_path::Handle<#image_path::Image>> = (&self.#field_name).into();
579                                    if let #FQOption::Some(handle) = handle {
580                                        images.get(handle).ok_or_else(|| #render_path::render_resource::AsBindGroupError::RetryNextUpdate)?.texture_view.clone()
581                                    } else {
582                                        #fallback_image.texture_view.clone()
583                                    }
584                                }
585                            )
586                        )
587                    });
588
589                    non_bindless_binding_layouts.push(quote! {
590                        #bind_group_layout_entries.push(
591                            #render_path::render_resource::BindGroupLayoutEntry {
592                                binding: #binding_index,
593                                visibility: #visibility,
594                                ty: #render_path::render_resource::BindingType::StorageTexture {
595                                    access: #render_path::render_resource::StorageTextureAccess::#access,
596                                    format: #render_path::render_resource::TextureFormat::#image_format,
597                                    view_dimension: #render_path::render_resource::#dimension,
598                                },
599                                count: #actual_bindless_slot_count,
600                            }
601                        );
602                    });
603                }
604
605                BindingType::Texture => {
606                    let TextureAttrs {
607                        dimension,
608                        sample_type,
609                        multisampled,
610                        visibility,
611                    } = tex_attrs.as_ref().unwrap();
612
613                    let visibility =
614                        visibility.hygienic_quote(&quote! { #render_path::render_resource });
615
616                    let fallback_image = get_fallback_image(&render_path, *dimension);
617
618                    // insert fallible texture-based entries at 0 so that if we fail here, we exit before allocating any buffers
619                    binding_impls.insert(0, quote! {
620                        (
621                            #binding_index,
622                            #render_path::render_resource::OwnedBindingResource::TextureView(
623                                #render_path::render_resource::#dimension,
624                                {
625                                    let handle: #FQOption<&#asset_path::Handle<#image_path::Image>> = (&self.#field_name).into();

647                                },
648                                count: #actual_bindless_slot_count,
649                            }
650                        );
651                    });
652
653                    let bindless_resource_type = match *dimension {
654                        BindingTextureDimension::D1 => {
655                            quote! {
656                                #render_path::render_resource::BindlessResourceType::Texture1d
657                            }
658                        }
659                        BindingTextureDimension::D2 => {
660                            quote! {
661                                #render_path::render_resource::BindlessResourceType::Texture2d
662                            }
663                        }
664                        BindingTextureDimension::D2Array => {
665                            quote! {
666                                #render_path::render_resource::BindlessResourceType::Texture2dArray
667                            }
668                        }
669                        BindingTextureDimension::Cube => {
670                            quote! {
671                                #render_path::render_resource::BindlessResourceType::TextureCube
672                            }
673                        }
674                        BindingTextureDimension::CubeArray => {
675                            quote! {
676                                #render_path::render_resource::BindlessResourceType::TextureCubeArray
677                            }
678                        }
679                        BindingTextureDimension::D3 => {
680                            quote! {
681                                #render_path::render_resource::BindlessResourceType::Texture3d
682                            }
683                        }
684                    };
685
686                    // Add the texture to the `BindlessResourceType` list in the
687                    // bindless descriptor.
688                    add_bindless_resource_type(
689                        &render_path,
690                        &mut bindless_resource_types,
691                        binding_index,
692                        bindless_resource_type,
693                    );
694                }
695
696                BindingType::Sampler => {
697                    let SamplerAttrs {
698                        sampler_binding_type,
699                        visibility,
700                        ..
701                    } = get_sampler_attrs(nested_meta_items)?;
702                    let TextureAttrs { dimension, .. } = tex_attrs
703                        .as_ref()
704                        .expect("sampler attribute must have matching texture attribute");
705
706                    let visibility =
707                        visibility.hygienic_quote(&quote! { #render_path::render_resource });
708
709                    let fallback_image = get_fallback_image(&render_path, *dimension);
710
711                    let expected_samplers = match sampler_binding_type {
712                        SamplerBindingType::Filtering => {
713                            quote!( [#render_path::render_resource::TextureSampleType::Float { filterable: true }] )
714                        }
715                        SamplerBindingType::NonFiltering => quote!([
716                            #render_path::render_resource::TextureSampleType::Float { filterable: false },
717                            #render_path::render_resource::TextureSampleType::Sint,
718                            #render_path::render_resource::TextureSampleType::Uint,
719                        ]),
720                        SamplerBindingType::Comparison => {
721                            quote!( [#render_path::render_resource::TextureSampleType::Depth] )
722                        }
723                    };
724
725                    // insert fallible texture-based entries at 0 so that if we fail here, we exit before allocating any buffers
726                    binding_impls.insert(0, quote! {
727                        (
728                            #binding_index,
729                            #render_path::render_resource::OwnedBindingResource::Sampler(

735                                    let image = images.get(handle).ok_or_else(|| #render_path::render_resource::AsBindGroupError::RetryNextUpdate)?;

738                                        return #FQResult::Err(#render_path::render_resource::AsBindGroupError::InvalidSamplerType(
739                                            #binding_index,
740                                            "None".to_string(),
741                                            format!("{:?}", #expected_samplers),
742                                        ));
743                                    };
744
745                                    let valid = #expected_samplers.contains(&sample_type);
746
747                                    if !valid {
748                                        return #FQResult::Err(#render_path::render_resource::AsBindGroupError::InvalidSamplerType(
749                                            #binding_index,
750                                            format!("{:?}", sample_type),
751                                            format!("{:?}", #expected_samplers),

753                                    }
754                                    image.sampler.clone()
755                                } else {

757                                }
758                            })
759                        )
760                    });
761

763

766                            #render_path::render_resource::BindGroupLayoutEntry {

773                    });

776                    // bindless descriptor.

955    // the `#[bindless(index_table(binding(B)))]` into account.

979                    }
980                };

988                            ) &&

990                                render_device.limits().max_binding_array_sampler_elements_per_shader_stage >=
991                                    (#sampler_binding_count * #bindless_count_syntax)
992                        }
993                };
994                let actual_bindless_slot_count_declaration = quote! {

1003                    BindlessSlabResourceLimitAttr::Auto => {

1007                            > {

1009                            }
1010                        }
1011                    }
1012                    BindlessSlabResourceLimitAttr::Limit(lit) => {

1016                            > {
1017                                #FQOption::Some(#render_path::render_resource::BindlessSlabResourceLimit::Custom(#lit))
1018                            }
1019                        }
1020                    }
1021                };
1022
1023                let bindless_buffer_descriptor_count = bindless_buffer_descriptors.len();

1026                // a `const fn`.

1034                    ]> = ::std::sync::LazyLock::new(|| {
1035                        [#(#bindless_buffer_descriptors),*]
1036                    });

1061            None => (
1062                TokenStream::new().into(),

1066        };

1146/// Adds a bindless resource type to the `BindlessResourceType` array in the
1147/// bindless descriptor we're building up.
1148///
1149/// See the `bevy_render::render_resource::bindless::BindlessResourceType`
1150/// documentation for more information.
1151fn add_bindless_resource_type(
1152    render_path: &syn::Path,
1153    bindless_resource_types: &mut Vec<proc_macro2::TokenStream>,
1154    binding_index: u32,
1155    bindless_resource_type: proc_macro2::TokenStream,
1156) {
1157    // If we need to grow the array, pad the unused fields with
1158    // `BindlessResourceType::None`.
1159    if bindless_resource_types.len() < (binding_index as usize + 1) {
1160        bindless_resource_types.resize_with(binding_index as usize + 1, || {
1161            quote! { #render_path::render_resource::BindlessResourceType::None }
1162        });
1163    }
1164
1165    // Assign the `BindlessResourceType`.
1166    bindless_resource_types[binding_index as usize] = bindless_resource_type;
1167}

1221/// Represents the arguments for any general binding attribute.
1222///
1223/// If parsed, represents an attribute
1224/// like `#[foo(LitInt, ...)]` where the rest is optional [`Meta`].
1225enum BindingMeta {
1226    IndexOnly(LitInt),
1227    IndexWithOptions(BindingIndexOptions),
1228}

1230/// Represents the arguments for an attribute with a list of arguments.
1231///
1232/// This represents, for example, `#[texture(0, dimension = "2d_array")]`.
1233struct BindingIndexOptions {
1234    lit_int: LitInt,
1235    _comma: Comma,
1236    meta_list: Punctuated<Meta, Comma>,
1237}
1238
1239impl Parse for BindingMeta {
1240    fn parse(input: ParseStream) -> Result<Self> {
1241        if input.peek2(Comma) {
1242            input.parse().map(Self::IndexWithOptions)
1243        } else {
1244            input.parse().map(Self::IndexOnly)
1245        }
1246    }
1247}

1249impl Parse for BindingIndexOptions {
1250    fn parse(input: ParseStream) -> Result<Self> {
1251        Ok(Self {
1252            lit_int: input.parse()?,
1253            _comma: input.parse()?,
1254            meta_list: input.parse_terminated(Meta::parse, Comma)?,
1255        })
1256    }
1257}
1258
1259impl Parse for UniformBindingMeta {
1260    // Parse syntax like `#[uniform(0, StandardMaterial, binding_array(10))]`.
1261    fn parse(input: ParseStream) -> Result<Self> {
1262        let lit_int = input.parse()?;
1263        input.parse::<Comma>()?;
1264        let ident = input.parse()?;
1265
1266        // Look for a `binding_array(BINDING_NUMBER)` declaration.
1267        let mut binding_array: Option<LitInt> = None;
1268        if input.parse::<Comma>().is_ok() {
1269            if input
1270                .parse::<syn::Path>()?
1271                .get_ident()
1272                .is_none_or(|ident| *ident != BINDING_ARRAY_MODIFIER_NAME)
1273            {
1274                return Err(Error::new_spanned(ident, "Expected `binding_array`"));
1275            }
1276            let parser;
1277            parenthesized!(parser in input);
1278            binding_array = Some(parser.parse()?);
1279        }
1280
1281        Ok(Self {
1282            lit_int,
1283            ident,
1284            binding_array,
1285        })
1286    }
1287}

1373impl ShaderStageVisibility {
1374    fn hygienic_quote(&self, path: &proc_macro2::TokenStream) -> proc_macro2::TokenStream {
1375        match self {
1376            ShaderStageVisibility::All => quote! {
1377                if cfg!(feature = "webgpu") {
1378                    todo!("Please use a more specific shader stage: https://github.com/gfx-rs/wgpu/issues/7708")

1392                }
1393                if flags.compute {
1394                    quoted.push(quote! { #path::ShaderStages::COMPUTE });
1395                }
1396
1397                quote! { #(#quoted)|* }
1398            }
1399        }
1400    }

1403const VISIBILITY: Symbol = Symbol("visibility");
1404const VISIBILITY_VERTEX: Symbol = Symbol("vertex");
1405const VISIBILITY_FRAGMENT: Symbol = Symbol("fragment");
1406const VISIBILITY_COMPUTE: Symbol = Symbol("compute");
1407const VISIBILITY_ALL: Symbol = Symbol("all");
1408const VISIBILITY_NONE: Symbol = Symbol("none");
1409
1410fn get_visibility_flag_value(meta_list: &MetaList) -> Result<ShaderStageVisibility> {
1411    let mut flags = Vec::new();
1412
1413    meta_list.parse_nested_meta(|meta| {
1414        flags.push(meta.path);
1415        Ok(())
1416    })?;
1417
1418    if flags.is_empty() {
1419        return Err(Error::new_spanned(
1420            meta_list,
1421            "Invalid visibility format. Must be `visibility(flags)`, flags can be `all`, `none`, or a list-combination of `vertex`, `fragment` and/or `compute`."
1422        ));
1423    }
1424
1425    if flags.len() == 1
1426        && let Some(flag) = flags.first()
1427    {
1428        if flag == VISIBILITY_ALL {
1429            return Ok(ShaderStageVisibility::All);
1430        } else if flag == VISIBILITY_NONE {
1431            return Ok(ShaderStageVisibility::None);
1432        }
1433    }
1434
1435    let mut visibility = VisibilityFlags::default();
1436
1437    for flag in flags {
1438        if flag == VISIBILITY_VERTEX {
1439            visibility.vertex = true;
1440        } else if flag == VISIBILITY_FRAGMENT {
1441            visibility.fragment = true;
1442        } else if flag == VISIBILITY_COMPUTE {
1443            visibility.compute = true;
1444        } else {
1445            return Err(Error::new_spanned(
1446                flag,
1447                "Not a valid visibility flag. Must be `all`, `none`, or a list-combination of `vertex`, `fragment` and/or `compute`."
1448            ));
1449        }
1450    }
1451
1452    Ok(ShaderStageVisibility::Flags(visibility))
1453}
