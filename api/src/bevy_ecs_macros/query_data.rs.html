39fn contiguous_item_struct(
40    path: &syn::Path,
41    fields: &Fields,
42    derive_macro_call: &proc_macro2::TokenStream,
43    struct_name: &Ident,
44    visibility: &Visibility,
45    item_struct_name: &Ident,
46    field_types: &Vec<Type>,
47    user_impl_generics_with_world_and_state: &ImplGenerics,
48    field_attrs: &Vec<Vec<Attribute>>,
49    field_visibilities: &Vec<Visibility>,
50    field_members: &Vec<Member>,
51    user_ty_generics: &TypeGenerics,
52    user_ty_generics_with_world_and_state: &TypeGenerics,
53    user_where_clauses_with_world_and_state: Option<&WhereClause>,
54) -> proc_macro2::TokenStream {
55    let item_attrs = quote! {
56        #[doc = concat!(
57            "Automatically generated [`ContiguousQueryData`](",
58            stringify!(#path),
59            "::fetch::ContiguousQueryData) item type for [`",
60            stringify!(#struct_name),
61            "`], returned when iterating over contiguous query results",
62        )]
63        #[automatically_derived]
64    };
65
66    match fields {
67        Fields::Named(_) if !fields.is_empty() => quote! {
68            #derive_macro_call
69            #item_attrs
70            #visibility struct #item_struct_name #user_impl_generics_with_world_and_state #user_where_clauses_with_world_and_state {
71                #(#(#field_attrs)* #field_visibilities #field_members: <#field_types as #path::query::ContiguousQueryData>::Contiguous<'__w, '__s>,)*
72            }
73        },
74        Fields::Unnamed(_) if !fields.is_empty() => quote! {
75            #derive_macro_call
76            #item_attrs
77            #visibility struct #item_struct_name #user_impl_generics_with_world_and_state(
78                #( #field_visibilities <#field_types as #path::query::ContiguousQueryData>::Contiguous<'__w, '__s>, )*
79            ) #user_where_clauses_with_world_and_state;
80        },
81        Fields::Unit | Fields::Named(_) | Fields::Unnamed(_) => quote! {
82            #item_attrs
83            #visibility type #item_struct_name #user_ty_generics_with_world_and_state = #struct_name #user_ty_generics;
84        },
85    }
86}

88fn contiguous_query_data_impl(
89    path: &syn::Path,
90    struct_name: &Ident,
91    contiguous_item_struct_name: &Ident,
92    field_types: &Vec<Type>,
93    user_impl_generics: &ImplGenerics,
94    user_ty_generics: &TypeGenerics,
95    user_ty_generics_with_world_and_state: &TypeGenerics,
96    field_members: &Vec<Member>,
97    field_aliases: &Vec<Ident>,
98    user_where_clauses: Option<&WhereClause>,
99) -> proc_macro2::TokenStream {
100    quote! {
101        impl #user_impl_generics #path::query::ContiguousQueryData for #struct_name #user_ty_generics #user_where_clauses {
102            type Contiguous<'__w, '__s> = #contiguous_item_struct_name #user_ty_generics_with_world_and_state;
103
104            unsafe fn fetch_contiguous<'__w, '__s>(
105                _state: &'__s <Self as #path::query::WorldQuery>::State,
106                _fetch: &mut <Self as #path::query::WorldQuery>::Fetch<'__w>,
107                _entities: &'__w [#path::entity::Entity],
108            ) -> Self::Contiguous<'__w, '__s> {
109                #contiguous_item_struct_name {
110                    #(
111                        #field_members:
112                        <#field_types>::fetch_contiguous(
113                            &_state.#field_aliases,
114                            &mut _fetch.#field_aliases,
115                            _entities,
116                        ),
117                    )*
118                }
119            }
120        }
121    }
122}

124pub fn derive_query_data_impl(input: TokenStream) -> TokenStream {
125    let tokens = input.clone();
126
127    let ast = parse_macro_input!(input as DeriveInput);
128    let visibility = ast.vis;
129
130    let mut attributes = QueryDataAttributes::default();
131    for attr in &ast.attrs {
132        if !attr.path().is_ident(QUERY_DATA_ATTRIBUTE_NAME) {
133            continue;
134        }
135
136        let result = attr.parse_nested_meta(|meta| {
137            if meta.path.is_ident(MUTABLE_ATTRIBUTE_NAME) {
138                attributes.is_mutable = true;
139                Ok(())
140            } else if meta.path.is_ident(DERIVE_ATTRIBUTE_NAME) {
141                meta.parse_nested_meta(|meta| {
142                    attributes.derive_args.push(Meta::Path(meta.path));
143                    Ok(())
144                })
145            } else if meta.path.is_ident(CONTIGUOUS_ATTRIBUTE_NAME) {
146                meta.parse_nested_meta(|meta| {
147                    if meta.path.is_ident("all") {
148                        attributes.is_contiguous_mutable = true;
149                        attributes.is_contiguous_immutable = true;
150                        Ok(())
151                    } else if meta.path.is_ident("mutable") {
152                        attributes.is_contiguous_mutable = true;
153                        Ok(())
154                    } else if meta.path.is_ident("immutable") {
155                        attributes.is_contiguous_immutable = true;
156                        Ok(())
157                    } else {
158                        Err(meta.error("invalid target, expected `all`, `mutable` or `immutable`"))
159                    }
160                })
161            } else {
162                Err(meta.error(format_args!("invalid attribute, expected `{MUTABLE_ATTRIBUTE_NAME}`, `{DERIVE_ATTRIBUTE_NAME}` or `{CONTIGUOUS_ATTRIBUTE_NAME}`")))
163            }
164        });
165
166        if let Err(err) = result {
167            return err.to_compile_error().into();
168        }
169    }
170
171    let path = bevy_ecs_path();
172
173    let user_generics = ast.generics.clone();
174    let (user_impl_generics, user_ty_generics, user_where_clauses) = user_generics.split_for_impl();
175    let user_where_predicates = user_where_clauses.map(|clause| &clause.predicates);
176    let user_generics_with_world = {
177        let mut generics = ast.generics.clone();
178        generics.params.insert(0, parse_quote!('__w));
179        generics
180    };
181    let (user_impl_generics_with_world, user_ty_generics_with_world, user_where_clauses_with_world) =
182        user_generics_with_world.split_for_impl();
183    let user_generics_with_world_and_state = {
184        let mut generics = ast.generics;
185        generics.params.insert(0, parse_quote!('__w));
186        generics.params.insert(1, parse_quote!('__s));
187        generics
188    };
189    let (
190        user_impl_generics_with_world_and_state,
191        user_ty_generics_with_world_and_state,
192        user_where_clauses_with_world_and_state,
193    ) = user_generics_with_world_and_state.split_for_impl();
194
195    let struct_name = ast.ident;
196    let read_only_struct_name = if attributes.is_mutable {
197        Ident::new(&format!("{struct_name}ReadOnly"), Span::call_site())
198    } else {
199        struct_name.clone()
200    };
201
202    let item_struct_name = Ident::new(&format!("{struct_name}Item"), Span::call_site());
203    let read_only_item_struct_name = if attributes.is_mutable {
204        Ident::new(&format!("{struct_name}ReadOnlyItem"), Span::call_site())
205    } else {
206        item_struct_name.clone()
207    };
208    let contiguous_item_struct_name = if attributes.is_contiguous_mutable {
209        Ident::new(&format!("{struct_name}ContiguousItem"), Span::call_site())
210    } else {
211        item_struct_name.clone()
212    };
213    let read_only_contiguous_item_struct_name = if attributes.is_contiguous_immutable {
214        Ident::new(
215            &format!("{struct_name}ReadOnlyContiguousItem"),
216            Span::call_site(),
217        )
218    } else {
219        item_struct_name.clone()
220    };
221
222    let fetch_struct_name = Ident::new(&format!("{struct_name}Fetch"), Span::call_site());
223    let fetch_struct_name = ensure_no_collision(fetch_struct_name, tokens.clone());
224    let read_only_fetch_struct_name = if attributes.is_mutable {
225        let new_ident = Ident::new(&format!("{struct_name}ReadOnlyFetch"), Span::call_site());
226        ensure_no_collision(new_ident, tokens.clone())
227    } else {
228        fetch_struct_name.clone()
229    };
230
231    let marker_name =
232        ensure_no_collision(format_ident!("_world_query_derive_marker"), tokens.clone());
233
234    // Generate a name for the state struct that doesn't conflict
235    // with the struct definition.
236    let state_struct_name = Ident::new(&format!("{struct_name}State"), Span::call_site());
237    let state_struct_name = ensure_no_collision(state_struct_name, tokens);
238
239    let fields = match get_struct_fields(&ast.data, "derive(QueryData)") {
240        Ok(fields) => fields,
241        Err(e) => return e.into_compile_error().into(),
242    };
243
244    let field_attrs = fields.iter().map(|f| f.attrs.clone()).collect();
245    let field_visibilities = fields.iter().map(|f| f.vis.clone()).collect();
246    let field_members = fields.members().collect();
247    let field_aliases = fields
248        .members()
249        .map(|m| format_ident!("field{}", m))
250        .collect();
251    let field_types: Vec<Type> = fields.iter().map(|f| f.ty.clone()).collect();
252    let read_only_field_types = field_types
253        .iter()
254        .map(|ty| parse_quote!(<#ty as #path::query::QueryData>::ReadOnly))
255        .collect();
256
257    let derive_args = &attributes.derive_args;
258    // `#[derive()]` is valid syntax
259    let derive_macro_call = quote! { #[derive(#derive_args)] };
260
261    let mutable_item_struct = item_struct(
262        &path,
263        fields,
264        &derive_macro_call,
265        &struct_name,
266        &visibility,
267        &item_struct_name,
268        &field_types,
269        &user_impl_generics_with_world_and_state,
270        &field_attrs,
271        &field_visibilities,
272        &field_members,
273        &user_ty_generics,
274        &user_ty_generics_with_world_and_state,
275        user_where_clauses_with_world_and_state,
276    );
277    let mutable_world_query_impl = world_query_impl(
278        &path,
279        &struct_name,
280        &visibility,
281        &fetch_struct_name,
282        &field_types,
283        &user_impl_generics,
284        &user_impl_generics_with_world,
285        &user_ty_generics,
286        &user_ty_generics_with_world,
287        &field_aliases,
288        &marker_name,
289        &state_struct_name,
290        user_where_clauses,
291        user_where_clauses_with_world,
292    );
293
294    let (mutable_contiguous_item_struct, mutable_contiguous_impl) =
295        if attributes.is_contiguous_mutable {
296            let contiguous_item_struct = contiguous_item_struct(
297                &path,
298                fields,
299                &derive_macro_call,
300                &struct_name,
301                &visibility,
302                &contiguous_item_struct_name,
303                &field_types,
304                &user_impl_generics_with_world_and_state,
305                &field_attrs,
306                &field_visibilities,
307                &field_members,
308                &user_ty_generics,
309                &user_ty_generics_with_world_and_state,
310                user_where_clauses_with_world_and_state,
311            );
312
313            let contiguous_impl = contiguous_query_data_impl(
314                &path,
315                &struct_name,
316                &contiguous_item_struct_name,
317                &field_types,
318                &user_impl_generics,
319                &user_ty_generics,
320                &user_ty_generics_with_world_and_state,
321                &field_members,
322                &field_aliases,
323                user_where_clauses,
324            );
325
326            (contiguous_item_struct, contiguous_impl)
327        } else {
328            (quote! {}, quote! {})
329        };
330
331    let (read_only_struct, read_only_impl) = if attributes.is_mutable {
332        // If the query is mutable, we need to generate a separate readonly version of some things
333        let readonly_item_struct = item_struct(
334            &path,
335            fields,
336            &derive_macro_call,
337            &read_only_struct_name,
338            &visibility,
339            &read_only_item_struct_name,
340            &read_only_field_types,
341            &user_impl_generics_with_world_and_state,
342            &field_attrs,
343            &field_visibilities,
344            &field_members,
345            &user_ty_generics,
346            &user_ty_generics_with_world_and_state,
347            user_where_clauses_with_world_and_state,
348        );
349        let readonly_world_query_impl = world_query_impl(
350            &path,
351            &read_only_struct_name,
352            &visibility,
353            &read_only_fetch_struct_name,
354            &read_only_field_types,
355            &user_impl_generics,
356            &user_impl_generics_with_world,
357            &user_ty_generics,
358            &user_ty_generics_with_world,
359            &field_aliases,
360            &marker_name,
361            &state_struct_name,
362            user_where_clauses,
363            user_where_clauses_with_world,
364        );
365        let read_only_structs = quote! {
366            #[doc = concat!(
367                "Automatically generated [`WorldQuery`](",
368                stringify!(#path),
369                "::query::WorldQuery) type for a read-only variant of [`",
370                stringify!(#struct_name),
371                "`]."
372            )]
373            #[automatically_derived]
374            #visibility struct #read_only_struct_name #user_impl_generics #user_where_clauses {
375                #(
376                    #[doc = "Automatically generated read-only field for accessing `"]
377                    #[doc = stringify!(#field_types)]
378                    #[doc = "`."]
379                    #field_visibilities #field_members: #read_only_field_types,
380                )*
381            }
382
383            #readonly_item_struct
384        };
385        (read_only_structs, readonly_world_query_impl)
386    } else {
387        (quote! {}, quote! {})
388    };
389
390    let (read_only_contiguous_item_struct, read_only_contiguous_impl) =
391        if attributes.is_mutable && attributes.is_contiguous_immutable {
392            let contiguous_item_struct = contiguous_item_struct(
393                &path,
394                fields,
395                &derive_macro_call,
396                &read_only_struct_name,
397                &visibility,
398                &read_only_contiguous_item_struct_name,
399                &read_only_field_types,
400                &user_impl_generics_with_world_and_state,
401                &field_attrs,
402                &field_visibilities,
403                &field_members,
404                &user_ty_generics,
405                &user_ty_generics_with_world_and_state,
406                user_where_clauses_with_world_and_state,
407            );
408
409            let contiguous_impl = contiguous_query_data_impl(
410                &path,
411                &read_only_struct_name,
412                &read_only_contiguous_item_struct_name,
413                &read_only_field_types,
414                &user_impl_generics,
415                &user_ty_generics,
416                &user_ty_generics_with_world_and_state,
417                &field_members,
418                &field_aliases,
419                user_where_clauses,
420            );
421
422            (contiguous_item_struct, contiguous_impl)
423        } else {
424            (quote! {}, quote! {})
425        };
426
427    let data_impl = {
428        let read_only_data_impl = if attributes.is_mutable {
429            quote! {
430                // SAFETY: we assert fields are readonly below
431                unsafe impl #user_impl_generics #path::query::QueryData
432                for #read_only_struct_name #user_ty_generics #user_where_clauses {
433                    const IS_READ_ONLY: bool = true;
434                    const IS_ARCHETYPAL: bool = true #(&& <#read_only_field_types as #path::query::QueryData>::IS_ARCHETYPAL)*;
435                    type ReadOnly = #read_only_struct_name #user_ty_generics;
436                    type Item<'__w, '__s> = #read_only_item_struct_name #user_ty_generics_with_world_and_state;
437
438                    fn shrink<'__wlong: '__wshort, '__wshort, '__s>(
439                        item: Self::Item<'__wlong, '__s>
440                    ) -> Self::Item<'__wshort, '__s> {
441                        #read_only_item_struct_name {
442                            #(
443                                #field_members: <#read_only_field_types>::shrink(item.#field_members),
444                            )*
445                        }
446                    }
447
448                    fn provide_extra_access(
449                        state: &mut Self::State,
450                        access: &mut #path::query::Access,
451                        available_access: &#path::query::Access,
452                    ) {
453                        #(<#field_types>::provide_extra_access(&mut state.#field_aliases, access, available_access);)*
454                    }
455
456                    /// SAFETY: we call `fetch` for each member that implements `Fetch`.
457                    #[inline(always)]
458                    unsafe fn fetch<'__w, '__s>(
459                        _state: &'__s Self::State,
460                        _fetch: &mut <Self as #path::query::WorldQuery>::Fetch<'__w>,
461                        _entity: #path::entity::Entity,
462                        _table_row: #path::storage::TableRow,
463                    ) -> #FQOption<Self::Item<'__w, '__s>> {
464                        #FQOption::Some(Self::Item {
465                            #(#field_members: <#read_only_field_types>::fetch(&_state.#field_aliases, &mut _fetch.#field_aliases, _entity, _table_row)?,)*
466                        })
467                    }
468
469                    fn iter_access(
470                        _state: &Self::State,
471                    ) -> impl #FQIterator<Item = #path::query::EcsAccessType<'_>> {
472                        ::core::iter::empty() #(.chain(<#field_types>::iter_access(&_state.#field_aliases)))*
473                    }
474                }
475
476                // SAFETY: Access is read-only
477                unsafe impl #user_impl_generics #path::query::IterQueryData
478                for #read_only_struct_name #user_ty_generics #user_where_clauses {}
479
480                // SAFETY: All fields only access the current entity
481                unsafe impl #user_impl_generics #path::query::SingleEntityQueryData
482                for #read_only_struct_name #user_ty_generics
483                // Make these HRTBs with an unused lifetime parameter to allow trivial constraints
484                // See https://github.com/rust-lang/rust/issues/48214
485                where #(for<'__a> #field_types: #path::query::QueryData<ReadOnly: #path::query::SingleEntityQueryData>,)* #user_where_predicates {}
486
487                impl #user_impl_generics #path::query::ReleaseStateQueryData
488                for #read_only_struct_name #user_ty_generics
489                // Make these HRTBs with an unused lifetime parameter to allow trivial constraints
490                // See https://github.com/rust-lang/rust/issues/48214
491                where #(for<'__a> #field_types: #path::query::QueryData<ReadOnly: #path::query::ReleaseStateQueryData>,)* #user_where_predicates {
492                    fn release_state<'__w>(_item: Self::Item<'__w, '_>) -> Self::Item<'__w, 'static> {
493                        Self::Item {
494                            #(#field_members: <#read_only_field_types>::release_state(_item.#field_members),)*
495                        }
496                    }
497                }
498
499                impl #user_impl_generics #path::query::ArchetypeQueryData
500                for #read_only_struct_name #user_ty_generics
501                // Make these HRTBs with an unused lifetime parameter to allow trivial constraints
502                // See https://github.com/rust-lang/rust/issues/48214
503                where #(for<'__a> #field_types: #path::query::ArchetypeQueryData,)* #user_where_predicates {}
504            }
505        } else {
506            quote! {}
507        };
508
509        let is_read_only = !attributes.is_mutable;
510
511        quote! {
512            // SAFETY: we assert fields are readonly below
513            unsafe impl #user_impl_generics #path::query::QueryData
514            for #struct_name #user_ty_generics #user_where_clauses {
515                const IS_READ_ONLY: bool = #is_read_only;
516                const IS_ARCHETYPAL: bool = true #(&& <#field_types as #path::query::QueryData>::IS_ARCHETYPAL)*;
517                type ReadOnly = #read_only_struct_name #user_ty_generics;
518                type Item<'__w, '__s> = #item_struct_name #user_ty_generics_with_world_and_state;
519
520                fn shrink<'__wlong: '__wshort, '__wshort, '__s>(
521                    item: Self::Item<'__wlong, '__s>
522                ) -> Self::Item<'__wshort, '__s> {
523                    #item_struct_name {
524                        #(
525                            #field_members: <#field_types>::shrink(item.#field_members),
526                        )*
527                    }
528                }
529
530                fn provide_extra_access(
531                    state: &mut Self::State,
532                    access: &mut #path::query::Access,
533                    available_access: &#path::query::Access,
534                ) {
535                    #(<#field_types>::provide_extra_access(&mut state.#field_aliases, access, available_access);)*
536                }
537
538                /// SAFETY: we call `fetch` for each member that implements `Fetch`.
539                #[inline(always)]
540                unsafe fn fetch<'__w, '__s>(
541                    _state: &'__s Self::State,
542                    _fetch: &mut <Self as #path::query::WorldQuery>::Fetch<'__w>,
543                    _entity: #path::entity::Entity,
544                    _table_row: #path::storage::TableRow,
545                ) -> #FQOption<Self::Item<'__w, '__s>> {
546                    #FQOption::Some(Self::Item {
547                        #(#field_members: <#field_types>::fetch(&_state.#field_aliases, &mut _fetch.#field_aliases, _entity, _table_row)?,)*
548                    })
549                }
550
551                fn iter_access(
552                    _state: &Self::State,
553                ) -> impl #FQIterator<Item = #path::query::EcsAccessType<'_>> {
554                    ::core::iter::empty() #(.chain(<#field_types>::iter_access(&_state.#field_aliases)))*
555                }
556            }
557
558            // SAFETY: All fields are iterable
559            unsafe impl #user_impl_generics #path::query::IterQueryData
560            for #struct_name #user_ty_generics
561            // Make these HRTBs with an unused lifetime parameter to allow trivial constraints
562            // See https://github.com/rust-lang/rust/issues/48214
563            where #(for<'__a> #field_types: #path::query::IterQueryData,)* #user_where_predicates {}
564
565            // SAFETY: All fields only access the current entity
566            unsafe impl #user_impl_generics #path::query::SingleEntityQueryData
567            for #struct_name #user_ty_generics
568            // Make these HRTBs with an unused lifetime parameter to allow trivial constraints
569            // See https://github.com/rust-lang/rust/issues/48214
570            where #(for<'__a> #field_types: #path::query::SingleEntityQueryData,)* #user_where_predicates {}
571
572            impl #user_impl_generics #path::query::ReleaseStateQueryData
573            for #struct_name #user_ty_generics
