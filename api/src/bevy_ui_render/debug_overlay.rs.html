35/// Configuration for the UI debug overlay
36///
37/// Can be added as a `Component` to individual UI node entities.
38/// This overwrites the default [`GlobalUiDebugOptions`] resource.
39#[derive(Component, Reflect, Copy, Clone)]
40#[reflect(Component)]
41pub struct UiDebugOptions {
42    /// Set to true to enable the UI debug overlay
43    pub enabled: bool,
44    /// Show outlines for the border boxes of UI nodes
45    pub outline_border_box: bool,
46    /// Show outlines for the padding boxes of UI nodes
47    pub outline_padding_box: bool,
48    /// Show outlines for the content boxes of UI nodes
49    pub outline_content_box: bool,
50    /// Show outlines for the scrollbar regions of UI nodes
51    pub outline_scrollbars: bool,
52    /// Width of the overlay's lines in logical pixels
53    pub line_width: f32,
54    /// Override Color for the overlay's lines
55    pub line_color_override: Option<LinearRgba>,
56    /// Show outlines for non-visible UI nodes
57    pub show_hidden: bool,
58    /// Show outlines for clipped sections of UI nodes
59    pub show_clipped: bool,
60    /// Draw outlines with sharp corners even if the UI nodes have border radii
61    pub ignore_border_radius: bool,
62}
63
64impl UiDebugOptions {
65    pub fn toggle(&mut self) {
66        self.enabled = !self.enabled;
67    }
68}

172pub fn extract_debug_overlay(
173    mut commands: Commands,
174    debug_options: Extract<Res<GlobalUiDebugOptions>>,
175    mut extracted_uinodes: ResMut<ExtractedUiNodes>,
176    uinode_query: Extract<
177        Query<(
178            Entity,
179            &ComputedNode,
180            &ComputedStackIndex,
181            &UiGlobalTransform,
182            &InheritedVisibility,
183            Option<&CalculatedClip>,
184            &ComputedUiTargetCamera,
185            Option<&UiDebugOptions>,
186        )>,
187    >,
188    ui_stack: Extract<Res<UiStack>>,
189    camera_map: Extract<UiCameraMap>,
190) {
191    let mut camera_mapper = camera_map.get_mapper();
192
193    for (entity, uinode, stack_index, transform, visibility, maybe_clip, computed_target, debug) in
194        &uinode_query
195    {
196        let debug_options = debug.copied().unwrap_or((*debug_options.as_ref()).into());
197        if !debug_options.enabled {
198            continue;
199        }
200        if !debug_options.show_hidden && !visibility.get() {
201            continue;
202        }
203
204        let Some(extracted_camera_entity) = camera_mapper.map(computed_target) else {
205            continue;
206        };
207
208        let color = debug_options
209            .line_color_override
210            .unwrap_or_else(|| Hsla::sequential_dispersed(entity.index_u32()).into());
211        let z_order = (ui_stack.uinodes.len() as u32 + stack_index.0) as f32;
212        let border = BorderRect::all(debug_options.line_width / uinode.inverse_scale_factor());
213        let transform = transform.affine();
214
215        let mut push_outline = |rect: Rect, radius: ResolvedBorderRadius| {
216            if rect.is_empty() {
217                return;
218            }
219
220            extracted_uinodes.uinodes.push(ExtractedUiNode {
221                render_entity: commands.spawn(TemporaryRenderEntity).id(),
222                // Keep all overlays above UI, and nudge each type slightly in Z so ordering is stable.
223                z_order,
224                clip: maybe_clip
225                    .filter(|_| !debug_options.show_clipped)
226                    .map(|clip| clip.clip),
227                image: AssetId::default(),
228                extracted_camera_entity,
229                transform: transform * Affine2::from_translation(rect.center()),
230                item: ExtractedUiItem::Node {
231                    color,
232                    rect: Rect {
233                        min: Vec2::ZERO,
234                        max: rect.size(),
235                    },
236                    atlas_scaling: None,
237                    flip_x: false,
238                    flip_y: false,
239                    border,
240                    border_radius: radius,
241                    node_type: NodeType::Border(shader_flags::BORDER_ALL),
242                },
243                main_entity: entity.into(),
244            });
245        };
246
247        let border_box = Rect::from_center_size(Vec2::ZERO, uinode.size);
248
249        if debug_options.outline_border_box {
250            push_outline(border_box, uinode.border_radius());
251        }
252
253        if debug_options.outline_padding_box {
254            let mut padding_box = border_box;
255            padding_box.min += uinode.border.min_inset;
256            padding_box.max -= uinode.border.max_inset;
257            push_outline(padding_box, uinode.inner_radius());
258        }
259
260        if debug_options.outline_content_box {
261            let mut content_box = border_box;
262            let content_inset = uinode.content_inset();
263            content_box.min += content_inset.min_inset;
264            content_box.max -= content_inset.max_inset;
265            push_outline(content_box, ResolvedBorderRadius::ZERO);
266        }
267
268        if debug_options.outline_scrollbars {
269            if let Some((gutter, [thumb_min, thumb_max])) = uinode.horizontal_scrollbar() {
270                push_outline(gutter, ResolvedBorderRadius::ZERO);
271                push_outline(
272                    Rect {
273                        min: Vec2::new(thumb_min, gutter.min.y),
274                        max: Vec2::new(thumb_max, gutter.max.y),
275                    },
276                    ResolvedBorderRadius::ZERO,
277                );
278            }
279            if let Some((gutter, [thumb_min, thumb_max])) = uinode.vertical_scrollbar() {
280                push_outline(gutter, ResolvedBorderRadius::ZERO);
281                push_outline(
282                    Rect {
283                        min: Vec2::new(gutter.min.x, thumb_min),
284                        max: Vec2::new(gutter.max.x, thumb_max),
285                    },
286                    ResolvedBorderRadius::ZERO,
287                );
288            }
289        }
290    }
