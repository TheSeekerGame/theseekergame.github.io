39impl Plugin for LineGizmo2dPlugin {
40    fn build(&self, app: &mut App) {
41        let Some(render_app) = app.get_sub_app_mut(RenderApp) else {
42            return;
43        };
44
45        render_app
46            .add_render_command::<Transparent2d, DrawLineGizmo2d>()
47            .add_render_command::<Transparent2d, DrawLineGizmo2dStrip>()
48            .add_render_command::<Transparent2d, DrawLineJointGizmo2d>()
49            .init_gpu_resource::<SpecializedRenderPipelines<LineGizmoPipeline>>()
50            .init_gpu_resource::<SpecializedRenderPipelines<LineJointGizmoPipeline>>()
51            .configure_sets(
52                Render,
53                GizmoRenderSystems::QueueLineGizmos2d
54                    .in_set(RenderSystems::Queue)
55                    .ambiguous_with(bevy_sprite_render::queue_sprites)
56                    .ambiguous_with(
57                        bevy_sprite_render::queue_material2d_meshes::<
58                            bevy_sprite_render::ColorMaterial,
59                        >,
60                    ),
61            )
62            .add_systems(
63                RenderStartup,
64                init_line_gizmo_pipelines
65                    .after(init_line_gizmo_uniform_bind_group_layout)
66                    .after(init_mesh_2d_pipeline),
67            )
68            .add_systems(
69                Render,
70                queue_line_and_joint_gizmos_2d
71                    .in_set(GizmoRenderSystems::QueueLineGizmos2d)
72                    .after(prepare_assets::<GpuLineGizmo>),
73            );
74    }
75}

109impl SpecializedRenderPipeline for LineGizmoPipeline {
110    type Key = LineGizmoPipelineKey;
111
112    fn specialize(&self, key: Self::Key) -> RenderPipelineDescriptor {
113        let format = key.mesh_key.target_format();
114
115        let shader_defs = vec![
116            #[cfg(all(feature = "webgl", target_arch = "wasm32", not(feature = "webgpu")))]
117            "SIXTEEN_BYTE_ALIGNMENT".into(),
118        ];
119
120        let layout = vec![
121            self.mesh_pipeline.view_layout.clone(),
122            self.uniform_layout.clone(),
123        ];
124
125        let fragment_entry_point = match key.line_style {
126            GizmoLineStyle::Solid => "fragment_solid",
127            GizmoLineStyle::Dotted => "fragment_dotted",
128            GizmoLineStyle::Dashed { .. } => "fragment_dashed",
129            _ => unimplemented!(),
130        };
131
132        RenderPipelineDescriptor {
133            vertex: VertexState {
134                shader: self.shader.clone(),
135                shader_defs: shader_defs.clone(),
136                buffers: line_gizmo_vertex_buffer_layouts(key.strip),
137                ..default()
138            },
139            fragment: Some(FragmentState {
140                shader: self.shader.clone(),
141                shader_defs,
142                entry_point: Some(fragment_entry_point.into()),
143                targets: vec![Some(ColorTargetState {
144                    format,
145                    blend: Some(BlendState::ALPHA_BLENDING),
146                    write_mask: ColorWrites::ALL,
147                })],
148            }),
149            layout,
150            depth_stencil: Some(DepthStencilState {
151                format: CORE_2D_DEPTH_FORMAT,
152                depth_write_enabled: Some(false),
153                depth_compare: Some(CompareFunction::Always),
154                stencil: StencilState {
155                    front: StencilFaceState::IGNORE,
156                    back: StencilFaceState::IGNORE,
157                    read_mask: 0,
158                    write_mask: 0,
159                },
160                bias: DepthBiasState {
161                    constant: 0,
162                    slope_scale: 0.0,
163                    clamp: 0.0,
164                },
165            }),
166            multisample: MultisampleState {
167                count: key.mesh_key.msaa_samples(),
168                mask: !0,
169                alpha_to_coverage_enabled: false,
170            },
171            label: Some("LineGizmo Pipeline 2D".into()),
172            ..default()
173        }
174    }
175}

185struct LineJointGizmoPipelineKey {

188}

190impl SpecializedRenderPipeline for LineJointGizmoPipeline {
191    type Key = LineJointGizmoPipelineKey;
192
193    fn specialize(&self, key: Self::Key) -> RenderPipelineDescriptor {
194        let format = key.mesh_key.target_format();
195
196        let shader_defs = vec![
197            #[cfg(all(feature = "webgl", target_arch = "wasm32", not(feature = "webgpu")))]
198            "SIXTEEN_BYTE_ALIGNMENT".into(),
199        ];
200
201        let layout = vec![
202            self.mesh_pipeline.view_layout.clone(),
203            self.uniform_layout.clone(),
204        ];
205
206        if key.joints == GizmoLineJoint::None {
207            error!("There is no entry point for line joints with GizmoLineJoints::None. Please consider aborting the drawing process before reaching this stage.");
208        };
209
210        let entry_point = match key.joints {
211            GizmoLineJoint::Miter => "vertex_miter",
212            GizmoLineJoint::Round(_) => "vertex_round",
213            GizmoLineJoint::None | GizmoLineJoint::Bevel => "vertex_bevel",
214        };
215
216        RenderPipelineDescriptor {
217            vertex: VertexState {
218                shader: self.shader.clone(),
219                entry_point: Some(entry_point.into()),
220                shader_defs: shader_defs.clone(),
221                buffers: line_joint_gizmo_vertex_buffer_layouts(),
222            },
223            fragment: Some(FragmentState {
224                shader: self.shader.clone(),
225                shader_defs,
226                targets: vec![Some(ColorTargetState {
227                    format,
228                    blend: Some(BlendState::ALPHA_BLENDING),
229                    write_mask: ColorWrites::ALL,
230                })],
231                ..default()
232            }),
233            layout,
234            primitive: PrimitiveState::default(),
235            depth_stencil: Some(DepthStencilState {
236                format: CORE_2D_DEPTH_FORMAT,
237                depth_write_enabled: Some(false),
238                depth_compare: Some(CompareFunction::Always),
239                stencil: StencilState {
240                    front: StencilFaceState::IGNORE,
241                    back: StencilFaceState::IGNORE,
242                    read_mask: 0,
243                    write_mask: 0,
244                },
245                bias: DepthBiasState {
246                    constant: 0,
247                    slope_scale: 0.0,
248                    clamp: 0.0,
249                },
250            }),
251            multisample: MultisampleState {
252                count: key.mesh_key.msaa_samples(),
253                mask: !0,
254                alpha_to_coverage_enabled: false,
255            },
256            label: Some("LineJointGizmo Pipeline 2D".into()),
257            ..default()
258        }
259    }

281fn queue_line_and_joint_gizmos_2d(
282    draw_functions: Res<DrawFunctions<Transparent2d>>,
283    line_gizmo_pipeline: Res<LineGizmoPipeline>,
284    line_joint_gizmo_pipeline: Res<LineJointGizmoPipeline>,
285    mut line_gizmo_pipelines: ResMut<SpecializedRenderPipelines<LineGizmoPipeline>>,
286    mut line_joint_gizmo_pipelines: ResMut<SpecializedRenderPipelines<LineJointGizmoPipeline>>,
287    pipeline_cache: Res<PipelineCache>,
288    line_gizmos: Query<(Entity, &GizmoMeshConfig)>,
289    line_gizmo_assets: Res<RenderAssets<GpuLineGizmo>>,
290    line_gizmo_entities: Res<LineGizmoEntities>,
291    mut transparent_render_phases: ResMut<ViewSortedRenderPhases<Transparent2d>>,
292    mut views: Query<(&ExtractedView, &Msaa, Option<&RenderLayers>)>,
293) {
294    let draw_function = draw_functions.read().get_id::<DrawLineGizmo2d>().unwrap();
295    let draw_line_function_strip = draw_functions
296        .read()
297        .get_id::<DrawLineGizmo2dStrip>()
298        .unwrap();
299    let draw_line_joint_function = draw_functions
300        .read()
301        .get_id::<DrawLineJointGizmo2d>()
302        .unwrap();
303
304    for (view, msaa, render_layers) in &mut views {
305        let Some(transparent_phase) = transparent_render_phases.get_mut(&view.retained_view_entity)
306        else {
307            continue;
308        };
309
310        let mesh_key = Mesh2dPipelineKey::from_msaa_samples(msaa.samples())
311            | Mesh2dPipelineKey::from_target_format(view.target_format);
312
313        let render_layers = render_layers.unwrap_or_default();
314        for (entity, config) in &line_gizmos {
315            if !config.render_layers.intersects(render_layers) {
316                continue;
317            }
318
319            let Some(line_gizmo) = line_gizmo_assets.get(&config.handle) else {
320                continue;
321            };
322
323            // Draw lines
324            if line_gizmo.list_vertex_count > 0 {
325                let pipeline = line_gizmo_pipelines.specialize(
326                    &pipeline_cache,
327                    &line_gizmo_pipeline,
328                    LineGizmoPipelineKey {
329                        mesh_key,
330                        strip: false,
331                        line_style: config.line_style,
332                    },
333                );
334                transparent_phase.add_transient(Transparent2d {
335                    entity: (entity, line_gizmo_entities.line_gizmo_renderer),
336                    draw_function,
337                    pipeline,
338                    sort_key: FloatOrd(f32::INFINITY),
339                    batch_range: 0..1,
340                    extra_index: PhaseItemExtraIndex::None,
341                    extracted_index: usize::MAX,
342                    indexed: false,
343                });
344            }
345
346            if line_gizmo.strip_vertex_count >= 2 {
347                let pipeline = line_gizmo_pipelines.specialize(
348                    &pipeline_cache,
349                    &line_gizmo_pipeline,
350                    LineGizmoPipelineKey {
351                        mesh_key,
352                        strip: true,
353                        line_style: config.line_style,
354                    },
355                );
356                transparent_phase.add_transient(Transparent2d {
357                    entity: (entity, line_gizmo_entities.line_strip_gizmo_renderer),
358                    draw_function: draw_line_function_strip,
359                    pipeline,
360                    sort_key: FloatOrd(f32::INFINITY),
361                    batch_range: 0..1,
362                    extra_index: PhaseItemExtraIndex::None,
363                    extracted_index: usize::MAX,
364                    indexed: false,
365                });
366            }
367
368            // Draw line joints
369            if line_gizmo.strip_vertex_count < 3 || config.line_joints == GizmoLineJoint::None {
370                continue;
371            }
372
373            let pipeline = line_joint_gizmo_pipelines.specialize(
374                &pipeline_cache,
375                &line_joint_gizmo_pipeline,
376                LineJointGizmoPipelineKey {
377                    mesh_key,
378                    joints: config.line_joints,
379                },
380            );
381            transparent_phase.add_transient(Transparent2d {
382                entity: (entity, line_gizmo_entities.line_joint_gizmo_renderer),
383                draw_function: draw_line_joint_function,
384                pipeline,
385                sort_key: FloatOrd(f32::INFINITY),
386                batch_range: 0..1,
387                extra_index: PhaseItemExtraIndex::None,
388                extracted_index: usize::MAX,
389                indexed: false,
390            });
391        }
392    }
