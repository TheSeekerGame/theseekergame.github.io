4use core::mem::Discriminant;
5use hashbrown::HashMap;
6
7type StyleProperty<Brush> = crate::StyleProperty<'static, Brush>;
8
9/// A long-lived collection of [`StyleProperties`](super::StyleProperty), containing at
10/// most one of each property.
11///
12/// This is used by [`PlainEditor`](crate::editing::PlainEditor) to provide a reasonably ergonomic
13/// mutable API for styles applied to all text managed by it.
14/// This can be accessed using [`PlainEditor::edit_styles`](crate::editing::PlainEditor::edit_styles).
15///
16/// These styles do not have a corresponding range, and are generally unsuited for rich text.
17#[derive(Clone, Debug)]
18pub struct StyleSet<Brush: crate::Brush>(
19    HashMap<Discriminant<StyleProperty<Brush>>, StyleProperty<Brush>>,
20);
21
22impl<Brush: crate::Brush> StyleSet<Brush> {
23    /// Create a new collection of styles.
24    ///
25    /// The font size will be `font_size`, and can be overwritten at runtime by
26    /// [inserting](Self::insert) a new [`FontSize`](crate::StyleProperty::FontSize).
27    pub fn new(font_size: f32) -> Self {
28        let mut this = Self(HashMap::default());
29        this.insert(StyleProperty::FontSize(font_size));
30        this
31    }
32
33    /// Add `style` to this collection, returning any overwritten value.
34    ///
35    /// Note: Adding a [`font-family`](crate::StyleProperty::FontFamily) to this collection is not
36    /// additive, and instead overwrites any previously added font stack.
37    pub fn insert(&mut self, style: StyleProperty<Brush>) -> Option<StyleProperty<Brush>> {
38        let discriminant = core::mem::discriminant(&style);
39        self.0.insert(discriminant, style)
40    }
41
42    /// [Retain](alloc::vec::Vec::retain) only the styles for which `f` returns true.
43    ///
44    /// Styles which are removed return to their default values.
45    ///
46    /// Removing the [font size](crate::StyleProperty::FontSize) is not recommended, as an unspecified
47    /// fallback font size will be used.
48    pub fn retain(&mut self, mut f: impl FnMut(&StyleProperty<Brush>) -> bool) {
49        self.0.retain(|_, v| f(v));
50    }
51
52    /// Remove the style with the discriminant `property`.
53    ///
54    /// Styles which are removed return to their default values.
55    ///
56    /// To get the discriminant requires constructing a valid `StyleProperty` for the
57    /// the desired property and passing it to [`core::mem::discriminant`].
58    /// Getting this discriminant is usually possible in a `const` context.
59    ///
60    /// Removing the [font size](crate::StyleProperty::FontSize) is not recommended, as an unspecified
61    /// fallback font size will be used.
62    pub fn remove(
63        &mut self,
64        property: Discriminant<StyleProperty<Brush>>,
65    ) -> Option<StyleProperty<Brush>> {
66        self.0.remove(&property)
67    }
68
69    /// Read the raw underlying storage of this.
70    ///
71    /// Write access is not provided due to the invariant that keys
72    /// are the discriminant of their corresponding value.
73    pub fn inner(&self) -> &HashMap<Discriminant<StyleProperty<Brush>>, StyleProperty<Brush>> {
74        &self.0
75    }
76}
