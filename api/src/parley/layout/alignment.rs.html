11/// Alignment of a layout.
12#[derive(Copy, Clone, Default, PartialEq, Eq, Debug)]
13#[repr(u8)]
14pub enum Alignment {
15    /// This is [`Alignment::Left`] for LTR text and [`Alignment::Right`] for RTL text.
16    #[default]
17    Start,
18    /// This is [`Alignment::Right`] for LTR text and [`Alignment::Left`] for RTL text.
19    End,
20    /// Align content to the left edge.
21    ///
22    /// For alignment that should be aware of text direction, use [`Alignment::Start`] or
23    /// [`Alignment::End`] instead.
24    Left,
25    /// Align each line centered within the container.
26    Center,
27    /// Align content to the right edge.
28    ///
29    /// For alignment that should be aware of text direction, use [`Alignment::Start`] or
30    /// [`Alignment::End`] instead.
31    Right,
32    /// Justify each line by spacing out content, except for the last line.
33    Justify,
34}
35
36/// Additional options to fine tune alignment
37#[derive(Debug, Clone, Copy)]
38pub struct AlignmentOptions {
39    /// If set to `true`, "end" and "center" alignment will apply even if the line contents are
40    /// wider than the alignment width. If it is set to `false`, all overflowing lines will be
41    /// [`Alignment::Start`] aligned.
42    pub align_when_overflowing: bool,
43}
44
45#[expect(
46    clippy::derivable_impls,
47    reason = "Make default values explicit rather than relying on the implicit default value of bool"
48)]
49impl Default for AlignmentOptions {
50    fn default() -> Self {
51        Self {
52            align_when_overflowing: false,
53        }
54    }
55}
56
57/// Align the layout.
58///
59/// If [`Alignment::Justify`] is requested, clusters' [`ClusterData::advance`] will be adjusted.
60/// Prior to re-line-breaking or re-aligning, [`unjustify`] has to be called.
61pub(crate) fn align<B: Brush>(
62    layout: &mut LayoutData<B>,
63    alignment: Alignment,
64    options: AlignmentOptions,
65) {
66    #[cfg(feature = "accesskit")]
67    {
68        layout.alignment = Some(alignment);
69    }
70    layout.is_aligned_justified = alignment == Alignment::Justify;
71
72    align_impl::<_, false>(layout, alignment, options);
73}
74
75/// Removes previous justification applied to clusters.
76///
77/// This is part of resetting state in preparation for re-line-breaking or re-aligning the same
78/// layout.
79pub(crate) fn unjustify<B: Brush>(layout: &mut LayoutData<B>) {
80    if layout.is_aligned_justified {
81        align_impl::<_, true>(layout, Alignment::Justify, AlignmentOptions::default());
82        layout.is_aligned_justified = false;
83    }
84}
85
86/// The actual alignment implementation.
87///
88/// This is const-generic over `UNDO_JUSTIFICATION`: justified alignment adjusts clusters'
89/// [`ClusterData::advance`], and this mutation has to be undone for re-line-breaking or
90/// re-aligning. `UNDO_JUSTIFICATION` indicates whether the adjustment has to be applied, or
91/// undone.
92///
93/// Writing a separate function for undoing justification would be faster, but not by much, and
94/// doing it this way we are sure the calculations performed are equivalent.
95fn align_impl<B: Brush, const UNDO_JUSTIFICATION: bool>(
96    layout: &mut LayoutData<B>,
97    alignment: Alignment,
98    options: AlignmentOptions,
99) {
100    // Whether the text base direction is right-to-left.
101    let is_rtl = layout.base_level & 1 == 1;
102
103    // Apply alignment to line items
104    for line in &mut layout.lines {
105        let indent = line.indent;
106
107        if is_rtl {
108            // In RTL text, trailing whitespace is on the left. As we hang that whitespace, offset
109            // the line to the left. Note: indent is not subtracted here because `free_space` below
110            // already accounts for it.
111            line.metrics.offset = -line.metrics.trailing_whitespace;
112        } else {
113            line.metrics.offset = indent;
114        }
115
116        // Compute free space.
117        let line_width = line.metrics.inline_max_coord - line.metrics.inline_min_coord;
118        let free_space =
119            line_width - indent - line.metrics.advance + line.metrics.trailing_whitespace;
120
121        if !options.align_when_overflowing && free_space <= 0.0 {
122            if is_rtl {
123                // In RTL text, right-align on overflow.
124                line.metrics.offset += free_space;
125            }
126            continue;
127        }
128
129        match (alignment, is_rtl) {
130            (Alignment::Left, _) | (Alignment::Start, false) | (Alignment::End, true) => {
131                // Do nothing
132            }
133            (Alignment::Right, _) | (Alignment::Start, true) | (Alignment::End, false) => {
134                line.metrics.offset += free_space;
135            }
136            (Alignment::Center, _) => {
137                line.metrics.offset += free_space * 0.5;
138            }
139            (Alignment::Justify, _) => {
140                // Justified alignment doesn't have any effect if free_space is negative or zero
141                if free_space <= 0.0 {
142                    continue;
143                }
144
145                // Justified alignment doesn't apply to the last line of a paragraph
146                // (`BreakReason::None`), (`BreakReason::Explicit`) or if there are no whitespace
147                // gaps to adjust. In that case, start-align, i.e., left-align for LTR text and
148                // right-align for RTL text.
149                if matches!(line.break_reason, BreakReason::None | BreakReason::Explicit)
150                    || line.num_spaces == 0
151                {
152                    if is_rtl {
153                        line.metrics.offset += free_space;
154                    }
155                    continue;
156                }
157
158                let adjustment =
159                    free_space / line.num_spaces as f32 * if UNDO_JUSTIFICATION { -1. } else { 1. };
160                let mut applied = 0;
161                // Iterate over text runs in the line and clusters in the text run
162                //   - Iterate forwards for even bidi levels (which represent LTR runs)
163                //   - Iterate backwards for odd bidi levels (which represent RTL runs)
164                let line_items: &mut dyn Iterator<Item = &LineItemData> = if is_rtl {
165                    &mut layout.line_items[line.item_range.clone()].iter().rev()
166                } else {
167                    &mut layout.line_items[line.item_range.clone()].iter()
168                };
169                line_items
170                    .filter(|item| item.is_text_run())
171                    .for_each(|line_item| {
172                        let clusters = &mut layout.clusters[line_item.cluster_range.clone()];
173                        let line_item_is_rtl = line_item.bidi_level & 1 != 0;
174                        let clusters: &mut dyn Iterator<Item = &mut ClusterData> =
175                            if line_item_is_rtl {
176                                &mut clusters.iter_mut().rev()
177                            } else {
178                                &mut clusters.iter_mut()
179                            };
180                        clusters.for_each(|cluster| {
181                            if applied == line.num_spaces {
182                                return;
183                            }
184                            if cluster.info.whitespace().is_space_or_nbsp() {
185                                cluster.advance += adjustment;
186                                applied += 1;
187                            }
188                        });
189                    });
190            }
191        }
192    }
193}
