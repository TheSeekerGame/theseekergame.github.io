28pub(crate) struct ShapeContext {

30    shape_instance_cache: LruCache<cache::ShapeInstanceId, harfrust::ShaperInstance>,
31    shape_plan_cache: LruCache<cache::ShapePlanId, harfrust::ShapePlan>,
32    unicode_buffer: Option<harfrust::UnicodeBuffer>,

37impl Default for ShapeContext {
38    fn default() -> Self {
39        const MAX_ENTRIES: usize = 16;
40        Self {
41            shape_data_cache: LruCache::new(MAX_ENTRIES),
42            shape_instance_cache: LruCache::new(MAX_ENTRIES),
43            shape_plan_cache: LruCache::new(MAX_ENTRIES),
44            unicode_buffer: Some(harfrust::UnicodeBuffer::new()),
45            features: Vec::new(),
46            char_cluster: CharCluster::default(),
47        }
48    }
49}

54    script: Script,
55    level: u8,

63#[allow(clippy::too_many_arguments)]
64pub(crate) fn shape_text<'a, B: Brush>(
65    rcx: &'a ResolveContext,
66    mut fq: Query<'a>,
67    styles: &'a [ResolvedStyle<B>],
68    inline_boxes: &[InlineBox],
69    infos: &[(CharInfo, u16)],
70    levels: &[u8],
71    scx: &mut ShapeContext,
72    mut text: &str,
73    layout: &mut Layout<B>,
74    analysis_data_sources: &AnalysisDataSources,
75) {
76    // If we have both empty text and no inline boxes, shape with a fake space
77    // to generate metrics that can be used to size a cursor.
78    if text.is_empty() && inline_boxes.is_empty() {
79        text = " ";
80    }
81    // Do nothing if there is no text or styles (there should always be a default style)
82    if text.is_empty() || styles.is_empty() {
83        // Process any remaining inline boxes whose index is greater than the length of the text
84        for box_idx in 0..inline_boxes.len() {
85            // Push the box to the list of items
86            layout.data.push_inline_box(box_idx);
87        }
88        return;
89    }
90
91    // Setup mutable state for iteration
92    let mut style = &styles[0];
93    let mut item = Item {
94        style_index: 0,
95        size: style.font_size,
96        level: levels.first().copied().unwrap_or(0),
97        script: infos
98            .iter()
99            .map(|x| x.0.script)
100            .find(|&script| real_script(script))
101            .unwrap_or(Script::Latin),
102        locale: style.locale,
103        variations: style.font_variations,
104        features: style.font_features,
105        word_spacing: style.word_spacing,
106        letter_spacing: style.letter_spacing,
107    };
108
109    let mut char_range = 0..0;
110    let mut text_range = 0..0;
111
112    let mut inline_box_iter = inline_boxes.iter().enumerate();
113    let mut current_box = inline_box_iter.next();
114
115    // Iterate over characters in the text
116    for ((char_index, (byte_index, ch)), (info, style_index)) in
117        text.char_indices().enumerate().zip(infos)
118    {
119        let mut break_run = false;
120        let mut script = info.script;
121        if !real_script(script) {
122            script = item.script;
123        }
124        let level = levels.get(char_index).copied().unwrap_or(0);
125        if item.style_index != *style_index {
126            item.style_index = *style_index;
127            style = &styles[*style_index as usize];
128            if !nearly_eq(style.font_size, item.size)
129                || style.locale != item.locale
130                || style.font_variations != item.variations
131                || style.font_features != item.features
132                || !nearly_eq(style.letter_spacing, item.letter_spacing)
133                || !nearly_eq(style.word_spacing, item.word_spacing)
134            {
135                break_run = true;
136            }
137        }
138
139        if level != item.level || script != item.script {
140            break_run = true;
141        }
142
143        // Check if there is an inline box at this index
144        // Note:
145        //   - We loop because there may be multiple boxes at this index
146        //   - We do this *before* processing the text run because we need to know whether we should
147        //     break the run due to the presence of an inline box.
148        let mut deferred_boxes: Option<RangeInclusive<usize>> = None;
149        while let Some((box_idx, inline_box)) = current_box {
150            if inline_box.index == byte_index {
151                break_run = true;
152                if let Some(boxes) = &mut deferred_boxes {
153                    deferred_boxes = Some((*boxes.start())..=box_idx);
154                } else {
155                    deferred_boxes = Some(box_idx..=box_idx);
156                };
157                // Update the current box to the next box
158                current_box = inline_box_iter.next();
159            } else {
160                break;
161            }
162        }
163
164        if break_run && !text_range.is_empty() {
165            shape_item(
166                &mut fq,
167                rcx,
168                styles,
169                &item,
170                scx,
171                text,
172                &text_range,
173                &char_range,
174                infos,
175                layout,
176                analysis_data_sources,
177            );
178            item.size = style.font_size;
179            item.level = level;
180            item.script = script;
181            item.locale = style.locale;
182            item.variations = style.font_variations;
183            item.features = style.font_features;
184            item.word_spacing = style.word_spacing;
185            item.letter_spacing = style.letter_spacing;
186            text_range.start = text_range.end;
187            char_range.start = char_range.end;
188        }
189
190        if let Some(deferred_boxes) = deferred_boxes {
191            for box_idx in deferred_boxes {
192                layout.data.push_inline_box(box_idx);
193            }
194        }
195
196        text_range.end += ch.len_utf8();
197        char_range.end += 1;
198    }
199
200    if !text_range.is_empty() {
201        shape_item(
202            &mut fq,
203            rcx,
204            styles,
205            &item,
206            scx,
207            text,
208            &text_range,
209            &char_range,
210            infos,
211            layout,
212            analysis_data_sources,
213        );
214    }
215
216    // Process any remaining inline boxes whose index is greater than the length of the text
217    if let Some((box_idx, _inline_box)) = current_box {
218        layout.data.push_inline_box(box_idx);
219    }
220    for (box_idx, _inline_box) in inline_box_iter {
221        layout.data.push_inline_box(box_idx);
222    }
223}

225// Rebuilds the provided `char_cluster` in-place using the existing allocation
226// for the given grapheme `segment_text`, consuming items from `item_infos_iter`.
227fn fill_cluster_in_place(
228    segment_text: &str,
229    item_infos_iter: &mut core::slice::Iter<'_, (CharInfo, u16)>,
230    code_unit_offset_in_string: &mut usize,
231    char_cluster: &mut CharCluster,
232) {
233    // Reset cluster but keep allocation
234    char_cluster.clear();
235
236    let mut force_normalize = false;
237    let mut is_emoji_or_pictograph = false;
238    let mut map_len: u8 = 0;
239    let start = *code_unit_offset_in_string as u32;
240
241    for ((_, ch), (info, style_index)) in segment_text.char_indices().zip(item_infos_iter.by_ref())
242    {
243        force_normalize |= info.force_normalize();
244        // TODO - make emoji detection more complete, as per (except using composite Trie tables as
245        //  much as possible:
246        //  https://github.com/conor-93/parley/blob/4637d826732a1a82bbb3c904c7f47a16a21cceec/parley/src/shape/mod.rs#L221-L269
247        is_emoji_or_pictograph |= info.is_emoji_or_pictograph();
248        *code_unit_offset_in_string += ch.len_utf8();
249
250        let contributes_to_shaping = info.contributes_to_shaping();
251        if contributes_to_shaping {
252            map_len += 1;
253        }
254
255        char_cluster.chars.push(Char {
256            ch,
257            contributes_to_shaping,
258            glyph_id: 0,
259            style_index: *style_index,
260            is_control_character: info.is_control(),
261        });
262    }
263
264    // Finalize cluster metadata
265    let end = *code_unit_offset_in_string as u32;
266    char_cluster.is_emoji = is_emoji_or_pictograph;
267    char_cluster.map_len = map_len;
268    char_cluster.start = start;
269    char_cluster.end = end;
270    char_cluster.force_normalize = force_normalize;
271}

273fn shape_item<'a, B: Brush>(
274    fq: &mut Query<'a>,
275    rcx: &'a ResolveContext,
276    styles: &'a [ResolvedStyle<B>],
277    item: &Item,
278    scx: &mut ShapeContext,
279    text: &str,
280    text_range: &core::ops::Range<usize>,
281    char_range: &core::ops::Range<usize>,
282    infos: &[(CharInfo, u16)],
283    layout: &mut Layout<B>,
284    analysis_data_sources: &AnalysisDataSources,
285) {
286    let item_text = &text[text_range.clone()];
287    let item_infos = &infos[char_range.start..char_range.end]; // Only process current item
288    let first_style_index = item_infos[0].1;
289    let fb_script = convert::script_to_fontique(item.script, analysis_data_sources);
290    let mut font_selector =
291        FontSelector::new(fq, rcx, styles, first_style_index, fb_script, item.locale);
292
293    let grapheme_cluster_boundaries = analysis_data_sources
294        .grapheme_segmenter()
295        .segment_str(item_text);
296    let mut item_infos_iter = item_infos.iter();
297    let mut code_unit_offset_in_string = text_range.start;
298    let char_cluster = &mut scx.char_cluster;
299
300    // Build an iterator of boundaries and consume the first segment to seed the loop
301    let mut boundaries_iter = grapheme_cluster_boundaries.skip(1);
302    let mut last_boundary = 0_usize;
303    let Some(mut current_boundary) = boundaries_iter.next() else {
304        return; // No clusters
305    };
306
307    fill_cluster_in_place(
308        &item_text[last_boundary..current_boundary],
309        &mut item_infos_iter,
310        &mut code_unit_offset_in_string,
311        char_cluster,
312    );
313
314    let mut current_font = font_selector.select_font(char_cluster, analysis_data_sources);
315
316    // Main segmentation loop (based on swash shape_clusters) - only within current item
317    while let Some(font) = current_font.take() {
318        // Collect all clusters for this font segment
319        let cluster_range = char_cluster.range();
320        let segment_start_offset = cluster_range.start as usize - text_range.start;
321        let mut segment_end_offset = cluster_range.end as usize - text_range.start;
322
323        for next_boundary in boundaries_iter.by_ref() {
324            // Build next cluster in-place
325            last_boundary = current_boundary;
326            current_boundary = next_boundary;
327            fill_cluster_in_place(
328                &item_text[last_boundary..current_boundary],
329                &mut item_infos_iter,
330                &mut code_unit_offset_in_string,
331                char_cluster,
332            );
333
334            if let Some(next_font) = font_selector.select_font(char_cluster, analysis_data_sources)
335            {
336                if next_font != font {
337                    current_font = Some(next_font);
338                    break;
339                } else {
340                    // Same font - add to current segment
341                    segment_end_offset = char_cluster.range().end as usize - text_range.start;
342                }
343            } else {
344                // No font determined, continue to next cluster
345                continue;
346            }
347        }
348
349        // Shape this font segment with harfrust
350        let segment_text = &item_text[segment_start_offset..segment_end_offset];
351        // Shape the entire segment text including newlines
352        // The line breaking algorithm will handle newlines automatically
353
354        // TODO: How do we want to handle errors like this?
355        let font_ref =
356            harfrust::FontRef::from_index(font.font.blob.as_ref(), font.font.index).unwrap();
357
358        // Create harfrust shaper
359        let shaper_data = scx.shape_data_cache.entry(
360            cache::ShapeDataKey::new(font.font.blob.id(), font.font.index),
361            || harfrust::ShaperData::new(&font_ref),
362        );
363        let instance = scx.shape_instance_cache.entry(
364            cache::ShapeInstanceKey::new(
365                font.font.blob.id(),
366                font.font.index,
367                &font.font.synthesis,
368                rcx.variations(item.variations),
369            ),
370            || {
371                harfrust::ShaperInstance::from_variations(
372                    &font_ref,
373                    variations_iter(&font.font.synthesis, rcx.variations(item.variations)),
374                )
375            },
376        );
377
378        let direction = if item.level & 1 != 0 {
379            harfrust::Direction::RightToLeft
380        } else {
381            harfrust::Direction::LeftToRight
382        };
383        let hb_script = script_to_harfrust(fb_script);
384        let language = item
385            .locale
386            .as_ref()
387            .and_then(|lang| lang.language().parse::<harfrust::Language>().ok());
388        scx.features.clear();
389        for feature in rcx.features(item.features).unwrap_or(&[]) {
390            scx.features.push(harfrust::Feature::new(
391                harfrust::Tag::new(&feature.tag.to_bytes()),
392                feature.value as u32,
393                ..,
394            ));
395        }
396        let harf_shaper = shaper_data
397            .shaper(&font_ref)
398            .instance(Some(instance))
399            .point_size(Some(item.size))
400            .build();
401        let shaper_plan = scx.shape_plan_cache.entry(
402            cache::ShapePlanKey::new(
403                font.font.blob.id(),
404                font.font.index,
405                &font.font.synthesis,
406                direction,
407                hb_script,
408                language.clone(),
409                &scx.features,
410                rcx.variations(item.variations),
411            ),
412            || {
413                harfrust::ShapePlan::new(
414                    &harf_shaper,
415                    direction,
416                    Some(hb_script),
417                    language.as_ref(),
418                    &scx.features,
419                )
420            },
421        );
422
423        // Prepare harfrust buffer
424        let mut buffer = mem::take(&mut scx.unicode_buffer).unwrap();
425        buffer.clear();
426
427        // Use the entire segment text including newlines
428        buffer.reserve(segment_text.len());
429        for (i, ch) in segment_text.chars().enumerate() {
430            // Ensure that each cluster's index matches the index into `infos`. This is required
431            // for efficient cluster lookup within `data.rs`.
432            //
433            // In other words, instead of using `buffer.push_str`, which iterates `segment_text`
434            // with `char_indices`, push each char individually via `.chars` with a cluster index
435            // that matches its `infos` counterpart. This allows us to lookup `infos` via cluster
436            // index in `data.rs`.
437            buffer.add(ch, i as u32);
438        }
439
440        buffer.set_direction(direction);
441
442        buffer.set_script(hb_script);
443
444        if let Some(lang) = language {
445            buffer.set_language(lang);
446        }
447
448        let glyph_buffer = harf_shaper.shape_with_plan(shaper_plan, buffer, &scx.features);
449
450        // Extract relevant CharInfo slice for this segment
451        let char_start = char_range.start + item_text[..segment_start_offset].chars().count();
452        let segment_char_start = char_start - char_range.start;
453        let segment_char_count = segment_text.chars().count();
454        let segment_infos =
455            &item_infos[segment_char_start..(segment_char_start + segment_char_count)];
456
457        // Push harfrust-shaped run for the entire segment
458        layout.data.push_run(
459            FontData::new(font.font.blob.clone(), font.font.index),
460            item.size,
461            font.attrs,
462            font.font.synthesis,
463            &glyph_buffer,
464            item.level,
465            item.style_index,
466            item.word_spacing,
467            item.letter_spacing,
468            segment_text,
469            segment_infos,
470            (text_range.start + segment_start_offset)..(text_range.start + segment_end_offset),
471            harf_shaper.coords(),
472        );
473
474        // Replace buffer to reuse allocation in next iteration.
475        scx.unicode_buffer = Some(glyph_buffer.clear());
476    }
477}

479fn real_script(script: Script) -> bool {
480    script != Script::Common && script != Script::Unknown && script != Script::Inherited
481}

483fn variations_iter<'a>(
484    synthesis: &'a fontique::Synthesis,
485    item: Option<&'a [FontVariation]>,
486) -> impl Iterator<Item = harfrust::Variation> + 'a {
487    synthesis
488        .variation_settings()
489        .iter()
490        .map(|(tag, value)| harfrust::Variation {
491            tag: *tag,
492            value: *value,
493        })
494        .chain(
495            item.unwrap_or(&[])
496                .iter()
497                .map(|variation| harfrust::Variation {
498                    tag: harfrust::Tag::new(&variation.tag.to_bytes()),
499                    value: variation.value,
500                }),
501        )
502}

504struct FontSelector<'a, 'b, B: Brush> {
505    query: &'b mut Query<'a>,
506    fonts_id: Option<usize>,
507    rcx: &'a ResolveContext,
508    styles: &'a [ResolvedStyle<B>],
509    style_index: u16,
510    attrs: fontique::Attributes,
511    variations: &'a [FontVariation],
512    features: &'a [FontFeature],
513}
514
515impl<'a, 'b, B: Brush> FontSelector<'a, 'b, B> {
516    fn new(
517        query: &'b mut Query<'a>,
518        rcx: &'a ResolveContext,
519        styles: &'a [ResolvedStyle<B>],
520        style_index: u16,
521        fb_script: fontique::Script,
522        locale: Option<Language>,
523    ) -> Self {
524        let style = &styles[style_index as usize];
525        let fonts_id = style.font_family.id();
526        let fonts = rcx.stack(style.font_family).unwrap_or(&[]);
527        let attrs = fontique::Attributes {
528            width: style.font_width,
529            weight: style.font_weight,
530            style: style.font_style,
531        };
532        let variations = rcx.variations(style.font_variations).unwrap_or(&[]);
533        let features = rcx.features(style.font_features).unwrap_or(&[]);
534        query.set_families(fonts.iter().copied());
535
536        query.set_fallbacks(fontique::FallbackKey::new(fb_script, locale.as_ref()));
537        query.set_attributes(attrs);
538
539        Self {
540            query,
541            fonts_id: Some(fonts_id),
542            rcx,
543            styles,
544            style_index,
545            attrs,
546            variations,
547            features,
548        }
549    }
550
551    fn select_font(
552        &mut self,
553        cluster: &mut CharCluster,
554        analysis_data_sources: &AnalysisDataSources,
555    ) -> Option<SelectedFont> {
556        let style_index = cluster.style_index();
557        let is_emoji = cluster.is_emoji;
558        if style_index != self.style_index || is_emoji || self.fonts_id.is_none() {
559            self.style_index = style_index;
560            let style = &self.styles[style_index as usize];
561
562            let fonts_id = style.font_family.id();
563            let fonts = self.rcx.stack(style.font_family).unwrap_or(&[]);
564            let fonts = fonts.iter().copied().map(QueryFamily::Id);
565            if is_emoji {
566                use core::iter::once;
567                let emoji_family = QueryFamily::Generic(fontique::GenericFamily::Emoji);
568                self.query.set_families(fonts.chain(once(emoji_family)));
569                self.fonts_id = None;
570            } else if self.fonts_id != Some(fonts_id) {
571                self.query.set_families(fonts);
572                self.fonts_id = Some(fonts_id);
573            }
574
575            let attrs = fontique::Attributes {
576                width: style.font_width,
577                weight: style.font_weight,
578                style: style.font_style,
579            };
580            if self.attrs != attrs {
581                self.query.set_attributes(attrs);
582                self.attrs = attrs;
583            }
584            self.variations = self.rcx.variations(style.font_variations).unwrap_or(&[]);
585            self.features = self.rcx.features(style.font_features).unwrap_or(&[]);
586        }
587        let mut selected_font = None;
588        self.query.matches_with(|font| {
589            let Some(charmap) = font.charmap() else {
590                return fontique::QueryStatus::Continue;
591            };
592
593            let map_status = cluster.map(
594                |ch| {
595                    charmap
596                        .map(ch)
597                        .map(|g| {
598                            // HACK: in reality, we're only computing coverage, so
599                            // we only care about whether the font  has a mapping
600                            // for a particular glyph. Any non-zero value indicates
601                            // the existence of a glyph so we can simplify this
602                            // without a fallible conversion from u32 to u16.
603                            (g != 0) as u16
604                        })
605                        .unwrap_or_default()
606                },
607                analysis_data_sources,
608            );
609
610            match map_status {
611                Status::Complete => {
612                    selected_font = Some(SelectedFont {
613                        font: font.clone(),
614                        attrs: self.attrs,
615                    });
616                    fontique::QueryStatus::Stop
617                }
618                Status::Keep => {
619                    selected_font = Some(SelectedFont {
620                        font: font.clone(),
621                        attrs: self.attrs,
622                    });
623                    fontique::QueryStatus::Continue
624                }
625                Status::Discard => {
626                    if selected_font.is_none() {
627                        selected_font = Some(SelectedFont {
628                            font: font.clone(),
629                            attrs: self.attrs,
630                        });
631                    }
632                    fontique::QueryStatus::Continue
633                }
634            }
635        });
636        selected_font
637    }
638}
639
640struct SelectedFont {
641    font: QueryFont,
642    attrs: fontique::Attributes,
643}
644
645impl PartialEq for SelectedFont {
646    fn eq(&self, other: &Self) -> bool {
647        self.font.family == other.font.family && self.font.synthesis == other.font.synthesis
648    }
