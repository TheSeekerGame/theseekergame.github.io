14/// Defines a range within a text layout.
15#[derive(Copy, Clone, Default, Debug)]
16pub struct Selection {
17    anchor: Cursor,
18    focus: Cursor,
19    anchor_base: AnchorBase,
20    h_pos: Option<f32>,
21}
22
23#[derive(Copy, Clone, Default, Debug)]
24enum AnchorBase {
25    #[default]
26    Cluster,
27    Word(Cursor, Cursor),
28    Line(Cursor, Cursor),
29}
30
31impl Selection {
32    /// Creates a new selection from the given anchor and focus cursors.
33    pub fn new(anchor: Cursor, focus: Cursor) -> Self {
34        Self {
35            anchor,
36            focus,
37            anchor_base: AnchorBase::default(),
38            h_pos: None,
39        }
40    }

53    /// Creates a new selection bounding the word at the given coordinates.
54    pub fn word_from_point<B: Brush>(layout: &Layout<B>, x: f32, y: f32) -> Self {
55        if let Some((mut cluster, _)) = Cluster::from_point(layout, x, y) {
56            if !cluster.is_word_boundary() {
57                if let Some(prev) = cluster.previous_logical_word() {
58                    cluster = prev;
59                }
60            }
61            let anchor = Cursor::from_cluster(layout, cluster.clone(), !cluster.is_rtl());
62            let focus = anchor.next_logical_word(layout);
63            Self {
64                anchor,
65                focus,
66                anchor_base: AnchorBase::Word(anchor, focus),
67                h_pos: None,
68            }
69        } else {
70            Cursor::from_byte_index(layout, layout.data.text_len, Affinity::Upstream).into()
71        }
72    }

74    /// Creates a new selection bounding the line at the given coordinates.
75    pub fn line_from_point<B: Brush>(layout: &Layout<B>, x: f32, y: f32) -> Self {
76        let Self { anchor, focus, .. } = Self::from_point(layout, x, y)
77            .line_start(layout, false)
78            .line_end(layout, true);
79        Self {
80            anchor,
81            focus,
82            anchor_base: AnchorBase::Line(anchor, focus),
83            h_pos: None,
84        }
85    }

143    /// Returns a new selection that is guaranteed to be within the bounds of
144    /// the given layout.
145    #[must_use]
146    pub fn refresh<B: Brush>(&self, layout: &Layout<B>) -> Self {
147        let anchor = self.anchor.refresh(layout);
148        let focus = self.focus.refresh(layout);
149        let anchor_base = match self.anchor_base {
150            AnchorBase::Cluster => AnchorBase::Cluster,
151            AnchorBase::Word(start, end) => {
152                AnchorBase::Word(start.refresh(layout), end.refresh(layout))
153            }
154            AnchorBase::Line(start, end) => {
155                AnchorBase::Line(start.refresh(layout), end.refresh(layout))
156            }
157        };
158        let h_pos = self.h_pos;
159        Self {
160            anchor,
161            focus,
162            anchor_base,
163            h_pos,
164        }
165    }

167    /// Returns the underlying text range of the selection.
168    pub fn text_range(&self) -> Range<usize> {
169        let start = self.anchor.index().min(self.focus.index());
170        let end = self.focus.index().max(self.anchor.index());
171        start..end
172    }

174    /// Returns a new selection with the focus at the next cluster in visual
175    /// order.
176    ///
177    /// If `extend` is `true` then the current anchor will be retained,
178    /// otherwise the new selection will be collapsed.
179    #[must_use]
180    pub fn next_visual<B: Brush>(&self, layout: &Layout<B>, extend: bool) -> Self {
181        if !self.is_collapsed() && !extend {
182            let anchor_geom = self.anchor.geometry(layout, 0.0);
183            let focus_geom = self.focus.geometry(layout, 0.0);
184            let new_focus = if (anchor_geom.y0, anchor_geom.x0) > (focus_geom.y0, focus_geom.x0) {
185                self.anchor
186            } else {
187                self.focus
188            };
189            new_focus.into()
190        } else {
191            self.maybe_extend(self.focus.next_visual(layout), extend)
192        }
193    }

256    /// Returns a new selection with the focus moved the specified number of
257    /// lines.
258    ///
259    /// The sign of the `delta` parameter determines the direction to move with
260    /// negative values moving toward previous lines and positive ones moving
261    /// toward next lines.
262    ///
263    /// If `extend` is `true` then the current anchor will be retained,
264    /// otherwise the new selection will be collapsed.
265    #[must_use]
266    pub fn move_lines<B: Brush>(&self, layout: &Layout<B>, delta: isize, extend: bool) -> Self {
267        if delta == 0 {
268            return *self;
269        }
270        let line_limit = layout.len().saturating_sub(1);
271        let geometry = self.focus.geometry(layout, 0.0);
272        let line_index = layout
273            .line_for_offset(geometry.y0 as f32)
274            .map(|(ix, _)| ix)
275            .unwrap_or(line_limit);
276        let new_line_index = line_index.saturating_add_signed(delta);
277        if delta < 0 && line_index.checked_add_signed(delta).is_none() {
278            return self
279                .move_to_line(layout, 0, extend)
280                .line_start(layout, extend);
281        } else if delta > 0 && new_line_index > line_limit {
282            return self
283                .move_to_line(layout, line_limit, extend)
284                .line_end(layout, extend);
285        }
286        self.move_to_line(layout, new_line_index, extend)
287    }

289    #[must_use]
290    fn move_to_line<B: Brush>(&self, layout: &Layout<B>, line_index: usize, extend: bool) -> Self {
291        let Some(line) = layout.get(line_index) else {
292            return *self;
293        };
294        let h_pos = self
295            .h_pos
296            .unwrap_or_else(|| self.focus.geometry(layout, 0.0).x0 as f32);
297        let y = line.metrics().block_max_coord - line.metrics().ascent * 0.5;
298        let new_focus = Cursor::from_point(layout, h_pos, y);
299        let h_pos = Some(h_pos);
300        if extend {
301            Self {
302                anchor: self.anchor,
303                focus: new_focus,
304                h_pos,
305                ..Default::default()
306            }
307        } else {
308            Self {
309                anchor: new_focus,
310                focus: new_focus,
311                h_pos,
312                ..Default::default()
313            }
314        }
315    }

324        if let Some((_, line)) = self.focus.line(layout) {
325            self.maybe_extend(
326                Cursor::from_byte_index(layout, line.text_range().start, Affinity::Downstream),
327                extend,
328            )
329        } else {

331        }

334    /// Returns a new selection with the focus moved to just after the previous hard line break.
335    ///
336    /// If `extend` is `true` then the current anchor will be retained,
337    /// otherwise the new selection will be collapsed.
338    #[must_use]
339    pub fn hard_line_start<B: Brush>(&self, layout: &Layout<B>, extend: bool) -> Self {
340        if let Some((mut hard_line_start_index, line)) = self.focus.line(layout) {
341            let mut result_byte_index = line.text_range().start;
342            loop {
343                if hard_line_start_index == 0 {
344                    break;
345                }
346                let prev_index = hard_line_start_index - 1;
347                let Some(line) = layout.get(prev_index) else {
348                    unreachable!(
349                        "{hard_line_start_index} is a valid line in the layout, but {prev_index} isn't, despite the latter being smaller.\n\
350                        The layout has {} lines.",
351                        layout.len()
352                    );
353                };
354                if matches!(line.break_reason(), BreakReason::Explicit) {
355                    // The start of the line 'hard_line_start_index' is the target point.
356                    break;
357                }
358                result_byte_index = line.text_range().start;
359                hard_line_start_index = prev_index;
360            }
361            self.maybe_extend(
362                Cursor::from_byte_index(layout, result_byte_index, Affinity::Downstream),
363                extend,
364            )
365        } else {
366            *self
367        }
368    }

370    /// Returns a new selection with the focus moved to the end of the
371    /// current line.
372    ///
373    /// If `extend` is `true` then the current anchor will be retained,
374    /// otherwise the new selection will be collapsed.
375    #[must_use]
376    pub fn line_end<B: Brush>(&self, layout: &Layout<B>, extend: bool) -> Self {
377        if let Some((_, line)) = self.focus.line(layout) {
378            let (index, affinity) = (line.break_reason() == BreakReason::Explicit)
379                .then(|| {
380                    Cluster::from_byte_index(layout, line.text_range().end - 1)
381                        .map(|cluster| (cluster.text_range().start, Affinity::Downstream))
382                })
383                .flatten()
384                .unwrap_or_else(|| (line.text_range().end, Affinity::Upstream));
385            self.maybe_extend(Cursor::from_byte_index(layout, index, affinity), extend)
386        } else {
387            *self
388        }
389    }

391    /// Returns a new selection with the focus moved to just before the next hard line break.
392    ///
393    /// If `extend` is `true` then the current anchor will be retained,
394    /// otherwise the new selection will be collapsed.
395    #[must_use]
396    pub fn hard_line_end<B: Brush>(&self, layout: &Layout<B>, extend: bool) -> Self {
397        if let Some((mut hard_line_end_index, line)) = self.focus.line(layout) {
398            let mut result_byte_index = line.text_range().end;
399            // If we're already on the last line of the hard line, use that.
400            if !matches!(line.break_reason(), BreakReason::Explicit) {
401                // Otherwise, check if any of the following lines are the last line of the hard line.
402                loop {
403                    let next_index = hard_line_end_index + 1;
404                    if let Some(line) = layout.get(next_index) {
405                        result_byte_index = line.text_range().end;
406                        hard_line_end_index = next_index;
407                        if matches!(line.break_reason(), BreakReason::Explicit) {
408                            // result_byte_index is the last byte of the previous line, so is the value we need
409                            break;
410                        }
411                    } else {
412                        // We hit the end of text. Select to the end of the "final" line, which was not an EOF.
413                        return self.maybe_extend(
414                            Cursor::from_byte_index(layout, result_byte_index, Affinity::Upstream),
415                            extend,
416                        );
417                    }
418                }
419            }
420
421            // We want to select to "before" the newline character in the hard line, so we have downstream affinity on the boundary before it.
422            self.maybe_extend(
423                Cursor::from_byte_index(layout, result_byte_index - 1, Affinity::Downstream),
424                extend,
425            )
426        } else {
427            *self
428        }
429    }

431    /// Returns a new selection with the focus extended to the given point.
432    ///
433    /// If the initial selection was created from a word or line, then the new
434    /// selection will be extended at the same granularity.
435    #[must_use]
436    pub fn extend_to_point<B: Brush>(&self, layout: &Layout<B>, x: f32, y: f32) -> Self {
437        match self.anchor_base {
438            AnchorBase::Cluster => Self::new(self.anchor, Cursor::from_point(layout, x, y)),
439            AnchorBase::Word(start, end) => {
440                let target = Self::word_from_point(layout, x, y);
441                let [anchor, focus] = extend_selection(target, [start, end]);
442                Self {
443                    anchor,
444                    focus,
445                    anchor_base: self.anchor_base,
446                    h_pos: None,
447                }
448            }
449            AnchorBase::Line(start, end) => {
450                let target = Self::line_from_point(layout, x, y);
451                let [anchor, focus] = extend_selection(target, [start, end]);
452                Self {
453                    anchor,
454                    focus,
455                    anchor_base: self.anchor_base,
456                    h_pos: None,
457                }
458            }
459        }
460    }

462    /// Returns a new selection with the focus extended to the given point.
463    #[must_use]
464    pub fn shift_click_extension<B: Brush>(&self, layout: &Layout<B>, x: f32, y: f32) -> Self {
465        let target = Cursor::from_point(layout, x, y);
466        match self.anchor_base {
467            AnchorBase::Cluster => Self::new(self.anchor, target),
468            AnchorBase::Word(start, end) | AnchorBase::Line(start, end) => {
469                // Place the focus where the user just clicked, and the anchor on the "far" side of the anchorbase.
470                let anchor = if target.index < start.index {
471                    end
472                } else {
473                    start
474                };
475                Self {
476                    anchor,
477                    focus: target,
478                    anchor_base: self.anchor_base,
479                    h_pos: None,
480                }
481            }
482        }
483    }

492    /// Returns a vector containing the rectangles which represent the visual
493    /// geometry of this selection for the given layout, and the indices of the
494    /// lines to which they belong.
495    ///
496    /// This is a convenience method built on [`geometry_with`](Self::geometry_with).
497    pub fn geometry<B: Brush>(&self, layout: &Layout<B>) -> Vec<(BoundingBox, usize)> {
498        let mut rects = Vec::new();
499        self.geometry_with(layout, |rect, line_idx| rects.push((rect, line_idx)));
500        rects
501    }

503    /// Invokes `f` with the sequence of rectangles which represent the visual
504    /// geometry of this selection for the given layout, and the indices of the
505    /// lines to which they belong.
506    ///
507    /// This avoids allocation if the intent is to render the rectangles
508    /// immediately.
509    pub fn geometry_with<B: Brush>(
510        &self,
511        layout: &Layout<B>,
512        mut f: impl FnMut(BoundingBox, usize),
513    ) {
514        const NEWLINE_WHITESPACE_WIDTH_RATIO: f64 = 0.25;
515        if self.is_collapsed() {
516            return;
517        }
518        let mut start = self.anchor;
519        let mut end = self.focus;
520        if start.index > end.index {
521            core::mem::swap(&mut start, &mut end);
522        }
523        let text_range = start.index..end.index;
524        let line_start_ix = start.line(layout).map(|(ix, _)| ix).unwrap_or(0);
525        let line_end_ix = end
526            .line(layout)
527            .map(|(ix, _)| ix)
528            .unwrap_or(layout.len() + 1);
529        for line_ix in line_start_ix..=line_end_ix {
530            let Some(line) = layout.get(line_ix) else {
531                continue;
532            };
533            let metrics = line.metrics();
534            let line_min = metrics.block_min_coord as f64;
535            let line_max = metrics.block_max_coord as f64;
536            // Trailing whitespace to indicate that the newline character at the
537            // end of this line is selected. It's based on the ascent and
538            // descent so it doesn't change with the line height.
539            //
540            // TODO: the width of this whitespace should be the width of a space
541            // (U+0020) character.
542            let newline_whitespace = if line.break_reason() == BreakReason::Explicit {
543                (metrics.ascent as f64 + metrics.descent as f64) * NEWLINE_WHITESPACE_WIDTH_RATIO
544            } else {
545                0.0
546            };
547            if line_ix == line_start_ix || line_ix == line_end_ix {
548                // We only need to run the expensive logic on the first and
549                // last lines
550                let mut start_x = metrics.offset as f64 + metrics.inline_min_coord as f64;
551                let mut cur_x = start_x;
552                let mut cluster_count = 0;
553                let mut box_advance = 0.0;
554                let mut have_seen_any_runs = false;
555                for item in line.items_nonpositioned() {
556                    match item {
557                        LineItem::Run(run) => {
558                            have_seen_any_runs = true;
559                            for cluster in run.visual_clusters() {
560                                let advance = cluster.advance() as f64 + box_advance;
561                                box_advance = 0.0;
562                                if text_range.contains(&cluster.text_range().start) {
563                                    cluster_count += 1;
564                                    cur_x += advance;
565                                } else {
566                                    if cur_x != start_x {
567                                        f(
568                                            BoundingBox::new(start_x, line_min, cur_x, line_max),
569                                            line_ix,
570                                        );
571                                    }
572                                    cur_x += advance;
573                                    start_x = cur_x;
574                                }
575                            }
576                        }
577                        LineItem::InlineBox(inline_box) => {
578                            box_advance += inline_box.width as f64;
579                            // HACK: Don't display selections for inline boxes
580                            // if they're the first thing in the line. This
581                            // makes the selection match the cursor position.
582                            if !have_seen_any_runs {
583                                cur_x += box_advance;
584                                box_advance = 0.0;
585                                start_x = cur_x;
586                            }
587                        }
588                    }
589                }
590                let mut end_x = cur_x;
591                if line_ix != line_end_ix || (cluster_count != 0 && metrics.advance == 0.0) {
592                    end_x += newline_whitespace;
593                }
594                if end_x != start_x {
595                    f(
596                        BoundingBox::new(start_x, line_min, end_x, line_max),
597                        line_ix,
598                    );
599                }
600            } else {
601                let x = metrics.offset as f64 + metrics.inline_min_coord as f64;
602                let width = metrics.advance as f64;
603                f(
604                    BoundingBox::new(x, line_min, x + width + newline_whitespace, line_max),
605                    line_ix,
606                );
607            }
608        }
609    }

611    pub(crate) fn maybe_extend(&self, focus: Cursor, extend: bool) -> Self {
612        if extend {
613            Self::new(self.anchor, focus)
614        } else {
615            focus.into()
616        }

621        &self,

645/// Given the anchor base and the target selection, returns the anchor and focus of the resulting selection extension.
646fn extend_selection(target_selection: Selection, anchor_base: [Cursor; 2]) -> [Cursor; 2] {
647    let extending_to_the_right = target_selection.anchor.index >= anchor_base[0].index;
648
649    let cursors = [
650        target_selection.anchor,
651        target_selection.focus,
652        anchor_base[0],
653        anchor_base[1],
654    ];
655
656    let mut min = cursors[0];
657    let mut max = cursors[0];
658    for pos in cursors {
659        if pos.index < min.index {
660            min = pos;
661        }
662        if pos.index > max.index {
663            max = pos;
664        }
665    }
666
667    if extending_to_the_right {
668        [min, max]
669    } else {
670        [max, min]
671    }
672}
