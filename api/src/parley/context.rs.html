20/// Shared scratch space used when constructing text layouts.
21///
22/// This type is designed to be a global resource with only one per-application (or per-thread).
23pub struct LayoutContext<B: Brush = [u8; 4]> {
24    pub(crate) rcx: ResolveContext,
25    pub(crate) style_table: Vec<ResolvedStyle<B>>,
26    pub(crate) style_runs: Vec<StyleRun>,
27    pub(crate) inline_boxes: Vec<InlineBox>,
28    pub(crate) bidi: BidiResolver,
29
30    // Reusable style builders (to amortise allocations)
31    pub(crate) ranged_style_builder: RangedStyleBuilder<B>,
32    pub(crate) tree_style_builder: TreeStyleBuilder<B>,
33
34    // u16: style index for character
35    pub(crate) info: Vec<(CharInfo, u16)>,
36    pub(crate) scx: ShapeContext,
37
38    // Unicode analysis data sources (provided by icu)
39    pub(crate) analysis_data_sources: AnalysisDataSources,
40}
41
42impl<B: Brush> LayoutContext<B> {
43    pub fn new() -> Self {
44        Self {
45            rcx: ResolveContext::default(),
46            style_table: vec![],
47            style_runs: vec![],
48            inline_boxes: vec![],
49            bidi: BidiResolver::new(),
50            ranged_style_builder: RangedStyleBuilder::default(),
51            tree_style_builder: TreeStyleBuilder::default(),
52            info: vec![],
53            analysis_data_sources: AnalysisDataSources::new(),
54            scx: ShapeContext::default(),
55        }
56    }
57
58    fn resolve_style_set(
59        &mut self,
60        font_ctx: &mut FontContext,
61        scale: f32,
62        raw_style: &TextStyle<'_, '_, B>,
63    ) -> ResolvedStyle<B> {
64        self.rcx
65            .resolve_entire_style_set(font_ctx, raw_style, scale)
66    }
67
68    /// Create a ranged style layout builder.
69    ///
70    /// Set `quantize` as `true` to have the layout coordinates aligned to pixel boundaries.
71    /// That is the easiest way to avoid blurry text and to receive ready-to-paint layout metrics.
72    ///
73    /// For advanced rendering use cases you can set `quantize` as `false` and receive
74    /// fractional coordinates. This ensures the most accurate results if you want to perform
75    /// some post-processing on the coordinates before painting. To avoid blurry text you will
76    /// still need to quantize the coordinates just before painting.
77    ///
78    /// Your should round at least the following:
79    /// * Glyph run baseline
80    /// * Inline box baseline
81    ///   - `box.y = (box.y + box.height).round() - box.height`
82    /// * Selection geometry's `y0` & `y1`
83    /// * Cursor geometry's `y0` & `y1`
84    ///
85    /// Keep in mind that for the simple `f32::round` to be effective,
86    /// you need to first ensure the coordinates are in physical pixel space.
87    pub fn ranged_builder<'a>(
88        &'a mut self,
89        fcx: &'a mut FontContext,
90        text: &'a str,
91        scale: f32,
92        quantize: bool,
93    ) -> RangedBuilder<'a, B> {
94        self.begin();
95
96        let resolved_root_style = self.resolve_style_set(fcx, scale, &TextStyle::default());
97        self.ranged_style_builder
98            .begin(resolved_root_style, text.len());
99
100        fcx.source_cache.prune(128, false);
101
102        RangedBuilder {
103            scale,
104            quantize,
105            lcx: self,
106            fcx,
107        }
108    }
109
110    /// Create a builder for constructing a layout from indexed style runs.
111    ///
112    /// Unlike [`Self::ranged_builder`], this builder expects callers to provide:
113    /// - a style table of fully specified [`TextStyle`] values (via [`StyleRunBuilder::push_style`])
114    /// - a complete sequence of **contiguous**, **non-overlapping** spans that cover
115    ///   `0..text.len()` and reference style indices (via [`StyleRunBuilder::push_style_run`])
116    ///
117    /// Parley then skips its internal range-splitting logic.
118    pub fn style_run_builder<'a>(
119        &'a mut self,
120        fcx: &'a mut FontContext,
121        text: &'a str,
122        scale: f32,
123        quantize: bool,
124    ) -> StyleRunBuilder<'a, B> {
125        self.begin();
126
127        fcx.source_cache.prune(128, false);
128
129        StyleRunBuilder {
130            scale,
131            quantize,
132            len: text.len(),
133            lcx: self,
134            fcx,
135            cursor: 0,
136        }
137    }
138
139    /// Create a tree style layout builder.
140    ///
141    /// Set `quantize` as `true` to have the layout coordinates aligned to pixel boundaries.
142    /// That is the easiest way to avoid blurry text and to receive ready-to-paint layout metrics.
143    ///
144    /// For advanced rendering use cases you can set `quantize` as `false` and receive
145    /// fractional coordinates. This ensures the most accurate results if you want to perform
146    /// some post-processing on the coordinates before painting. To avoid blurry text you will
147    /// still need to quantize the coordinates just before painting.
148    ///
149    /// Your should round at least the following:
150    /// * Glyph run baseline
151    /// * Inline box baseline
152    ///   - `box.y = (box.y + box.height).round() - box.height`
153    /// * Selection geometry's `y0` & `y1`
154    /// * Cursor geometry's `y0` & `y1`
155    ///
156    /// Keep in mind that for the simple `f32::round` to be effective,
157    /// you need to first ensure the coordinates are in physical pixel space.
158    pub fn tree_builder<'a>(
159        &'a mut self,
160        fcx: &'a mut FontContext,
161        scale: f32,
162        quantize: bool,
163        root_style: &TextStyle<'_, '_, B>,
164    ) -> TreeBuilder<'a, B> {
165        self.begin();
166
167        let resolved_root_style = self.resolve_style_set(fcx, scale, root_style);
168        self.tree_style_builder.begin(resolved_root_style);
169
170        fcx.source_cache.prune(128, false);
171
172        TreeBuilder {
173            scale,
174            quantize,
175            lcx: self,
176            fcx,
177        }
178    }
179
180    fn begin(&mut self) {
181        self.rcx.clear();
182        self.style_table.clear();
183        self.style_runs.clear();
184        self.inline_boxes.clear();
185        self.info.clear();
186        self.bidi.clear();
187    }
188}
189
190impl<B: Brush> Default for LayoutContext<B> {
191    fn default() -> Self {
192        Self::new()
193    }
194}
195
196impl<B: Brush> Clone for LayoutContext<B> {
197    fn clone(&self) -> Self {
198        // None of the internal state is visible so just return a new instance.
199        Self::new()
200    }
201}
