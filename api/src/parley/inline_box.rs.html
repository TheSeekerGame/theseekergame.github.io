4/// A box to be laid out inline with text
5#[derive(PartialEq, Debug, Clone)]
6pub struct InlineBox {
7    /// User-specified identifier for the box, which can be used by the user to determine which box in
8    /// parley's output corresponds to which box in its input.
9    pub id: u64,
10    /// Whether the box is in-flow (takes up space in the layout) or out-of-flow (e.g. absolutely positioned or floated)
11    pub kind: InlineBoxKind,
12    /// The byte offset into the underlying text string at which the box should be placed.
13    /// This must not be within a Unicode code point.
14    pub index: usize,
15    /// The width of the box in pixels
16    pub width: f32,
17    /// The height of the box in pixels
18    pub height: f32,
19}
20
21/// Whether a box is in-flow (takes up space in the layout) or out-of-flow (e.g. absolutely positioned)
22/// or custom-out-of-flow (line-breaking should yield control flow)
23#[derive(PartialEq, Debug, Clone, Copy)]
24pub enum InlineBoxKind {
25    /// `InFlow` boxes take up space in the layout and flow in line with text
26    ///
27    /// They correspond to `display: inline-block` boxes in CSS.
28    InFlow,
29    /// `OutOfFlow` boxes are assigned a position as if they were a zero-sized inline box, but
30    /// do not take up space in the layout.
31    ///
32    /// They correspond to `position: absolute` boxes in CSS.
33    OutOfFlow,
34    /// `CustomOutOfFlow` boxes also do not take up space in the layout, but they are not assigned a position
35    /// by Parley. When they are encountered, control flow is yielded back to the caller who is then responsible
36    /// for laying out the box.
37    ///
38    /// They can be used to implement advanced layout modes such as CSS's `float`
39    CustomOutOfFlow,
40}
